11/// What the default exclusion rules skipped, in the detail a report shows.
12///
13/// Kept beside [`crate::policy::Coverage`] rather than inside it: the counts a report needs
14/// are derivable from `Coverage`, and putting the reasons and the sample in
15/// their own type means adding detail later never changes `Coverage`'s
16/// public shape. New fields here are additive.
17#[derive(Debug, Clone, Default, PartialEq, Eq)]
18#[non_exhaustive]
19pub struct Exclusions {
20    /// Excluded paths by reason (`dir:vendor`, `size>500k`, `minified`,
21    /// `hidden-dir`). A directory counts once, so dependency directories
22    /// appear here without inflating `files_excluded`.
23    pub by_reason: std::collections::BTreeMap<String, usize>,
24    /// Supported source files the exclusions hid. This is the number
25    /// [`crate::policy::Coverage`] adds to its denominator.
26    pub files_excluded: usize,
27    /// First [`COVERAGE_SAMPLE`] excluded paths, those holding source files
28    /// first.
29    pub sample: Vec<ExcludedPath>,
30}
31
32impl Exclusions {
33    /// Summarize one walk's excluded paths for the report.
34    #[must_use]
35    pub fn from_paths(excluded: &[ExcludedPath]) -> Self {
36        let mut by_reason: std::collections::BTreeMap<String, usize> =
37            std::collections::BTreeMap::new();
38        let mut files_excluded = 0usize;
39        for e in excluded {
40            *by_reason.entry(e.reason.clone()).or_insert(0) += 1;
41            files_excluded += e.source_files.unwrap_or(0);
42        }
43        let mut sample: Vec<ExcludedPath> = excluded.to_vec();
44        // Paths holding source first, then by path: the sample leads with the
45        // exclusions that matter.
46        sample.sort_by(|a, b| {
47            (a.source_files.unwrap_or(0) == 0, &a.path)
48                .cmp(&(b.source_files.unwrap_or(0) == 0, &b.path))
49        });
50        sample.truncate(COVERAGE_SAMPLE);
51        Self {
52            by_reason,
53            files_excluded,
54            sample,
55        }
56    }
57}
