5//! `--explain`: why the engine stayed silent - the "engine limitation" map.
6//!
7//! Collected under `--explain` so users can see *where* analysis stopped.
8//! The reporting policy (which findings become advisories, and the
9//! severity/confidence floor) lives in [`crate::policy`].
10
11use crate::policy::reports_finding;

15/// How a sink argument slot reads to a person: the receiver of a method call
16/// is stored as `usize::MAX`, which must never reach the screen as a number.
17#[must_use]
18pub(crate) fn slot_label(slot: usize) -> String {
19    if slot == frensense_engine::graph::svfg::RECEIVER_ARG_INDEX {
20        "receiver".to_string()
21    } else {
22        format!("arg {slot}")
23    }
24}

26/// One bundle sanitizer fact that suppressed a flow, as shown by
27/// `--show-suppressed`.
28#[derive(Debug, Clone, PartialEq, Eq)]
29pub struct SuppressedFact {
30    /// Sanitizing call name the walk met.
31    pub call: String,
32    /// Classes the fact was proven for (empty = legacy universal).
33    pub classes: Vec<String>,
34}
35
36impl SuppressedFact {
37    /// One-line audit form: `X (class Z)`.
38    #[must_use]
39    pub fn describe(&self) -> String {
40        let class = if self.classes.is_empty() {
41            "all classes (legacy unscoped)".to_string()
42        } else {
43            self.classes.join(", ")
44        };
45        format!("{} (class {class})", self.call)
46    }
47}

49/// One explored sink argument that did NOT become an advisory, with the
50/// reason the engine stayed silent. Collected under `--explain` so users
51/// can see *where* analysis stopped - the "engine limitation" map.
52///
53/// `#[non_exhaustive]`: an evolving diagnostic DTO; new fields are minor
54/// releases.
55#[derive(Debug, Clone)]
56#[non_exhaustive]
57pub struct ExplainRecord {
58    pub file: String,
59    pub line: u32,
60    pub sink: String,
61    pub arg_slot: usize,
62    /// Short verdict label: `"no-source"`, `"sanitized"`, `"unknown"`, or
63    /// `"safe-channel"` (vulnerable flow into a non-dangerous slot).
64    pub verdict: String,
65    /// Distinct silent stop sites hit during the walk (`"guard"`,
66    /// `"value-const"`, `"sanitizer"`, `"unresolvable"`); empty when the
67    /// walk simply exhausted its predecessors without a source.
68    pub cut_reasons: Vec<&'static str>,
69    /// Human summary of why nothing was reported.
70    pub reason: String,
71    /// The source access path, when the walk did reach one (e.g. a
72    /// `Vulnerable` flow into a safe channel).
73    pub source_desc: Option<String>,
74    /// Function holding the sink call.
75    pub function: String,
76    /// Where the walk met a value it could not resolve (the actionable half
77    /// of the `unresolvable` reason): callee, parameter or global, with its
78    /// file and line.
79    pub cut_sites: Vec<LocatedCutSite>,
80    /// Bundle sanitizer facts that suppressed this flow, with provenance.
81    pub suppressed_by: Vec<SuppressedFact>,
82}
83
84impl ExplainRecord {
85    /// Machine-readable form, for `--explain-json` and downstream tooling.
86    #[must_use]
87    pub fn to_json(&self) -> serde_json::Value {
88        serde_json::json!({
89            "file": self.file,
90            "line": self.line,
91            "function": self.function,
92            "sink": self.sink,
93            "arg_slot": self.arg_slot,
94            "arg": slot_label(self.arg_slot),
95            "verdict": self.verdict,
96            "cut_reasons": self.cut_reasons,
97            "reason": self.reason,
98            "source": self.source_desc,
99            "cut_sites": self.cut_sites.iter().map(|c| serde_json::json!({
100                "kind": c.kind,
101                "function": c.function,
102                "detail": c.detail,
103                "file": c.file,
104                "line": c.line,
105            })).collect::<Vec<_>>(),
106            "suppressed_by": self.suppressed_by.iter().map(|s| serde_json::json!({
107                "sanitizer": s.call,
108                "classes": s.classes,
109            })).collect::<Vec<_>>(),
110        })
111    }
112}

114fn verdict_label(v: &BackwardVerdict) -> &'static str {
115    match v {
116        BackwardVerdict::Vulnerable => "vulnerable",
117        BackwardVerdict::Sanitized => "sanitized",
118        BackwardVerdict::Unknown => "unknown",
119        // The walk found no definition feeding the argument. That is a
120        // statement about the walk, not a claim that the code is safe.
121        BackwardVerdict::Clean => "no-source",
122    }
123}
124
125fn cut_summary(cuts: &[&'static str]) -> String {
126    if cuts.is_empty() {
127        return "no attacker source reached the sink; the backward walk exhausted its predecessors (flow died or was never tainted)".to_string();
128    }
129    let names: Vec<&str> = cuts
130        .iter()
131        .map(|c| match *c {
132            "guard" => "a validating guard dominated the use",
133            "value-const" => "the value folded to an attacker-independent constant",
134            "sanitizer" => "a sanitizer transformed the value",
135            "unresolvable" => "the flow hit a value the engine cannot resolve",
136            other => other,
137        })
138        .collect();
139    names.join("; ")
140}

142/// Build an [`ExplainRecord`] for a located finding that will not be
143/// reported. Returns `None` for findings that *are* reported (callers
144/// should only pass silent ones).
145#[must_use]
146pub fn explain_silent(loc: &LocatedFinding) -> Option<ExplainRecord> {
147    if reports_finding(&loc.finding) {
148        return None;
149    }
150    let f = &loc.finding;
151    let (verdict, reason) = if f.verdict == BackwardVerdict::Vulnerable && f.alert.is_some() {
152        // Vulnerable on a dangerous slot, but the sink resolves to a
153        // fallback rule card: it names no vulnerability class, so the
154        // reporting policy drops it (nothing to fix, suppress or trend).
155        (
156            "unclassified".to_string(),
157            format!(
158                "taint reaches `{}` but the sink names no vulnerability class, so no advisory is reported",
159                f.sink
160            ),
161        )
162    } else if f.verdict == BackwardVerdict::Vulnerable {
163        // Vulnerable but alert is None: the flow landed in a safe channel.
164        (
165            "safe-channel".to_string(),
166            format!(
167                "taint reaches `{}` but its {} is a safe (non-injection) channel",
168                f.sink,
169                slot_label(f.arg_slot)
170            ),
171        )
172    } else {
173        (
174            verdict_label(&f.verdict).to_string(),
175            cut_summary(&f.cut_reasons),
176        )
177    };
178    // A flow that ended without a source may have crossed library calls the
179    // engine has no model for: name them, since a missing source or
180    // propagation entry for one of them is the usual reason for silence.
181    let passed: Vec<String> = loc
182        .cut_sites
183        .iter()
184        .filter(|c| c.kind == "passed-through")
185        .map(|c| format!("`{}`", c.detail))
186        .collect();
187    let reason = if passed.is_empty() || f.verdict == BackwardVerdict::Vulnerable {
188        reason
189    } else {
190        format!(
191            "{reason}; the walk assumed {} carries its arguments into the result - if one of them is where request data enters, no source is modelled for it",
192            passed.join(", ")
193        )
194    };
195    let suppressed_by: Vec<SuppressedFact> = f
196        .suppressions
197        .iter()
198        .map(|s| SuppressedFact {
199            call: s.call.clone(),
200            classes: s.classes.clone(),
201        })
202        .collect();
203    let reason = if suppressed_by.is_empty() {
204        reason
205    } else {
206        format!(
207            "{reason}; suppressed by bundle sanitizer {}",
208            suppressed_by
209                .iter()
210                .map(SuppressedFact::describe)
211                .collect::<Vec<_>>()
212                .join("; ")
213        )
214    };
215    Some(ExplainRecord {
216        file: loc.file.clone(),
217        line: loc.line,
218        sink: f.sink.clone(),
219        arg_slot: f.arg_slot,
220        verdict,
221        cut_reasons: f.cut_reasons.clone(),
222        reason,
223        source_desc: f.source_desc.clone(),
224        function: f.function.clone(),
225        cut_sites: loc.cut_sites.clone(),
226        suppressed_by,
227    })
228}

230/// Build the `--show-suppressed` record for a finding whose flow was silenced
231/// by a bundle sanitizer fact. Returns `None` for findings with no such fact,
232/// so the switch never adds ordinary silent-sink noise.
233#[must_use]
234pub fn explain_suppressed(loc: &LocatedFinding) -> Option<ExplainRecord> {
235    if loc.finding.suppressions.is_empty() {
236        return None;
237    }
238    explain_silent(loc)
239}

241#[cfg(test)]
242mod explain_tests {
243    use super::*;
244    use frensense_engine::analysis::forward::SinkAlert;
245    use frensense_engine::analysis::taint::engine::{FindingClass, SinkFinding};
246    use frensense_engine::analysis::taint::path::TaintPath;
247    use frensense_engine::analysis::taint::role::SinkRole;
248
249    fn loc(
250        verdict: BackwardVerdict,
251        alert: Option<SinkAlert>,
252        cuts: Vec<&'static str>,
253    ) -> LocatedFinding {
254        LocatedFinding {
255            finding: SinkFinding::builder()
256                .function("handler")
257                .sink("exec")
258                .arg_slot(0)
259                .alert(alert)
260                .verdict(verdict)
261                .finding_class(FindingClass::Injection)
262                .role(SinkRole::Execution)
263                // A class the sink names: unlabeled Execution resolves to the
264                // generic card and would never be reported.
265                .label(Some("command".to_string()))
266                .path(TaintPath::default())
267                .cut_reasons(cuts)
268                .build(),
269            file: "app.py".to_string(),
270            line: 42,
271            column: 1,
272            cut_sites: Vec::new(),
273        }
274    }
275
276    #[test]
277    fn the_receiver_slot_reads_as_receiver() {
278        assert_eq!(slot_label(0), "arg 0");
279        assert_eq!(
280            slot_label(frensense_engine::graph::svfg::RECEIVER_ARG_INDEX),
281            "receiver"
282        );
283        // The JSON carries both the raw slot and the label.
284        let record = explain_silent(&loc(BackwardVerdict::Clean, None, vec!["value-const"]))
285            .expect("a silent finding is explained");
286        let json = record.to_json();
287        assert_eq!(json["arg"], "arg 0");
288    }
289
290    #[test]
291    fn vulnerable_with_alert_is_not_explained() {
292        let alert = Some(SinkAlert {
293            sink: "exec".to_string(),
294            slot: 0,
295            function: "handler".to_string(),
296            class: FindingClass::Injection,
297        });
298        let l = loc(BackwardVerdict::Vulnerable, alert, Vec::new());
299        assert!(explain_silent(&l).is_none());
300    }
301
302    #[test]
303    fn unclassified_vulnerable_sink_is_explained() {
304        let mut l = loc(
305            BackwardVerdict::Vulnerable,
306            Some(SinkAlert {
307                sink: "exec".to_string(),
308                slot: 0,
309                function: "handler".to_string(),
310                class: FindingClass::Injection,
311            }),
312            Vec::new(),
313        );
314        l.finding.label = None;
315        let rec = explain_silent(&l).expect("record");
316        assert_eq!(rec.verdict, "unclassified");
317        assert!(
318            rec.reason.contains("names no vulnerability class"),
319            "{}",
320            rec.reason
321        );
322    }

324    #[test]
325    fn vulnerable_without_alert_is_safe_channel() {
326        let l = loc(BackwardVerdict::Vulnerable, None, Vec::new());
327        let rec = explain_silent(&l).expect("record");
328        assert_eq!(rec.verdict, "safe-channel");
329        assert!(rec.reason.contains("safe"));
330    }

348    fn sanitized_verdict_label() {
349        let l = loc(BackwardVerdict::Sanitized, None, vec!["sanitizer"]);

351        assert_eq!(rec.verdict, "sanitized");

354    #[test]
355    fn unknown_record_carries_its_stop_sites_in_json() {
356        let mut l = loc(BackwardVerdict::Unknown, None, vec!["unresolvable"]);
357        l.cut_sites = vec![LocatedCutSite {
358            kind: "external-call",
359            function: "handler".into(),
360            detail: "os.path.join".into(),
361            file: "app.py".into(),
362            line: 40,
363        }];
364        let rec = explain_silent(&l).expect("record");
365        let j = rec.to_json();
366        assert_eq!(j["function"], "handler");
367        assert_eq!(j["cut_sites"][0]["kind"], "external-call");
368        assert_eq!(j["cut_sites"][0]["detail"], "os.path.join");
369        assert_eq!(j["cut_sites"][0]["line"], 40);
370    }
