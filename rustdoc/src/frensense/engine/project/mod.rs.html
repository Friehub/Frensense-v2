7use crate::{FileId, Severity};

10/// The scan engine.
11///
12/// Configuration is minimal by design: the analysis knowledge lives in the
13/// language specs (`frensense-lang`) and the `.frc` bundle's learned facts,
14/// not in knobs on this struct.
15#[derive(Debug)]
16pub struct Engine {
17    /// Optional `.frc` bundle bytes (static) whose learned facts are merged
18    /// over the built-in fact table.
19    pub(crate) corpus_bundle: Option<&'static [u8]>,
20    /// Optional `.frc` bundle loaded from disk at scan time.
21    pub(crate) corpus_bundle_path: Option<PathBuf>,
22    pub(crate) severity_filter: Option<Severity>,
23    pub(crate) language_filter: Option<Vec<&'static str>>,
24    pub(crate) min_confidence: f64,
25    /// When `true`, the runner records every explored sink argument that
26    /// did NOT become an advisory (see [`crate::explain::ExplainRecord`]),
27    /// surfaced via [`Engine::take_explain`] under `--explain`.
28    pub(crate) explain: bool,
29    /// When `true` (`--show-suppressed`), also record flows silenced by a
30    /// bundle sanitizer fact, with its family and class provenance, even
31    /// without `--explain`.
32    pub(crate) show_suppressed: bool,
33    /// When `true` (`--show-flows`), the taint advisory TEXT also embeds
34    /// the full source→sink flow. `taint_steps`/SARIF `codeFlows` carry
35    /// the path unconditionally now (WI-54); this knob only governs the
36    /// text observation, the debug-grade detail of a terse text report.
37    pub(crate) show_flows: bool,
38    /// Precision profile governing reportable finding tiers.
39    pub(crate) profile: crate::Profile,
40    /// When `true` (`--hygiene`), hygiene findings (such as `function_complexity`)
41    /// are reported even if not included in the active scan profile.
42    pub(crate) enable_hygiene: bool,
43    /// Optional cyclomatic complexity limit override (`--max-complexity`).
44    pub(crate) max_complexity: Option<usize>,
45    /// Rule ids suppressed by `--disable-rule` or `[rules] disable`.
46    pub(crate) disabled_rules: std::collections::BTreeSet<String>,
47    /// Overrides for the default file-collection exclusions.
48    pub(crate) collect_options: crate::engine::files::CollectOptions,
49    /// When set, only these rule ids are reported (`--only`).
50    pub(crate) only_rules: Option<std::collections::BTreeSet<String>>,
51    /// Silent-finding records from the most recent `run` (when `explain`).
52    pub(crate) explain_records: Vec<crate::explain::ExplainRecord>,
53    /// Functions the per-function work budget pushed into the degraded
54    /// (flow-insensitive) form, accumulated across `run` calls and surfaced
55    /// via [`Engine::take_incomplete`] so the report can list them under
56    /// `analysis_incomplete` instead of silently weakening those results.
57    pub(crate) analysis_incomplete: Vec<frensense_engine::scan::IncompleteFunction>,
58    /// Source files skipped by the most recent `run` because the engine has
59    /// no language spec for them. Surfaced via
60    /// [`Engine::take_skipped_unsupported`] so the report can say which
61    /// languages it did not analyze instead of describing a partial scan as
62    /// complete.
63    pub(crate) skipped_unsupported: Vec<crate::engine::files::UnsupportedSourceFile>,
64    /// Hidden finding counts grouped by tier.
65    pub(crate) hidden_by_tier: std::collections::BTreeMap<crate::RuleTier, usize>,
66    /// Accumulated telemetry across scanned language groups.
67    pub(crate) telemetry: frensense_engine::telemetry::ScanTelemetry,
68    /// What the most recent `run` covered: source files found versus
69    /// analyzed, sink sites inspected, files with none, and sites the walk
70    /// left undecided. Surfaced via [`Engine::take_coverage`] so a "no
71    /// findings" report can say whether it means "inspected, no finding" or
72    /// "nothing recognized".
73    pub(crate) coverage: crate::policy::Coverage,
74    /// Paths the most recent run skipped by the default exclusion rules, with
75    /// the source files each hid. Surfaced via [`Engine::take_excluded`].
76    pub(crate) excluded: Vec<crate::engine::files::ExcludedPath>,
77}
78
79impl Engine {
80    #[must_use]
81    pub fn new() -> Self {
82        Self {
83            corpus_bundle: None,
84            corpus_bundle_path: None,
85            severity_filter: None,
86            language_filter: None,
87            min_confidence: 0.0,
88            explain: false,
89            show_suppressed: false,
90            show_flows: false,
91            profile: crate::Profile::Default,
92            enable_hygiene: false,
93            max_complexity: None,
94            disabled_rules: std::collections::BTreeSet::new(),
95            only_rules: None,
96            collect_options: crate::engine::files::CollectOptions::default(),
97            explain_records: Vec::new(),
98            analysis_incomplete: Vec::new(),
99            skipped_unsupported: Vec::new(),
100            hidden_by_tier: std::collections::BTreeMap::new(),
101            telemetry: frensense_engine::telemetry::ScanTelemetry::default(),
102            coverage: crate::policy::Coverage::default(),
103            excluded: Vec::new(),
104        }
105    }
106
107    /// Provide a `.frc` bundle (static bytes) whose learned facts teach the
108    /// engine additional sinks/sanitizers/guards.
109    pub fn set_corpus_bundle(&mut self, bundle: &'static [u8]) {
110        self.corpus_bundle = Some(bundle);
111    }
112
113    /// Provide a `.frc` bundle loaded from disk (mut + scan workflow).
114    pub fn set_corpus_bundle_path(&mut self, path: PathBuf) {
115        self.corpus_bundle_path = Some(path);
116    }
117
118    pub fn set_severity_filter(&mut self, severity: Option<Severity>) {
119        self.severity_filter = severity;
120    }
121
122    pub fn set_language_filter(&mut self, extensions: Vec<&'static str>) {
123        self.language_filter = Some(extensions);
124    }
125
126    pub fn set_min_confidence(&mut self, min: f64) {
127        self.min_confidence = min;
128    }
129
130    /// Enable `--explain`: record every silent sink argument (not reported)
131    /// with the reason the engine stayed silent, for later retrieval via
132    /// [`Engine::take_explain`].
133    pub fn set_explain(&mut self, on: bool) {
134        self.explain = on;
135    }
136
137    /// Adjust the default file exclusions (`--include-dir`,
138    /// `--no-default-excludes`).
139    pub fn set_collect_options(&mut self, include_dirs: Vec<String>, no_default_excludes: bool) {
140        self.collect_options = crate::engine::files::CollectOptions {
141            include_dirs,
142            no_default_excludes,
143        };
144    }
145
146    /// Set the user's rule filters (`--disable-rule`, `--only`, config).
147    pub fn set_rule_filters(&mut self, disable: Vec<String>, only: Vec<String>) {
148        self.disabled_rules = disable.into_iter().collect();
149        if only.is_empty() {
150            self.only_rules = None;
151        } else {
152            self.only_rules = Some(only.into_iter().collect());
153        }
154    }
155
156    /// Enable `--show-suppressed`: record flows a bundle sanitizer fact
157    /// silenced, naming the fact, family and class.
158    pub fn set_show_suppressed(&mut self, on: bool) {
159        self.show_suppressed = on;
160    }
161
162    /// Enable `--show-flows`: taint advisories embed the full source→sink
163    /// flow in the observation text as well. `taint_steps` (and SARIF
164    /// codeFlows) are always populated since WI-54; this only widens the
165    /// text observation.
166    pub fn set_show_flows(&mut self, on: bool) {
167        self.show_flows = on;
168    }
169
170    /// Take the explain records collected by the most recent `run`.
171    /// Returns an empty vec when `--explain` was not enabled.
172    #[must_use]
173    pub fn take_explain(&mut self) -> Vec<crate::explain::ExplainRecord> {
174        std::mem::take(&mut self.explain_records)
175    }
176
177    /// Take the functions analyzed in degraded form because their
178    /// per-function work budget expired (accumulated across `run` calls).
179    /// Empty when every function was analyzed fully.
180    #[must_use]
181    pub fn take_incomplete(&mut self) -> Vec<frensense_engine::scan::IncompleteFunction> {
182        std::mem::take(&mut self.analysis_incomplete)
183    }
184
185    /// Take the source files the most recent `run` skipped because their
186    /// language has no engine spec. A scan that reports findings while
187    /// silently ignoring the Python half of a repo is not complete, and the
188    /// caller is expected to say so.
189    #[must_use]
190    pub fn take_skipped_unsupported(&mut self) -> Vec<crate::engine::files::UnsupportedSourceFile> {
191        std::mem::take(&mut self.skipped_unsupported)
192    }
193
194    /// Set the scan profile governing which rule tiers are reported.
195    pub fn set_profile(&mut self, profile: crate::Profile) {
196        self.profile = profile;
197    }
198
199    /// Enable hygiene findings (such as function_complexity) under any profile.
200    pub fn set_enable_hygiene(&mut self, on: bool) {
201        self.enable_hygiene = on;
202    }
203
204    /// Set a custom cyclomatic complexity ceiling for the function_complexity check.
205    /// Setting this automatically enables hygiene rules.
206    pub fn set_max_complexity(&mut self, max: Option<usize>) {
207        self.max_complexity = max;
208        if max.is_some() {
209            self.enable_hygiene = true;
210        }
211    }
212
213    /// The active scan profile.
214    #[must_use]
215    pub fn profile(&self) -> crate::Profile {
216        self.profile
217    }
218
219    /// Take the counts of hidden findings by tier collected by the most recent `run`.
220    #[must_use]
221    pub fn take_hidden_by_tier(&mut self) -> std::collections::BTreeMap<crate::RuleTier, usize> {
222        std::mem::take(&mut self.hidden_by_tier)
223    }
224
225    /// Take the scan telemetry collected by the most recent `run`.
226    #[must_use]
227    pub fn take_telemetry(&mut self) -> frensense_engine::telemetry::ScanTelemetry {
228        std::mem::take(&mut self.telemetry)
229    }
230
231    /// Take what the most recent `run` covered: files found versus analyzed,
232    /// sink sites inspected, which files yielded none, and which sites it
233    /// left undecided. A consumer that reports "no findings" must consult
234    /// this, or it cannot tell "inspected, no finding" from "nothing
235    /// recognized".
236    #[must_use]
237    pub fn take_coverage(&mut self) -> crate::policy::Coverage {
238        std::mem::take(&mut self.coverage)
239    }
240
241    /// Take the paths the most recent `run` skipped by the default exclusion
242    /// rules, with the reason and the source files each hid. [`Coverage`]
243    /// counts them; this names them.
244    ///
245    /// [`Coverage`]: crate::policy::Coverage
246    #[must_use]
247    pub fn take_excluded(&mut self) -> Vec<crate::engine::files::ExcludedPath> {
248        std::mem::take(&mut self.excluded)
249    }
250
251    /// Register a synthetic file id for a scanned path (used by the runner
252    /// when converting findings to advisories).
253    #[allow(dead_code)]
254    pub(crate) fn next_file_id(seq: usize) -> FileId {
255        FileId(u32::try_from(seq).unwrap_or(u32::MAX))
256    }
257}
258
259impl Default for Engine {
260    fn default() -> Self {
261        Self::new()
262    }
263}
