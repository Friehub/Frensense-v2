5//! Watch mode: poll a directory tree, re-scan files whose mtime changed,
6//! and print only findings that are NEW since the previous run.
7//!
8//! Architecture split (deliberate): the watcher is a CLI delivery feature
9//! over the existing scan API. The engine knows nothing about watching —
10//! `Engine::run(path)` stays the only analysis entry point, invoked here
11//! per changed file. This module owns three concerns, each independently
12//! testable without filesystem events or a real engine:
13//!
14//! 1. [`snapshot_mtimes`] — the observed state (path → mtime seconds) using
15//!    the same `collect_files` ignore rules as a normal scan;
16//! 2. [`FileWatcher::changed_files`] — the diff between observations;
17//! 3. [`new_advisories`] — fingerprint-set diff between scan rounds so a
18//!    file whose findings didn't change is silent.
19//!
20//! Polling (500ms) rather than an inotify/FSEvents dependency: zero new
21//! crates, deterministic in tests, and the per-file scan is fast enough
22//! that re-stat-ing the tree is not the bottleneck. The upgrade path is
23//! swapping [`snapshot_mtimes`] for a notify-debounced event source
24//! without touching the loop below.
25
26use crate::Advisory;
27use crate::engine::files::collect_files;
28use rustc_hash::FxHashMap;
29use std::path::{Path, PathBuf};
30use std::time::{Duration, SystemTime};
31
32/// Poll interval between tree observations.
33pub const POLL_INTERVAL: Duration = Duration::from_millis(500);
34
35/// One tree observation: supported files → mtime (second resolution).
36/// Second resolution is deliberate: it matches what filesystems and git
37/// guarantee, and sub-second churn within one poll tick collapses into a
38/// single change detection on the next tick anyway.
39pub fn snapshot_mtimes(
40    root: &Path,
41    language_filter: Option<&Vec<&'static str>>,
42) -> FxHashMap<PathBuf, u64> {
43    let mut snap = FxHashMap::default();
44    for path in collect_files(root, language_filter) {
45        if let Ok(meta) = std::fs::metadata(&path)
46            && let Ok(mtime) = meta.modified()
47            && let Ok(d) = mtime.duration_since(SystemTime::UNIX_EPOCH)
48        {
49            snap.insert(path, d.as_secs());
50        }
51    }
52    snap
53}
54
55/// Rolling watcher state: the last observation.
56#[derive(Debug, Default)]
57pub struct FileWatcher {
58    last: FxHashMap<PathBuf, u64>,
59}
60
61impl FileWatcher {
62    /// Take the first observation without reporting it as "changed" —
63    /// the initial scan is not an edit.
64    pub fn prime(&mut self, root: &Path, language_filter: Option<&Vec<&'static str>>) {
65        self.last = snapshot_mtimes(root, language_filter);
66    }
67
68    /// Observe the tree again; return files that are new or modified since
69    /// the previous observation, sorted for deterministic output. Deleted
70    /// files drop out of the snapshot silently — their findings age out
71    /// through the baseline merge in the loop.
72    pub fn changed_files(
73        &mut self,
74        root: &Path,
75        language_filter: Option<&Vec<&'static str>>,
76    ) -> Vec<PathBuf> {
77        let current = snapshot_mtimes(root, language_filter);
78        let mut changed = Vec::new();
79        for (path, mtime) in &current {
80            match self.last.get(path) {
81                None => changed.push(path.clone()),
82                Some(prev) if prev != mtime => changed.push(path.clone()),
83                Some(_) => {}
84            }
85        }
86        changed.sort();
87        self.last = current;
88        changed
89    }
90}

92/// Advisories present in `current` whose fingerprint was absent from
93/// `previous` — the watch loop prints exactly these, so an edit that
94/// doesn't change findings stays silent and a newly introduced bug
95/// surfaces immediately. Findings without a fingerprint (should not happen
96/// for dataflow advisories) are always treated as new: fail loud, not
97/// silent.
98pub fn new_advisories(previous: &[Advisory], current: &[Advisory]) -> Vec<Advisory> {
99    let seen: std::collections::HashSet<&str> =
100        previous.iter().map(|a| a.fingerprint.as_str()).collect();
101    current
102        .iter()
103        .filter(|a| !seen.contains(a.fingerprint.as_str()))
104        .cloned()
105        .collect()
106}
107
108/// Render one new advisory as a terminal line (watch mode is a human- and
109/// agent-facing stream, not a report file).
110pub fn format_watch_finding(adv: &Advisory) -> String {
111    let sev = match adv.severity {
112        crate::Severity::Critical => "CRITICAL",
113        crate::Severity::Warning => "WARNING",
114        crate::Severity::Info => "INFO",
115    };
116    format!(
117        "[{sev}] {} ({}:{}) :: {}",
118        adv.title, adv.file_path, adv.line, adv.observation
119    )
120}

122/// The watch loop. Prime → initial full scan → poll → re-scan changed
123/// files → print only new findings. Returns when `should_stop` fires
124/// (SIGINT handling in production; tests drive it directly).
125///
126/// The engine is injected as two closures so tests drive the loop without
127/// a real engine; production passes `|| Engine::configured()` and
128/// `|e, p| e.run(p)`.
129pub fn run_watch_loop<S, B, F>(
130    root: &Path,
131    language_filter: Option<&Vec<&'static str>>,
132    build_engine: B,
133    mut scan: F,
134    mut should_stop: impl FnMut() -> bool,
135) -> crate::Result<()>
136where
137    B: Fn() -> S,
138    F: FnMut(&mut S, &Path) -> crate::Result<Vec<Advisory>>,
139{
140    let mut watcher = FileWatcher::default();
141    watcher.prime(root, language_filter);
142
143    // Initial round: full scan of the root, establishes the baseline
144    // fingerprint set so pre-existing findings stay silent.
145    let mut engine = build_engine();
146    let mut previous = scan(&mut engine, root)?;
147    eprintln!(
148        "[watch] {} known finding(s) (baseline). Watching {}...",
149        previous.len(),
150        root.display()
151    );
152
153    loop {
154        if should_stop() {
155            return Ok(());
156        }
157        std::thread::sleep(POLL_INTERVAL);
158        if should_stop() {
159            return Ok(());
160        }
161
162        let changed = watcher.changed_files(root, language_filter);
163        if changed.is_empty() {
164            continue;
165        }
166        eprintln!("[watch] {} file(s) changed", changed.len());
167
168        let mut engine = build_engine();
169        let mut round_current: Vec<Advisory> = Vec::new();
170        for file in &changed {
171            match scan(&mut engine, file) {
172                Ok(found) => {
173                    round_current.extend(found);
174                }
175                Err(e) => {
176                    // A half-written file can fail to parse; report and
177                    // keep watching rather than dying.
178                    eprintln!("[watch] scan failed for {}: {e}", file.display());
179                }
180            }
181        }
182
183        let fresh = new_advisories(&previous, &round_current);
184        for adv in &fresh {
185            println!("{}", format_watch_finding(adv));
186        }
187        if !fresh.is_empty() {
188            use std::io::Write;
189            let _ = std::io::stdout().flush();
190        }
191
192        // Merge: the new baseline is old findings from untouched files plus
193        // everything this round produced from the changed files. Findings
194        // whose file was edited are refreshed wholesale — including ones
195        // that disappeared (fixed) or moved lines (refingerprinted).
196        let touched: std::collections::HashSet<String> = changed
197            .iter()
198            .map(|p| p.to_string_lossy().into_owned())
199            .collect();
200        let mut merged: Vec<Advisory> = previous
201            .into_iter()
202            .filter(|a| !touched.contains(a.file_path.as_str()))
203            .collect();
204        merged.extend(round_current);
205        previous = merged;
206    }
207}
