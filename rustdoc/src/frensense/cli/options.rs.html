8/// CLI options for the deterministic dataflow compiler scan.
9///
10/// `#[non_exhaustive]`: an evolving options DTO. New flags (e.g. `explain` in
11/// 0.8.0) are added as minor releases; construct via [`super::parse_options`],
12/// not literals.
13#[derive(Debug, Clone, PartialEq)]
14#[non_exhaustive]
15pub struct CliOptions {
16    /// Positional scan target paths (files or directories).
17    pub paths: Vec<PathBuf>,
18    /// Report format: "text", "json", "sarif", "github".
19    pub format: String,
20    /// Optional output file to write the report to.
21    pub output_path: Option<PathBuf>,
22    /// Fail with exit code 1 if any findings are found.
23    pub is_strict: bool,
24    /// Minimum severity floor: critical, high, medium, warning, low, info.
25    /// Default: warning.
26    pub severity_filter: Option<Severity>,
27    /// Language filter (e.g., rust, go, typescript, javascript, python, c).
28    pub language_filter: Option<String>,
29    /// .frc bundle providing learned facts and advisory patterns.
30    pub corpus_bundle_path: Option<PathBuf>,
31    /// Save current findings as a baseline JSON artifact.
32    pub emit_baseline_path: Option<String>,
33    /// Compare findings against a baseline and fail on new findings.
34    pub compare_baseline_path: Option<String>,
35    /// Only scan files modified relative to git HEAD.
36    pub diff_only: bool,
37    /// Record every silent sink argument (why the engine stayed quiet).
38    pub explain: bool,
39    /// Show flows silenced by a bundle sanitizer fact, with the fact's
40    /// family and class provenance (`--show-suppressed`).
41    pub show_suppressed: bool,
42    /// Write the `--explain` records as JSON to this file (implies `--explain`).
43    pub explain_json_path: Option<PathBuf>,
44    /// Include the taint-path flow (source→sink SSA steps) in the text
45    /// observation. JSON/SARIF always carry the path (`taint_steps` / SARIF
46    /// `codeFlows`); `--show-flows` additionally embeds it in the text
47    /// observation for terminal reports.
48    pub show_flows: bool,
49    /// Precision profile: default (stable), recommended (stable + beta),
50    /// or strict (all tiers, including hygiene and experimental).
51    pub profile: crate::Profile,
52    /// Report code hygiene rules (e.g. function complexity) under any profile.
53    pub enable_hygiene: bool,
54    /// Maximum cyclomatic complexity ceiling before emitting function_complexity.
55    pub max_complexity: Option<usize>,
56    /// Exit with code 2 if the scan was incomplete, degraded, or had errors.
57    pub fail_on_incomplete: bool,
58    /// Destination path to write profiling/telemetry JSON.
59    pub profile_output: Option<PathBuf>,
60    /// Initialize tracing subscriber for spans and logs.
61    pub trace: bool,
62    /// Disable real-time progress bar and updates.
63    pub no_progress: bool,
64    /// Rule ids to suppress (`--disable-rule`, repeatable or comma-separated).
65    pub disable_rules: Vec<String>,
66    /// When non-empty, report only these rule ids (`--only`).
67    pub only_rules: Vec<String>,
68    /// Project config file (`--config`); defaults to `.frensense.toml` in cwd.
69    pub config_path: Option<PathBuf>,
70    /// Directory names to scan even though a default rule skips them
71    /// (`--include-dir`, repeatable or comma-separated).
72    pub include_dirs: Vec<String>,
73    /// Scan without the default directory, size and minified-file
74    /// exclusions (`--no-default-excludes`).
75    pub no_default_excludes: bool,
76}

78/// Parse CLI arguments into structured scan options.
79pub fn parse_options(args: &[String]) -> std::result::Result<CliOptions, String> {
80    let mut options = CliOptions {
81        paths: Vec::new(),
82        format: "text".to_string(),
83        output_path: None,
84        is_strict: false,
85        severity_filter: Some(Severity::Warning),
86        language_filter: None,
87        corpus_bundle_path: None,
88        emit_baseline_path: None,
89        compare_baseline_path: None,
90        diff_only: false,
91        explain: false,
92        show_suppressed: false,
93        explain_json_path: None,
94        show_flows: false,
95        profile: crate::Profile::Default,
96        enable_hygiene: false,
97        max_complexity: None,
98        fail_on_incomplete: false,
99        profile_output: None,
100        trace: false,
101        no_progress: false,
102        disable_rules: Vec::new(),
103        only_rules: Vec::new(),
104        config_path: None,
105        include_dirs: Vec::new(),
106        no_default_excludes: false,
107    };
108
109    let mut i = 1;
110    while i < args.len() {
111        let arg = &args[i];
112        match arg.as_str() {
113            "-o" | "--output" => {
114                let val = args
115                    .get(i + 1)
116                    .ok_or_else(|| format!("missing value for {arg}"))?;
117                options.output_path = Some(PathBuf::from(val));
118                i += 1;
119            }
120            "--json" => options.format = "json".to_string(),
121            "--sarif" => options.format = "sarif".to_string(),
122            "--github" | "--github-annotations" => {
123                options.format = "github".to_string();
124            }
125            "-f" | "--format" => {
126                let val = args
127                    .get(i + 1)
128                    .ok_or_else(|| format!("missing value for {arg}"))?;
129                match val.to_lowercase().as_str() {
130                    "text" | "json" | "sarif" | "github" => {
131                        options.format = val.to_lowercase();
132                    }
133                    other => {
134                        return Err(format!(
135                            "unknown format '{other}' (expected: text, json, sarif, github)"
136                        ));
137                    }
138                }
139                i += 1;
140            }
141            "--strict" => options.is_strict = true,
142            "--diff-only" => options.diff_only = true,
143            "--explain" => options.explain = true,
144            "--show-suppressed" => options.show_suppressed = true,
145            "--no-default-excludes" => options.no_default_excludes = true,
146            "--include-dir" => {
147                let val = args
148                    .get(i + 1)
149                    .ok_or_else(|| format!("missing value for {arg}"))?;
150                for name in val.split(',').map(str::trim).filter(|s| !s.is_empty()) {
151                    options.include_dirs.push(name.to_string());
152                }
153                i += 1;
154            }
155            "--disable-rule" => {
156                let val = args
157                    .get(i + 1)
158                    .ok_or_else(|| format!("missing value for {arg}"))?;
159                for id in val.split(',').map(str::trim).filter(|s| !s.is_empty()) {
160                    options.disable_rules.push(id.to_string());
161                }
162                i += 1;
163            }
164            "--only" => {
165                let val = args
166                    .get(i + 1)
167                    .ok_or_else(|| format!("missing value for {arg}"))?;
168                for id in val.split(',').map(str::trim).filter(|s| !s.is_empty()) {
169                    options.only_rules.push(id.to_string());
170                }
171                i += 1;
172            }
173            "--config" => {
174                let val = args
175                    .get(i + 1)
176                    .ok_or_else(|| format!("missing value for {arg}"))?;
177                options.config_path = Some(PathBuf::from(val));
178                i += 1;
179            }
180            "--explain-json" => {
181                let val = args
182                    .get(i + 1)
183                    .ok_or_else(|| format!("missing value for {arg}"))?;
184                options.explain = true;
185                options.explain_json_path = Some(PathBuf::from(val));
186                i += 1;
187            }
188            "--show-flows" => options.show_flows = true,
189            "--hygiene" => options.enable_hygiene = true,
190            "--max-complexity" => {
191                let val = args
192                    .get(i + 1)
193                    .ok_or_else(|| format!("missing value for {arg}"))?;
194                let n = val.parse::<usize>().map_err(|_| {
195                    format!("invalid value for {arg}: '{val}' must be a positive integer")
196                })?;
197                options.max_complexity = Some(n);
198                options.enable_hygiene = true;
199                i += 1;
200            }
201            "--fail-on-incomplete" => options.fail_on_incomplete = true,
202            "--trace" => options.trace = true,
203            "--no-progress" => options.no_progress = true,
204            "--profile-out" | "--profile-output" => {
205                let val = args
206                    .get(i + 1)
207                    .ok_or_else(|| format!("missing value for {arg}"))?;
208                options.profile_output = Some(PathBuf::from(val));
209                i += 1;
210            }
211            "-p" | "--profile" => {
212                let val = args
213                    .get(i + 1)
214                    .ok_or_else(|| format!("missing value for {arg}"))?;
215                if val.ends_with(".json") {
216                    options.profile_output = Some(PathBuf::from(val));
217                } else {
218                    options.profile = crate::Profile::parse(val)?;
219                }
220                i += 1;
221            }
222            "-s" | "--severity" => {
223                let level = args
224                    .get(i + 1)
225                    .ok_or_else(|| format!("missing value for {arg}"))?;
226                options.severity_filter = Some(crate::policy::severity_from_label(level)?);
227                i += 1;
228            }
229            "-l" | "--lang" | "--language" => {
230                let val = args
231                    .get(i + 1)
232                    .ok_or_else(|| format!("missing value for {arg}"))?;
233                options.language_filter = Some(val.clone());
234                i += 1;
235            }
236            "-b" | "--bundle" | "--corpus-bundle" => {
237                let val = args
238                    .get(i + 1)
239                    .ok_or_else(|| format!("missing value for {arg}"))?;
240                options.corpus_bundle_path = Some(PathBuf::from(val));
241                i += 1;
242            }
243            "--emit-baseline" => {
244                let val = args
245                    .get(i + 1)
246                    .ok_or_else(|| format!("missing value for {arg}"))?;
247                options.emit_baseline_path = Some(val.clone());
248                i += 1;
249            }
250            "--compare-baseline" => {
251                let val = args
252                    .get(i + 1)
253                    .ok_or_else(|| format!("missing value for {arg}"))?;
254                options.compare_baseline_path = Some(val.clone());
255                i += 1;
256            }
257            other if other.starts_with('-') => {
258                return Err(format!("unknown flag: {other}"));
259            }
260            pos => {
261                let p = PathBuf::from(pos);
262                if p.exists() {
263                    options.paths.push(p);
264                } else {
265                    return Err(format!(
266                        "path '{pos}' does not exist - specify a valid file or directory"
267                    ));
268                }
269            }
270        }
271        i += 1;
272    }
273
274    if options.paths.is_empty() {
275        options.paths.push(PathBuf::from("."));
276    }
277
278    // Project rule config: `[rules] disable = [...]` / `only = [...]`.
279    // An explicit `--config` must exist; the implicit `.frensense.toml` is
280    // optional and silently absent outside a configured project.
281    let explicit = options.config_path.is_some();
282    let config_path = options.config_path.clone().or_else(|| {
283        let default = PathBuf::from(".frensense.toml");
284        default.exists().then_some(default)
285    });
286    if let Some(path) = config_path {
287        let (disable, only) = read_rule_config(&path).map_err(|e| {
288            if explicit {
289                e
290            } else {
291                format!("failed to read {}: {e}", path.display())
292            }
293        })?;
294        options.disable_rules.extend(disable);
295        options.only_rules.extend(only);
296    }
297
298    Ok(options)
299}

301/// Read the `[rules]` section of a project config file.
302///
303/// Deliberately a small, dependency-free reader: the file is a list of rule
304/// ids, and a full TOML parser adds a dependency for one section. Accepts
305/// both `disable = ["a", "b"]` and a bare comma-separated value.
306pub fn read_rule_config(
307    path: &std::path::Path,
308) -> std::result::Result<(Vec<String>, Vec<String>), String> {
309    let text = std::fs::read_to_string(path).map_err(|e| e.to_string())?;
310    let mut in_rules = false;
311    let mut disable = Vec::new();
312    let mut only = Vec::new();
313    for line in text.lines() {
314        let line = line.split('#').next().unwrap_or("").trim();
315        if line.is_empty() {
316            continue;
317        }
318        if line.starts_with('[') {
319            in_rules = line.trim_matches(['[', ']', ' '].as_slice()) == "rules";
320            continue;
321        }
322        if !in_rules {
323            continue;
324        }
325        let Some((key, value)) = line.split_once('=') else {
326            continue;
327        };
328        let target = match key.trim() {
329            "disable" => &mut disable,
330            "only" => &mut only,
331            _ => continue,
332        };
333        let value = value.trim().trim_matches(['[', ']'].as_slice());
334        for id in value
335            .split(',')
336            .map(str::trim)
337            .map(|s| s.trim_matches(['"', '\''].as_slice()))
338        {
339            if !id.is_empty() && !target.iter().any(|x: &String| x == id) {
340                target.push(id.to_string());
341            }
342        }
343    }
344    Ok((disable, only))
345}

347/// All positional input paths: extracted from parsed options.
348#[must_use]
349pub fn get_input_paths(args: &[String]) -> Vec<PathBuf> {
350    parse_options(args)
351        .map(|opts| opts.paths)
352        .unwrap_or_else(|_| vec![std::env::current_dir().unwrap_or_else(|_| PathBuf::from("."))])
353}
