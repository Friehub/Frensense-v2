5//! LSP `Content-Length` framing: read one JSON-RPC message from a byte
6//! stream, write one out. Kept separate from the server logic so the
7//! framing round-trip is testable without a real editor.
8
9use std::io::{BufRead, Write};
10
11/// Read headers + body; returns the raw JSON text. `None` on clean EOF.
12///
13/// # Errors
14/// `Err` on malformed headers (missing `Content-Length`) or a truncated
15/// body — both mean the transport is broken and the loop should stop.
16pub fn read_message(reader: &mut impl BufRead) -> std::io::Result<Option<String>> {
17    let mut content_length: Option<usize> = None;
18    loop {
19        let mut line = String::new();
20        let n = reader.read_line(&mut line)?;
21        if n == 0 {
22            return Ok(None); // EOF
23        }
24        let line = line.trim_end();
25        if line.is_empty() {
26            break; // headers done
27        }
28        if let Some(value) = line
29            .strip_prefix("Content-Length:")
30            .or_else(|| line.strip_prefix("content-length:"))
31        {
32            content_length = value.trim().parse::<usize>().ok();
33        }
34    }
35    let Some(len) = content_length else {
36        return Err(std::io::Error::new(
37            std::io::ErrorKind::InvalidData,
38            "missing Content-Length header",
39        ));
40    };
41    let mut body = vec![0u8; len];
42    reader.read_exact(&mut body)?;
43    String::from_utf8(body)
44        .map(Some)
45        .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e))
46}
47
48/// Write one framed message.
49///
50/// # Errors
51/// Propagates write failures (broken pipe: the client went away).
52pub fn write_message(writer: &mut impl Write, body: &str) -> std::io::Result<()> {
53    write!(writer, "Content-Length: {}\r\n\r\n", body.len())?;
54    writer.write_all(body.as_bytes())?;
55    writer.flush()
56}
