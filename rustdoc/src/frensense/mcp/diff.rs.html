5//! `frensense_diff` - scan a patch/diff and report only findings whose
6//! reported line falls inside the patch's added-line ranges.
7//!
8//! Why the range intersection rather than re-parsing the patch into code:
9//! the engine's source of truth is the file *on disk* - the working tree
10//! already contains the change. Feeding synthetic patch content into the
11//! parser would fork the analysis path (patch context, hunk offsets,
12//! partial functions). Instead this module observes the patch, computes
13//! which line ranges it touched per file, and intersects those ranges
14//! with the advisories a normal `Engine::run` produced. One lowering
15//! path, one taint engine, zero divergence between "what CI scans" and
16//! "what the diff reports".
17//!
18//! Scope discipline (same doctrine as `scan_file`): this module owns
19//! only patch observation and line-range filtering. Language knowledge
20//! stays in `frensense-lang`, analysis stays in the engine.
21

28/// What one diff scan produced: the parsed added ranges, the advisories
29/// that landed on them, and how many diff files the engine actually ran.
30pub type DiffScanOutcome = (BTreeMap<String, Vec<AddedRange>>, Vec<Advisory>, usize);

32/// A diff scan plus how far the analysis behind it could see.
33///
34/// Coverage travels with the findings because a diff gate is the one place
35/// where an empty advisory list is consumed as a decision: an agent that
36/// treats "no advisories" from a file the engine recognized nothing in as a
37/// pass has been told the wrong thing.
38#[derive(Debug)]
39pub struct DiffScan {
40    pub ranges: BTreeMap<String, Vec<AddedRange>>,
41    pub advisories: Vec<Advisory>,
42    pub scanned: usize,
43    pub coverage: crate::policy::Coverage,
44    pub telemetry: frensense_engine::telemetry::ScanTelemetry,
45    pub incomplete: Vec<frensense_engine::scan::IncompleteFunction>,
46}

87/// One added-line range in a file, inclusive on both ends, 1-based.
88#[derive(Debug, Clone, PartialEq, Eq)]
89pub struct AddedRange {
90    pub start: u32,
91    pub end: u32,
92}
93
94impl AddedRange {
95    fn contains(&self, line: u32) -> bool {
96        line >= self.start && line <= self.end
97    }
98}

100/// Extract the `+++ b/<path>` target paths from a unified diff, in order
101/// of appearance, deduplicated. Handles `a/`/`b/` prefixes and `/dev/null`
102/// (new files whose target we still want - `+++ b/path` form always
103/// carries the path). Quoted paths (`"\b/path with spaces"`) are
104/// unquoted.
105#[must_use]
106pub fn diff_file_paths(diff: &str) -> Vec<String> {
107    let mut seen = std::collections::BTreeSet::new();
108    let mut out = Vec::new();
109    for line in diff.lines() {
110        let Some(path) = line.strip_prefix("+++ ") else {
111            continue;
112        };
113        let path = path.trim();
114        if path == "/dev/null" {
115            continue;
116        }
117        let path = unquote(path);
118        let path = path.strip_prefix("b/").unwrap_or(path);
119        if seen.insert(path.to_string()) {
120            out.push(path.to_string());
121        }
122    }
123    out
124}

126/// Parse a unified diff into per-file added-line ranges.
127///
128/// Only `+` lines inside hunks count (context lines and `-` lines do not):
129/// a finding belongs to "the change" when its line exists *because* of
130/// the patch. Hunk headers (`@@ -l,c +l,c @@`) carry the new-file line
131/// numbering; the parser tracks the cursor through each hunk body so
132/// multi-hunk files produce disjoint ranges. `No newline at end of file`
133/// markers are ignored.
134#[must_use]
135pub fn added_ranges(diff: &str) -> BTreeMap<String, Vec<AddedRange>> {
136    let mut result: BTreeMap<String, Vec<AddedRange>> = BTreeMap::new();
137    let mut current_file: Option<String> = None;
138    let mut current_line: u32 = 0;
139    let mut in_hunk = false;
140    // New-side line budget promised by the current hunk header
141    // (`@@ -a,b +c,d @@` → d, defaulting to 1 when omitted). Grounds the
142    // empty-line ambiguity check below in the patch's own metadata.
143    let mut remaining_new: u32 = 0;
144
145    for line in diff.lines() {
146        if let Some(path) = line.strip_prefix("+++ ") {
147            let path = path.trim();
148            if path != "/dev/null" {
149                let p = unquote(path);
150                let p = p.strip_prefix("b/").unwrap_or(p).to_string();
151                current_file = Some(p);
152            }
153            in_hunk = false;
154            continue;
155        }
156        if line.starts_with("@@") {
157            // @@ -oldStart,oldCount +newStart,newCount @@ ...
158            in_hunk = true;
159            current_line = parse_new_start(line).unwrap_or(0);
160            remaining_new = parse_new_count(line);
161            continue;
162        }
163        if !in_hunk {
164            continue;
165        }
166        if line.starts_with('+') {
167            // `diff.lines()` already split on newlines, so each `+` line
168            // here is exactly one added line (a bare "+" is an added
169            // empty line). Consecutive additions merge into one range:
170            // a five-line added block is one finding region, not five.
171            let file = current_file.clone().unwrap_or_default();
172            let line_no = current_line;
173            let ranges = result.entry(file).or_default();
174            match ranges.last_mut() {
175                Some(last) if last.end + 1 == line_no => last.end = line_no,
176                _ => ranges.push(AddedRange {
177                    start: line_no,
178                    end: line_no,
179                }),
180            }
181            current_line += 1;
182            remaining_new = remaining_new.saturating_sub(1);
183        } else if line.starts_with('-') || line.starts_with('\\') {
184            // Removed lines / "\ No newline": don't advance the new cursor.
185        } else if line.starts_with(' ') {
186            // Context line.
187            current_line += 1;
188            remaining_new = remaining_new.saturating_sub(1);
189        } else if line.is_empty() {
190            // Ambiguous: an empty `.lines()` item is either a context
191            // line whose content is empty (git emits it as " \n" which
192            // some tools strip the leading space from) or a separator
193            // between top-level file sections of a hand-written patch.
194            // Distinguish by the hunk's own line budget: track how many
195            // new-side lines the header promised; empty input inside
196            // that budget is context, anything after it is not.
197            if remaining_new > 0 {
198                current_line += 1;
199                remaining_new -= 1;
200            }
201        }
202    }
203    result.retain(|_, ranges| !ranges.is_empty());
204    result
205}

207/// New-side line count from a hunk header; git omits `,1` for
208/// single-line hunks, so the default is 1.
209fn parse_new_count(header: &str) -> u32 {
210    let plus = match header.find('+') {
211        Some(i) => i,
212        None => return 1,
213    };
214    let rest = &header[plus + 1..];
215    let after_comma = rest.split_once(',').map(|(_, r)| r);
216    match after_comma {
217        Some(r) => {
218            let end = r.find([' ', '@', '\r']).unwrap_or(r.len());
219            r[..end].parse().unwrap_or(1)
220        }
221        None => 1,
222    }
223}
224
225fn parse_new_start(header: &str) -> Option<u32> {
226    // Second +/- span: "@@ -1,3 +2,5 @@"
227    let plus = header.find('+')?;
228    let rest = &header[plus + 1..];
229    let end = rest.find([',', ' ', '@'])?;
230    rest[..end].parse().ok()
231}
232
233fn unquote(s: &str) -> &str {
234    let s = s.trim();
235    if s.len() >= 2 && s.starts_with('"') && s.ends_with('"') {
236        &s[1..s.len() - 1]
237    } else {
238        s
239    }
240}

242/// True when the advisory's reported line is inside one of the patch's
243/// added ranges for its file. Findings from files the diff doesn't touch
244/// never qualify.
245#[must_use]
246pub fn in_added_lines(adv: &Advisory, ranges: &BTreeMap<String, Vec<AddedRange>>) -> bool {
247    let adv_path = normalize(&adv.file_path);
248    let file_ranges = ranges
249        .iter()
250        .find(|(diff_path, _)| path_matches(&adv_path, diff_path))
251        .map(|(_, rs)| rs);
252    let Some(file_ranges) = file_ranges else {
253        return false;
254    };
255    file_ranges.iter().any(|r| r.contains(adv.line))
256}

258/// Diff paths are repo-relative; engine advisory paths may be absolute
259/// or repo-relative depending on how the repo path was passed. Match on
260/// suffix boundaries so `src/foo.py` matches `<repo>/src/foo.py` but not
261/// `mysrc/foo.py`.
262fn normalize(p: &str) -> String {
263    p.replace('\\', "/")
264}
265
266fn path_matches(advisory_path: &str, diff_path: &str) -> bool {
267    let adv = normalize(advisory_path);
268    let d = normalize(diff_path);
269    if adv == d {
270        return true;
271    }
272    // Absolute advisory path (`/repo/src/app.py`) vs diff-relative
273    // (`src/app.py`): match on a `/`-delimited suffix boundary.
274    adv.ends_with(&format!("/{d}")) || d.ends_with(&format!("/{adv}"))
275}

277/// Severity/confidence filter. Thin wrapper over
278/// [`crate::policy::apply`], the one reporting policy every surface
279/// shares.
280#[must_use]
281pub fn filter_advisories(
282    advisories: Vec<Advisory>,
283    severity_floor: Severity,
284    min_confidence: f64,
285) -> Vec<Advisory> {
286    crate::policy::apply(advisories, min_confidence, Some(severity_floor))
287}

289/// Run `git diff HEAD` in `repo` and append synthetic added-file hunks
290/// for untracked files (`git diff` never shows them, but a brand-new
291/// file is exactly the change a diff gate must not miss). Returns empty
292/// output (not an error) when the repo has no changes.
293fn git_diff_head(repo: &Path) -> Result<String, String> {
294    let run = |args: &[&str]| -> Result<String, String> {
295        let output = std::process::Command::new("git")
296            .args(args)
297            .current_dir(repo)
298            .output()
299            .map_err(|e| format!("git {args:?} failed: {e}"))?;
300        if !output.status.success() {
301            return Err(format!(
302                "git {args:?} failed: {}",
303                String::from_utf8_lossy(&output.stderr).trim()
304            ));
305        }
306        Ok(String::from_utf8_lossy(&output.stdout).into_owned())
307    };
308
309    let mut diff = run(&["diff", "HEAD"])?;
310
311    let untracked = run(&["ls-files", "--others", "--exclude-standard"])?;
312    for file in untracked.lines().filter(|l| !l.trim().is_empty()) {
313        match std::fs::read_to_string(repo.join(file)) {
314            Ok(content) => {
315                diff.push_str(&format!("\ndiff --git a/{file} b/{file}\n"));
316                diff.push_str("--- /dev/null\n");
317                diff.push_str(&format!("+++ b/{file}\n"));
318                diff.push_str(&format!("@@ -0,0 +1,{} @@\n", content.lines().count()));
319                for line in content.lines() {
320                    diff.push('+');
321                    diff.push_str(line);
322                    diff.push('\n');
323                }
324            }
325            // Unreadable/unusual untracked file: skip it silently -
326            // the engine would skip it too.
327            Err(_) => continue,
328        }
329    }
330    Ok(diff)
331}

333/// Core: scan the files named in the diff via the normal engine path and
334/// keep only findings on added lines.
335///
336/// # Errors
337/// Engine failure, unreadable repo, or `git diff` failure (when
338/// `diff_text` is None).
339pub fn run_diff(
340    repo: &Path,
341    diff_text: Option<&str>,
342    severity_threshold: &str,
343    min_confidence: f64,
344    corpus_bundle: Option<&str>,
345) -> crate::Result<DiffScanOutcome> {
346    let scan = run_diff_detailed(
347        repo,
348        diff_text,
349        severity_threshold,
350        min_confidence,
351        corpus_bundle,
352    )?;
353    Ok((scan.ranges, scan.advisories, scan.scanned))
354}

356/// [`run_diff`] plus the coverage and telemetry behind its answer.
357///
358/// # Errors
359/// Same as [`run_diff`].
360pub fn run_diff_detailed(
361    repo: &Path,
362    diff_text: Option<&str>,
363    severity_threshold: &str,
364    min_confidence: f64,
365    corpus_bundle: Option<&str>,
366) -> crate::Result<DiffScan> {
367    // Resolve the one shared vocabulary before touching the tree: an unknown
368    // threshold is a caller error, not a reason to fall back to `info` (the
369    // least restrictive floor) and silently report everything.
370    let severity_floor = crate::policy::severity_from_label(severity_threshold)
371        .map_err(crate::FrensenseError::Config)?;
372    let diff = match diff_text {
373        Some(d) => d.to_string(),
374        None => git_diff_head(repo).map_err(crate::FrensenseError::Config)?,
375    };
376    let ranges = added_ranges(&diff);
377
378    // Scan each file the diff touches with a full engine run of its own
379    // directory scope (the file itself). Files the engine can't handle
380    // (unsupported ext, deleted files that no longer exist) are skipped
381    // by collect_files' own rules.
382    let mut all = Vec::new();
383    let mut scanned = 0usize;
384    let mut unsupported_touched = 0usize;
385    let mut coverage = crate::policy::Coverage::default();
386    let mut telemetry = frensense_engine::telemetry::ScanTelemetry::default();
387    let mut incomplete = Vec::new();
388    let mut engine = super::scan_file::build_engine(min_confidence, corpus_bundle);
389    for path in ranges.keys() {
390        let abs = repo.join(path);
391        if !is_supported(&abs) {
392            // A touched source file in a language with no spec (or a deleted
393            // file) is part of the change; it belongs in the denominator
394            // even though the engine cannot scan it.
395            unsupported_touched += 1;
396            continue;
397        }
398        // A file run emits advisories whose file_path is the path we
399        // passed; re-point them at the diff-relative name so intersection
400        // and output stay diff-scoped.
401        if let Ok(found) = engine.run(&abs) {
402            scanned += 1;
403            for mut adv in found {
404                adv.file_path = path.clone();
405                all.push(adv);
406            }
407            coverage.merge(engine.take_coverage());
408            telemetry.merge(engine.take_telemetry());
409            incomplete.extend(engine.take_incomplete());
410        }
411    }
412    telemetry.finalize();
413    coverage.files_found += unsupported_touched;
414    coverage.files_unsupported += unsupported_touched;
415
416    let all = filter_advisories(all, severity_floor, min_confidence);
417    let on_added: Vec<Advisory> = all
418        .into_iter()
419        .filter(|adv| {
420            ranges.keys().any(|d| path_matches(&adv.file_path, d)) && in_added_lines(adv, &ranges)
421        })
422        .collect();
423    Ok(DiffScan {
424        ranges,
425        advisories: on_added,
426        scanned,
427        coverage,
428        telemetry,
429        incomplete,
430    })
431}

443/// Build the JSON-RPC `tools/call` result value with coverage.
444#[must_use]
445pub(crate) fn result_payload_detailed(
446    ranges: &BTreeMap<String, Vec<AddedRange>>,
447    advisories: &[Advisory],
448    scanned: usize,
449    coverage: Option<&crate::policy::Coverage>,
450    telemetry: Option<&frensense_engine::telemetry::ScanTelemetry>,
451    incomplete: &[frensense_engine::scan::IncompleteFunction],
452) -> Value {
453    let empty_telem = frensense_engine::telemetry::ScanTelemetry::default();
454    let telem = telemetry.unwrap_or(&empty_telem);
455    let cov = coverage.cloned().unwrap_or_default();
456    let analysis = super::scan_file::mcp_analysis(&cov, telem, incomplete);
457    let files: Vec<Value> = ranges
458        .iter()
459        .map(|(f, rs)| {
460            json!({
461                "path": f,
462                "added_ranges": rs.iter()
463                    .map(|r| json!({"start": r.start, "end": r.end}))
464                    .collect::<Vec<_>>(),
465            })
466        })
467        .collect();
468    json!({
469        "files_touched": files.len(),
470        "files_scanned": scanned,
471        "added_ranges": files,
472        // Stable IDs of the on-added-line findings: agents gate a change
473        // by comparing IDs across iterations of the same change.
474        "stable_ids": advisories.iter().map(|a| a.stable_id()).collect::<Vec<_>>(),
475        "advisories": advisories,
476        "analysis": analysis,
477        "message": if !advisories.is_empty() {
478            ""
479        } else if !cov.recognized_anything() {
480            "added lines introduce no findings; no sink site was recognized in the touched files, so the gate cannot vouch for them"
481        } else if !cov.fully_resolved() {
482            "added lines introduce no findings; some sink sites could not be decided"
483        } else if cov.files_not_analyzed() > 0 {
484            "added lines introduce no findings; some touched files were not analyzed, so they cannot be ruled out"
485        } else {
486            "added lines introduce no findings"
487        },
488    })
489}

491/// Full tool invocation: argument parsing, diff sourcing, scanning,
492/// filtering, shaping. Returns the `tools/call` result value.
493#[must_use]
494pub fn run_diff_tool(args: &Value) -> Value {
495    let repo_str = args.get("repo").and_then(Value::as_str).unwrap_or(".");
496    let repo = Path::new(repo_str);
497    if !repo.exists() {
498        return json!({
499            "advisories": [],
500            "error": format!("repo path does not exist: {repo_str}")
501        });
502    }
503
504    let diff_text = args.get("diff_text").and_then(Value::as_str);
505    let severity_threshold = args
506        .get("severity_threshold")
507        .and_then(Value::as_str)
508        .unwrap_or("info");
509    if let Err(error) = crate::policy::severity_from_label(severity_threshold) {
510        return json!({
511            "advisories": [],
512            "error": error
513        });
514    }
515    let min_confidence = args
516        .get("min_confidence")
517        .and_then(Value::as_f64)
518        .unwrap_or(0.0)
519        .clamp(0.0, 1.0);
520    let corpus_bundle = args.get("corpus_bundle").and_then(Value::as_str);
521    if let Some(bundle) = corpus_bundle
522        && !Path::new(bundle).exists()
523    {
524        return json!({
525            "advisories": [],
526            "error": format!("corpus bundle does not exist: {bundle}")
527        });
528    }
529
530    if diff_text.is_none() {
531        // Only hit git when asked to; an explicit diff_text must work
532        // outside any repository.
533        let probe = std::process::Command::new("git")
534            .args(["rev-parse", "--is-inside-work-tree"])
535            .current_dir(repo)
536            .output();
537        match probe {
538            Ok(o) if String::from_utf8_lossy(&o.stdout).trim() != "true" => {
539                return json!({
540                    "advisories": [],
541                    "error": "not a git repository: pass diff_text or run inside a repo"
542                });
543            }
544            Err(_) => {
545                return json!({
546                    "advisories": [],
547                    "error": "git unavailable: pass diff_text explicitly"
548                });
549            }
550            _ => {}
551        }
552    }
553
554    match run_diff_detailed(
555        repo,
556        diff_text,
557        severity_threshold,
558        min_confidence,
559        corpus_bundle,
560    ) {
561        Ok(outcome) => result_payload_detailed(
562            &outcome.ranges,
563            &outcome.advisories,
564            outcome.scanned,
565            Some(&outcome.coverage),
566            Some(&outcome.telemetry),
567            &outcome.incomplete,
568        ),
569        Err(e) => json!({
570            "advisories": [],
571            "error": format!("analysis error: {e}")
572        }),
573    }
574}
