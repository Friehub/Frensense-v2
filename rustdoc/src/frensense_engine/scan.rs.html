27/// Scan a set of source files under a config and fact table.
28///
29/// `files` maps file path → extension (used to pick the language spec).
30/// Files are lowered one at a time through [`ProgramBuilder`] and then
31/// analyzed as one interprocedural program.
32///
33/// ```
34/// use frensense_engine::analysis::taint::config::TaintConfig;
35/// use frensense_engine::analysis::taint::engine::BackwardVerdict;
36/// use frensense_engine::analysis::taint::facts::FactTable;
37/// use frensense_engine::scan::scan;
38///
39/// let config = TaintConfig {
40///     sources: ["req.body".to_string()].into_iter().collect(),
41///     sinks: ["query".to_string()].into_iter().collect(),
42///     sanitizers: ["escape".to_string()].into_iter().collect(),
43/// };
44/// let facts = FactTable::from_config(&config);
45///
46/// let file = |path: &str, src: &str| (path.to_string(), src.to_string(), "ts".to_string());
47/// let vulnerable = |r: &frensense_engine::scan::ScanResult| {
48///     r.located.iter().any(|l| l.finding.verdict == BackwardVerdict::Vulnerable)
49/// };
50///
51/// // Source and sink in one file.
52/// let direct = scan(
53///     &[file("a.ts", "declare const db: any\nexport function h(req: any) {\n  const id = req.body.id\n  db.query('SELECT ' + id)\n}\n")],
54///     &config,
55///     &facts,
56/// );
57/// assert!(vulnerable(&direct));
58///
59/// // The same flow through a configured sanitizer is not reported as vulnerable.
60/// let sanitized = scan(
61///     &[file("a.ts", "declare const db: any\ndeclare function escape(s: string): string\nexport function h(req: any) {\n  const id = escape(req.body.id)\n  db.query('SELECT ' + id)\n}\n")],
62///     &config,
63///     &facts,
64/// );
65/// assert!(!vulnerable(&sanitized));
66///
67/// // Source in one file, sink in another: the finding is placed at the sink.
68/// let cross = scan(
69///     &[
70///         file("lib.ts", "declare const db: any\nexport function run(q: string) { db.query('SELECT ' + q) }\n"),
71///         file("app.ts", "import { run } from './lib'\nexport function h(req: any) { run(req.body.id) }\n"),
72///     ],
73///     &config,
74///     &facts,
75/// );
76/// let hit = cross
77///     .located
78///     .iter()
79///     .find(|l| l.finding.verdict == BackwardVerdict::Vulnerable)
80///     .expect("cross-file flow is found");
81/// assert_eq!(hit.file, "lib.ts");
82/// assert_eq!(hit.finding.function, "run");
83/// ```
84pub fn scan(
85    files: &[(String, String, String)], // (path, source, ext)
86    config: &TaintConfig,
87    facts: &FactTable,
88) -> ScanResult {
89    let started = Instant::now();
90    let mut builder = ProgramBuilder::new();
91    add_sorted(&mut builder, files, Some(facts));
92    let mut result = scan_prepared(&builder.finish(), config, facts);
93    result.telemetry.phases.push(PhaseTiming {
94        phase: "total".into(),
95        ms: millis_since(started),
96    });
97    result
98}

100/// Scan one source file: the single-file capability.
101///
102/// Only flows whose source and sink both live in `source` can be found. A
103/// flow that starts here and ends in another file is the program
104/// capability's job ([`scan`] / [`scan_prepared`]); a frontend picks the one
105/// that matches what it has in scope (an editor buffer, one path from
106/// `frensense_scan_file`, or a whole repository).
107///
108/// ```
109/// use frensense_engine::analysis::taint::config::TaintConfig;
110/// use frensense_engine::analysis::taint::engine::BackwardVerdict;
111/// use frensense_engine::analysis::taint::facts::FactTable;
112/// use frensense_engine::scan::scan_file;
113///
114/// let config = TaintConfig {
115///     sources: ["req.body".to_string()].into_iter().collect(),
116///     sinks: ["query".to_string()].into_iter().collect(),
117///     sanitizers: ["escape".to_string()].into_iter().collect(),
118/// };
119/// let facts = FactTable::from_config(&config);
120///
121/// // The sink is in `lib.ts`, which this call never sees, so the flow that
122/// // starts in this file is not found.
123/// let result = scan_file(
124///     "app.ts",
125///     "import { run } from './lib'\nexport function h(req: any) { run(req.body.id) }\n",
126///     "ts",
127///     &config,
128///     &facts,
129/// );
130/// assert!(result.located.iter().all(|l| l.finding.verdict != BackwardVerdict::Vulnerable));
131///
132/// // With source and sink in the same file it is found.
133/// let result = scan_file(
134///     "a.ts",
135///     "declare const db: any\nexport function h(req: any) {\n  db.query(req.body.id)\n}\n",
136///     "ts",
137///     &config,
138///     &facts,
139/// );
140/// assert!(result.located.iter().any(|l| l.finding.verdict == BackwardVerdict::Vulnerable));
141/// ```
142pub fn scan_file(
143    path: &str,
144    source: &str,
145    ext: &str,
146    config: &TaintConfig,
147    facts: &FactTable,
148) -> ScanResult {
149    let started = Instant::now();
150    let mut builder = ProgramBuilder::new();
151    builder.add_file(path, source, ext, Some(facts));
152    let mut result = scan_prepared(&builder.finish(), config, facts);
153    result.telemetry.phases.push(PhaseTiming {
154        phase: "total".into(),
155        ms: millis_since(started),
156    });
157    result
158}

160/// A program lowered once, ready for repeated scans under different fact
161/// tables. The replay gate lowers each family exactly once and re-scans it
162/// per candidate fact, lowering dominates scan cost, so this turns the
163/// gate's O(candidates × families) lowerings into O(families).
164pub struct PreparedProgram {
165    irs: FxHashMap<String, FunctionIR>,
166    fn_file: FxHashMap<String, String>,
167    file_source: FxHashMap<String, String>,
168    /// Per file, the local names that an import binds to a program function.
169    pub file_aliases: FxHashMap<String, FxHashMap<String, String>>,
170    /// Per file, the local names that an import binds to an external package.
171    pub(crate) file_package_imports:
172        FxHashMap<String, FxHashMap<String, crate::graph::module::PackageImport>>,
173    /// Import declarations per file (see `ModuleResolution`).
174    ///
175    /// `ModuleResolution`: crate::graph::module::ModuleResolution
176    pub file_imports: FxHashMap<String, Vec<Import>>,
177    /// Export/re-export declarations per file, with visibility and kind.
178    pub file_exports: FxHashMap<String, Vec<Export>>,
179    /// Every top-level declaration per file, exported or not.
180    pub file_module_items: FxHashMap<String, Vec<ModuleItem>>,
181    /// Files that could not be lowered, with the reason. The scan continues
182    /// without them and says so; it never drops one silently.
183    pub lower_errors: Vec<String>,
184    /// Files whose parse tree contained errors, so they were analyzed from
185    /// whatever tree-sitter recovered rather than from an intact tree.
186    pub parse_partial: Vec<String>,
187    /// Number of source inputs attempted by this prepared program.
188    pub submitted: usize,
189    /// Safety cap events that happened during per-file lowering.
190    pub(crate) lower_caps: Vec<CapHit>,
191    /// Paths that failed lowering, independent of error message formatting.
192    pub lower_failed_files: Vec<String>,
193    /// Lowering cost per file, feeding the slowest-file report.
194    pub(crate) file_timings: Vec<FileTiming>,
195}

207/// Stack size for the per-file lowering thread (the default 4 MiB overflows
208/// on deeply nested real-world functions; one such overflow used to abort the
209/// process and lose every file's findings).
210const LOWER_STACK_BYTES: usize = 32 * 1024 * 1024;
211
212/// Lower one file on its own thread with a larger stack.
213///
214/// Returns the extracted functions and whether the file only partially
215/// parsed. A panic is contained here and reported as that file's error.
216type IsolatedLowering = (FxHashMap<String, FunctionIR>, bool, Vec<CapHit>);
217
218#[allow(
219    clippy::disallowed_methods,
220    reason = "lowering runs on a dedicated 32 MiB thread because real-world deeply nested functions overflow the default 4 MiB stack; stacker cannot recover a stack overflow that happens inside the tree-sitter lowering call frames"
221)]
222fn lower_isolated(
223    path: &str,
224    source: &str,
225    ext: &str,
226    facts: Option<&FactTable>,
227) -> Result<IsolatedLowering, String> {
228    type Lowered = Result<(FxHashMap<String, FunctionIR>, bool), String>;
229    type JoinOutcome = Result<(Lowered, Vec<CapHit>), Box<dyn std::any::Any + Send + 'static>>;
230    let outcome: Result<JoinOutcome, std::io::Error> = std::thread::scope(|scope| {
231        let handle = std::thread::Builder::new()
232            .stack_size(LOWER_STACK_BYTES)
233            .spawn_scoped(scope, || {
234                let lowered = crate::harness::lower_source_detailed(path, source, ext, facts);
235                let caps = crate::telemetry::drain_caps();
236                (lowered, caps)
237            })?;
238        Ok(handle.join())
239    });
240    match outcome {
241        Ok(Ok((Ok((functions, partial)), caps))) => Ok((functions, partial, caps)),
242        Ok(Ok((Err(error), _))) => Err(error),
243        Ok(Err(_)) => Err("lowering worker panicked; see the file path in scan accounting".into()),
244        Err(_) => Err("failed to spawn lowering worker".into()),
245    }
246}

248/// One file lowered by a [`ProgramBuilder`], before the program-wide merge.
249#[derive(Debug)]
250struct LoweredFile<'a> {
251    path: String,
252    source: &'a str,
253    ext: String,
254    functions: FxHashMap<String, FunctionIR>,
255    parse_partial: bool,
256    caps: Vec<CapHit>,
257    ms: u128,
258    error: Option<String>,

261/// Lowering outcome for one file, returned by [`ProgramBuilder::add_file`].
262///
263/// This is the per-file feedback a frontend renders as progress: which file
264/// finished, how many functions it produced, whether the parse tree was
265/// broken, and why lowering failed when it did.
266#[derive(Debug, Clone)]
267pub struct FileReport {
268    /// Path exactly as submitted.
269    pub path: String,
270    /// Functions the file lowered to (`0` when lowering failed).
271    pub functions: usize,
272    /// Milliseconds spent lowering this file.
273    pub ms: u128,
274    /// The parse tree had errors; analysis used the recovered tree.
275    pub parse_partial: bool,
276    /// Why lowering failed. The file is left out of the program and the
277    /// reason is recorded in the scan's error accounting too.
278    pub error: Option<String>,
279}

281/// Builds a [`PreparedProgram`] one file at a time.
282///
283/// This is the per-file capability behind [`scan`]. A frontend submits files
284/// itself. The caller owns the loop, the counter, any parallelism and early
285/// cancellation, and acts on each [`FileReport`] immediately.
286/// [`finish`](Self::finish) then merges the submitted files deterministically
287/// and resolves modules. Lowering is local, so submission order does not
288/// affect the result.
289///
290/// The builder borrows each file's source text for the lifetime of the
291/// buffer: a scan never keeps a second copy of the corpus in memory.
292#[derive(Debug, Default)]
293pub struct ProgramBuilder<'a> {
294    files: Vec<LoweredFile<'a>>,
295}
296
297impl<'a> ProgramBuilder<'a> {
298    /// A builder with no files.
299    #[must_use]
300    pub fn new() -> Self {
301        Self::default()
302    }
303
304    /// Lower one source file and remember it for [`finish`](Self::finish).
305    ///
306    /// A lowering failure is contained here: the file is retained as an
307    /// error in the accounting and reported through [`FileReport::error`],
308    /// exactly as [`scan`] reports a failed file.
309    pub fn add_file(
310        &mut self,
311        path: &str,
312        source: &'a str,
313        ext: &str,
314        facts: Option<&FactTable>,
315    ) -> FileReport {
316        let started = Instant::now();
317        let lowered = lower_isolated(path, source, ext, facts);
318        let ms = millis_since(started);
319        match lowered {
320            Ok((functions, parse_partial, caps)) => {
321                let count = functions.len();
322                self.files.push(LoweredFile {
323                    path: path.to_owned(),
324                    source,
325                    ext: ext.to_owned(),
326                    functions,
327                    parse_partial,
328                    caps,
329                    ms,
330                    error: None,
331                });
332                FileReport {
333                    path: path.to_owned(),
334                    functions: count,
335                    ms,
336                    parse_partial,
337                    error: None,
338                }
339            }
340            Err(error) => {
341                self.files.push(LoweredFile {
342                    path: path.to_owned(),
343                    source,
344                    ext: ext.to_owned(),
345                    functions: FxHashMap::default(),
346                    parse_partial: false,
347                    caps: Vec::new(),
348                    ms,
349                    error: Some(error.clone()),
350                });
351                FileReport {
352                    path: path.to_owned(),
353                    functions: 0,
354                    ms,
355                    parse_partial: false,
356                    error: Some(error),
357                }
358            }
359        }
360    }
361
362    /// Merge every submitted file into one program.
363    ///
364    /// Submission order does not matter: files are merged path-sorted, so a
365    /// colliding function name rekeys the same way on every run.
366    #[must_use]
367    pub fn finish(mut self) -> PreparedProgram {
368        // Deterministic merge order (path-sorted): which twin of a colliding
369        // name keeps the bare key must not depend on submission order.
370        self.files.sort_by(|a, b| a.path.cmp(&b.path));
371        let submitted = self.files.len();
372        // Module resolution consumes the whole input list, including files
373        // whose lowering failed: a failed twin still declares imports that
374        // another file's resolution may need to see. Paths and extensions are
375        // owned; sources stay borrowed from the caller's buffers.
376        let program: Vec<(String, &'a str, String)> = self
377            .files
378            .iter()
379            .map(|f| (f.path.clone(), f.source, f.ext.clone()))
380            .collect();
381
382        let mut irs: FxHashMap<String, FunctionIR> = FxHashMap::default();
383        let mut fn_file: FxHashMap<String, String> = FxHashMap::default();
384        let mut file_source: FxHashMap<String, String> = FxHashMap::default();
385        let mut lower_errors: Vec<String> = Vec::new();
386        let mut lower_failed_files: Vec<String> = Vec::new();
387        let mut lower_caps: Vec<CapHit> = Vec::new();
388        let mut parse_partial_files: Vec<String> = Vec::new();
389        let mut file_timings: Vec<FileTiming> = Vec::new();
390
391        for file in self.files {
392            if let Some(error) = file.error {
393                lower_failed_files.push(file.path);
394                lower_errors.push(error);
395                continue;
396            }
397            let LoweredFile {
398                path,
399                source,
400                functions,
401                parse_partial,
402                caps,
403                ms,
404                ..
405            } = file;
406            lower_caps.extend(caps);
407            if parse_partial {
408                parse_partial_files.push(path.clone());
409            }
410            file_timings.push(FileTiming {
411                file: path.clone(),
412                ms,
413                functions: functions.len(),
414            });
415            for (name, mut ir) in functions {
416                file_source
417                    .entry(path.clone())
418                    .or_insert_with(|| source.to_owned());
419                // Function keys are only unique per file: byte-offset names
420                // (`<fn@505>`) and common method names (`set`) collide across
421                // files. First-wins dropped the other file's whole function
422                // silently (the forgedReview/noSqlReviews miss); rekey the
423                // incoming twin under its file instead. `ir.name` tracks the
424                // key so findings, `fn_file`, and checker locations stay in
425                // lockstep.
426                let key = if irs.contains_key(&name) {
427                    format!("{path}::{name}")
428                } else {
429                    name
430                };
431                ir.name = key.clone();
432                fn_file.entry(key.clone()).or_insert_with(|| path.clone());
433                irs.insert(key, ir);
434            }
435        }
436
437        let ir_names: rustc_hash::FxHashSet<String> = irs.keys().cloned().collect();
438        let program_refs: Vec<(&str, &str, &str)> = program
439            .iter()
440            .map(|(path, source, ext)| (path.as_str(), *source, ext.as_str()))
441            .collect();
442        let res = crate::graph::module::ModuleAliasResolver::build_from_refs(
443            &program_refs,
444            &ir_names,
445            &fn_file,
446        );
447
448        PreparedProgram {
449            irs,
450            fn_file,
451            file_source,
452            file_aliases: res.file_aliases,
453            file_package_imports: res.file_package_imports,
454            file_imports: res.file_imports,
455            file_exports: res.file_exports,
456            file_module_items: res.file_module_items,
457            lower_errors,
458            parse_partial: parse_partial_files,
459            submitted,
460            lower_caps,
461            lower_failed_files,
462            file_timings,
463        }
464    }
465}

467/// Lower every file once, remembering which file each function came from
468/// (functions are keyed by name; first file wins on collision).
469///
470/// # Errors
471/// Lowering failures are recorded in the returned program's error accounting
472/// rather than returned: a scan continues without the file and says so.
473pub fn prepare(files: &[(String, String, String)]) -> Result<PreparedProgram, String> {
474    prepare_with_facts(files, None)
475}

477/// Lower every file once with dynamic bundle facts.
478///
479/// # Errors
480/// Lowering failures are recorded in the returned program's error accounting
481/// rather than returned: a scan continues without the file and says so.
482pub fn prepare_with_facts(
483    files: &[(String, String, String)],
484    facts: Option<&FactTable>,
485) -> Result<PreparedProgram, String> {
486    let mut builder = ProgramBuilder::new();
487    add_sorted(&mut builder, files, facts);
488    Ok(builder.finish())
489}

491/// Submit `files` in path-sorted order, the order the pre-builder lowering
492/// loop used. Determinism lives in [`ProgramBuilder::finish`]; this only
493/// keeps the convenience entry points byte-for-byte compatible.
494fn add_sorted<'a>(
495    builder: &mut ProgramBuilder<'a>,
496    files: &'a [(String, String, String)],
497    facts: Option<&FactTable>,
498) {
499    let mut ordered: Vec<&'a (String, String, String)> = files.iter().collect();
500    ordered.sort_by(|a, b| a.0.cmp(&b.0));
501    for (path, source, ext) in ordered {
502        builder.add_file(path, source, ext, facts);
503    }
504}

506impl PreparedProgram {
507    /// Borrow the IRs for the program graph's lifetime.
508    fn ir_views(&self) -> FxHashMap<String, &FunctionIR> {
509        self.irs
510            .iter()
511            .map(|(name, ir)| (name.clone(), ir))
512            .collect()
513    }
514
515    /// Every file the program parsed, sorted.
516    ///
517    /// The module surface is the complete list (every parsed file gets an
518    /// entry, even one that declares no function); `file_source` is the
519    /// fallback for a program built before that layer ran.
520    #[must_use]
521    pub fn files(&self) -> Vec<&str> {
522        let mut files: Vec<&str> = if self.file_imports.is_empty() {
523            self.file_source.keys().map(String::as_str).collect()
524        } else {
525            self.file_imports.keys().map(String::as_str).collect()
526        };
527        files.sort_unstable();
528        files
529    }
530
531    /// The export surface of one file; empty when the file declares none or
532    /// its language has no export syntax the spec models.
533    #[must_use]
534    #[cfg(test)]
535    pub(crate) fn exports_of(&self, file: &str) -> &[Export] {
536        self.file_exports.get(file).map_or(&[], Vec::as_slice)
537    }
538
539    /// Every top-level declaration of one file, exported or not.
540    #[must_use]
541    #[cfg(test)]
542    pub(crate) fn module_items_of(&self, file: &str) -> &[ModuleItem] {
543        self.file_module_items.get(file).map_or(&[], Vec::as_slice)
544    }
545
546    /// The import declarations of one file.
547    #[must_use]
548    #[cfg(test)]
549    pub(crate) fn imports_of(&self, file: &str) -> &[Import] {
550        self.file_imports.get(file).map_or(&[], Vec::as_slice)
551    }
552}

554/// Scan a [`PreparedProgram`] under a config and fact table. Same semantics
555/// as [`scan`]; the lowering is reused across calls.
556#[allow(
557    clippy::disallowed_macros,
558    reason = "the FRENSdbg_MEM probes are process-level debug output, printed only when that debug flag is enabled"
559)]
560pub fn scan_prepared(
561    prepared: &PreparedProgram,
562    config: &TaintConfig,
563    facts: &FactTable,
564) -> ScanResult {
565    // Bundle-learned source patterns widen the config: a `.frc` bundle
566    // teaches new framework sources without touching the built-in tables.
567    // Local mutable copy; the caller's config stays shared and unchanged.
568    let mut config = config.clone();
569    config.sources.extend(facts.learned_sources.iter().cloned());
570    let config = &config;
571
572    // The program graph borrows the program's own IRs for the duration of
573    // this scan. No `Box::leak`: a leaked IR set accumulated per program in
574    // every long-lived consumer (MCP/LSP servers build one `PreparedProgram`
575    // per scan; the bundler's replay gate holds one per family, re-scanned
576    // per candidate fact), so resident memory grew without bound across
577    // scans. Borrowing is scoped to the call instead.
578    let statics = prepared.ir_views();
579    let statics = &statics;
580
581    // Phase accounting. Every phase below is timed; the reader of a report
582    // needs to know where the minutes went, not just that there were some.
583    let mut phases: Vec<PhaseTiming> = Vec::new();
584
585    let mem_probe = crate::debug_flags::DebugFlags::get().mem;
586    if mem_probe {
587        eprintln!(
588            "[mem] after_lower files={} irs={} rss_kb={}",
589            prepared.files().len(),
590            prepared.irs.len(),
591            crate::debug_flags::rss_kb()
592        );
593    }
594
595    // The program graph is built first: non-dataflow checks that walk
596    // interprocedural value flow (UAF free/use pairs) reuse it, and the
597    // taint engine below gets the same instance.
598    let prog = {
599        let t = Instant::now();
600        let prog = ProgramSvfg::new_with_module_aliases(
601            statics,
602            config,
603            facts,
604            Some(&prepared.file_aliases),
605            Some(&prepared.fn_file),
606        );
607        if mem_probe {
608            eprintln!(
609                "[mem] after_svfg fns={} rss_kb={}",
610                prog.functions.len(),
611                crate::debug_flags::rss_kb()
612            );
613        }
614        phases.push(PhaseTiming {
615            phase: "graph".into(),
616            ms: millis_since(t),
617        });
618        prog
619    };
620    // Non-dataflow policy checks run on the same lowered IR, no taint
621    // needed, so weak-crypto/config bugs surface even with zero taint paths.
622    // `facts` also carries corpus-learned checks installed by the bundle.
623    // Package-scope context: policies carrying `package: Some(pkg)` only
624    // fire in functions whose file imports `pkg`.
625    let pkg_scope = checks::policy::PackageScope {
626        fn_file: &prepared.fn_file,
627        file_package_imports: &prepared.file_package_imports,
628    };
629
630    let (checker_findings, check_telemetry) = {
631        let t = Instant::now();
632        let (found, check_telemetry) = checks::check_all_with_telemetry(
633            statics.values().copied(),
634            facts,
635            Some(&prog),
636            Some(&pkg_scope),
637        );
638        if mem_probe {
639            eprintln!(
640                "[mem] after_checks findings={} rss_kb={}",
641                found.len(),
642                crate::debug_flags::rss_kb()
643            );
644        }
645        phases.push(PhaseTiming {
646            phase: "checks".into(),
647            ms: millis_since(t),
648        });
649        (found, check_telemetry)
650    };
651
652    let fn_file = &prepared.fn_file;
653    let file_source = &prepared.file_source;
654    let mut engine = BackwardTaintEngine::new(&prog, config)
655        .with_fact_table(facts)
656        .with_fn_file(fn_file)
657        .with_file_package_imports(&prepared.file_package_imports)
658        // Route anchoring for per-call-site framing (schema §5.1):
659        // consult the same predicate the harness lowered the file with.
660        .with_route_registration(crate::harness::is_route_registration);
661    let taint_timing = {
662        let t = Instant::now();
663        engine.run();
664        millis_since(t)
665    };
666
667    if mem_probe {
668        eprintln!(
669            "[mem] after_taint findings={} rss_kb={}",
670            engine.findings.len(),
671            crate::debug_flags::rss_kb()
672        );
673    }
674    phases.push(PhaseTiming {
675        phase: "taint".into(),
676        ms: taint_timing,
677    });
678
679    // Which functions ate the taint phase: the ten slowest, with their file
680    // and how many sink arguments each one explored.
681    let slowest_functions = slowest_functions(&engine, &prepared.fn_file);
682
683    // Resolve each finding to a file + line/column from the sink's byte span.
684    // A finding whose function has no file (or whose file has no source text)
685    // cannot be pinned to a location; it is counted, never silently dropped.
686    let (located, unlocated_sinks) = locate_findings(&engine.findings, fn_file, file_source);
687
688    // Locate checker findings the same way: function → file, span → line/col.
689    let (located_checker, unlocated_checks) =
690        locate_checker_findings(&checker_findings, fn_file, file_source);
691    let unlocated = unlocated_sinks + unlocated_checks;
692
693    // Functions the per-function work budget pushed into the degraded
694    // (flow-insensitive) form during lowering: the report names them under
695    // `analysis_incomplete` rather than silently returning weaker results
696    // for them. Sorted so the JSON is stable across runs.
697    let mut analysis_incomplete: Vec<IncompleteFunction> = prepared
698        .irs
699        .iter()
700        .filter(|(_, ir)| ir.analysis_incomplete)
701        .map(|(name, _ir)| IncompleteFunction {
702            function: name.clone(),
703            file: prepared.fn_file.get(name).cloned().unwrap_or_default(),
704        })
705        .collect();
706    analysis_incomplete.sort_by(|a, b| (&a.file, &a.function).cmp(&(&b.file, &b.function)));
707
708    // Lowering already happened before this function; its cost is carried
709    // on the prepared program so `scan` and `scan_prepared` agree.
710    phases.insert(
711        0,
712        PhaseTiming {
713            phase: "lower".into(),
714            ms: prepared.file_timings.iter().map(|f| f.ms).sum(),
715        },
716    );
717
718    let failed_files = prepared.lower_failed_files.clone();
719    let mut caps = prepared.lower_caps.clone();
720    caps.extend(crate::telemetry::drain_caps());
721    // Real sizes of what the taint phase walked, and where its call sites left
722    // the analysis: measured here, never derived from another counter.
723    let graph = crate::telemetry::GraphSize::new(
724        prog.functions.len(),
725        prog.functions.iter().map(|f| f.svfg.nodes.len()).sum(),
726        prog.functions.iter().map(|f| f.svfg.edge_count()).sum(),
727        prog.cross_edges.values().map(Vec::len).sum(),
728    );
729    let funnel = engine.funnel.clone();
730    let mut telemetry = ScanTelemetry {
731        phases,
732        slowest_files: prepared.file_timings.clone(),
733        slowest_functions,
734        caps,
735        checks: check_telemetry,
736        suppressed: if unlocated > 0 {
737            vec![SuppressionCount {
738                reason: crate::telemetry::UNLOCATED_FINDING.into(),
739                count: unlocated,
740            }]
741        } else {
742            Vec::new()
743        },
744        errors: prepared.lower_errors.clone(),
745        input: InputAccounting {
746            submitted: prepared.submitted,
747            scanned: prepared.file_timings.len(),
748            failed: prepared.lower_failed_files.len(),
749            unsupported: 0,
750            failed_files,
751            parse_partial: prepared.parse_partial.len(),
752            parse_partial_files: prepared.parse_partial.clone(),
753            functions: prepared.irs.len(),
754        },
755        taint: engine.stats,
756        graph,
757        funnel,
758    };
759    telemetry
760        .slowest_files
761        .sort_by_key(|b| std::cmp::Reverse(b.ms));
762    telemetry
763        .slowest_files
764        .truncate(crate::telemetry::SLOW_SAMPLE);
765    // Files that failed to lower are errors, not silence: they land in both
766    // places so a report with no `errors` key really did read everything.
767    let errors = prepared.lower_errors.clone();
768
769    ScanResult {
770        findings: engine.findings,
771        located,
772        located_checker,
773        checker: checker_findings,
774        errors,
775        analysis_incomplete,
776        telemetry,
777    }
778}

780/// The `SLOW_SAMPLE` functions whose sink exploration took longest, each
781/// with the file it came from.
782fn slowest_functions(
783    engine: &BackwardTaintEngine<'_>,
784    fn_file: &FxHashMap<String, String>,
785) -> Vec<FunctionTiming> {
786    let mut slowest = engine.fn_timings.clone();
787    for entry in &mut slowest {
788        if entry.file.is_empty() {
789            entry.file = fn_file.get(&entry.function).cloned().unwrap_or_default();

791    }

793    slowest.truncate(crate::telemetry::SLOW_SAMPLE);

809/// The result of one scan: sink findings of every verdict, checker findings, non-fatal errors and telemetry.
810#[derive(Debug, Default)]
811pub struct ScanResult {
812    /// All sink findings (every verdict); retain only findings whose verdict
813    /// is `Vulnerable` to obtain the alert set.
814    ///
815    /// A method call's receiver is walked, because an object constructed from
816    /// tainted data (`new Package(taint)`) hands that data to every method
817    /// acting on it, but only a *tainted* receiver is reported: a clean or
818    /// undecided receiver adds no finding, so an all-args signature (what
819    /// [`FactTable::from_config`](crate::analysis::taint::facts::FactTable::from_config)
820    /// builds) reports the same slots as one with explicit arguments.
821    ///
822    /// ```
823    /// use frensense_engine::analysis::taint::config::TaintConfig;
824    /// use frensense_engine::analysis::taint::facts::{FactTable, SinkSignature};
825    /// use frensense_engine::scan::scan;
826    ///
827    /// let config = TaintConfig {
828    ///     sources: ["req.body".to_string()].into_iter().collect(),
829    ///     sinks: ["query".to_string()].into_iter().collect(),
830    ///     sanitizers: Default::default(),
831    /// };
832    /// let src = "declare const db: any\nexport function h(req: any) { db.query('SELECT ' + req.body.id) }\n";
833    /// let files = [("a.ts".to_string(), src.to_string(), "ts".to_string())];
834    /// let slots = |sig: SinkSignature| -> Vec<usize> {
835    ///     let mut facts = FactTable::from_config(&config);
836    ///     facts.sink_signatures.insert("query".into(), sig);
837    ///     let mut s: Vec<usize> = scan(&files, &config, &facts).findings.iter().map(|f| f.arg_slot).collect();
838    ///     s.sort();
839    ///     s
840    /// };
841    /// assert_eq!(slots(SinkSignature::with_args("query", &[0])), vec![0]);
842    /// // The receiver `db` is not an argument, and an all-args signature
843    /// // arms arguments only, so both signatures report the same slot.
844    /// assert_eq!(slots(SinkSignature::all_args("query")), vec![0]);
845    /// ```
846    pub findings: Vec<SinkFinding>,
847    /// The same findings resolved to source locations (file/line/column).
848    pub located: Vec<LocatedFinding>,
849    /// Non-dataflow policy violations (weak crypto, insecure config),
850    /// resolved to file/line/column.
851    pub located_checker: Vec<LocatedChecker>,
852    /// Non-dataflow policy violations (weak crypto, insecure config).
853    /// These have no taint path by definition; they are policy assertions
854    /// over the API/constant choices in the lowered IR.
855    pub checker: Vec<CheckerFinding>,
856    /// Errors collected during the scan (for example files that failed to lower).
857    pub errors: Vec<String>,
858    /// Functions analyzed in degraded form because their per-function work
859    /// budget expired (see [`IncompleteFunction`]). Empty for a scan in
860    /// which every function was analyzed fully.
861    pub analysis_incomplete: Vec<IncompleteFunction>,
862    /// What the scan measured and what stopped it: phase timings, the files
863    /// and functions that ate the time, every safety valve that fired, and
864    /// how much of the input was actually readable. See
865    /// [`crate::telemetry::ScanTelemetry`].
866    pub telemetry: ScanTelemetry,
867}

869/// A [`CheckerFinding`] pinned to its source location.
870#[derive(Debug, Clone)]
871pub struct LocatedChecker {
872    /// The policy violation.
873    pub finding: CheckerFinding,
874    /// Path of the file containing the violating call.
875    pub file: String,
876    /// 1-based line of the violating call (0 when no span recorded).
877    pub line: u32,
878    /// 1-based column (0 when no span recorded).
879    pub column: u32,
880    /// 1-based last line of the finding's span (>= `line`). Multi-line
881    /// spans (allowlist definitions) report their full extent.
882    pub end_line: u32,
883}

885/// 1-based line of byte `offset` in `source` (clamped to its end).
886fn line_at(source: &str, offset: usize) -> u32 {
887    source[..offset.min(source.len())].matches('\n').count() as u32 + 1
888}

890/// Resolve sink findings to file/line/column.
891///
892/// Returns the located findings and how many could not be located (their
893/// function has no file entry, or their file has no source text). The two
894/// always sum to `findings.len()`: nothing is dropped without being counted.
895fn locate_findings(
896    findings: &[SinkFinding],
897    fn_file: &FxHashMap<String, String>,
898    file_source: &FxHashMap<String, String>,
899) -> (Vec<LocatedFinding>, usize) {
900    let mut located = Vec::with_capacity(findings.len());
901    for f in findings {
902        let Some(file) = fn_file.get(&f.function) else {
903            continue;
904        };
905        let Some(source) = file_source.get(file) else {
906            continue;
907        };
908        let (line, col) = f
909            .sink_span
910            .map(|(start, _)| {
911                let before = &source[..start.min(source.len())];
912                let line = before.matches('\n').count() as u32 + 1;
913                let col = before
914                    .rfind('\n')
915                    .map_or(before.len(), |i| before.len() - i - 1)
916                    as u32
917                    + 1;
918                (line, col)
919            })
920            .unwrap_or((0, 0));
921        let cut_sites = f
922            .cut_sites
923            .iter()
924            .map(|c| {
925                let cfile = fn_file.get(&c.function).cloned().unwrap_or_default();
926                let cline = c
927                    .span
928                    .and_then(|(start, _)| Some(line_at(file_source.get(&cfile)?, start)))
929                    .unwrap_or(0);
930                LocatedCutSite {
931                    kind: c.kind,
932                    function: c.function.clone(),
933                    detail: c.detail.clone(),
934                    file: cfile,
935                    line: cline,
936                }
937            })
938            .collect();
939        located.push(LocatedFinding {
940            finding: f.clone(),
941            file: file.clone(),
942            line,
943            column: col,
944            cut_sites,
945        });
946    }
947    let unlocated = findings.len() - located.len();
948    (located, unlocated)
949}

951/// Resolve checker findings to file/line/column; see [`locate_findings`].
952fn locate_checker_findings(
953    findings: &[CheckerFinding],
954    fn_file: &FxHashMap<String, String>,
955    file_source: &FxHashMap<String, String>,
956) -> (Vec<LocatedChecker>, usize) {
957    let mut located = Vec::with_capacity(findings.len());
958    for c in findings {
959        let Some(file) = fn_file.get(&c.function) else {
960            continue;
961        };
962        let Some(source) = file_source.get(file) else {
963            continue;
964        };
965        let (line, col, end_line) = c
966            .span
967            .map(|(start, end)| {
968                let before = &source[..start.min(source.len())];
969                let line = before.matches('\n').count() as u32 + 1;
970                let col = before
971                    .rfind('\n')
972                    .map_or(before.len(), |i| before.len() - i - 1)
973                    as u32
974                    + 1;
975                (line, col, line_at(source, end))
976            })
977            .unwrap_or((0, 0, 0));
978        located.push(LocatedChecker {
979            finding: c.clone(),
980            file: file.clone(),
981            line,
982            column: col,
983            end_line: end_line.max(line),
984        });
985    }
986    let unlocated = findings.len() - located.len();
987    (located, unlocated)
988}
