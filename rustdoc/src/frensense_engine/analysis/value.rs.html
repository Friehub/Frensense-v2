5//! Per-function constant/interval value lattice (an abstract-interpretation
6//! pass over the lowered IR).
7//!
8//! Taint answers "whose data is this?"; this module answers "what data is
9//! this?". The two questions are orthogonal, and the second unlocks bug
10//! classes taint cannot see: integer bounds, weak crypto key sizes, literal
11//! selectors, constant-condition dead code, and hard evidence that a value
12//! is attacker-independent (taint verdict promotion).
13//!
14//! The lattice per variable is the classic constant-propagation domain lifted
15//! to a two-dimensional interval for integers:
16//!
17//! ```text
18//!            Const(i64)  |  Const(String)  |  Const(bool)      (precise)
19//!            Range(lo..=hi)                                   (interval)
20//!            Top                                              (unknown)
21//! ```
22//!
23//! Join is value-wise: `Const(a) ⊔ Const(b) = Top` unless equal;
24//! intervals widen to their hull; anything else degrades to `Top`. The
25//! transfer functions cover `Assign`, `BinaryOp`, `UnaryOp`, `Cast`, and the
26//! block-level `Phi` merge; every other definition (calls, loads, externals)
27//! yields `Top` for its destination, which is the sound over-approximation.
28//!
29//! The analysis runs a standard worklist fixpoint over the CFG with edge-state
30//! branch sharpening: conditional branches narrow numeric intervals along
31//! true and false paths (e.g. `x < 10` refines `x` to `[MIN, 9]` on true and
32//! `[10, MAX]` on false).
33
34use rustc_hash::{FxHashMap, FxHashSet};
35use std::collections::VecDeque;

39/// One variable's abstract value at a program point.
40#[derive(Debug, Clone, PartialEq, Eq)]
41pub enum Value {
42    /// A provably constant integer.
43    IntConst(i64),
44    /// A provably constant string (quotes already stripped by the lowering).
45    StrConst(String),
46    /// A provably constant boolean.
47    BoolConst(bool),
48    /// A known integer interval (inclusive). `Const(n)` is the degenerate
49    /// `Range(n, n)`; the separate `IntConst` variant keeps precise equality
50    /// joins cheap and makes callers' intent explicit.
51    Range(i64, i64),
52    /// Unknown / not provably constant. Sound default.
53    Top,
54}
55
56impl Value {
57    /// The interval hull of this value, for range arithmetic and joins.
58    /// Non-integer values have no interval (`None`).
59    pub fn interval(&self) -> Option<(i64, i64)> {
60        match self {
61            Value::IntConst(n) => Some((*n, *n)),
62            Value::Range(lo, hi) => Some((*lo, *hi)),
63            _ => None,
64        }
65    }
66
67    /// Least upper bound (the join). Associative and commutative; `Top` is
68    /// the identity for soundness (unknown absorbs anything).
69    pub fn join(&self, other: &Value) -> Value {
70        use Value::*;
71        match (self, other) {
72            (IntConst(a), IntConst(b)) if a == b => IntConst(*a),
73            (IntConst(a), IntConst(b)) => Range((*a).min(*b), (*a).max(*b)),
74            (StrConst(a), StrConst(b)) if a == b => StrConst(a.clone()),
75            (BoolConst(a), BoolConst(b)) if a == b => BoolConst(*a),
76            (Range(a, b), Range(c, d)) => Range((*a).min(*c), (*b).max(*d)),
77            (IntConst(a), Range(c, d)) | (Range(c, d), IntConst(a)) => {
78                Range((*c).min(*a), (*d).max(*a))
79            }
80            _ => Top,
81        }
82    }
83
84    /// Intersect an integer value/range with an inclusive interval `[min_lo, max_hi]`.
85    pub(crate) fn intersect_interval(&self, min_lo: Option<i64>, max_hi: Option<i64>) -> Value {
86        match self {
87            Value::IntConst(n) => {
88                if min_lo.is_none_or(|lo| *n >= lo) && max_hi.is_none_or(|hi| *n <= hi) {
89                    Value::IntConst(*n)
90                } else {
91                    self.clone()
92                }
93            }
94            Value::Range(lo, hi) => {
95                let new_lo = min_lo.map_or(*lo, |m| (*lo).max(m));
96                let new_hi = max_hi.map_or(*hi, |m| (*hi).min(m));
97                if new_lo > new_hi {
98                    self.clone()
99                } else if new_lo == new_hi {
100                    Value::IntConst(new_lo)
101                } else {
102                    Value::Range(new_lo, new_hi)
103                }
104            }
105            Value::Top => match (min_lo, max_hi) {
106                (Some(lo), Some(hi)) if lo == hi => Value::IntConst(lo),
107                (Some(lo), Some(hi)) if lo <= hi => Value::Range(lo, hi),
108                (Some(lo), None) => Value::Range(lo, i64::MAX),
109                (None, Some(hi)) => Value::Range(i64::MIN, hi),
110                _ => Value::Top,
111            },
112            _ => self.clone(),
113        }
114    }
115}

117/// Abstract state: one entry per var that has a non-`Top` value.
118pub(crate) type AbsState = FxHashMap<VarId, Value>;
119
120/// Merge state `b` into `a` using the lattice join operation.
121fn merge_states(a: &mut AbsState, b: &AbsState) {
122    for (v, val) in b {
123        let merged = match a.get(v) {
124            None => val.clone(),
125            Some(cur) => cur.join(val),
126        };
127        if merged == Value::Top {
128            a.remove(v);
129        } else {
130            a.insert(*v, merged);
131        }
132    }
133}

135/// Evaluate one `Operand` under a state.
136fn operand_value(state: &AbsState, op: &Operand) -> Value {
137    match op {
138        Operand::IntLiteral(n) => Value::IntConst(*n),
139        Operand::StringLiteral(s) => Value::StrConst(s.clone()),
140        Operand::BoolLiteral(b) => Value::BoolConst(*b),
141        Operand::Var(v) => state.get(v).cloned().unwrap_or(Value::Top),
142        Operand::FloatLiteral(_) | Operand::Null | Operand::Unknown => Value::Top,
143    }
144}

146/// Apply a binary operator to two integer intervals.
147fn binary_interval(op: &str, (al, ah): (i64, i64), (bl, bh): (i64, i64)) -> Value {
148    let add = |x: i64, y: i64| x.saturating_add(y);
149    match op {
150        "+" | "concat" => Value::Range(add(al, bl), add(ah, bh)),
151        "-" => Value::Range(al.saturating_sub(bh), ah.saturating_sub(bl)),
152        "*" => {
153            // Widen to i128 so wide ranges cannot wrap (or panic in
154            // debug): the corner products are clamped back to i64,
155            // which over-approximates the true interval.
156            let p = [
157                i128::from(al) * i128::from(bl),
158                i128::from(al) * i128::from(bh),
159                i128::from(ah) * i128::from(bl),
160                i128::from(ah) * i128::from(bh),
161            ];
162            let clamp = |v: i128| v.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
163            Value::Range(
164                clamp(*p.iter().min().unwrap_or(&i128::from(al))),
165                clamp(*p.iter().max().unwrap_or(&i128::from(ah))),
166            )
167        }
168        "<<" if bl == bh && (0..=32).contains(&bl) => Value::Range(al << bl, ah << bh),
169        ">>" if bl == bh && (0..=63).contains(&bl) => Value::Range(al >> bl, ah >> bh),
170        _ => Value::Top,
171    }
172}

174/// Transfer function for one instruction.
175fn transfer(state: &AbsState, instr: &Instruction) -> Option<(VarId, Value)> {
176    match instr {
177        Instruction::Assign { dest, src } => Some((*dest, operand_value(state, src))),
178        Instruction::BinaryOp { dest, op, lhs, rhs } => {
179            let lv = operand_value(state, lhs);
180            let rv = operand_value(state, rhs);
181            let out = match op.as_str() {
182                "==" | "===" | "!=" | "!==" => {
183                    let eq = match (&lv, &rv) {
184                        (Value::IntConst(a), Value::IntConst(b)) => Some(a == b),
185                        (Value::StrConst(a), Value::StrConst(b)) => Some(a == b),
186                        (Value::BoolConst(a), Value::BoolConst(b)) => Some(a == b),
187                        _ => None,
188                    };
189                    eq.map(|e| Value::BoolConst(if op.starts_with('!') { !e } else { e }))
190                        .unwrap_or(Value::Top)
191                }
192                "<" | ">" | "<=" | ">=" => match (lv.interval(), rv.interval()) {
193                    (Some((al, ah)), Some((bl, bh))) => match op.as_str() {
194                        "<" if ah < bl => Value::BoolConst(true),
195                        "<" if al >= bh => Value::BoolConst(false),
196                        "<=" if ah <= bl => Value::BoolConst(true),
197                        "<=" if al > bh => Value::BoolConst(false),
198                        ">" if al > bh => Value::BoolConst(true),
199                        ">" if ah <= bl => Value::BoolConst(false),
200                        ">=" if al >= bh => Value::BoolConst(true),
201                        ">=" if ah < bl => Value::BoolConst(false),
202                        _ => Value::Top,
203                    },
204                    _ => Value::Top,
205                },
206                _ => match (lv.interval(), rv.interval()) {
207                    (Some(a), Some(b)) => binary_interval(op, a, b),
208                    _ => Value::Top,
209                },
210            };
211            Some((*dest, out))
212        }
213        Instruction::UnaryOp { dest, op, src } => {
214            let v = operand_value(state, src);
215            let out = match op.as_str() {
216                "-" => v.interval().map_or(Value::Top, |(lo, hi)| {
217                    let n_hi = if hi == i64::MIN { i64::MAX } else { -hi };
218                    let n_lo = if lo == i64::MIN { i64::MAX } else { -lo };
219                    Value::Range(n_hi.min(n_lo), n_hi.max(n_lo))
220                }),
221                "!" => match v {
222                    Value::BoolConst(b) => Value::BoolConst(!b),
223                    _ => Value::Top,
224                },
225                "~" => match v {
226                    Value::IntConst(n) => Value::IntConst(!n),
227                    _ => Value::Top,
228                },
229                _ => Value::Top,
230            };
231            Some((*dest, out))
232        }
233        Instruction::Cast { dest, src, .. } => Some((*dest, operand_value(state, src))),
234        _ => None,
235    }
236}

238/// Does `instr` define a variable whose *value* is a boolean?
239///
240/// Only these shapes may carry a `BoolConst` refinement. A branch proves its
241/// condition is truthy, and for a boolean-valued variable truthiness IS the
242/// value; for anything else (`raw = request.args.get("id"); if raw:`) the
243/// truthiness says nothing about the payload, and recording a boolean there
244/// was unsound: the lattice then claimed `q = raw` was a constant and the
245/// backward walk cut a real source-to-sink flow as attacker-independent.
246fn is_boolean_producing(instr: &Instruction) -> bool {
247    match instr {
248        Instruction::BinaryOp { op, .. } => matches!(
249            op.as_str(),
250            "==" | "!=" | "===" | "!==" | "<" | "<=" | ">" | ">=" | "&&" | "||"
251        ),
252        Instruction::UnaryOp { op, .. } => matches!(op.as_str(), "!" | "not"),
253        Instruction::Assign {
254            src: Operand::BoolLiteral(_),
255            ..
256        } => true,
257        _ => false,
258    }
259}

261/// Sharpen variable bounds along a branch condition.
262fn sharpen_branch(
263    mut state: AbsState,
264    cond: &Operand,
265    is_true: bool,
266    defs: &FxHashMap<VarId, &Instruction>,
267) -> AbsState {
268    let mut cur = cond;
269    let mut sense = is_true;
270    loop {
271        match cur {
272            Operand::Var(c) => {
273                // Truthiness == value only for a boolean-valued variable.
274                let boolean_valued = matches!(state.get(c), Some(Value::BoolConst(_)))
275                    || defs.get(c).is_some_and(|d| is_boolean_producing(d));
276                if boolean_valued {
277                    state.insert(*c, Value::BoolConst(sense));
278                }
279                match defs.get(c) {
280                    Some(Instruction::UnaryOp { op, src, .. })
281                        if op == "!" || op == "neg" || op == "not" =>
282                    {
283                        cur = src;
284                        sense = !sense;
285                    }
286                    Some(Instruction::Assign { src, .. }) => cur = src,
287                    Some(Instruction::BinaryOp { op, lhs, rhs, .. })
288                        if (op == "||" && !sense) || (op == "&&" && sense) =>
289                    {
290                        state = sharpen_branch(state, lhs, sense, defs);
291                        return sharpen_branch(state, rhs, sense, defs);
292                    }
293                    Some(Instruction::BinaryOp { op, lhs, rhs, .. }) => {
294                        let (var, norm_op, bound) = match (lhs, rhs) {
295                            (Operand::Var(v), r) => match operand_value(&state, r) {
296                                Value::IntConst(k) => (*v, op.as_str(), k),
297                                _ => return state,
298                            },
299                            (l, Operand::Var(v)) => match operand_value(&state, l) {
300                                Value::IntConst(k) => {
301                                    let inv = match op.as_str() {
302                                        "<" => ">",
303                                        "<=" => ">=",
304                                        ">" => "<",
305                                        ">=" => "<=",
306                                        eq => eq,
307                                    };
308                                    (*v, inv, k)
309                                }
310                                _ => return state,
311                            },
312                            _ => return state,
313                        };
314
315                        let (lo, hi) = match (norm_op, sense) {
316                            ("<", true) | (">=", false) => (None, bound.checked_sub(1)),
317                            ("<=", true) | (">", false) => (None, Some(bound)),
318                            (">", true) | ("<=", false) => (bound.checked_add(1), None),
319                            (">=", true) | ("<", false) => (Some(bound), None),
320                            ("==" | "===", true) | ("!=" | "!==", false) => {
321                                (Some(bound), Some(bound))
322                            }
323                            _ => (None, None),
324                        };
325
326                        if lo.is_some() || hi.is_some() {
327                            let cur_val = state.get(&var).cloned().unwrap_or(Value::Top);
328                            let narrowed = cur_val.intersect_interval(lo, hi);
329                            if narrowed == Value::Top {
330                                state.remove(&var);
331                            } else {
332                                state.insert(var, narrowed);
333                            }
334                        }
335                        return state;
336                    }
337                    _ => return state,
338                }
339            }
340            _ => return state,
341        }
342    }
343}

345/// Compute fixpoint abstract states with edge-sensitive branch sharpening.
346/// Returns the block-entry states plus whether the worklist ran to
347/// convergence (`false` when the iteration cap cut the fixpoint short -
348/// states may then hold narrower-than-true values and must not be used
349/// for soundness-critical pruning).
350fn fixpoint(ir: &FunctionIR) -> (FxHashMap<BlockId, AbsState>, bool) {
351    // Pre-index definition instructions for fast predicate inspection.
352    let mut defs: FxHashMap<VarId, &Instruction> = FxHashMap::default();
353    for blk in ir.blocks.values() {
354        for instr in &blk.instructions {
355            match instr {
356                Instruction::Assign { dest, .. }
357                | Instruction::BinaryOp { dest, .. }
358                | Instruction::UnaryOp { dest, .. }
359                | Instruction::Cast { dest, .. } => {
360                    defs.insert(*dest, instr);
361                }
362                _ => {}
363            }
364        }
365    }
366
367    // Reverse postorder from entry block.
368    let mut po = Vec::new();
369    let mut seen: FxHashSet<BlockId> = FxHashSet::default();
370    let mut stack = vec![(ir.entry_block, 0usize)];
371    while let Some((b, i)) = stack.pop() {
372        if i == 0 {
373            if !seen.insert(b) {
374                continue;
375            }
376            stack.push((b, 1));
377            if let Some(blk) = ir.blocks.get(&b) {
378                for &s in &blk.successors {
379                    stack.push((s, 0));
380                }
381            }
382        } else {
383            po.push(b);
384        }
385    }
386    let rpo: Vec<BlockId> = po.into_iter().rev().collect();
387
388    let mut in_state: FxHashMap<BlockId, AbsState> = FxHashMap::default();
389    let mut edge_state: FxHashMap<(BlockId, BlockId), AbsState> = FxHashMap::default();
390
391    let mut queue: VecDeque<BlockId> = VecDeque::new();
392    let mut queued: FxHashSet<BlockId> = FxHashSet::default();
393    let mut iterations: usize = 0;
394    let mut converged = true;
395    const MAX_ITERATIONS: usize = 64;
396
397    for &b in &rpo {
398        queue.push_back(b);
399        queued.insert(b);
400    }
401
402    while let Some(bid) = queue.pop_front() {
403        iterations += 1;
404        if iterations > MAX_ITERATIONS * (ir.blocks.len() + 1) {
405            converged = false;
406            crate::telemetry::Cap::ValueIterations.record();
407            break;
408        }
409        queued.remove(&bid);
410
411        // 1. Entry state = join of incoming predecessor edge states.
412        let mut entry = AbsState::default();
413        let preds = ir
414            .blocks
415            .get(&bid)
416            .map(|b| b.predecessors.as_slice())
417            .unwrap_or(&[]);
418        let mut first = true;
419        for &p in preds {
420            if let Some(pstate) = edge_state.get(&(p, bid)) {
421                if first {
422                    entry = pstate.clone();
423                    first = false;
424                } else {
425                    merge_states(&mut entry, pstate);
426                }
427            }
428        }
429
430        // Phi merges: join over incoming operands.
431        if let Some(blk) = ir.blocks.get(&bid) {
432            for phi in &blk.phis {
433                let mut acc: Option<Value> = None;
434                for (_, v) in &phi.incoming {
435                    let val = entry.get(v).cloned().unwrap_or(Value::Top);
436                    acc = Some(acc.map_or(val.clone(), |a| a.join(&val)));
437                }
438                match acc.unwrap_or(Value::Top) {
439                    Value::Top => entry.remove(&phi.dest),
440                    v => entry.insert(phi.dest, v),
441                };
442            }
443        }
444        in_state.insert(bid, entry.clone());
445
446        // 2. Transfer through the block's instructions.
447        let mut exit = entry;
448        if let Some(blk) = ir.blocks.get(&bid) {
449            for instr in &blk.instructions {
450                if let Some((dest, val)) = transfer(&exit, instr) {
451                    if val == Value::Top {
452                        exit.remove(&dest);
453                    } else {
454                        exit.insert(dest, val);
455                    }
456                }
457            }
458        }
459
460        // 3. Propagate refined edge states to CFG successors.
461        if let Some(blk) = ir.blocks.get(&bid) {
462            let edges: Vec<(BlockId, AbsState)> = match &blk.terminator {
463                Terminator::Jump(s) => vec![(*s, exit)],
464                Terminator::Branch {
465                    cond,
466                    true_block,
467                    false_block,
468                } => {
469                    if true_block == false_block {
470                        vec![(*true_block, exit)]
471                    } else {
472                        vec![
473                            (*true_block, sharpen_branch(exit.clone(), cond, true, &defs)),
474                            (*false_block, sharpen_branch(exit, cond, false, &defs)),
475                        ]
476                    }
477                }
478                Terminator::Switch {
479                    cases,
480                    default_block,
481                    ..
482                } => {
483                    let mut e: Vec<_> = cases.iter().map(|(_, b)| (*b, exit.clone())).collect();
484                    e.push((*default_block, exit));
485                    e
486                }
487                _ => vec![],
488            };
489
490            for (succ, new_state) in edges {
491                let edge_key = (bid, succ);
492                let changed = match edge_state.get(&edge_key) {
493                    None => true,
494                    Some(old) => old != &new_state,
495                };
496                if changed {
497                    edge_state.insert(edge_key, new_state);
498                    if queued.insert(succ) {
499                        queue.push_back(succ);
500                    }
501                }
502            }
503        }
504    }
505
506    (in_state, converged)
507}

509/// The per-function analysis result: each variable's value at the point
510/// where its defining instruction executes, plus block-entry states.
511#[derive(Debug)]
512pub struct ValueInfo {
513    /// var → abstract value (only non-`Top` entries).
514    pub values: FxHashMap<VarId, Value>,
515    /// Number of CFG blocks analysed (diagnostics).
516    pub blocks: usize,
517    /// Block-entry abstract states (keyed by block ID).
518    pub block_entry_states: FxHashMap<BlockId, AbsState>,
519    /// Whether the fixpoint worklist ran to convergence. `false` means the
520    /// iteration cap cut the analysis short: states may hold
521    /// narrower-than-true values, so consumers that prune or promote based
522    /// on these values must treat the whole result as `Top`.
523    pub converged: bool,
524}
525
526/// Analyse one function.
527pub fn analyze(ir: &FunctionIR) -> ValueInfo {
528    let (in_states, converged) = fixpoint(ir);
529
530    let mut values: FxHashMap<VarId, Value> = FxHashMap::default();
531    for (bid, mut state) in in_states.clone() {
532        let Some(blk) = ir.blocks.get(&bid) else {
533            continue;
534        };
535        for phi in &blk.phis {
536            if let Some(val) = state.get(&phi.dest) {
537                values.insert(phi.dest, val.clone());
538            }
539        }
540        for instr in &blk.instructions {
541            if let Some((dest, val)) = transfer(&state, instr) {
542                values.insert(dest, val.clone());
543                if val == Value::Top {
544                    state.remove(&dest);
545                } else {
546                    state.insert(dest, val);
547                }
548            }
549        }
550    }
551    values.retain(|_, v| *v != Value::Top);
552    ValueInfo {
553        values,
554        blocks: ir.blocks.len(),
555        block_entry_states: in_states,
556        converged,
557    }
558}

560impl ValueInfo {
561    /// The constant value of `var`, if provably constant.
562    pub fn const_int(&self, var: VarId) -> Option<i64> {
563        match self.values.get(&var)? {
564            Value::IntConst(n) => Some(*n),
565            Value::Range(lo, hi) if lo == hi => Some(*lo),
566            _ => None,
567        }
568    }
569
570    /// The string constant of `var`, if provably constant.
571    pub fn const_str(&self, var: VarId) -> Option<&str> {
572        match self.values.get(&var)? {
573            Value::StrConst(s) => Some(s.as_str()),
574            _ => None,
575        }
576    }
577
578    /// The boolean constant of `var`, if provably constant.
579    pub fn const_bool(&self, var: VarId) -> Option<bool> {
580        match self.values.get(&var)? {
581            Value::BoolConst(b) => Some(*b),
582            _ => None,
583        }
584    }
585
586    /// The interval of `var`, if an integer with a known range.
587    pub(crate) fn range(&self, var: VarId) -> Option<(i64, i64)> {
588        self.values.get(&var).and_then(Value::interval)
589    }
590
591    /// The constant integer value of `var` at the entry of `block`.
592    pub(crate) fn const_int_at(&self, block: BlockId, var: VarId) -> Option<i64> {
593        match self.block_entry_states.get(&block)?.get(&var)? {
594            Value::IntConst(n) => Some(*n),
595            Value::Range(lo, hi) if lo == hi => Some(*lo),
596            _ => None,
597        }
598    }
599
600    /// The interval of `var` at the entry of `block`.
601    pub(crate) fn range_at(&self, block: BlockId, var: VarId) -> Option<(i64, i64)> {
602        self.block_entry_states
603            .get(&block)?
604            .get(&var)
605            .and_then(Value::interval)
606    }
