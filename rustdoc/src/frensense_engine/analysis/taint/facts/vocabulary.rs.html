5//! Fact-vocabulary fingerprint: the wire compatibility contract for `.frc`
6//! bundles.
7//!
8//! A bundle stores `LearnedFactEntry` values and `HintKind` discriminants
9//! by *position* (bincode is positional and non-self-describing). An older
10//! binary therefore cannot decode a bundle that uses a variant added after
11//! it was built, and the failure surfaces as an opaque decode error deep
12//! inside serde rather than as a version mismatch a user can act on.
13//!
14//! This module replaces that silence with one number. It hashes the ordered
15//! serialization surface - hint kinds and fact-entry variants - with a
16//! stable FNV-1a so it is reproducible from source without a runtime
17//! dependency. A bundle records the fingerprint it was built against; the
18//! engine compares it at load time and refuses, by name, to decode
19//! vocabulary it does not know ([`frensense-bundler`'s header check]).
20//!
21//! The ratchet is mechanical, not honour-based:
22//!
23//! * `HintKind::index` and `LearnedFactEntry::wire_index` are exhaustive
24//!   `match`es, so *adding* a variant is a compile error that lands the
25//!   developer next to the vocabulary lists.
26//! * `tests::vocabulary_lists_are_ratcheted` fails until the new variant is
27//!   appended to `HintKind::ALL` / `LearnedFactEntry::VARIANT_TAGS`, and
28//!   `tests::vocabulary_fingerprint_is_committed` fails until the change is
29//!   acknowledged by updating [`FACT_VOCABULARY_FINGERPRINT`] - which is also
30//!   the cue to bump `BUNDLE_VERSION` and regenerate the default pack.
31//!
32//! Struct *field* additions are not covered here (they are caught by the
33//! bundler's committed golden bundle, which stops decoding once the layout
34//! moves); this fingerprint covers enum order and naming.
35
36use super::{HintKind, LearnedFactEntry};
37use std::fmt::Write as _;
38
39/// Canonical description of the serialized fact vocabulary: one line per
40/// hint kind and fact-entry variant, in wire order, with its index.
41///
42/// Two vocabularies that can decode each other's bundles produce identical
43/// text; any append, reorder, insert or rename changes it.
44#[must_use]
45pub(crate) fn fact_vocabulary_canonical() -> String {
46    // `write!` into the buffer (not `format!` + `push_str`): the output is
47    // byte-identical, but this is a deterministic wire-format serializer,
48    // not user-facing prose, so it must not count against the engine-prose
49    // ratchet. Every line below is a stable serialization token.
50    let mut out = String::with_capacity(4096);
51    let _ = writeln!(out, "frensense-fact-vocabulary/1");
52    let _ = writeln!(out, "hint:{}", HintKind::ALL.len());
53    for (i, kind) in HintKind::ALL.iter().enumerate() {
54        let _ = writeln!(out, "{i} {}", kind.name());
55    }
56    let _ = writeln!(out, "entry:{}", LearnedFactEntry::VARIANT_TAGS.len());
57    for (i, tag) in LearnedFactEntry::VARIANT_TAGS.iter().enumerate() {
58        let _ = writeln!(out, "{i} {tag}");
59    }
60    out
61}
62
63/// FNV-1a 64 over `fact_vocabulary_canonical`.
64///
65/// FNV rather than a cryptographic hash: this is a compatibility token, not
66/// a security property, and it must be reproducible from source with no
67/// dependency and no build step.
68#[must_use]
69pub fn fact_vocabulary_fingerprint() -> u64 {
70    const OFFSET: u64 = 0xcbf2_9ce4_8422_2325;
71    const PRIME: u64 = 0x0000_0100_0000_01b3;
72    let mut hash = OFFSET;
73    for byte in fact_vocabulary_canonical().as_bytes() {
74        hash ^= u64::from(*byte);
75        hash = hash.wrapping_mul(PRIME);
76    }
77    hash
78}

80/// The committed fingerprint for this build. Recorded in every `.frc` the
81/// bundler writes and checked by the engine at load time.
82///
83/// When this value changes, so does every published bundle's compatibility:
84/// bump `BUNDLE_VERSION`, regenerate the default pack, and rebuild any
85/// corpus bundle. `docs/FORMATS.md` carries the same rule in prose.
86pub const FACT_VOCABULARY_FINGERPRINT: u64 = 0x4994_27a8_ed94_e470;
87
88/// Call-name decorators that libraries append to a base operation to name a
89/// sibling in the same family: adm-zip's `extractAllTo` / `extractAllToAsync`,
90/// Node's `readFile` / `readFileSync`, and the callback forms of the same
91/// pattern.
92///
93/// Facts are matched by call name, so without this a corpus family that
94/// teaches the base operation silently misses every decorated sibling - the
95/// held-out pair then reports a miss even though the underlying defect is
96/// identical. The suffix is only ever stripped for lookup, never published:
97/// a fact still names exactly one call.
98pub(crate) const MIRROR_CALL_SUFFIXES: &[&str] = &["Async", "Sync", "Callback", "Promise"];
99
100/// The base operation a decorated call name belongs to.
101///
102/// `extractAllToAsync` -> `extractAllTo`. Returns `None` when the name carries
103/// no known decorator, or when stripping it would leave no base at all.
104#[must_use]
105pub(crate) fn mirror_call_base(call: &str) -> Option<&str> {
106    MIRROR_CALL_SUFFIXES
107        .iter()
108        .find_map(|suffix| call.strip_suffix(suffix).filter(|base| !base.is_empty()))
109}

111#[cfg(test)]
112mod tests {
113    use super::*;
114
115    /// Every `HintKind` is listed, exactly once, in wire order.
116    #[test]
117    fn hint_kinds_are_ratcheted() {
118        let all = HintKind::ALL;
119        assert_eq!(
120            all.len(),
121            HintKind::WeakHashDisclaimer.index() + 1,
122            "HintKind::ALL is stale: a variant was added or removed without \
123             appending it to ALL (and to the canonical vocabulary)"
124        );
125        for kind in &all {
126            assert_eq!(
127                kind.index(),
128                all.iter()
129                    .position(|other| other.name() == kind.name())
130                    .expect("duplicate HintKind name"),
131                "HintKind::ALL is not in wire order"
132            );
133        }
134    }
135
136    /// Every `LearnedFactEntry` variant is tagged, once, in wire order.
137    #[test]
138    fn vocabulary_lists_are_ratcheted() {
139        // Constructing the highest-index variant: its `wire_index()` is the
140        // count of variants, so a new arm in that exhaustive match fails
141        // this assertion until VARIANT_TAGS grows too.
142        let last = LearnedFactEntry::IntegerOverflowUnboundedRule {
143            rule: String::new(),
144            wrap_threshold: 0,
145            severity: String::new(),
146            message: String::new(),
147        };
148        assert_eq!(
149            LearnedFactEntry::VARIANT_TAGS.len(),
150            last.wire_index() + 1,
151            "LearnedFactEntry::VARIANT_TAGS is stale: a variant was added \
152             without appending its tag"
153        );
154        let mut seen = std::collections::BTreeSet::new();
155        for tag in LearnedFactEntry::VARIANT_TAGS {
156            assert!(seen.insert(tag), "duplicate fact-entry tag {tag}");
157        }
158    }
159
160    /// The fingerprint must equal the committed constant. Failing here means
161    /// the serialized vocabulary changed: append nothing silently - bump
162    /// `BUNDLE_VERSION`, update `FACT_VOCABULARY_FINGERPRINT`, regenerate the
163    /// default pack, and rebuild corpus bundles.
164    #[test]
165    fn vocabulary_fingerprint_is_committed() {
166        assert_eq!(
167            fact_vocabulary_fingerprint(),
168            FACT_VOCABULARY_FINGERPRINT,
169            "fact vocabulary changed (canonical text: {})",
170            fact_vocabulary_canonical()
171        );
172    }
173
174    /// The canonical text is stable and covers every list.
175    #[test]
176    fn canonical_text_covers_the_whole_surface() {
177        let text = fact_vocabulary_canonical();
178        assert!(text.starts_with("frensense-fact-vocabulary/1\n"));
179        for tag in LearnedFactEntry::VARIANT_TAGS {
180            assert!(text.contains(&format!("{tag}\n")), "missing {tag}");
181        }
182        for kind in HintKind::ALL {
183            assert!(
184                text.contains(&format!("{}\n", kind.name())),
185                "missing {kind:?}"
186            );
187        }
188    }
189}
