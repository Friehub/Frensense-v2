5//! `AuthoredPolicyEntry`: hand-authored policy rules persisted in a
6//! bundle's `policy_pack` section (this module keeps only the entry type;
7//! the table itself lives in `table.rs`, the learned entries in
8//! `entries.rs`).
9
10use super::*;
11
12/// One hand-authored rule as persisted in a bundle's `policy_pack` section
13/// (`.frc` v5). The section is reserved (D2 cancelled: corpus-only
14/// authoring); presence in the section is what marks an entry's
15/// provenance [`Provenance::Authored`], mirroring `learned_facts` ->
16/// [`Provenance::Learned`].
17#[derive(Debug, Clone, PartialEq, Eq)]
18#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
19pub enum AuthoredPolicyEntry {
20    /// A generalized co-occurrence policy: same fields as [`PolicyFact`],
21    /// carried whole so the author's severity/message travel with it.
22    Policy(PolicyFact),
23    /// An allocation-size integer-overflow rule: same fields as
24    /// [`LearnedFactEntry::IntegerOverflowRule`] (the hand-authored
25    /// precedent: `[frensense] wrap-max:` corpus metadata).
26    IntegerOverflowRule {
27        /// Prover rule id.
28        rule: String,
29        /// Largest value the size computation may reach without wrapping; a computation whose proven maximum exceeds it is reported.
30        wrap_threshold: u128,
31        /// Advisory severity hint.
32        severity: String,
33        /// Advisory text shown to the user.
34        message: String,
35    },
36}
