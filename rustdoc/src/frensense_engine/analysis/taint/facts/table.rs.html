11/// The merged fact table: built-in language tables + bundle-learned facts.
12///
13/// Built from a [`TaintConfig`] (name sets, backward compatible) plus
14/// optional signatures/sanitizer facts supplied by `frensense-lang` specs or
15/// a `.frc` bundle.
16#[derive(Debug, Clone, Default)]
17pub struct FactTable {
18    /// By call name. Absent = default "all args dangerous" for configured
19    /// sinks.
20    pub sink_signatures: FxHashMap<String, SinkSignature>,
21    /// Language-keyed sink signatures: `language -> call -> signature`.
22    ///
23    /// The default pack declares its provider vocabulary per language, and a
24    /// bare call name is not portable between them: Rust's `File::create`
25    /// and a JavaScript `prisma.order.create(...)` share the last segment
26    /// `create`, so installing both into one name-keyed map makes the Rust
27    /// fact fire on the TypeScript call. Measured on a real repo: a single
28    /// `.rs` file in a TypeScript-only package took the scan from 6 findings
29    /// to 40, all of them `create`/`output` calls in `.ts` files.
30    ///
31    /// Facts therefore stay under the language that declared them, and
32    /// matching consults the call site's language first
33    /// ([`FactTable::sink_signature_for_language`]). [`sink_signatures`]
34    /// keeps only genuinely language-agnostic knowledge (`"*"` pack group,
35    /// bundle facts, hand-built configs).
36    ///
37    /// [`sink_signatures`]: FactTable::sink_signatures
38    pub lang_sink_signatures: FxHashMap<String, FxHashMap<String, SinkSignature>>,
39    /// Language-keyed taint sources: `language -> patterns`.
40    ///
41    /// The source mirror of [`lang_sink_signatures`]: conventional parameter
42    /// names (`req`, `input`, ...) and framework source patterns are declared
43    /// per language, so a JavaScript-only name must not seed taint in a Rust
44    /// file. `input` is a request source in TypeScript and an unrelated local
45    /// in Rust; sharing one set made `info!(..)` a source for every Rust
46    /// `tokio::spawn` call in the same repo.
47    ///
48    /// [`lang_sink_signatures`]: FactTable::lang_sink_signatures
49    pub lang_sources: FxHashMap<String, FxHashSet<String>>,
50    /// Sanitizer facts keyed by call name.
51    pub sanitizer_facts: FxHashMap<String, SanitizerFact>,
52    /// Scoped sink signatures: (package, call_name) -> SinkSignature.
53    pub scoped_sink_signatures: FxHashMap<(String, String), SinkSignature>,
54    /// Scoped sanitizers: (package, call_name) -> SanitizerFact.
55    pub scoped_sanitizer_facts: FxHashMap<(String, String), SanitizerFact>,
56    /// Scoped taint sources: (package, pattern).
57    pub scoped_sources: FxHashSet<(String, String)>,
58    /// Corpus-verified non-dataflow checks (order-preserving; dedup on
59    /// `(rule, call)` at merge time, provenance-ordered per entry).
60    pub learned_checks: Vec<(LearnedCheckFact, Provenance)>,
61    /// Corpus-verified co-occurrence policies (the generalized check fact).
62    /// Evaluated by `checks::policy`; legacy `learned_checks` entries ALSO
63    /// evaluate there after `PolicyFact::from_legacy` conversion, so bundles
64    /// never need to migrate to keep firing. Dedup on
65    /// `(rule, when_call)` at merge time, provenance-ordered per entry.
66    pub policy_facts: Vec<(PolicyFact, Provenance)>,
67    /// Corpus-verified interprocedural memory contracts (allocators / deallocators).
68    pub memory_contracts: Vec<MemoryContractFact>,
69    /// Corpus-verified weak cryptography rules.
70    pub weak_crypto_rules: Vec<WeakCryptoFact>,
71    /// Containment callees (for allowlist bypass checks) with per-entry
72    /// provenance: default-pack-seeded (Spec), bundle-applied (Learned).
73    pub containment_callees: FxHashMap<String, Provenance>,
74    /// Credential setter/hasher sinks with per-entry provenance (pack ->
75    /// Spec, bundle -> Learned).
76    pub credential_sinks: FxHashMap<String, Provenance>,
77    /// Parameter names identifying credentials with per-entry provenance
78    /// (pack -> Spec, bundle -> Learned).
79    pub credential_params: FxHashMap<String, Provenance>,
80    /// Schema builder methods with per-entry provenance (pack -> Spec,
81    /// bundle -> Learned).
82    pub schema_builders: FxHashMap<String, Provenance>,
83    /// Corpus-verified schema enforcer methods (default-pack-seeded).
84    pub schema_enforcers: FxHashSet<String>,
85    /// Bound keywords with per-entry provenance (pack -> Spec, bundle ->
86    /// Learned).
87    pub schema_keywords: FxHashMap<String, Provenance>,
88    /// URL/redirect parameter-name hints (default-pack-seeded).
89    pub url_param_hints: FxHashSet<String>,
90    /// URL-ish guard argument hints (default-pack-seeded).
91    pub url_arg_hints: FxHashSet<String>,
92    /// Absolute-URL literal hints (default-pack-seeded).
93    pub url_literal_hints: FxHashSet<String>,
94    /// Security-context name hints (default-pack-seeded).
95    pub security_context_hints: FxHashSet<String>,
96    /// Auth-guard call-path hints (default-pack-seeded); the idor emission
97    /// gate suppresses identity findings behind a guard branch over these
98    /// calls.
99    pub auth_guard_hints: FxHashSet<String>,
100    /// JWT algorithm-operation call-path hints (default-pack-seeded).
101    pub jwt_algorithm_hints: FxHashSet<String>,
102    /// Credential-context name hints (default-pack-seeded).
103    pub credential_context_hints: FxHashSet<String>,
104    /// Receiver-name fragments of password-hashing libraries.
105    pub kdf_receiver_hints: FxHashSet<String>,
106    /// Lowercase name prefixes of calls that create a bound-parameter
107    /// statement handle (`Prepare`, `prepareStatement`, `prepare_cached`).
108    ///
109    /// A handle from one of these factories executes text that was fixed
110    /// when it was created, so the arguments of its later calls are bindings,
111    /// not SQL. Which factory names exist is pack knowledge; the engine only
112    /// walks the def chain and matches the prefix against this set.
113    pub prepared_statement_prefixes: FxHashSet<String>,
114    /// Argument tokens (lowercase) that disclaim security use of a weak
115    /// digest: a literal `false` flag in a non-payload position marks the
116    /// digest as a checksum or cache key, not as credential protection.
117    /// Pack knowledge; the engine only decides which positions are flags.
118    pub weak_hash_disclaimer_tokens: FxHashSet<String>,
119    /// Weak-hash policy rules (default-pack-seeded).
120    pub weak_hash_rules: Vec<WeakPrimitiveRule>,
121    /// Corpus-extendable allocation-size overflow rules (CWE-190 -> CWE-680;
122    /// default-pack-seeded, bundle extension via
123    /// `LearnedFactEntry::IntegerOverflowRule` from a family's
124    /// `[frensense] check-rule:` declaration).
125    pub integer_overflow_rules: Vec<(IntegerOverflowRule, Provenance)>,
126    /// Insecure config selectors (default-pack-seeded).
127    pub insecure_config_selectors: Vec<InsecureConfigRule>,
128    /// Key-size policy rules (default-pack-seeded).
129    pub key_size_rules: Vec<KeySizeRule>,
130    /// Suspicious hash-wrapper names (default-pack-seeded).
131    pub suspicious_hash_wrappers: FxHashSet<String>,
132    /// Last segments that come ONLY from dotted client sinks (`got.get`,
133    /// `axios.post`, ...). These verbs are ambiguous, `Map.get`, router
134    /// `app.post`, LRU `.put` all share the names, so they match
135    /// receiver-aware: the call only counts as a sink when the receiver's
136    /// root is a known client (see [`Self::client_roots`]).
137    pub verb_sinks: FxHashSet<String>,
138    /// Receiver roots that make `verb_sinks` real client calls, derived
139    /// from the first segment of dotted verb sinks (`got`, `axios`, `http`).
140    pub client_roots: FxHashSet<String>,
141    /// Receiver-specific roles for dotted sinks that share their last
142    /// segment with a different-role bare entry:
143    /// `(receiver_root, method) -> role`.
144    ///
145    /// Example: `KVNamespace.put` is StorageWrite while bare `run` is
146    /// SqlInjection; `shell.run` is CommandInjection. When the engine sees
147    /// `kv.put(...)` and the receiver root resolves to `kv`'s declared
148    /// dotted root, THIS role wins over the bare entry's, the receiver is
149    /// the disambiguator, exactly like `verb_sinks` for match/no-match.
150    pub receiver_roles: FxHashMap<(String, String), crate::analysis::taint::role::SinkRole>,
151    /// Receiver-specific label slugs: (receiver_root, method) -> label slug.
152    pub receiver_labels: FxHashMap<(String, String), String>,
153    /// Last segments that are ONLY sinks under a declared receiver root:
154    /// `last -> declared roots`.
155    ///
156    /// A dotted declaration (`pickle.loads`, `urllib.request.urlopen`)
157    /// publishes a bare shadow of its last segment, because the walker
158    /// matches a virtual call by method name and cannot see the module the
159    /// method was reached through. Without this map the shadow is
160    /// receiver-agnostic: `json.loads(t)` inherited unsafe-deserialization
161    /// semantics from `pickle.loads`, and `client.urlopen(t)` inherited
162    /// SSRF from `urllib.request.urlopen`.
163    ///
164    /// The map turns "declared qualified" into an engine-enforced
165    /// requirement: a name listed in this map arms only under a receiver root (or
166    /// an import origin) the declaration named. The engine stays
167    /// mechanism-only - which roots qualify which name is pack/bundle
168    /// knowledge, not engine vocabulary.
169    pub receiver_qualified_sinks: FxHashMap<String, FxHashSet<String>>,
170    /// Last segments declared BARE (a declaration with no dot in it).
171    ///
172    /// A bare declaration is the receiver-agnostic form: it is the
173    /// language builtin's own name (`eval`, `exec`, `open`). A name that
174    /// is only ever declared qualified is gated by
175    /// `receiver_qualified_sinks`; a name declared bare is not - it keeps
176    /// matching any receiver, which is correct for a builtin and wrong for
177    /// a library function.
178    pub bare_sink_names: FxHashSet<String>,
179    /// Receiver roots of trusted session stores (`authenticatedUsers`):
180    /// `store.get(token)` returns a server-issued session object or
181    /// undefined, values read off the result are not attacker-controlled
182    /// (the pack ships these with session-trust sanitizer facts).
183    pub session_roots: FxHashSet<String>,
184    /// Taint-source patterns learned from a `.frc` bundle: call names or

188    pub learned_sources: FxHashSet<String>,
189    /// Dynamic node role classifications learned from a `.frc` bundle:
190    /// `(language, node_kind) -> role`. Stored as the owned

214    /// Stack-frame allocators (`alloca`, ...), default-pack-seeded

217    /// Route-registration call patterns per language

260impl FactTable {
261    /// Build from a plain [`TaintConfig`]: every configured sink gets the
262    /// all-args signature; every configured sanitizer a default fact.
263    pub fn from_config(config: &TaintConfig) -> Self {
264        let mut t = Self::default();
265        for s in &config.sinks {
266            t.sink_signatures
267                .insert(s.clone(), SinkSignature::all_args(s));
268        }
269        for s in &config.sanitizers {
270            t.sanitizer_facts.insert(
271                s.clone(),
272                SanitizerFact {
273                    call: s.clone(),
274                    package: None,
275                    kind: DEFAULT_SANITIZER_KIND.into(),
276                    sanitizes_args: Default::default(),
277                    guard_style: false,
278                },
279            );
280        }
281        t
282    }

284    /// Merge bundle-learned facts over the current table (learned wins on
285    /// name collision, bundle facts are more specific than built-ins).
286    ///
287    /// All-args signatures never overwrite slot-restricted ones on collision:
288    /// a slot-restricted signature is strictly more specific knowledge, so an
289    /// all-args merge (e.g. another language spec declaring the same bare
290    /// call name) must not widen it back to "everything dangerous".
291    /// Merge another table's language-keyed section under `language`.
292    ///
293    /// Used by [`crate::analysis::taint::facts::apply_language_entries`]
294    /// to install one pack language group: the group is built in a fresh
295    /// table (exactly like the pre-scoping per-spec seed merge) and then
296    /// filed under its own language instead of the shared name space, so
297    /// cross-language key collisions can no longer resolve last-merge-wins.
298    /// The signature-widening rule is the same one [`Self::merge`] applies.
299    pub(crate) fn merge_language(&mut self, language: &str, other: &FactTable) {
300        let slot = self
301            .lang_sink_signatures
302            .entry(language.to_string())
303            .or_default();
304        for (k, v) in &other.sink_signatures {
305            let widens = v.dangerous_args.is_empty()
306                && !v.binding_args_safe
307                && slot
308                    .get(k)
309                    .map(|cur| !cur.dangerous_args.is_empty() || cur.binding_args_safe)
310                    .unwrap_or(false);
311            if widens {
312                continue;
313            }
314            slot.insert(k.clone(), v.clone());
315        }
316    }

323    /// Sink signature for `call` in the language it was written in.
324    ///
325    /// Precedence: the language-agnostic lane first (a consumer bundle and
326    /// the `"*"` pack group are the last seeding layer, so they win over a
327    /// language section), then the call site's own language - except that a
328    /// blanket all-args placeholder (`SinkSignature::is_blanket`, what
329    /// `TaintConfig::sinks` seeds for every configured name) never shadows
330    /// the language's own entry, which knows the call's dangerous slots and
331    /// identity-payload keys. A qualified callee
332    /// (`std::fs::read_to_string`) also matches shorter declared shapes
333    /// through its `::` suffixes, exact matches first - the same rule
334    /// [`FactTable::sink_signature`] applies.
335    ///
336    /// `language: None` is the legacy "no call-site context" form: it
337    /// over-approximates across every installed language, which is why the
338    /// real scan path always passes the IR's language.
339    pub fn sink_signature_for_language(
340        &self,
341        call: &str,
342        language: Option<&str>,
343    ) -> Option<&SinkSignature> {
344        let universal = Self::lookup_sink_shape(&|c: &str| self.sink_signatures.get(c), call);
345        let scoped = match language {
346            Some(language) => self
347                .lang_sink_signatures
348                .get(language)
349                .and_then(|table| Self::lookup_sink_shape(&|c: &str| table.get(c), call)),
350            None => self
351                .lang_sink_signatures
352                .values()
353                .find_map(|table| Self::lookup_sink_shape(&|c: &str| table.get(c), call)),
354        };
355        match (universal, scoped) {
356            (Some(u), Some(l)) if u.is_blanket() => Some(l),
357            (Some(u), _) => Some(u),
358            (None, scoped) => scoped,
359        }
360    }

362    /// Exact, then `::`-suffix, then decorated-mirror lookup over one
363    /// signature source.
364    fn lookup_sink_shape<'a>(
365        lookup: &dyn Fn(&str) -> Option<&'a SinkSignature>,
366        call: &str,
367    ) -> Option<&'a SinkSignature> {
368        if let Some(sig) = lookup(call) {
369            return Some(sig);
370        }
371        if call.contains("::") {
372            let parts: Vec<&str> = call.split("::").collect();
373            for i in 1..parts.len() {
374                let suffix = parts[i..].join("::");
375                if let Some(sig) = lookup(&suffix) {
376                    return Some(sig);
377                }
378            }
379        }
380        // A decorated mirror of a taught operation (`extractAllToAsync` for
381        // `extractAllTo`) is the same API under a conventional suffix.
382        if let Some(base) = super::mirror_call_base(call) {
383            return lookup(base);
384        }
385        None
386    }

388    /// Merges `other` into this table. A blanket sink signature never widens an existing slot-restricted one, and for provenance-tracked entries the higher provenance wins.
389    pub fn merge(&mut self, other: &FactTable) {
390        for (k, v) in &other.sink_signatures {
391            let widens = v.dangerous_args.is_empty()
392                && !v.binding_args_safe
393                && self
394                    .sink_signatures
395                    .get(k)
396                    .map(|cur| !cur.dangerous_args.is_empty() || cur.binding_args_safe)
397                    .unwrap_or(false);
398            if widens {
399                continue;
400            }
401            self.sink_signatures.insert(k.clone(), v.clone());
402        }
403        for (k, v) in &other.sanitizer_facts {
404            self.sanitizer_facts.insert(k.clone(), v.clone());
405        }
406        for (lang, section) in &other.lang_sink_signatures {
407            let slot = self.lang_sink_signatures.entry(lang.clone()).or_default();
408            for (k, v) in section {
409                slot.insert(k.clone(), v.clone());
410            }
411        }
412        for (lang, section) in &other.lang_sources {
413            self.lang_sources
414                .entry(lang.clone())
415                .or_default()
416                .extend(section.iter().cloned());
417        }
418        for (k, v) in &other.scoped_sink_signatures {
419            let widens = self
420                .scoped_sink_signatures
421                .get(k)
422                .map(|cur| !cur.dangerous_args.is_empty() || cur.binding_args_safe)
423                .unwrap_or(false);
424            if widens {
425                continue;
426            }
427            self.scoped_sink_signatures.insert(k.clone(), v.clone());
428        }
429        for (k, v) in &other.scoped_sanitizer_facts {
430            self.scoped_sanitizer_facts.insert(k.clone(), v.clone());
431        }
432        self.scoped_sources
433            .extend(other.scoped_sources.iter().cloned());
434        // Provenance-ordered (Phase 5.3): learned checks accumulate with
435        // dedup on (rule, call), but the higher-ranked provenance wins the
436        // collision (Spec < Authored < Learned) and ties keep the existing
437        // entry, so merge is order-independent across provenance tiers.
438        for (c, prov) in &other.learned_checks {
439            upsert_by_provenance(&mut self.learned_checks, c.clone(), *prov, |a, b| {
440                a.rule == b.rule && a.call == b.call
441            });
442        }
443        // Policy facts order the same way, dedup on (rule, when_call).
444        for (p, prov) in &other.policy_facts {
445            upsert_by_provenance(&mut self.policy_facts, p.clone(), *prov, |a, b| {
446                a.rule == b.rule && a.when_call == b.when_call
447            });
448        }
449        // Memory contracts accumulate, with newer/learned contracts replacing older ones on collision.
450        for mc in &other.memory_contracts {
451            if let Some(existing) = self.memory_contracts.iter_mut().find(|c| c.name == mc.name) {
452                *existing = mc.clone();
453            } else {
454                self.memory_contracts.push(mc.clone());
455            }
456        }
457        for wc in &other.weak_crypto_rules {
458            if let Some(existing) = self
459                .weak_crypto_rules
460                .iter_mut()
461                .find(|r| r.rule_id == wc.rule_id && r.call == wc.call)
462            {
463                *existing = wc.clone();
464            } else {
465                self.weak_crypto_rules.push(wc.clone());
466            }
467        }
468        // Merged entries keep their own provenance; on collision the
469        // higher-ranked provenance wins (Spec < Authored < Learned), ties
470        // keep the existing entry - order-independent across tiers.
471        for (k, v) in &other.containment_callees {
472            if self.containment_callees.get(k).is_none_or(|cur| *v > *cur) {
473                self.containment_callees.insert(k.clone(), *v);
474            }
475        }
476        for (k, v) in &other.credential_sinks {
477            if self.credential_sinks.get(k).is_none_or(|cur| *v > *cur) {
478                self.credential_sinks.insert(k.clone(), *v);
479            }
480        }
481        for (k, v) in &other.credential_params {
482            if self.credential_params.get(k).is_none_or(|cur| *v > *cur) {
483                self.credential_params.insert(k.clone(), *v);
484            }
485        }
486        for (k, v) in &other.schema_builders {
487            if self.schema_builders.get(k).is_none_or(|cur| *v > *cur) {
488                self.schema_builders.insert(k.clone(), *v);
489            }
490        }
491        self.schema_enforcers
492            .extend(other.schema_enforcers.iter().cloned());
493        for (k, v) in &other.schema_keywords {
494            if self.schema_keywords.get(k).is_none_or(|cur| *v > *cur) {
495                self.schema_keywords.insert(k.clone(), *v);
496            }
497        }
498        self.url_param_hints
499            .extend(other.url_param_hints.iter().cloned());
500        self.url_arg_hints
501            .extend(other.url_arg_hints.iter().cloned());
502        self.url_literal_hints
503            .extend(other.url_literal_hints.iter().cloned());
504        self.security_context_hints
505            .extend(other.security_context_hints.iter().cloned());
506        self.auth_guard_hints
507            .extend(other.auth_guard_hints.iter().cloned());
508        self.jwt_algorithm_hints
509            .extend(other.jwt_algorithm_hints.iter().cloned());
510        self.credential_context_hints
511            .extend(other.credential_context_hints.iter().cloned());
512        self.kdf_receiver_hints
513            .extend(other.kdf_receiver_hints.iter().cloned());
514        self.prepared_statement_prefixes
515            .extend(other.prepared_statement_prefixes.iter().cloned());
516        self.weak_hash_disclaimer_tokens
517            .extend(other.weak_hash_disclaimer_tokens.iter().cloned());
518        // Same identity rules as `LearnedFactEntry::apply_with`: a later
519        // table overrides an earlier one on the rule's call coverage, so a
520        // consumer bundle can replace a learned/default conclusion without
521        // appending a duplicate the first-match loop never reaches.
522        for r in &other.weak_hash_rules {
523            if let Some(existing) = self.weak_hash_rules.iter_mut().find(|e| e.same_identity(r)) {
524                *existing = r.clone();
525            } else {
526                self.weak_hash_rules.push(r.clone());
527            }
528        }
529        // Provenance-ordered like the policy tables: a consumer bundle
530        // redefining a spec-seeded rule now overrides it (Learned >
531        // Spec), ties keep the existing entry.
532        for (r, p) in &other.integer_overflow_rules {
533            upsert_by_provenance(&mut self.integer_overflow_rules, r.clone(), *p, |a, b| {
534                a.rule_id == b.rule_id && a.wrap_threshold == b.wrap_threshold
535            });
536        }
537        for s in &other.insecure_config_selectors {
538            if let Some(existing) = self
539                .insecure_config_selectors
540                .iter_mut()
541                .find(|e| e.same_identity(s))
542            {
543                *existing = s.clone();
544            } else {
545                self.insecure_config_selectors.push(s.clone());
546            }
547        }
548        for r in &other.key_size_rules {
549            if let Some(existing) = self.key_size_rules.iter_mut().find(|e| e.same_identity(r)) {
550                existing.override_with(r);
551            } else {
552                self.key_size_rules.push(r.clone());
553            }
554        }
555        self.suspicious_hash_wrappers
556            .extend(other.suspicious_hash_wrappers.iter().cloned());
557
558        // Verb-sink bookkeeping accumulates too: dotted client entries
559        // (`got.get`, `axios.post`) from any merged spec/bundle widen the
560        // receiver-aware sets.
561        self.verb_sinks.extend(other.verb_sinks.iter().cloned());
562        self.client_roots.extend(other.client_roots.iter().cloned());
563        // Receiver-qualification bookkeeping accumulates like the other
564        // receiver-aware sets: two merged specs can declare different
565        // roots for the same method, and both must survive.
566        for (k, roots) in &other.receiver_qualified_sinks {
567            self.receiver_qualified_sinks
568                .entry(k.clone())
569                .or_default()
570                .extend(roots.iter().cloned());
571        }
572        // A bare declaration wins over a qualified one for the same name:
573        // the bare form is the receiver-agnostic one.
574        self.bare_sink_names
575            .extend(other.bare_sink_names.iter().cloned());
576        // Receiver-specific roles accumulate like the other receiver-aware
577        // sets: two merged specs can declare different dotted roots for the
578        // same method, and both disambiguation rules must survive.
579        for (k, v) in &other.receiver_roles {
580            self.receiver_roles.entry(k.clone()).or_insert(*v);
581        }
582        for (k, v) in &other.receiver_labels {
583            self.receiver_labels
584                .entry(k.clone())
585                .or_insert_with(|| v.clone());
586        }
587        self.session_roots
588            .extend(other.session_roots.iter().cloned());
589        self.learned_sources
590            .extend(other.learned_sources.iter().cloned());
591        for (k, v) in &other.grammar_roles {
592            self.grammar_roles.insert(k.clone(), v.clone());
593        }
594        for (lang, pats) in &other.route_patterns {
595            let slot = self.route_patterns.entry(lang.clone()).or_default();
596            for p in pats {
597                if !slot.contains(p) {
598                    slot.push(p.clone());
599                }
600            }
601        }
602        for (k, v) in &other.grammar_features {
603            self.grammar_features
604                .entry(k.clone())
605                .or_default()
606                .extend(v.iter().copied());
607        }
608        self.custom_allocators
609            .extend(other.custom_allocators.iter().cloned());
610        self.custom_deallocators
611            .extend(other.custom_deallocators.iter().cloned());
612        if !other.memory_functions.is_empty() {
613            self.memory_functions = other.memory_functions.clone();
614        }
615        if !other.buffer_builtins.is_empty() {
616            self.buffer_builtins = other.buffer_builtins.clone();
617        }
618        if !other.stack_allocators.is_empty() {
619            self.stack_allocators = other.stack_allocators.clone();
620        }
621        self.collection_constructors
622            .extend(other.collection_constructors.iter().cloned());
623        self.schema_describe_methods
624            .extend(other.schema_describe_methods.iter().cloned());
625        self.null_tokens.extend(other.null_tokens.iter().cloned());
626        self.session_accessors
627            .extend(other.session_accessors.iter().cloned());
628        self.receiver_params
629            .extend(other.receiver_params.iter().cloned());
630        self.idor_keys.extend(other.idor_keys.iter().cloned());
631        self.idor_finder_sinks
632            .extend(other.idor_finder_sinks.iter().cloned());
633        for (k, v) in &other.propagators {
634            self.propagators.insert(k.clone(), v.clone());
635        }
636        for (k, v) in &other.propagator_blocks_receiver {
637            self.propagator_blocks_receiver.insert(k.clone(), *v);
638        }
639        for p in &other.guard_denylist_patterns {
640            if !self.guard_denylist_patterns.contains(p) {
641                self.guard_denylist_patterns.push(p.clone());
642            }
643        }
644        if other.max_complexity.is_some() {
645            self.max_complexity = other.max_complexity;
646        }
647    }

649    /// Look up a dynamic AST node classification role learned from a `.frc` bundle.
650    pub fn get_grammar_role(
651        &self,
652        language: &str,
653        node_kind: &str,
654    ) -> Option<frensense_lang::NodeRole> {
655        let lang = language.to_lowercase();
656        self.grammar_roles
657            .get(&(lang, node_kind.to_string()))
658            .or_else(|| {
659                self.grammar_roles
660                    .get(&("*".to_string(), node_kind.to_string()))
661            })
662            .map(|r| r.to_node_role())
663    }

665    /// Check if a dynamic AST grammar feature is learned from a `.frc` bundle.
666    pub fn has_grammar_feature(
667        &self,
668        language: &str,
669        node_kind: &str,
670        feature: GrammarFeature,
671    ) -> Option<bool> {
672        let lang = language.to_lowercase();
673        if let Some(set) = self.grammar_features.get(&(lang, node_kind.to_string()))
674            && set.contains(&feature)
675        {
676            return Some(true);
677        }
678        if let Some(set) = self
679            .grammar_features
680            .get(&("*".to_string(), node_kind.to_string()))
681            && set.contains(&feature)
682        {
683            return Some(true);
684        }
685        None
686    }

792    /// Does a spec rule declare that `call`'s receiver does not taint its
793    /// return (`tainted_receiver: false`)? Keyed like `propagators`: full
794    /// name first, then last segment; a key with no declaration defaults to
795    /// pass-through (not blocked).
796    pub fn propagator_blocks_receiver(&self, call: &str) -> bool {
797        if let Some(&blocked) = self.propagator_blocks_receiver.get(call) {
798            return blocked;
799        }
800        let last = call.rsplit('.').next().unwrap_or(call);
801        last != call
802            && self
803                .propagator_blocks_receiver
804                .get(last)
805                .copied()
806                .unwrap_or(false)
807    }
