5//! Taint-path representation (R3c).
6//!
7//! A taint path is the ordered chain of SVFG nodes the
8//! [`super::engine::BackwardTaintEngine`] walked from a
9//! sink argument back to a source. It is intentionally *data*: the engine only
10//! records it; rendering (text/SARIF/LSP) lives in the consumer (see the
11//! root crate's `reporter` module). Keeping capture and rendering apart means
12//! new output formats never touch the engine.
13
14use rustc_hash::FxHashMap;
15
16use super::engine::BackwardTaintEngine;
17use crate::graph::svfg::{NodeKey, NodeKind};
18use crate::ir::function::{FunctionIR, Instruction};

20// ---------------------------------------------------------------------------
21// Steps
22// ---------------------------------------------------------------------------
23
24/// One step on a taint path, in **source → sink** order (the backward engine
25/// walk runs the other way; the engine's private path reconstruction follows
26/// parent links from the source toward the sink and the private constructor keeps
27/// that order).
28#[derive(Debug, Clone, PartialEq, Eq)]
29pub enum PathStep {
30    /// The origin: a configured source (access path or call name).
31    Source {
32        /// Description of the source: an access path or call name.
33        description: String,
34    },
35    /// A `phi` merging several incoming values.
36    Phi {
37        /// Function containing the phi.
38        function: String,
39        /// Number of the variable the phi defines.
40        variable: u32,
41    },
42    /// A load from an object field (`x = obj.f`).
43    FieldLoad {
44        /// Function containing the load.
45        function: String,
46        /// Name of the object the field is read from.
47        base: String,
48        /// Name of the field.
49        field: String,
50    },
51    /// A value returned by a call (interprocedural hop into the callee).
52    CallReturn {
53        /// Function containing the call.
54        function: String,
55        /// Name of the called function.
56        callee: String,
57        /// `true` when the callee could not be resolved in-program.
58        external: bool,
59    },
60    /// A value passed as an argument at a call site (hop into the callee).
61    CallArgument {
62        /// Function containing the call.
63        function: String,
64        /// Name of the called function.
65        callee: String,
66        /// Argument slot the value is passed in.
67        slot: usize,
68    },
69    /// A value bound to a callee's formal parameter (callee side of the hop).
70    FormalParam {
71        /// Callee the parameter belongs to.
72        function: String,
73        /// Parameter name.
74        param: String,
75    },
76    /// A value returned from a callee back to its call site (callee→caller).
77    ReturnToCaller {
78        /// Function the value returns to.
79        caller: String,
80    },
81    /// Any other def→use propagation inside one function.
82    Assignment {
83        /// Function containing the step.
84        function: String,
85        /// Number of the variable defined.
86        variable: u32,
87    },
88}
89
90impl PathStep {
91    /// Function this step lives in (source steps have no function).
92    pub fn function(&self) -> Option<&str> {
93        match self {
94            PathStep::Source { .. } => None,
95            PathStep::ReturnToCaller { .. } => None,
96            PathStep::Phi { function, .. }
97            | PathStep::FieldLoad { function, .. }
98            | PathStep::CallReturn { function, .. }
99            | PathStep::CallArgument { function, .. }
100            | PathStep::FormalParam { function, .. }
101            | PathStep::Assignment { function, .. } => Some(function),
102        }
103    }
104}

110/// The full source→sink chain for one vulnerable finding.
111#[derive(Debug, Clone, Default)]
112pub struct TaintPath {
113    /// Steps in source→sink order. Empty for non-vulnerable findings.
114    pub steps: Vec<PathStep>,
115    /// (function, byte-range) of each step, aligned with `steps` where the
116    /// lowering recorded a span. Sources have no function, so their entry is
117    /// `None`.
118    pub spans: Vec<Option<(String, (usize, usize))>>,
119}
120
121impl TaintPath {
122    /// Build a path from the walked chain. `chain` yields
123    /// `(function_ir, node, step)` in source→sink order (see
124    /// [`BackwardTaintEngine::reconstruct_path`]); the leading
125    /// [`PathStep::Source`] must already carry its description.
126    /// `fn_file` maps function names to their source files so step spans
127    /// point at real files (unmapped functions, engine-internal tests
128    /// without files, keep the function name as a pseudo-file).
129    ///
130    /// The chain borrows each function's IR instead of taking a clone: a
131    /// whole-program walk can carry tens of thousands of steps, and cloning
132    /// a `FunctionIR` per step (blocks, var metadata, string tables) turned
133    /// one path reconstruction into a multi-gigabyte allocation - the redis
134    /// directory scan died at 5.6 GiB here. The IR is only read for the
135    /// step's byte span, which is computed immediately and owned afterwards.
136    ///
137    /// ORDERING CONTRACT: steps are stored source → sink. The sink itself is
138    /// not a step (the finding already reports it), so the last step is the
139    /// propagation that feeds the sink argument.
140    pub(crate) fn new(
141        chain: Vec<(&FunctionIR, NodeKey, PathStep)>,
142        source_desc: Option<String>,
143        source_span: Option<(String, (usize, usize))>,
144        fn_file: &FxHashMap<String, String>,
145    ) -> Self {
146        let mut steps = Vec::with_capacity(chain.len() + 1);
147        let mut spans = Vec::with_capacity(chain.len() + 1);
148        steps.push(PathStep::Source {
149            description: source_desc.unwrap_or_else(|| "unknown origin".into()),
150        });
151        // The origin carries a location like every other step: a flow whose
152        // first hop is unlocatable cannot be opened in a SARIF viewer, and
153        // clients render step 1 as a bare string.
154        spans.push(source_span);
155        // The chain is already source→sink; keep it (reversing it here used
156        // to render flows sink→source, the "step 2 is the sink" bug).
157        for (ir, key, step) in chain.iter() {
158            let file = fn_file
159                .get(&ir.name)
160                .cloned()
161                .unwrap_or_else(|| ir.name.clone());
162            let span = span_of(ir, key).map(|r| (file, r));
163            steps.push(step.clone());
164            spans.push(span);
165        }
166        Self { steps, spans }
167    }
168
169    /// Stable identity of the path's *shape*: the sequence of step kinds with
170    /// function names, but no variable numbers. Two findings whose paths
171    /// differ only in SSA numbering dedup to the same shape.
172    /// Presentation-friendly compaction of the raw walked path.
173    ///
174    /// The backward walk records every SVFG node it stepped through, which
175    /// floods human-facing output with noise:
176    ///
177    /// 1. *Adjacent duplicate steps*: the walk visits both the def node and
178    ///    the use node of one instruction, producing two identical entries
179    ///    back to back;
180    /// 2. *Runs of `Assignment` hops*: SSA def→use renames inside one
181    ///    function (`flows through vNNN`) carry no user-visible content;
182    ///    value-shaping events (loads, phis, calls, interprocedural
183    ///    boundaries) are never `Assignment` steps.
184    ///
185    /// `compacted()` collapses both, keeping spans aligned: a collapsed step
186    /// keeps the first span of the collapsed run. Raw paths stay available
187    /// for shape dedup (`shape_id`) and machine consumers; call this only
188    /// where a path is rendered for humans.
189    #[must_use]
190    pub fn compacted(&self) -> TaintPath {
191        let mut steps: Vec<PathStep> = Vec::with_capacity(self.steps.len());
192        let mut spans: Vec<Option<(String, (usize, usize))>> = Vec::with_capacity(self.spans.len());
193        for (step, span) in self.steps.iter().zip(self.spans.iter()) {
194            // (1) identical neighbor (same kind, function, variable, the
195            // def/use pair of one instruction).
196            if steps.last().is_some_and(|prev| prev == step) {
197                continue;
198            }
199            // (2) collapse consecutive plain assignments inside one
200            // function to their first hop.
201            let merged = matches!(
202                (steps.last(), step),
203                (
204                    Some(PathStep::Assignment { function: prev_fn, .. }),
205                    PathStep::Assignment { function: cur_fn, .. }
206                ) if prev_fn == cur_fn,
207            );
208            if merged {
209                continue;
210            }
211            steps.push(step.clone());
212            spans.push(span.clone());
213        }
214        TaintPath { steps, spans }
215    }
216
217    /// Stable identity of the path's *shape*: the sequence of step kinds with
218    /// function names, but no variable numbers. Two findings whose paths
219    /// differ only in SSA numbering dedup to the same shape.
220    pub fn shape_id(&self) -> String {
221        use std::hash::{Hash, Hasher};
222        let mut h = std::collections::hash_map::DefaultHasher::new();
223        for s in &self.steps {
224            match s {
225                PathStep::Source { description: _ } => "src".hash(&mut h),
226                PathStep::Phi { .. } => "phi".hash(&mut h),
227                PathStep::FieldLoad {
228                    function, field, ..
229                } => format!("fld:{function}.{field}").hash(&mut h),
230                PathStep::CallReturn {
231                    function, callee, ..
232                } => format!("ret:{function}>{callee}").hash(&mut h),
233                PathStep::CallArgument {
234                    function,
235                    callee,
236                    slot,
237                } => format!("arg:{function}>{callee}#{slot}").hash(&mut h),
238                PathStep::FormalParam { function, .. } => format!("par:{function}").hash(&mut h),
239                PathStep::ReturnToCaller { caller } => format!("back:{caller}").hash(&mut h),
240                PathStep::Assignment { .. } => "asn".hash(&mut h),
241            }
242        }
243        format!("{:016x}", h.finish())
244    }
245}

247/// Dedup helper: keep one representative per shape, preferring the finding
248/// with the most steps (most informative path). Returns indices into `paths`.
249#[cfg(test)]
250pub(crate) fn dedup_by_shape(paths: &[TaintPath]) -> Vec<usize> {
251    let mut best: FxHashMap<String, usize> = FxHashMap::default();
252    for (i, p) in paths.iter().enumerate() {
253        match best.get(&p.shape_id()) {
254            Some(&j) if paths[j].steps.len() >= p.steps.len() => {}
255            _ => {
256                best.insert(p.shape_id(), i);
257            }
258        }
259    }
260    let mut kept: Vec<usize> = best.into_values().collect();
261    kept.sort_unstable();
262    kept
263}

265/// Location of a node: the instruction's span, or the declaring site of its
266/// value. Parameter sentinels carry no instruction, so without the fallback
267/// an origin that is a function parameter has no location at all and SARIF
268/// renders its first step as `location: null`.
269fn span_with_decl(ir: &FunctionIR, key: &NodeKey) -> Option<(usize, usize)> {
270    span_of(ir, key).or_else(|| ir.var_metadata.get(&key.var).and_then(|m| m.byte_range))
271}
272
273/// Byte range of the instruction (or param sentinel) a node points at.
274pub(crate) fn span_of(ir: &FunctionIR, key: &NodeKey) -> Option<(usize, usize)> {
275    let idx = key.instr_idx?;
276    let b = ir.blocks.get(&key.block)?;
277    let instr = b.instructions.get(idx)?;
278    match instr {
279        Instruction::CallStatic { dest, .. }
280        | Instruction::CallVirtual { dest, .. }
281        | Instruction::CallPointer { dest, .. } => dest
282            .and_then(|d| ir.var_metadata.get(&d))
283            .and_then(|m| m.byte_range)
284            .or_else(|| ir.var_metadata.get(&key.var).and_then(|m| m.byte_range)),
285        _ => ir.var_metadata.get(&key.var).and_then(|m| m.byte_range),
286    }
287}

289// ---------------------------------------------------------------------------
290// Reconstruction of a walked source->sink chain (engine impl lives here so
291// rendering stays next to the path types).
292// ---------------------------------------------------------------------------
293
294impl<'a> BackwardTaintEngine<'a> {
295    /// Classify one walked node as a [`PathStep`] for path reporting.
296    fn step_of(&self, cf: usize, key: &NodeKey) -> PathStep {
297        let fe = &self.prog.functions[cf];
298        let ir = fe.ir;
299        let fname = ir.name.clone();
300        let Some(node) = fe.svfg.node(key) else {
301            return PathStep::Assignment {
302                function: fname,
303                variable: key.var.0 as u32,
304            };
305        };
306        match &node.kind {
307            NodeKind::Phi => PathStep::Phi {
308                function: fname,
309                variable: key.var.0 as u32,
310            },
311            NodeKind::FormalParam => {
312                let name = ir
313                    .var_metadata
314                    .get(&key.var)
315                    .and_then(|m| m.source_name.clone())
316                    .unwrap_or_else(|| format!("v{}", key.var.0));
317                PathStep::FormalParam {
318                    function: fname,
319                    param: name,
320                }
321            }
322            NodeKind::ActualArg {
323                call_site,
324                arg_index,
325            } => PathStep::CallArgument {
326                function: fname,
327                callee: self.callee_display(ir, call_site),
328                slot: *arg_index,
329            },
330            NodeKind::ActualRet { call_site } => {
331                let callee = self.callee_display(ir, call_site);
332                let external = !self.prog.functions.iter().any(|f| f.ir.name == callee);
333                PathStep::CallReturn {
334                    function: fname,
335                    callee,
336                    external,
337                }
338            }
339            NodeKind::FormalRet => PathStep::ReturnToCaller { caller: fname },
340            NodeKind::InstrDef | NodeKind::InstrUse => self.instr_step(ir, key, fname),
341        }
342    }
343
344    /// Display name of the callee at a call-site node key.
345    fn callee_display(&self, ir: &FunctionIR, call_site: &NodeKey) -> String {
346        let Some(idx) = call_site.instr_idx else {
347            return "?".into();
348        };
349        match ir
350            .blocks
351            .get(&call_site.block)
352            .and_then(|b| b.instructions.get(idx))
353        {
354            Some(Instruction::CallStatic { func, .. }) => func.clone(),
355            Some(Instruction::CallVirtual { method, .. }) => method.clone(),
356            _ => "?".into(),
357        }
358    }
359
360    /// Instruction-level step: field loads become `FieldLoad`, everything
361    /// else a generic `Assignment` (variable-level flow).
362    fn instr_step(&self, ir: &FunctionIR, key: &NodeKey, fname: String) -> PathStep {
363        let Some(idx) = key.instr_idx else {
364            return PathStep::Assignment {
365                function: fname,
366                variable: key.var.0 as u32,
367            };
368        };
369        let Some(b) = ir.blocks.get(&key.block) else {
370            return PathStep::Assignment {
371                function: fname,
372                variable: key.var.0 as u32,
373            };
374        };
375        match b.instructions.get(idx) {
376            Some(Instruction::LoadField { base, field, .. }) => {
377                let base_name = ir
378                    .var_metadata
379                    .get(base)
380                    .and_then(|m| m.source_name.clone())
381                    .unwrap_or_else(|| format!("v{}", base.0));
382                PathStep::FieldLoad {
383                    function: fname,
384                    base: base_name,
385                    field: field.clone(),
386                }
387            }
388            _ => PathStep::Assignment {
389                function: fname,
390                variable: key.var.0 as u32,
391            },
392        }
393    }
394
395    /// Walk the parent map recorded during the backward BFS and build the
396    /// source→sink [`TaintPath`]. Each visited node's parent is the node one
397    /// step *closer to the sink* (the node it was discovered from), so
398    /// following parents from the source walks toward the sink in source→
399    /// sink order. The sink argument node itself is not a step: the finding
400    /// already reports it.
401    pub(super) fn reconstruct_path(
402        &self,
403        source_node: Option<(usize, NodeKey)>,
404        source_desc: Option<String>,
405    ) -> TaintPath {
406        let mut chain: Vec<(&FunctionIR, NodeKey, PathStep)> = Vec::new();
407        // The parent map maps each visited node → its BFS parent (the node
408        // one step closer to the sink). The chain therefore runs from the
409        // source (which has no parent entry, BFS stopped there) to the node
410        // just before the sink argument. The sink node itself is not a step:
411        // the finding already reports it.
412        let Some((sf, skey)) = source_node else {
413            return TaintPath::new(chain, source_desc, None, &self.fn_file);
414        };
415        let source_span = {
416            let ir = self.prog.functions[sf].ir;
417            span_with_decl(ir, &skey).map(|r| {
418                let file = self
419                    .fn_file
420                    .get(&ir.name)
421                    .cloned()
422                    .unwrap_or_else(|| ir.name.clone());
423                (file, r)
424            })
425        };
426        let mut cur = (sf, skey);
427        for _ in 0..self.current_parents.len() + 1 {
428            let (cf, key) = cur;
429            // Borrow, never clone: the IR is only read for the step's span
430            // inside `TaintPath::new`, and a clone per step on a deep path
431            // is what made whole-directory C scans run out of memory.
432            let ir = self.prog.functions[cf].ir;
433            let step = self.step_of(cf, &key);
434            chain.push((ir, key, step));
435            match self.current_parents.get(&cur) {
436                Some(&parent) => cur = parent,
437                None => break,
438            }
439        }
440        // The leading walk nodes ARE the origin: the chain starts at the
441        // source node, and its def/use pair each restates step 1 at the same
442        // byte range. The Source step now carries that location, so drop the
443        // echoes instead of printing the origin two or three times.
444        let origin = source_span.as_ref().map(|(_, r)| *r);
445        while let Some((ir, key, _)) = chain.first() {
446            if span_with_decl(ir, key) != origin {
447                break;
448            }
449            chain.remove(0);
450        }
451        TaintPath::new(chain, source_desc, source_span, &self.fn_file)
452    }
453}
