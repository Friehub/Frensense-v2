20/// Constant-fold a variable's definition chain to an integer value.
21///
22/// Handles the literal-arithmetic shapes that decide ternary/if branches:
23/// `Assign` from an `IntLiteral`, and `BinaryOp` arithmetic over literal
24/// operands (`7 * 18 + num > 200` folds bottom-up). Bounded hops keep
25/// pathological chains cheap. `None` = not a constant.
26fn const_eval(ir: &FunctionIR, var: VarId) -> Option<i64> {
27    fn operand_const(ir: &FunctionIR, op: &Operand, hops: u8) -> Option<i64> {
28        match op {
29            Operand::IntLiteral(i) => Some(*i),
30            Operand::Var(v) => const_eval_hops(ir, *v, hops),
31            _ => None,
32        }
33    }
34    fn const_eval_hops(ir: &FunctionIR, var: VarId, hops: u8) -> Option<i64> {
35        if hops == 0 {
36            return None;
37        }
38        let def = defs_of_var(ir, var)?;
39        match def {
40            Instruction::Assign {
41                src: Operand::IntLiteral(i),
42                ..
43            } => Some(*i),
44            Instruction::BinaryOp { op, lhs, rhs, .. } => {
45                let l = operand_const(ir, lhs, hops - 1)?;
46                let r = operand_const(ir, rhs, hops - 1)?;
47                // The lowering maps `+` to op "concat" (the generic
48                // additive fold); arithmetic on IntLiterals is still
49                // integer addition there.
50                Some(match op.as_str() {
51                    "concat" | "+" => l.wrapping_add(r),
52                    "-" => l.wrapping_sub(r),
53                    "*" => l.wrapping_mul(r),
54                    "/" => {
55                        if r == 0 {
56                            return None;
57                        }
58                        l.wrapping_div(r)
59                    }
60                    "%" => {
61                        if r == 0 {
62                            return None;
63                        }
64                        l.wrapping_rem(r)
65                    }
66                    _ => return None,
67                })
68            }
69            _ => None,
70        }
71    }
72    const_eval_hops(ir, var, 8)
73}

75/// Find the single instruction defining `var` (bounded scan).
76fn defs_of_var(ir: &FunctionIR, var: VarId) -> Option<&Instruction> {
77    for b in ir.blocks.values() {
78        for i in &b.instructions {
79            let dests: Vec<VarId> = match i {
80                Instruction::Assign { dest, .. }
81                | Instruction::BinaryOp { dest, .. }
82                | Instruction::UnaryOp { dest, .. }
83                | Instruction::LoadField { dest, .. }
84                | Instruction::LoadElement { dest, .. }
85                | Instruction::LoadGlobal { dest, .. }
86                | Instruction::Cast { dest, .. } => vec![*dest],
87                Instruction::CallStatic { dest: Some(d), .. }
88                | Instruction::CallVirtual { dest: Some(d), .. } => vec![*d],
89                _ => Vec::new(),
90            };
91            if dests.contains(&var) {
92                return Some(i);
93            }
94        }
95    }
96    None
97}

99/// Evaluate a comparison over two constant operands.
100fn compare_consts(op: &str, l: i64, r: i64) -> Option<bool> {
101    Some(match op {
102        ">" => l > r,
103        "<" => l < r,
104        ">=" => l >= r,
105        "<=" => l <= r,
106        "==" => l == r,
107        "!=" => l != r,
108        _ => return None,
109    })
110}

112/// Record a silent stop site on the current root's exploration, keeping
113/// the list free of duplicates (a walk can hit the same cut on many nodes).
114fn record_cut(cuts: &mut Vec<&'static str>, reason: &'static str) {
115    if !cuts.contains(&reason) {
116        cuts.push(reason);
117    }
118}

120/// Most stop sites kept per sink argument: enough to name the blind spot,
121/// bounded so a wide fan-in cannot bloat every finding.
122const MAX_CUT_SITES: usize = 4;
123
124/// [`CutSite::kind`] for a call with no analysed callee that the walk flowed
125/// *through* (its arguments were followed, assuming the result carries them).
126const PASSED_THROUGH: &str = "passed-through";
127
128/// Standard-library adapters, sorted: they are defined to forward their
129/// operands into the result, so naming one as a pass-through tells a reader
130/// nothing. Keeping them out leaves the list to calls the engine genuinely
131/// has no model for (`mystery.read(req.params)`), which is the point of it.
132/// Detection never consults this: it only filters the explanation.
133const STD_FORWARDERS: &[&str] = &[
134    "and_then",
135    "any",
136    "as_deref",
137    "as_mut",
138    "as_ref",
139    "as_str",
140    "clone",
141    "collect",
142    "filter",
143    "filter_map",
144    "flat_map",
145    "fold",
146    "into",
147    "into_iter",
148    "iter",
149    "iter_mut",
150    "keys",
151    "map",
152    "map_err",
153    "ok_or",
154    "ok_or_else",
155    "or",
156    "or_else",
157    "take",
158    "then",
159    "then_some",
160    "unwrap_or",
161    "unwrap_or_default",
162    "unwrap_or_else",
163    "values",
164];
165
166/// Whether `callee` is a standard-library adapter (see [`STD_FORWARDERS`]).
167fn is_std_forwarder(callee: &str) -> bool {
168    STD_FORWARDERS.binary_search(&callee).is_ok()
169}

171/// Keep stop sites distinct and bounded.
172fn record_cut_site(sites: &mut Vec<CutSite>, site: CutSite) {
173    // Pass-throughs are context, not stops: they get their own budget so a
174    // long chain of them cannot crowd out the stop that made the verdict.
175    let same_class = |s: &&CutSite| (s.kind == PASSED_THROUGH) == (site.kind == PASSED_THROUGH);
176    if sites.iter().filter(same_class).count() < MAX_CUT_SITES
177        && !sites
178            .iter()
179            .any(|s| s.kind == site.kind && s.function == site.function && s.detail == site.detail)
180    {
181        sites.push(site);
182    }
183}

185/// For the Phi `dest_var` in `merge_block`, return the incoming edges whose
186/// arm block is unreachable because the branch condition is a constant
187/// comparison with a decided outcome. Empty = nothing prunable.
188fn phi_infeasible_incoming(
189    ir: &FunctionIR,
190    merge_block: BlockId,
191    dest_var: VarId,
192) -> Vec<(BlockId, VarId)> {
193    let block = match ir.blocks.get(&merge_block) {
194        Some(b) => b,
195        None => return Vec::new(),
196    };
197    let Some(phi) = block.phis.iter().find(|p| p.dest == dest_var) else {
198        return Vec::new();
199    };
200    // An incoming (pred_block, var) is infeasible when pred_block is
201    // reached by a Branch whose condition is a decided constant and the
202    // edge taken is the OTHER side.
203    let mut out = Vec::new();
204    for &(pred_block, inc_var) in &phi.incoming {
205        // Find the branch that targets pred_block.
206        for b in ir.blocks.values() {
207            let (cond, true_block, false_block) = match &b.terminator {
208                Terminator::Branch {
209                    cond: Operand::Var(c),
210                    true_block,
211                    false_block,
212                } => (*c, *true_block, *false_block),
213                _ => continue,
214            };
215            if true_block != pred_block && false_block != pred_block {
216                continue;
217            }
218            let taken = match defs_of_var(ir, cond) {
219                Some(Instruction::BinaryOp { op, lhs, rhs, .. }) => {
220                    let l = match lhs {
221                        Operand::IntLiteral(i) => Some(*i),
222                        Operand::Var(v) => const_eval(ir, *v),
223                        _ => None,
224                    };
225                    let r = match rhs {
226                        Operand::IntLiteral(i) => Some(*i),
227                        Operand::Var(v) => const_eval(ir, *v),
228                        _ => None,
229                    };
230                    match (l, r) {
231                        (Some(l), Some(r)) => compare_consts(op, l, r),
232                        _ => None,
233                    }
234                }
235                Some(Instruction::Assign {
236                    src: Operand::IntLiteral(i),
237                    ..
238                }) => Some(*i != 0),
239                _ => None,
240            };
241            let Some(taken) = taken else {
242                continue;
243            };
244            let feasible_block = if taken { true_block } else { false_block };
245            if feasible_block != pred_block {
246                out.push((pred_block, inc_var));
247            }
248            break;
249        }
250    }
251    out
252}

254/// What the engine knows about one sink site before exploring its arguments.
255#[derive(Clone, Copy)]
256struct SinkSite<'a> {
257    /// Sink name as configured or declared (`innerHTML`, `send`, `write`).
258    name: &'a str,
259    /// Role the finding is ranked under.
260    role: crate::analysis::taint::role::SinkRole,
261    /// Human-readable label from the vocabulary, when one is declared.
262    label: Option<&'a str>,
263    /// Package the sink signature was scoped to, when known.
264    origin_package: Option<&'a str>,
265    /// Language of the site, for language-keyed sink facts.
266    language: Option<&'a str>,
267}

291#[cfg(test)]
292mod forwarder_tests {
293    use super::{STD_FORWARDERS, is_std_forwarder};
294
295    /// `is_std_forwarder` binary-searches [`STD_FORWARDERS`], so an entry
296    /// inserted out of order (or duplicated) is not a compile error - the
297    /// lookup just stops finding it, and std adapters creep back into the
298    /// pass-through list. `windows(2)` with a strict `<` fails on both.
299    #[test]
300    fn the_std_forwarder_list_is_sorted_and_unique() {
301        for pair in STD_FORWARDERS.windows(2) {
302            assert!(
303                pair[0] < pair[1],
304                "`{}` must sort before `{}`; binary_search silently stops \
305                 filtering once the list is out of order",
306                pair[0],
307                pair[1]
308            );
309        }
310    }
311
312    /// The list is the whole answer: every entry is found, and a call the
313    /// engine has no model for is not treated as an adapter.
314    #[test]
315    fn every_entry_is_found_and_unknown_calls_are_not() {
316        for name in STD_FORWARDERS {
317            assert!(is_std_forwarder(name), "`{name}` is in the list");
318        }
319        assert!(is_std_forwarder("iter"));
320        assert!(!is_std_forwarder("mystery.read"));
321        assert!(!is_std_forwarder(""));
322    }
