5use super::*;
6
7impl<'a> BackwardTaintEngine<'a> {
8    /// The `(caller_fn, call_site)` through which the function under walk was
9    /// entered, or `None` for the empty (k=0) context.
10    pub(super) fn ctx_last(&self, ctx: u32) -> Option<(usize, NodeKey)> {
11        if ctx == 0 {
12            return None;
13        }
14        let link = self.ctx_chains.get(ctx as usize)?;
15        Some((link.caller_fn, link.site))
16    }
17
18    /// Push a call-site entry when the walk enters a callee. Returns 0
19    /// (k=0 collapse) past the depth/count caps - sound, just unfiltered.
20    pub(super) fn ctx_enter(&mut self, parent: u32, caller_fn: usize, site: NodeKey) -> u32 {
21        let key = (parent, caller_fn, site);
22        if let Some(&id) = self.ctx_interner.get(&key) {
23            return id;
24        }
25        let mut depth = 0usize;
26        let mut p = parent;
27        while p != 0 && depth < MAX_CTX_DEPTH {
28            p = self.ctx_chains[p as usize].parent;
29            depth += 1;
30        }
31        if p != 0 {
32            // Depth ceiling: this walk's call-site stack is deeper than the
33            // model tracks, so it becomes k=0 from here.
34            crate::telemetry::Cap::CallContextDepth.record();
35            return 0;
36        }
37        if self.ctx_chains.len() >= MAX_CTX_CHAINS {
38            crate::telemetry::Cap::CallContextChains.record();
39            return 0;
40        }
41        let id = self.ctx_chains.len() as u32;
42        self.ctx_chains.push(CtxLink {
43            parent,
44            caller_fn,
45            site,
46        });
47        self.ctx_interner.insert(key, id);
48        id
49    }
50
51    /// Pop the innermost call-site entry (unwinding out of a callee).
52    pub(super) fn ctx_exit(&self, ctx: u32) -> u32 {
53        if ctx == 0 {
54            0
55        } else {
56            self.ctx_chains[ctx as usize].parent
57        }
58    }
59}
