3//! The backward walk itself: node-by-node exploration from a sink
4//! argument to a source, with its cut sites and stop reasons.
5//!
6//! Split out of `walk.rs`. One step of the walk is one call: `walk_stops`
7//! decides whether the node ends the branch, `gather_predecessors` collects
8//! its edges and `enqueue_predecessors` pushes them with their context.
9
10use super::*;
11
12/// One step of the backward walk: the node being explored plus the state the
13/// k=1 context stack and the field-demand mode carry for it.
14#[derive(Clone, Copy)]
15struct WalkStep<'a> {
16    /// Function the node lives in.
17    cf: usize,
18    /// IR of that function, borrowed for the whole walk.
19    ir: &'a FunctionIR,
20    /// The node being explored.
21    key: NodeKey,
22    /// Block whose guard map decides whether the value is used under a guard.
23    use_block: BlockId,
24    /// Whether the walk seeks one field/element of a container.
25    field_demand: bool,
26    /// k=1 call-site context encoded by the engine's context stack.
27    ctx: u32,
28    /// Call site the walk enters when this node is a call destination.
29    enter_call: Option<(usize, NodeKey)>,
30    /// Whether this node is a formal parameter (context filters its preds).
31    is_formal_param: bool,
32}
33
34/// A queued walk node: function, node, use block, field demand, context.
35pub(super) type WalkNode = (usize, NodeKey, BlockId, bool, u32);
36
37/// Local and cross-function predecessors of one walk step.
38type Preds = (Vec<NodeKey>, Vec<(usize, NodeKey)>);
39
40impl<'a> BackwardTaintEngine<'a> {
41    /// Does the root argument's own definition read one specific
42    /// field/element? Such walks carry *field demand*: the value they seek
43    /// can only come from field-matching stores, never from the
44    /// whole-container fill edges of an allocation.
45    pub(super) fn root_reads_field(&self, fi: usize, root: &NodeKey) -> bool {
46        let ir = self.prog.functions[fi].ir;
47        for b in ir.blocks.values() {
48            for phi in &b.phis {
49                if phi.dest == root.var {
50                    return false;
51                }
52            }
53            for ins in &b.instructions {
54                if ins.dest() == Some(root.var) {
55                    return matches!(
56                        ins,
57                        Instruction::LoadField { .. } | Instruction::LoadElement { .. }
58                    );
59                }
60            }
61        }
62        false
63    }
64
65    /// Is `cur` the defining node of an `Allocate` instruction?
66    pub(super) fn def_is_allocation(&self, fi: usize, cur: &NodeKey) -> bool {
67        let NodeKey {
68            block,
69            instr_idx: Some(i),
70            var,
71            ..
72        } = *cur
73        else {
74            return false;
75        };
76        self.prog.functions[fi]
77            .ir
78            .blocks
79            .get(&block)
80            .and_then(|b| b.instructions.get(i))
81            .is_some_and(|ins| matches!(ins, Instruction::Allocate { dest, .. } if *dest == var))
82    }

84    pub(super) fn explore_from(&mut self, fi: usize, root: NodeKey, state: &mut ExploreState) {
85        // Field demand at the root: the sink argument's own definition reads
86        // one specific field/element, so whole-container fill edges stay off
87        // limits for this whole walk (see the allocation stop below).
88        let root_field = self.root_reads_field(fi, &root);
89        let root_ctx: u32 = 0;
90        state
91            .visited
92            .insert((fi, root, root.block, root_field, root_ctx));
93        let mut queue: VecDeque<WalkNode> = VecDeque::new();
94        queue.push_back((fi, root, root.block, root_field, root_ctx));
95
96        let dbg = DebugFlags::get();
97        while let Some((cf, cur, use_block, field_demand, ctx)) = queue.pop_front() {
98            let (is_formal_param, enter_call) = self.classify_node(cf, &cur);
99            dbg_trace!(
100                dbg.walk,
101                "[walk] cf={cf} key={cur:?} kind={:?}",
102                self.prog.functions[cf].svfg.node(&cur).map(|n| &n.kind)
103            );
104            let step = WalkStep {
105                cf,
106                ir: self.prog.functions[cf].ir,
107                key: cur,
108                use_block,
109                field_demand,
110                ctx,
111                enter_call,
112                is_formal_param,
113            };
114
115            if self.walk_stops(fi, &step, root.block, state) {
116                continue;
117            }
118
119            let (local_preds, cross_preds) = self.gather_predecessors(&step);
120
121            if local_preds.is_empty() && cross_preds.is_empty() {
122                dbg_trace!(
123                    dbg.deadend,
124                    "[deadend] cf={cf} key={cur:?} kind={:?}",
125                    self.prog.functions[cf].svfg.node(&cur).map(|n| &n.kind)
126                );
127                // Dead end. Classify: unresolvable roots (formal params never
128                // fed by an analysed call site, or environment-defined values
129                // like LoadGlobal / Dereference / CallPointer) are "unknown",
130                // we can't prove them clean. Everything else (literals,
131                // allocations) is a genuinely clean root.
132                if self.node_is_unresolvable(cf, &cur) {
133                    state.saw_unknown = true;
134                    record_cut(&mut state.cuts, "unresolvable");
135                    if let Some(site) = self.cut_site(cf, &cur) {
136                        record_cut_site(&mut state.cut_sites, site);
137                    }
138                }
139                // Otherwise: clean root, stop.
140                continue;
141            }
142
143            // An external call whose result the walk follows into its
144            // arguments: sound only as an assumption (the library might
145            // sanitise or might not propagate), so name it.
146            if enter_call.is_some()
147                && cross_preds.is_empty()
148                && !local_preds.is_empty()
149                && let Some(mut site) = self.cut_site(cf, &cur)
150                && site.kind == "external-call"
151                && !is_std_forwarder(&site.detail)
152            {
153                site.kind = PASSED_THROUGH;
154                record_cut_site(&mut state.cut_sites, site);
155            }
156
157            self.enqueue_predecessors(&step, &cross_preds, local_preds, state, &mut queue);
158        }
159    }

161    /// Classify the node the walk stands on, once per step: a formal parameter
162    /// filters its cross predecessors by the entry call site, and a call
163    /// destination (`ActualRet`) pushes that site when the walk steps into the
164    /// callee.
165    fn classify_node(&self, cf: usize, cur: &NodeKey) -> (bool, Option<(usize, NodeKey)>) {
166        match self.prog.functions[cf].svfg.node(cur).map(|n| &n.kind) {
167            Some(NodeKind::FormalParam) => (true, None),
168            Some(NodeKind::ActualRet { call_site }) => (false, Some((cf, *call_site))),
169            _ => (false, None),
170        }
171    }
172
173    /// Every stop the walk evaluates before following predecessors. Returns
174    /// `true` when the node ends this walk, after recording why in `state`.
175    fn walk_stops(
176        &mut self,
177        fi: usize,
178        step: &WalkStep<'_>,
179        root_block: BlockId,
180        state: &mut ExploreState,
181    ) -> bool {
182        let WalkStep {
183            cf,
184            ir,
185            key,
186            use_block,
187            ..
188        } = *step;
189
190        // Stop: guard check. If this node's variable is checked by a
191        // guard-style call (`.test(x)` / `includes(x)`) whose branch dominates
192        // the block where the value is (re)used, the value was validated on
193        // every path reaching here, sanitized (task 8.3 guard map). Pure
194        // value-flow cannot see this: the guard is a sibling use, not a link
195        // in the chain.
196        let guard_hit = self.guard_maps.get(&cf).and_then(|gm| {
197            let class = state.sink_class.as_deref();
198            gm.guard_hit(key.var, use_block, class).or_else(|| {
199                (cf == fi)
200                    .then(|| gm.guard_hit(key.var, root_block, class))
201                    .flatten()
202            })
203        });
204        if let Some(hit) = guard_hit {
205            // Name the fact behind the guard so `--show-suppressed` audits a
206            // guard-style cut the way it audits a call-style one.
207            if let Some(GuardFact { call, classes }) = hit {
208                let suppression = SanitizerSuppression { call, classes };
209                if !state.suppressions.contains(&suppression) {
210                    state.suppressions.push(suppression);
211                }
212            }
213            state.saw_sanitized_root_only = true;
214            record_cut(&mut state.cuts, "guard");
215            return true;
216        }
217
218        // Stop: source reached, vulnerable path confirmed.
219        if is_source(ir, self.config, &self.facts, &key) {
220            state.saw_source = true;
221            if state.source_node.is_none() {
222                state.source_node = Some((cf, key));
223                state.source_desc = source_description(ir, self.config, &self.facts, &key);
224            }
225            return true;
226        }
227
228        // Stop: provably attacker-independent value (constant, string literal,
229        // boolean, or bounded integer range from the per-function value
230        // lattice). Every non-`Top` lattice value derives from literals,
231        // literal arithmetic, or branch sharpening - no source payload can be
232        // hiding behind this node - so the branch is cut as clean
233        // (unrealizable-path elimination) instead of exploring its
234        // predecessors, which may include over-approximated edges. Gated on
235        // fixpoint convergence: a truncated fixpoint may hold
236        // narrower-than-true values, and pruning on those could drop a real
237        // flow.
238        let provably_independent = self
239            .value_info_for(cf)
240            .is_some_and(|info| info.converged && info.values.contains_key(&key.var));
241        if provably_independent {
242            record_cut(&mut state.cuts, "value-const");
243            return true;
244        }
245
246        // Stop: sanitizer use, this branch is clean. Fact-table aware: includes
247        // learned methods like .replace()/.test() guards, qualified by sink
248        // role. Package-scoped sanitizer facts additionally require the call's
249        // own import origin to name their package; unknown origin leaves them
250        // inert.
251        let san_origin = self.sanitizer_origin_package(cf, &key);
252        if is_sanitizer_effective_for_role(
253            ir,
254            self.config,
255            &self.facts,
256            &key,
257            state.sink_role,
258            state.sink_label.as_deref(),
259            Some(state.sink_name.as_str()),
260            san_origin.as_deref(),
261        ) {
262            state.saw_sanitized_root_only = true;
263            if let Some(fact) = crate::analysis::forward::sanitizer_fact_defeating(
264                ir,
265                &self.facts,
266                &key,
267                state.sink_role,
268                state.sink_label.as_deref(),
269                Some(state.sink_name.as_str()),
270                san_origin.as_deref(),
271            ) {
272                let suppression = SanitizerSuppression {
273                    call: fact.call.clone(),
274                    classes: self
275                        .facts
276                        .sanitizer_classes(&fact.call, san_origin.as_deref()),
277                };
278                if !state.suppressions.contains(&suppression) {
279                    state.suppressions.push(suppression);
280                }
281            }
282            record_cut(&mut state.cuts, "sanitizer");
283            return true;
284        }
285
286        // Stop: a field-demand walk that reached a container's allocation. The
287        // requested field's value can only arrive through field-matching store
288        // edges, entered directly at the load's destination (same-function
289        // Pass 3, cross-function heap edges). Following the whole-container
290        // fill edges here would union every stored property - `sink(p.other)`
291        // must not see the store of `p.data` in the caller just because both
292        // share the container.
293        step.field_demand && self.def_is_allocation(cf, &key)
294    }
295
296    /// The predecessors of the step's node: local value-flow edges (phi arms
297    /// pruned by branch feasibility) and cross-function edges (a formal
298    /// parameter's actual arguments, a call destination's callee returns),
299    /// restricted to the entry call site when the k=1 context names one.
300    fn gather_predecessors(&self, step: &WalkStep<'_>) -> Preds {
301        let WalkStep {
302            cf,
303            ir,
304            key,
305            ctx,
306            is_formal_param,
307            ..
308        } = *step;
309
310        // Local predecessors (optionally honouring suppression).
311        let mut local_preds: Vec<NodeKey> = if self.honour_suppression {
312            self.prog.local_predecessors(cf, &key)
313        } else {
314            match self.prog.functions[cf].svfg.node(&key) {
315                Some(n) => n.predecessors(),
316                None => Vec::new(),
317            }
318        };
319
320        // Branch feasibility at phis: a Phi merges values from sibling arms;
321        // when the branch condition is a constant comparison, the infeasible
322        // arm never executes and its definition must not feed the merge
323        // (`bar = const if 7*18+106 > 200 else param` - the tainted else arm is
324        // dead). Filter phi predecessors here rather than in the SVFG:
325        // feasibility is a walk-time question and the graph stays a pure
326        // value-flow structure.
327        let is_phi = self.prog.functions[cf]
328            .svfg
329            .node(&key)
330            .is_some_and(|n| n.kind == NodeKind::Phi);
331        if is_phi {
332            let infeasible = phi_infeasible_incoming(ir, key.block, key.var);
333            if !infeasible.is_empty() {
334                local_preds.retain(|pk| {
335                    // Post-SSA each incoming var is defined in its arm block;
336                    // the def node key carries that block.
337                    !infeasible
338                        .iter()
339                        .any(|(blk, var)| *blk == pk.block && *var == pk.var)
340                });
341            }
342        }
343
344        // Cross predecessors (reverse index): who feeds this node from another
345        // function? For a FormalParam that means actual-args at caller call
346        // sites; for an ActualRet that means the callee's FormalRet nodes.
347        let mut cross_preds: Vec<(usize, NodeKey)> = self
348            .reverse_cross
349            .get(&(cf, key))
350            .cloned()
351            .unwrap_or_default();
352
353        // k=1 call-site context (phase 5): a formal parameter's value arrived
354        // through ONE invocation - the call site the walk entered this function
355        // by. Only that site's actual arguments explain it; sibling callers'
356        // args are a different invocation entirely. Empty context (root
357        // function, heap-edge entry, cap collapse): no filter - the sound k=0
358        // over-approximation.
359        if is_formal_param && let Some((f_expect, site_expect)) = self.ctx_last(ctx) {
360            cross_preds.retain(|(pf, pk)| {
361                *pf == f_expect
362                    && matches!(
363                        self.prog.functions[*pf].svfg.node(pk).map(|n| &n.kind),
364                        Some(NodeKind::ActualArg { call_site, .. })
365                            if *call_site == site_expect
366                    )
367            });
368        }
369
370        (local_preds, cross_preds)
371    }
372
373    /// Push every predecessor the walk has not seen, carrying the context each
374    /// edge implies.
375    fn enqueue_predecessors(
376        &mut self,
377        step: &WalkStep<'_>,
378        cross_preds: &[(usize, NodeKey)],
379        local_preds: Vec<NodeKey>,
380        state: &mut ExploreState,
381        queue: &mut VecDeque<WalkNode>,
382    ) {
383        let WalkStep {
384            cf,
385            key,
386            ctx,
387            enter_call,
388            field_demand,
389            ..
390        } = *step;
391        for &(pf, pk) in cross_preds {
392            // Context transition for the k=1 stack:
393            //  - landing on an ActualArg unwinds out of a callee → pop;
394            //  - leaving an ActualRet steps into a callee → push its site;
395            //  - any other cross jump (heap edges) has no call/return
396            //    shape → reset to the empty context (k=0 where it lands,
397            //    sound: never wrong-strict).
398            let pred_is_actual_arg = self.prog.functions[pf]
399                .svfg
400                .node(&pk)
401                .is_some_and(|n| matches!(n.kind, NodeKind::ActualArg { .. }));
402            let new_ctx = if pred_is_actual_arg {
403                self.ctx_exit(ctx)
404            } else if let Some((caller_fn, site)) = enter_call {
405                self.ctx_enter(ctx, caller_fn, site)
406            } else {
407                0
408            };
409            if state
410                .visited
411                .insert((pf, pk, pk.block, field_demand, new_ctx))
412            {
413                state.seen_nodes.insert((pf, pk));
414                if self.capture_paths {
415                    self.record_parent((pf, pk), (cf, key));
416                }
417                queue.push_back((pf, pk, pk.block, field_demand, new_ctx));
418            }
419        }
420        for pk in local_preds {
421            if state.visited.insert((cf, pk, key.block, field_demand, ctx)) {
422                state.seen_nodes.insert((cf, pk));
423                if self.capture_paths {
424                    self.record_parent((cf, pk), (cf, key));
425                }
426                queue.push_back((cf, pk, key.block, field_demand, ctx));
427            }
428        }
429    }
430}

432impl<'a> BackwardTaintEngine<'a> {
433    /// Describe an unresolvable dead-end node (see [`Self::node_is_unresolvable`])
434    /// as a [`CutSite`]: what the engine could not see, and where.
435    pub(super) fn cut_site(&self, fi: usize, key: &NodeKey) -> Option<CutSite> {
436        let ir = self.prog.functions[fi].ir;
437        let span = ir.var_metadata.get(&key.var).and_then(|m| m.byte_range);
438        let named = |kind: &'static str, detail: String| CutSite {
439            kind,
440            function: ir.name.clone(),
441            detail,
442            span,
443        };
444        let node = self.prog.functions[fi].svfg.node(key)?;
445        if node.kind == crate::graph::svfg::NodeKind::FormalParam {
446            let name = ir
447                .var_metadata
448                .get(&key.var)
449                .and_then(|m| m.source_name.clone())
450                .unwrap_or_default();
451            return Some(named("param-no-caller", name));
452        }
453        let idx = key.instr_idx?;
454        let ins = ir.blocks.get(&key.block)?.instructions.get(idx)?;
455        Some(match ins {
456            Instruction::CallStatic { func, .. } => named("external-call", func.clone()),
457            Instruction::CallVirtual { method, .. } => named("external-call", method.clone()),
458            Instruction::CallPointer { .. } => named("indirect-call", String::new()),
459            Instruction::LoadGlobal { name, .. } => named("global", name.clone()),
460            Instruction::Dereference { .. } => named("pointer-deref", String::new()),
461            _ => return None,
462        })
463    }
464
465    /// A node is unresolvable if we cannot prove its value clean within the
466    /// analysed program:
467    ///   * a `FormalParam` with no cross-edge feeders (external entry point,
468    ///     or called only with literal args), or
469    ///   * a def produced by an environment-read instruction (`LoadGlobal`,
470    ///     `Dereference`, `CallPointer`), its value comes from outside the
471    ///     local value-flow graph.
472    pub(super) fn node_is_unresolvable(&self, fi: usize, key: &NodeKey) -> bool {
473        let fe = &self.prog.functions[fi];
474        let Some(node) = fe.svfg.node(key) else {
475            return false;
476        };
477        if node.kind == crate::graph::svfg::NodeKind::FormalParam {
478            // Unresolvable iff there are no cross edges feeding it.
479            return !self.reverse_cross.contains_key(&(fi, *key));
480        }
481        // A call result with no cross-edge feeding it is a call to an
482        // unknown/external callee (library API): its semantics are unknown,
483        // so "clean" would be unsound, classify as unknown. Resolved
484        // internal callees have ActualRet→FormalRet reverse edges and never
485        // reach this dead-end classification.
486        if matches!(node.kind, crate::graph::svfg::NodeKind::ActualRet { .. }) {
487            dbg_trace!(
488                DebugFlags::get().unres,
489                "[unresolvable] ActualRet key={key:?} has_cross={} block={:?} idx={:?}",
490                self.reverse_cross.contains_key(&(fi, *key)),
491                key.block,
492                key.instr_idx
493            );
494            let NodeKey {
495                block,
496                instr_idx: Some(idx),
497                var,
498            } = *key
499            else {
500                return false;
501            };
502            if let Some(b) = fe.ir.blocks.get(&block)
503                && idx < b.instructions.len()
504                && matches!(
505                    &b.instructions[idx],
506                    Instruction::CallStatic { dest: Some(d), .. }
507                        | Instruction::CallVirtual { dest: Some(d), .. }
508                        | Instruction::CallPointer { dest: Some(d), .. }
509                    if *d == var
510                )
511            {
512                return true;
513            }
514            return false;
515        }
516        if node.kind == crate::graph::svfg::NodeKind::InstrDef {
517            let Some(instr) = key.instruction(fe.ir) else {
518                return false;
519            };
520            if matches!(
521                instr,
522                Instruction::LoadGlobal { .. }
523                    | Instruction::Dereference { .. }
524                    | Instruction::CallPointer { .. }
525            ) {
526                return true;
527            }
528            // A call result with no cross-edge feeding it is a call to an
529            // unknown/external callee (library API): its semantics are
530            // unknown, so "clean" would be unsound, classify as unknown.
531            // Resolved internal callees have ActualRet→FormalRet reverse
532            // edges and never reach this dead-end classification.
533            if matches!(node.kind, crate::graph::svfg::NodeKind::ActualRet { .. })
534                && matches!(
535                    instr,
536                    Instruction::CallStatic { dest: Some(d), .. }
537                        | Instruction::CallVirtual { dest: Some(d), .. }
538                    if *d == key.var
539                )
540            {
541                return true;
542            }
543        }
544        false
545    }
