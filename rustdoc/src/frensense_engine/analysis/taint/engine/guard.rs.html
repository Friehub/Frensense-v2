11/// One registered guard: the block it makes safe, the classes it was proven
12/// for (empty = a structural guard valid for every class) and, when a
13/// sanitizer fact registered it, that fact's call name.
14#[derive(Debug)]
15struct GuardEntry {
16    block: BlockId,
17    classes: Vec<String>,
18    call: Option<String>,
19}
20
21impl GuardEntry {
22    /// A guard with no sanitizer fact behind it (direct truthiness or a
23    /// comparison): valid for every class, nothing to name when audited.
24    fn structural(block: BlockId) -> Self {
25        Self {
26            block,
27            classes: Vec::new(),
28            call: None,
29        }
30    }
31}
32
33/// The sanitizer fact behind a guard that cut a walk, for `--show-suppressed`.
34#[derive(Debug, Clone, PartialEq, Eq)]
35pub(super) struct GuardFact {
36    pub call: String,
37    pub classes: Vec<String>,
38}
39
40/// Per-function guard analysis: which variables are *checked* by a
41/// guard-style call (`.test(x)`, `includes(x)`, allowlist `has(x)`) in which
42/// blocks, plus dominator sets.
43///
44/// A value-flow graph alone cannot see guards: `if (SAFE.test(x)) sink(x)`
45/// has the guard on a *sibling* use of `x`, so a backward walk from the sink
46/// never passes through it. The fix is structural: the branch condition is
47/// derived from a guard call on `x`; if that branch dominates the block the
48/// sink lives in, any `x` value flowing there was checked.
49#[derive(Debug, Default)]
50pub(super) struct GuardMap {
51    /// var → blocks whose terminator branch is derived from a guard-style
52    /// call on that var, together with the classes that guard was proven for
53    /// (empty = an unqualified structural guard, valid for every class).
54    guards: FxHashMap<VarId, Vec<GuardEntry>>,
55    /// block → set of blocks that dominate it (reflexive), computed once.
56    dominators: FxHashMap<BlockId, FxHashSet<BlockId>>,
57}
58
59impl GuardMap {
60    /// `file_package_imports`: the owning file's package-import table (or
61    /// the single-file fallback table), used to arm package-scoped
62    /// guard-style sanitizer facts through their import origin, a
63    /// scoped guard only registers when the callee's origin names its
64    /// package; unknown origin leaves it inert (recall-safe: an
65    /// unproven guard must never validate a value).
66    pub(super) fn build(
67        ir: &FunctionIR,
68        facts: &FactTable,
69        config: &TaintConfig,
70        file_package_imports: Option<&FxHashMap<String, crate::graph::module::PackageImport>>,
71    ) -> Self {
72        let mut guards: FxHashMap<VarId, Vec<GuardEntry>> = FxHashMap::default();
73
74        // Walk: cond var → def instruction, up to 3 hops, collecting
75        // guard-call vars along the way (handles `!ALLOWED.has(x)`,
76        // `x != null && SAFE.test(x)` shape fragments).
77        // Find guard calls and the vars they check.
78        // A guard call is a CallVirtual/CallStatic whose method has a
79        // guard-style SanitizerFact. The checked var is the first Var arg
80        // (predicate on an argument) or the receiver (predicate on the
81        // receiver, e.g. `file.includes('..')`).
82        let checked_vars = |instr: &Instruction| -> (Vec<VarId>, Vec<String>, String) {
83            let (method, receiver, args, origin) = match instr {
84                Instruction::CallVirtual {
85                    method,
86                    receiver,
87                    args,
88                    ..
89                } => {
90                    // Receiver import alias: `ALLOWLIST.has(x)` where
91                    // ALLOWLIST was imported from a package.
92                    let origin = match receiver {
93                        Operand::Var(r) => FactTable::receiver_root(ir, *r)
94                            .and_then(|root| file_package_imports.and_then(|m| m.get(&root)))
95                            .map(|p| p.package.clone()),
96                        _ => None,
97                    };
98                    (method, Some(receiver), args, origin)
99                }
100                Instruction::CallStatic { func, args, .. } => {
101                    // Static import: `import { isSafe } from 'pkg'`.
102                    let origin = file_package_imports
103                        .and_then(|m| m.get(func))
104                        .map(|p| p.package.clone());
105                    (func, None, args, origin)
106                }
107                _ => return (Vec::new(), Vec::new(), String::new()),
108            };
109            let Some(f) = facts.sanitizer_fact_scoped(method, origin.as_deref()) else {
110                return (Vec::new(), Vec::new(), String::new());
111            };
112            if !f.guard_style {
113                return (Vec::new(), Vec::new(), String::new());
114            }
115            let classes = facts.sanitizer_classes(method, origin.as_deref());
116            // The call is guard-style; also confirm the method really is one
117            // of the configured/known guards (fact table already ensures
118            // this). Collect checked vars.
119            let mut out = Vec::new();
120            if let Some(Operand::Var(r)) = receiver {
121                out.push(*r);
122            }
123            for a in args {
124                if let Operand::Var(v) = a {
125                    out.push(*v);
126                    let mut cur_v = *v;
127                    for _ in 0..3 {
128                        if let Some(def) = ir.def_instruction(cur_v) {
129                            match def {
130                                Instruction::AddressOf { src, .. } => {
131                                    out.push(*src);
132                                    cur_v = *src;
133                                }
134                                Instruction::Assign {
135                                    src: Operand::Var(s),
136                                    ..
137                                } => {
138                                    out.push(*s);
139                                    cur_v = *s;
140                                }
141                                _ => break,
142                            }
143                        } else {
144                            break;
145                        }
146                    }
147                }
148            }
149            (out, classes, method.clone())
150        };
151
152        let exits = |bid: BlockId| -> bool {
153            let mut cur = bid;
154            for _ in 0..5 {
155                match ir.blocks.get(&cur).map(|b| &b.terminator) {
156                    Some(Terminator::Return { .. } | Terminator::Throw { .. }) => return true,
157                    Some(Terminator::Jump(next)) => cur = *next,
158                    _ => return false,
159                }
160            }
161            false
162        };
163
164        let mut block_ids: Vec<BlockId> = ir.blocks.keys().copied().collect();
165        block_ids.sort_by_key(|b| b.0);
166        for &bid in &block_ids {
167            let blk = &ir.blocks[&bid];
168            let (cond_var, true_block, false_block) = match &blk.terminator {
169                Terminator::Branch {
170                    cond: Operand::Var(c),
171                    true_block,
172                    false_block,
173                } => (Some(*c), *true_block, *false_block),
174                _ => continue,
175            };
176            let Some(cond_var) = cond_var else { continue };
177            // Direct truthiness rejection: exactly one arm of this branch
178            // exits (`if (!t) { ...; return; }` / `if (t) { return; } ...`).
179            // A variable referenced directly in the condition - the
180            // condition var itself, or one reached through a `!` chain
181            // only - has been validated by the author's rejection test on
182            // the surviving arm: whatever reaches the dominated blocks
183            // passed the check (falsy/typeof-narrowed values are exactly
184            // what these guards reject). Without this, early-return
185            // validation on the value itself is invisible to the walk -
186            // only recognized sanitizer/predicate CALLS cut - which is
187            // the mre05 truthy/type-guard false-positive class.
188            let true_exits = exits(true_block);
189            let false_exits = false_block != true_block && exits(false_block);
190            let direct_safe = if true_exits && !false_exits {
191                Some(false_block)
192            } else if false_exits && !true_exits {
193                Some(true_block)
194            } else {
195                None
196            };
197            if let Some(safe) = direct_safe {
198                guards
199                    .entry(cond_var)
200                    .or_default()
201                    .push(GuardEntry::structural(safe));
202            }
203            // Hop 0..3 up the def chain from the condition, tracking polarity/sense.
204            let mut frontier = vec![(cond_var, true)];
205            for _hop in 0..3 {
206                let mut next = Vec::new();
207                for (v, sense) in &frontier {
208                    let Some(instr) = ir.def_instruction(*v) else {
209                        continue;
210                    };
211                    match instr {
212                        Instruction::CallStatic { .. } | Instruction::CallVirtual { .. } => {
213                            let (vars, classes, guard_call) = checked_vars(instr);
214                            for cv in vars {
215                                let true_exits = exits(true_block);
216                                let false_exits = exits(false_block);
217                                let safe_block = if true_exits && !false_exits {
218                                    false_block
219                                } else if (false_exits && !true_exits) || *sense {
220                                    true_block
221                                } else {
222                                    false_block
223                                };
224                                guards.entry(cv).or_default().push(GuardEntry {
225                                    block: safe_block,
226                                    classes: classes.clone(),
227                                    call: Some(guard_call.clone()),
228                                });
229                            }
230                        }
231                        Instruction::UnaryOp {
232                            op,
233                            src: Operand::Var(u),
234                            ..
235                        } => {
236                            let new_sense = if op == "!" || op == "neg" || op == "not" {
237                                !*sense
238                            } else {
239                                *sense
240                            };
241                            // `!`-chain member: part of the direct rejection
242                            // test, validated on the surviving arm.
243                            if let Some(safe) = direct_safe {
244                                guards
245                                    .entry(*u)
246                                    .or_default()
247                                    .push(GuardEntry::structural(safe));
248                            }
249                            next.push((*u, new_sense));
250                        }
251                        Instruction::BinaryOp { op, lhs, rhs, .. } => {
252                            let literal_sibling = matches!(
253                                (lhs, rhs),
254                                (Operand::StringLiteral(_), _)
255                                    | (_, Operand::StringLiteral(_))
256                                    | (Operand::IntLiteral(_), _)
257                                    | (_, Operand::IntLiteral(_))
258                            );
259                            let is_comparison = op.contains("in")
260                                || op == "=="
261                                || op == "==="
262                                || op == "!="
263                                || op == "!=="
264                                || op == "not";
265                            if literal_sibling && is_comparison {
266                                let is_inverted = op == "!=" || op == "!==" || op == "not in";
267                                let is_denylist = match (lhs, rhs) {
268                                    (Operand::StringLiteral(s), _)
269                                        if facts.is_guard_denylist(s) =>
270                                    {
271                                        true
272                                    }
273                                    (_, Operand::StringLiteral(s))
274                                        if facts.is_guard_denylist(s) =>
275                                    {
276                                        true
277                                    }
278                                    _ => false,
279                                };
280                                let effective_sense = if is_inverted ^ is_denylist {
281                                    !*sense
282                                } else {
283                                    *sense
284                                };
285                                let safe_block = if effective_sense {
286                                    true_block
287                                } else {
288                                    false_block
289                                };
290                                for op in [lhs, rhs] {
291                                    if let Operand::Var(u) = op {
292                                        guards
293                                            .entry(*u)
294                                            .or_default()
295                                            .push(GuardEntry::structural(safe_block));
296                                        if let Some(Instruction::UnaryOp {
297                                            op: uop,
298                                            src: Operand::Var(inner),
299                                            ..
300                                        }) = ir.def_instruction(*u)
301                                            && uop == "typeof"
302                                        {
303                                            guards
304                                                .entry(*inner)
305                                                .or_default()
306                                                .push(GuardEntry::structural(safe_block));
307                                        }
308                                    }
309                                }
310                            }
311                            for op in [lhs, rhs] {
312                                if let Operand::Var(u) = op {
313                                    next.push((*u, *sense));
314                                }
315                            }
316                        }
317                        _ => {}
318                    }
319                }
320                if next.is_empty() {
321                    break;
322                }
323                frontier = next;
324            }
325        }
326        let _ = config;
327
328        // Dominators: shared structural module (also serves policy guard
329        // dominance and control-dependence queries in checks).
330        let dominators = crate::ir::control::dominators(ir);
331
332        Self { guards, dominators }
333    }
334
335    /// The guard that dominates `sink_block` and was proven for this sink's
336    /// class, as `(call, classes)` when a sanitizer fact registered it.
337    ///
338    /// The outer `Option` is "is guarded"; the inner one is "which fact":
339    /// a direct-truthiness or comparison guard has no fact behind it, so
340    /// there is nothing to name in `--show-suppressed`.
341    pub(super) fn guard_hit(
342        &self,
343        var: VarId,
344        sink_block: BlockId,
345        sink_class: Option<&str>,
346    ) -> Option<Option<GuardFact>> {
347        let gs = self.guards.get(&var)?;
348        let doms = self.dominators.get(&sink_block)?;
349        gs.iter()
350            .find(|e| {
351                doms.contains(&e.block)
352                    && (e.classes.is_empty()
353                        || e.classes.iter().any(|c| c == "*")
354                        || sink_class.is_some_and(|class| e.classes.iter().any(|c| c == class)))
355            })
356            .map(|e| {
357                e.call.clone().map(|call| GuardFact {
358                    call,
359                    classes: e.classes.clone(),
360                })
361            })
362    }
363}
