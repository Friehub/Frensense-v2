5//! Sink-role model: what a sink *does* with the data it receives.
6//!
7//! The type itself lives in `frensense-lang` ([`SinkRole`]): the
8//! `SinkLabel` -> role mapping, the default role->tier ladder and the
9//! stable reporting tag are classification/reporting knowledge, and both
10//! the bundler's replay gate and the CLI need the ladder without going
11//! through the engine (ENGINE_PURITY_REFACTOR 1.3 / D4). The engine keeps
12//! only the type on its findings - zero tier strings.
13
14pub use frensense_lang::severity::SinkRole;
15
16/// Parse the snake_case [`SinkRole`] name the default-pack generator
17/// serializes on [`crate::analysis::taint::facts::LearnedFactEntry::LanguageSink`]
18/// (the serde wire form). Unknown names keep the conservative default.
19#[must_use]
20pub(crate) fn sink_role_from_name(name: &str) -> SinkRole {
21    let role_part = name.split_once(':').map(|(r, _)| r).unwrap_or(name);
22    match role_part {
23        "execution" => SinkRole::Execution,
24        "resource" => SinkRole::Resource,
25        "storage" => SinkRole::Storage,
26        "crypto" => SinkRole::Crypto,
27        "response" => SinkRole::Response,
28        "xss" => SinkRole::Xss,
29        "validation" => SinkRole::Validation,
30        _ => SinkRole::Other,
31    }
32}
33
34/// Extract the optional label slug serialized after the colon in a role string
35/// (e.g. `"execution:prototype"` -> `Some("prototype")`).
36#[must_use]
37pub(crate) fn sink_label_slug_from_role_str(name: &str) -> Option<&str> {
38    name.split_once(':').map(|(_, slug)| slug)
39}
40
41/// The snake_case wire name of a [`SinkRole`] - the inverse of
42/// `sink_role_from_name`. The default-pack generator serializes this
43/// onto [`crate::analysis::taint::facts::LearnedFactEntry::LanguageSink`],
44/// so the pack stores role names instead of serde enum indices (bundle
45/// stability across enum reorders).
46#[must_use]
47pub fn sink_role_name(role: SinkRole) -> &'static str {
48    match role {
49        SinkRole::Execution => "execution",
50        SinkRole::Resource => "resource",
51        SinkRole::Storage => "storage",
52        SinkRole::Crypto => "crypto",
53        SinkRole::Response => "response",
54        SinkRole::Xss => "xss",
55        SinkRole::Validation => "validation",
56        SinkRole::Other => "other",
57    }
58}
