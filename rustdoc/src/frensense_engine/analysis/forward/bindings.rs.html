5use super::*;

7// ---------------------------------------------------------------------------
8// Call-site bindings
9// ---------------------------------------------------------------------------
10
11/// One resolved call site inside a function.
12#[derive(Debug, Clone)]
13#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
14pub struct CallBinding {
15    /// Sentinel key identifying the call instruction (var = `VarId(usize::MAX)`).
16    pub call_site: NodeKey,
17    /// Indices of resolved callees in [`ProgramSvfg::functions`], sorted and
18    /// deduped. Empty when every target is external/unresolved, dynamic
19    /// dispatch and callbacks may resolve to several callees at once; the
20    /// cross-edge installer links all of them (sound: any may run).
21    /// Resolution comes from `callgraph` (aliases, receiver classes,
22    /// higher-order callbacks); bare-name matching is the fallback.
23    pub callees: Vec<usize>,
24    /// Best-known callee name for diagnostics and external fact lookup
25    /// (source/sink classification); `<indirect>` when unresolved.
26    pub callee_name: String,
27    /// The caller's receiver node, if this is a virtual call on a receiver.
28    pub receiver: Option<NodeKey>,
29    /// `(arg use-node key, arg index)`. Positional arguments indexed from 0.
30    pub args: Vec<(NodeKey, usize)>,
31    /// The caller's `ActualRet` def node, if the call has a destination.
32    pub ret_node: Option<NodeKey>,
33}
34
35impl<'a> ProgramSvfg<'a> {
36    // -----------------------------------------------------------------------
37    // Step 1: discover call-site bindings
38    // -----------------------------------------------------------------------
39
40    pub(super) fn discover_bindings(&mut self, callgraph: &CallGraph) {
41        for fi in 0..self.functions.len() {
42            let ir = self.functions[fi].ir;
43            let fname = self.functions[fi].name.clone();
44            let mut bindings: Vec<CallBinding> = Vec::new();
45            let mut arg_slots: FxHashMap<NodeKey, (usize, usize)> = FxHashMap::default();
46
47            let mut block_ids: Vec<BlockId> = ir.blocks.keys().copied().collect();
48            block_ids.sort_by_key(|b| b.0);
49
50            for &block in &block_ids {
51                let bdata = &ir.blocks[&block];
52                for (idx, instr) in bdata.instructions.iter().enumerate() {
53                    // Call-shape extraction. `CallPointer` sites participate
54                    // when the callgraph resolved them (aliases, callbacks);
55                    // unresolved ones keep the sound local pass-through
56                    // fallback instead of a binding.
57                    let (callee_name, receiver, args, dest): (
58                        String,
59                        Option<&Operand>,
60                        &Vec<Operand>,
61                        Option<VarId>,
62                    ) = match instr {
63                        Instruction::CallStatic {
64                            func, args, dest, ..
65                        } => (func.clone(), None, args, *dest),
66                        Instruction::CallVirtual {
67                            method,
68                            receiver,
69                            args,
70                            dest,
71                            ..
72                        } => (method.clone(), Some(receiver), args, *dest),
73                        Instruction::CallPointer { args, dest, .. } => {
74                            let targets = callgraph.targets_at(&fname, block.0, idx);
75                            if targets.is_empty() {
76                                continue; // unresolved indirect → sound fallback
77                            }
78                            let name = targets
79                                .iter()
80                                .find_map(|t| t.internal_key())
81                                .unwrap_or_else(|| "<indirect>".to_string());
82                            (name, None, args, *dest)
83                        }
84                        _ => continue,
85                    };
86
87                    let call_site = NodeKey::instr(block, idx, VarId(usize::MAX));
88                    let mut bargs: Vec<(NodeKey, usize)> = Vec::new();
89                    let recv_node = if let Some(Operand::Var(r)) = receiver {
90                        Some(NodeKey::instr(block, idx, *r))
91                    } else {
92                        None
93                    };
94                    let is_higher_order = matches!(
95                        callee_name.as_str(),
96                        "map"
97                            | "and_then"
98                            | "then"
99                            | "flatMap"
100                            | "flat_map"
101                            | "forEach"
102                            | "for_each"
103                            | "filter"
104                    );
105                    if is_higher_order && let Some(r_node) = recv_node {
106                        bargs.push((r_node, 0));
107                    }
108                    for (slot, a) in args.iter().enumerate() {
109                        if let Operand::Var(v) = a {
110                            let s = if is_higher_order { slot + 1 } else { slot };
111                            bargs.push((NodeKey::instr(block, idx, *v), s));
112                        }
113                    }
114                    let ret_node = dest.map(|d| NodeKey::instr(block, idx, d));
115
116                    // Resolve targets: callgraph first (aliases, class-
117                    // qualified methods, callbacks), bare-name fallback.
118                    // A module-level data binding (`const pkg =
119                    // require("pkg")`) is not a callable: `pkg(req)` calls
120                    // into the library. Resolving it here gave the site an
121                    // empty parameter list and a vacuous summary, whose
122                    // pass-through suppression deleted the only taint edge
123                    // out of a `require` binding - every such call read
124                    // Unknown while the same call to an undefined global
125                    // propagated fine.
126                    let mut callees: Vec<usize> = callgraph
127                        .targets_at(&fname, block.0, idx)
128                        .iter()
129                        .filter_map(|t| t.internal_key())
130                        .filter_map(|k| self.func_index.get(&k).copied())
131                        .filter(|&gi| !self.functions[gi].ir.is_static_data)
132                        .collect();
133                    // A builtin method name (`m.get(k)`, `buf.write(s)`) on an
134                    // unresolved receiver is the standard library's, never a
135                    // program function that happens to share the bare name.
136                    // The table is language mechanism, so it lives in
137                    // frensense-lang beside the grammar specs - not here and
138                    // not in the pack (packgen owns sink vocabulary).
139                    let builtin_verb =
140                        receiver.is_some() && frensense_lang::is_container_verb(&callee_name);
141                    if callees.is_empty()
142                        && !builtin_verb
143                        && let Some(gi) = self.func_index.get(&callee_name).copied()
144                        && !self.functions[gi].ir.is_static_data
145                    {
146                        callees.push(gi);
147                    }
148                    callees.sort_unstable();
149                    callees.dedup();
150
151                    bindings.push(CallBinding {
152                        call_site,
153                        callees,
154                        callee_name,
155                        receiver: recv_node,
156                        args: bargs,
157                        ret_node,
158                    });
159                    let bi = bindings.len() - 1;
160                    if let Some(r) = bindings[bi].receiver {
161                        arg_slots.insert(r, (bi, usize::MAX));
162                    }
163                    for (k, s) in &bindings[bi].args {
164                        arg_slots.insert(*k, (bi, *s));
165                    }
166                }
167            }
168
169            self.functions[fi].bindings = bindings;
170            self.functions[fi].arg_slots = arg_slots;
171        }
172    }
173}
