11/// The source vocabulary visible at one call site: the language-agnostic
12/// `TaintConfig` set plus the call site's own language section.
13///
14/// Source facts are declared per language (TypeScript's `req`/`input`
15/// parameter names, Rust's `std::env::var`, ...) and a bare name is not
16/// portable between them: `input` is caller data in TypeScript and an
17/// unrelated local in Rust. Sharing one set across a mixed-language scan
18/// seeds taint from names that mean nothing in the file being analyzed.
19pub(crate) struct SourceScope<'a> {
20    universal: &'a rustc_hash::FxHashSet<String>,
21    language: Option<&'a rustc_hash::FxHashSet<String>>,
22    /// Names that are sources only in the program's entry point, withheld
23    /// here because this function is not it (see [`ENTRY_ONLY_PARAMS`]).
24    withheld: &'static [&'static str],
25    /// The IR's own language spec, when that grammar is compiled in. Its
26    /// `classify_param_taint` hook is the provider's knowledge of which
27    /// handler parameters the framework fills with request data.
28    spec: Option<&'static dyn frensense_lang::LanguageSpec>,
29}
30
31impl<'a> SourceScope<'a> {
32    pub(crate) fn for_ir(config: &'a TaintConfig, facts: &'a FactTable, ir: &FunctionIR) -> Self {
33        Self {
34            universal: &config.sources,
35            language: facts.language_sources(&ir.language),
36            withheld: entry_only_names_withheld_for(ir),
37            spec: frensense_lang::LanguageRegistry::global().for_name(&ir.language),
38        }
39    }
40
41    /// True when `name` is, or is a field path under, a withheld name.
42    fn is_withheld(&self, name: &str) -> bool {
43        self.withheld.iter().any(|w| {
44            name == *w || (name.starts_with(w) && name.as_bytes().get(w.len()) == Some(&b'.'))
45        })
46    }
47
48    pub(crate) fn contains(&self, name: &str) -> bool {
49        !self.is_withheld(name)
50            && (self.universal.contains(name)
51                || self.language.is_some_and(|set| set.contains(name)))
52    }
53
54    /// Does the language spec classify this function parameter as
55    /// attacker-controlled input?
56    ///
57    /// This is the spec-level counterpart to the pack's name/type source
58    /// vocabulary: `classify_param_taint` is where a provider states that a
59    /// framework fills a handler parameter with request data (`def h(x:
60    /// Request)`, `fn h(r *http.Request)`). The check is name-agnostic for
61    /// typed languages because the type, not the author's identifier,
62    /// decides. A parameter that entry-point rules withhold (`argv` outside
63    /// `main`) is never classified, so the classifier cannot re-widen the
64    /// name-only matches those rules exist to suppress.
65    ///
66    /// Only `UserInput` origins seed backward taint: the engine models one
67    /// kind of untrusted value, and the environment/filesystem/network
68    /// categories are classifications the analysis does not yet distinguish.
69    /// Honouring them here would turn every parameter named `config` or `env`
70    /// into a source.
71    fn classifies_param(&self, name: &str, type_name: Option<&str>) -> bool {
72        if self.is_withheld(name) {
73            return false;
74        }
75        let Some(spec) = self.spec else {
76            return false;
77        };
78        matches!(
79            spec.classify_param_taint(Some(name), type_name),
80            Some(frensense_lang::spec::TaintOrigin::UserInput)
81        )
82    }
83
84    pub(crate) fn iter(&self) -> impl Iterator<Item = &'a String> {
85        self.universal
86            .iter()
87            .chain(self.language.into_iter().flatten())
88            .filter(|s| !self.is_withheld(s))
89    }
90}

92/// Entry-point-only parameter names, per language family.
93///
94/// `argv`/`argc` are the process's command line only in the program's entry
95/// point. Everywhere else the same name is an ordinary parameter: redis
96/// alone has `robj **argv` (a client command), `sds *argv` (a config line)
97/// and `char *argv[]` (a test) in functions that never see the process
98/// arguments, and treating every one as attacker input produced 48
99/// path-traversal findings with the wrong source. Reached from `main`, the
100/// value arrives through the call-argument edge instead, which keeps the
101/// real flow and drops the name-only ones.
102const ENTRY_ONLY_PARAMS: &[(&str, &[&str], &str)] = &[
103    ("c", &["argc", "argv"], "main"),
104    ("cpp", &["argc", "argv"], "main"),
105];
106
107/// The entry-only names this function must NOT treat as sources (empty for
108/// the entry point itself and for languages with no such rule).
109fn entry_only_names_withheld_for(ir: &FunctionIR) -> &'static [&'static str] {
110    for (lang, names, entry) in ENTRY_ONLY_PARAMS {
111        if ir.language == *lang {
112            let base = ir.name.rsplit("::").next().unwrap_or(&ir.name);
113            return if base == *entry { &[] } else { names };
114        }
115    }
116    &[]
117}

119/// True if the node is a definition whose instruction is a configured source.
120pub(crate) fn is_source(
121    ir: &FunctionIR,
122    config: &TaintConfig,
123    facts: &FactTable,
124    key: &NodeKey,
125) -> bool {
126    let sources = SourceScope::for_ir(config, facts, ir);
127    let NodeKey {
128        block,
129        instr_idx: Some(idx),
130        var,
131    } = *key
132    else {
133        return false;
134    };
135    let Some(b) = ir.blocks.get(&block) else {
136        return false;
137    };
138    if idx >= b.instructions.len() {
139        // Sentinel node: function parameter. Three ways a parameter is a
140        // source:
141        //   1. its conventional NAME is a source ("req", "input", "body");
142        //   2. its TYPE annotation names one of the framework's request
143        //      value types (`Query<P>`, `web::Query<P>`, `*http.Request`);
144        //   3. the language spec classifies it as user input from either the
145        //      name or the annotation (`classify_param_taint`).
146        // (2) exists because for typed languages the extractor type - not
147        // the author's variable name - decides that the argument carries
148        // attacker input. (3) is the spec's own framework knowledge: it is
149        // what makes `def h(x: Request)` a source when `x` is the author's
150        // rename and the pack lists only member paths under `request`.
151        if let Some(meta) = ir.var_metadata.get(&var) {
152            let classified = meta.source_name.as_deref().is_some_and(|name| {
153                sources.contains(name) || sources.classifies_param(name, meta.type_name.as_deref())
154            });
155            if classified {
156                return true;
157            }
158            return meta.type_name.as_deref().is_some_and(|ty| {
159                source_path_matches(&sources, ty, ty.split("::").next().unwrap_or(ty))
160            });
161        }
162        return false;
163    }
164    match &b.instructions[idx] {
165        Instruction::CallStatic {
166            func,
167            dest: Some(d),
168            ..
169        } => {
170            sources.contains(func)
171                && !config.sinks.contains(func)
172                && !facts.is_sink_call_scoped(
173                    func,
174                    None,
175                    None,
176                    (!ir.language.is_empty()).then_some(ir.language.as_str()),
177                )
178                && *d == var
179        }
180        Instruction::CallVirtual {
181            method,
182            dest: Some(d),
183            receiver,
184            ..
185        } if *d == var => {
186            // Module-qualified calls (`random.randint(...)`) lower as
187            // CallVirtual with the module as receiver, while verb-named
188            // methods (`res.json(...)`) carry an object receiver. Match the
189            // full member path `receiver.method` against dotted source
190            // facts first, then the bare method name (same over-approximate
191            // last-segment semantics as sinks).
192            if sources.contains(method) {
193                return true;
194            }
195            if let Operand::Var(r) = receiver
196                && let Some(path_sans_method) = FactTable::receiver_access_path(ir, *r)
197            {
198                let path = format!("{path_sans_method}.{method}");
199                let root = path_sans_method.split('.').next().unwrap_or("");
200                // Use `source_path_matches` (not a raw prefix rule): it
201                // suppresses bare-root re-widening when granular field
202                // rules exist for that root, so `req.replace(...)` (a
203                // String method) is NOT a source just because `req` is
204                // registered as a bare root. The raw prefix rule matched
205                // `req.replace` because `"req.replace".starts_with("req")`.
206                //
207                // The receiver's FULL access path is reconstructed (not
208                // just its root): `request.args.get('cmd')` must match the
209                // granular fact `request.args.get`, which the truncated
210                // root (`request` + method → `request.get`) can never
211                // reach, and the bare-root widening suppression would
212                // then correctly refuse to re-widen `request` over its
213                // granular children.
214                return source_path_matches(&sources, &path, root);
215            }
216            false
217        }
218        Instruction::LoadField { base, field, .. } => {
219            // Member-expression source: `req.body` lowers to LoadField
220            // chains, the *dest* of each link has no source_name, so the
221            // access path must be reconstructed by walking the base chain
222            // through the IR until a named root var is reached:
223            //   LoadField(dest2, base1, "args"); LoadField(dest1, req, "body")
224            // reconstructs "req.body.args".
225            let path = member_access_path(ir, *base, field);
226            let root = path.split('.').next().unwrap_or("");
227            crate::dbg_trace!(
228                crate::debug_flags::DebugFlags::get().is_source,
229                "[is_source] path={path:?} hit={}",
230                sources.contains(&path)
231            );
232            source_path_matches(&sources, &path, root)
233        }
234        _ => false,
235    }
236}

238/// Reconstruct a member-access path like `req.body.args` by walking
239/// LoadField base chains backwards to a named root variable. Bounded to
240/// prevent pathological IR from looping (alias chains are acyclic by SSA
241/// construction, but the bound is cheap insurance).
242pub(crate) fn member_access_path(ir: &FunctionIR, mut base: VarId, last_field: &str) -> String {
243    let mut segments = vec![last_field.to_string()];
244    for _ in 0..16 {
245        match ir
246            .var_metadata
247            .get(&base)
248            .and_then(|m| m.source_name.clone())
249        {
250            Some(name) => {
251                segments.push(name);
252                break;
253            }
254            None => {
255                // Continue through the LoadField that defines `base`.
256                match ir.loadfield_def(base) {
257                    Some((b2, field)) => {
258                        segments.push(field);
259                        base = b2;
260                    }
261                    None => break,
262                }
263            }
264        }
265    }
266    segments.reverse();
267    // Drop leading mem-state artifacts if any leaked in.
268    while segments.len() > 1 && segments[0] == crate::ir::function::INITIAL_HEAP_STATE {
269        segments.remove(0);
270    }
271    segments.join(".")
272}

274/// Does a reconstructed member-access `path` (whose first segment is `root`)
275/// match a configured source rule?
276///
277/// Exact matches and prefix-subtree rules ("req.body" covers "req.body.args")
278/// apply unconditionally. A whole-value registration of the bare root ("ctx")
279/// covers its subtree only while the config declares no granular field rules
280/// for that root ("ctx.state"): the granular list is the intended precise set,
281/// so a bare root must not silently re-widen it - otherwise `ctx.env.SECRET`
282/// becomes a source just because `ctx.state` was registered.
283///
284/// (The previous `sources.contains(root)` clause was redundant with the
285/// prefix rule for dotted paths and would defeat this suppression, so it is
286/// not carried over; dot-less paths hit `sources.contains(path)` exactly.)
287pub(crate) fn source_path_matches(sources: &SourceScope<'_>, path: &str, root: &str) -> bool {
288    sources.contains(path)
289        || sources.iter().any(|s| {
290            if !path.starts_with(s.as_str()) || path.as_bytes().get(s.len()) != Some(&b'.') {
291                return false;
292            }
293            if s.as_str() != root {
294                // Sub-path prefix rule ("ctx.request" → "ctx.request.body").
295                return true;
296            }
297            // Whole-value root registration: live only when no granular
298            // field rules exist for that root.
299            !sources
300                .iter()
301                .any(|g| g.starts_with(root) && g.as_bytes().get(root.len()) == Some(&b'.'))
302        })
303}

305/// C calls whose result does not carry the taint of their arguments:
306/// lengths (`strlen`), and allocators in every spelling the C ecosystem uses
307/// (`malloc`, `calloc`, `alloca`, `zmalloc`, `s_trymalloc_usable`,
308/// `xcalloc`, `g_malloc`). `realloc` is excluded: its result is the old
309/// object's contents.
310pub(crate) fn c_result_is_independent_of_args(func: &str) -> bool {
311    let last = func.rsplit(['.', ':']).next().unwrap_or(func);
312    let l = last.to_ascii_lowercase();
313    if matches!(l.as_str(), "strlen" | "strnlen" | "wcslen" | "alloca") {
314        return true;
315    }
316    if l.contains("realloc") {
317        return false;
318    }
319    let l = l.strip_suffix("_usable").unwrap_or(&l);
320    l.ends_with("malloc") || l.ends_with("calloc")
321}

323/// True if the node is a use inside a configured sanitizer call.
324pub(crate) fn is_sanitizer_use(ir: &FunctionIR, config: &TaintConfig, key: &NodeKey) -> bool {
325    let Some(instr) = key.instruction(ir) else {
326        return false;
327    };
328    match instr {
329        Instruction::CallStatic { func, .. } => config.sanitizers.contains(func),
330        Instruction::CallVirtual { method, .. } => config.sanitizers.contains(method),
331        _ => false,
332    }
333}

338/// String-rewrite methods whose first argument is the subsequence removed.
339///
340/// `value.replace("'", "&apos;")` is the inline escaping idiom for a value
341/// interpolated into a quoted XPath/SQL/HTML string; `replaceAll` is the ES2021
342/// spelling. The method name alone proves nothing (a rewrite may remove
343/// anything), so [`removes_injection_delimiter`] additionally requires the
344/// removed literal to be a guard-denylist token.
345const REWRITE_METHODS: &[&str] = &["replace", "replaceAll"];
346
347/// Is this rewrite removing the delimiter an injection needs?
348///
349/// The vocabulary is the pack's guard-denylist set - the same quote tokens an
350/// early `if "'" in value: return` guard rejects - and the mechanism is the
351/// literal first argument of a string-rewrite method. Removing `'` (or `"`)
352/// from a value that is then interpolated into a delimited string leaves it no
353/// way to break out of the literal, so the check is exact rather than
354/// heuristic: the delimiter is provably absent from the result.
355fn removes_injection_delimiter(args: &[Operand], facts: &FactTable) -> bool {
356    matches!(args.first(), Some(Operand::StringLiteral(s)) if facts.is_guard_denylist(s))
357}

359/// Fact-table-aware sanitizer check qualified by the sink's role and label.
360/// Returns true if this node is inside a sanitizer that actually defeats the candidate sink.
361///
362/// Package-scoped sanitizer facts additionally require `origin_package`
363/// to name their package (import resolution), pass `None` when no
364/// origin is known; scoped facts then stay inert (recall-safe: they may
365/// only cut a flow under a confirmed origin).
366pub(crate) fn is_sanitizer_effective_for_role(
367    ir: &FunctionIR,
368    config: &TaintConfig,
369    facts: &FactTable,
370    key: &NodeKey,
371    role: SinkRole,
372    label: Option<&str>,
373    sink_name: Option<&str>,
374    origin_package: Option<&str>,
375) -> bool {
376    // `config` is retained for API stability with callers; the legacy
377    // flat `config.sanitizers` fallback was removed (it cut sink-
378    // agnostically, bypassing role checks). Fact-table sanitizers
379    // carry a `kind` slug that `from_slug` resolves.
380    let _ = config;
381    let Some(instr) = key.instruction(ir) else {
382        return false;
383    };
384    match instr {
385        Instruction::CallStatic { func, .. } => {
386            // A fresh allocation or a length is not the data it was sized
387            // from. Allocator wrappers carry project-specific names
388            // (`zmalloc`, `s_trymalloc_usable`, `xcalloc`), so the shape of
389            // the name decides rather than a per-project list.
390            if matches!(ir.language.as_str(), "c" | "cpp") && c_result_is_independent_of_args(func)
391            {
392                return true;
393            }
394            if let Some(fact) = facts.sanitizer_fact_scoped(func, origin_package) {
395                if let Some(kind) = SanitizerKind::from_slug(&fact.kind) {
396                    let classes = facts.sanitizer_classes(func, origin_package);
397                    return kind.defeats_sink(role, label, sink_name, &classes);
398                }
399                // Unknown kind slug: do NOT cut sink-agnostically. The
400                // previous `return true` fallback let any unrecognized
401                // label defeat every role (residual #15: "encode" from
402                // from_config silenced Execution sinks). Legacy
403                // "encode"/"allowlist" are now mapped to Full in
404                // `from_slug`; anything else is genuinely unknown.
405                return false;
406            }
407            false
408        }
409        Instruction::CallVirtual {
410            method,
411            receiver,
412            args,
413            ..
414        } => {
415            // Inline delimiter removal neutralizes exactly the delimited-string
416            // sinks: an execution sink (SQL/XPath/command), an XSS sink, or a
417            // response writer. Other roles (path, URL, deserialization) are
418            // not delimiter-sensitive and keep the flow.
419            if matches!(
420                role,
421                SinkRole::Execution | SinkRole::Xss | SinkRole::Response
422            ) && REWRITE_METHODS.contains(&method.as_str())
423                && removes_injection_delimiter(args, facts)
424            {
425                return true;
426            }
427            if let Operand::Var(r) = receiver {
428                let path = FactTable::receiver_access_path(ir, *r);
429                if let Some(ref p) = path {
430                    let full = format!("{p}.{method}");
431                    if let Some(fact) = facts.sanitizer_fact_scoped(&full, origin_package) {
432                        if let Some(kind) = SanitizerKind::from_slug(&fact.kind) {
433                            let classes = facts.sanitizer_classes(&full, origin_package);
434                            return kind.defeats_sink(role, label, sink_name, &classes);
435                        }
436                        return false;
437                    }
438                }
439                if facts.is_session_path(method, path.as_deref()) {
440                    return true;
441                }
442            }
443            if let Some(fact) = facts.sanitizer_fact_scoped(method, origin_package) {
444                if let Some(kind) = SanitizerKind::from_slug(&fact.kind) {
445                    let classes = facts.sanitizer_classes(method, origin_package);
446                    return kind.defeats_sink(role, label, sink_name, &classes);
447                }
448                return false;
449            }
450            false
451        }
452        _ => false,
453    }
454}

456/// The sanitizer fact that cuts this node for a sink of `(role, label)`, when
457/// one does.
458///
459/// The boolean [`is_sanitizer_effective_for_role`] is the mechanism's answer;
460/// this variant also names *which bundle fact* silenced the flow, so
461/// `--show-suppressed` can report `suppressed by bundle sanitizer X (class
462/// Z)` on a real project.
463pub(crate) fn sanitizer_fact_defeating<'a>(
464    ir: &FunctionIR,
465    facts: &'a FactTable,
466    key: &NodeKey,
467    role: SinkRole,
468    label: Option<&str>,
469    sink_name: Option<&str>,
470    origin_package: Option<&str>,
471) -> Option<&'a crate::analysis::taint::facts::SanitizerFact> {
472    let instr = key.instruction(ir)?;
473    let defeating = |call: &str, fact: &'a crate::analysis::taint::facts::SanitizerFact| -> bool {
474        let classes = facts.sanitizer_classes(call, origin_package);
475        SanitizerKind::from_slug(&fact.kind)
476            .is_some_and(|kind| kind.defeats_sink(role, label, sink_name, &classes))
477    };
478    match instr {
479        Instruction::CallStatic { func, .. } => facts
480            .sanitizer_fact_scoped(func, origin_package)
481            .filter(|f| defeating(func, f)),
482        Instruction::CallVirtual {
483            method, receiver, ..
484        } => {
485            let receiver_path = match receiver {
486                Operand::Var(r) => FactTable::receiver_access_path(ir, *r),
487                _ => None,
488            };
489            if let Some(path) = receiver_path {
490                // Built without `format!`: the purity ratchet counts prose
491                // formatting in the engine, and this is a lookup key.
492                let full = path + "." + method;
493                if let Some(fact) = facts.sanitizer_fact_scoped(&full, origin_package) {
494                    return defeating(&full, fact).then_some(fact);
495                }
496            }
497            facts
498                .sanitizer_fact_scoped(method, origin_package)
499                .filter(|f| defeating(method, f))
500        }
501        _ => None,
502    }
503}

505/// Fact-table-aware sanitizer check: configured sanitizers PLUS any call with
506/// a [`SanitizerFact`] in the table (e.g. `.replace()`, `.test()` guard
507/// methods learned from `frensense-lang` tables or a bundle).
508///
509/// Fact lookups pass no import origin (this path serves the forward
510/// relevance engine and summaries, which carry no per-file import
511/// context), so package-scoped sanitizer facts are inert here, the
512/// recall-safe direction: universal sanitizers still cut, a bundle-
513/// scoped one never silences a flow it was not proven on.
514pub(crate) fn is_sanitizer_use_with_facts(
515    ir: &FunctionIR,
516    config: &TaintConfig,
517    facts: &FactTable,
518    key: &NodeKey,
519) -> bool {
520    if is_sanitizer_use(ir, config, key) {
521        return true;
522    }
523    let Some(instr) = key.instruction(ir) else {
524        return false;
525    };
526    match instr {
527        Instruction::CallStatic { func, .. } => facts.sanitizer_fact(func).is_some(),
528        Instruction::CallVirtual {
529            method, receiver, ..
530        } => {
531            if let Operand::Var(r) = receiver {
532                let path = FactTable::receiver_access_path(ir, *r);
533                if let Some(ref p) = path {
534                    let full = format!("{p}.{method}");
535                    if facts.sanitizer_fact(&full).is_some() {
536                        return true;
537                    }
538                }
539                if facts.is_session_path(method, path.as_deref()) {
540                    return true;
541                }
542            }
543            facts.sanitizer_fact(method).is_some()
544        }
545        _ => false,
546    }
547}

549/// A structured sink alert: which sink fired, in which function, in which
550/// argument slot, and how it classifies. Consumers render their own prose;
551/// the engine never formats messages.

555    pub sink: String,
556    /// Formal slot of the tainted argument (`usize::MAX` = receiver).
557    pub slot: usize,
558    /// Function in which the sink call lives.
559    pub function: String,
560    /// Shape classification (Idor for identity-payload query objects).
561    pub class: crate::analysis::taint::engine::FindingClass,
562}

564/// Slot a tainted value occupies when a property store names an armed sink.
565///
566/// `el.innerHTML = v` and a JSX attribute write taint into a field rather than
567/// into a call: the field name is the sink and the stored value is argument 0.
568fn store_field_sink<'a>(
569    ir: &FunctionIR,
570    config: &TaintConfig,
571    facts: &FactTable,
572    base: VarId,
573    field: &'a String,
574    origin_package: Option<&str>,
575    language: Option<&str>,
576) -> Option<(usize, &'a String)> {
577    let root = FactTable::receiver_root(ir, base);
578    facts
579        .sink_name_armed(
580            &config.sinks,
581            field,
582            root.as_deref(),
583            origin_package,
584            language,
585            true,
586        )
587        .then_some((0usize, field))
588}

590/// Whether a receiver-slot alert is really a source read rather than a sink.
591///
592/// A tainted receiver whose method name ends a configured source pattern
593/// (Go's `r.URL.Query().Get("id")`) reads taint instead of consuming it.
594fn receiver_reads_a_source(
595    ir: &FunctionIR,
596    config: &TaintConfig,
597    facts: &FactTable,
598    key: &NodeKey,
599    name: &str,
600) -> bool {
601    if is_source(ir, config, facts, key) {
602        return true;
603    }
604    SourceScope::for_ir(config, facts, ir)
605        .iter()
606        .any(|s| s.rsplit('.').next() == Some(name))
607}

621/// Alert if the node is a tainted use inside a configured sink call.
622/// Honours [`FactTable`] sink signatures when one matches: a tainted arg in a
623/// *non-dangerous* slot (e.g. a parameterized query's binding array) does not
624/// alert. Callers build the [`FactTable`] once and pass it in. Package-scoped
625/// sink facts additionally require `origin_package` to name their package
626/// (import resolution), pass `None` when no origin is known.
627pub(crate) fn sink_alert_with_facts(
628    ir: &FunctionIR,
629    config: &TaintConfig,
630    facts: &FactTable,
631    key: &NodeKey,
632    origin_package: Option<&str>,
633    language: Option<&str>,
634) -> Option<SinkAlert> {
635    let NodeKey {
636        block,
637        instr_idx: Some(idx),
638        var,
639    } = *key
640    else {
641        return None;
642    };
643    let b = ir.blocks.get(&block)?;
644    if idx >= b.instructions.len() {
645        return None;
646    }
647    let slot_and_name = match &b.instructions[idx] {
648        Instruction::CallStatic { func, args, .. }
649            if facts.sink_name_armed(
650                &config.sinks,
651                func,
652                None,
653                origin_package,
654                language,
655                false,
656            ) =>
657        {
658            args.iter()
659                .position(|a| *a == Operand::Var(var))
660                .map(|pos| (pos, func))
661        }
662        Instruction::CallVirtual {
663            method,
664            args,
665            receiver,
666            ..
667        } if {
668            let root = match receiver {
669                Operand::Var(r) => FactTable::receiver_root(ir, *r),
670                _ => None,
671            };
672            facts.sink_name_armed(
673                &config.sinks,
674                method,
675                root.as_deref(),
676                origin_package,
677                language,
678                true,
679            )
680        } =>
681        {
682            if receiver == &Operand::Var(var) {
683                Some((usize::MAX, method)) // receiver slot
684            } else {
685                args.iter()
686                    .position(|a| *a == Operand::Var(var))
687                    .map(|pos| (pos, method))
688            }
689        }
690        // A property assignment (`el.innerHTML = x`, a JSX attribute) is a
691        // sink the program writes to, not a call it makes.
692        Instruction::StoreField {
693            base,
694            field,
695            src: Operand::Var(sv),
696            ..
697        } if *sv == var => {
698            store_field_sink(ir, config, facts, *base, field, origin_package, language)
699        }
700        _ => None,
701    };
702    let (slot, name) = slot_and_name?;
703    // Arguments to a call on a prepared statement are bound parameters: the
704    // statement text was fixed when the handle was created, so `stmt.Exec(a,
705    // b)` carries values, not SQL. The creating call (`db.Prepare(query)`)
706    // is judged on its own.
707    if slot != usize::MAX && binds_prepared_parameters(ir, &b.instructions[idx], facts) {
708        return None;
709    }
710    // Source-accessor precedence (RECEIVER-TAINT ALERTS ONLY): when the
711    // alert comes from a tainted *receiver* and the method name is the last
712    // segment of a configured dotted source pattern, the call READS taint
713    // rather than consuming it, Go's canonical `r.URL.Query().Get("id")`
714    // (both `Query` and `Get` are also sink names) must not alert just
715    // because its receiver carries taint.
716    //
717    // This must NOT apply to argument-slot alerts: `req.query` is a source
718    // pattern, so a blanket rule would suppress `pool.query(sql)`, a real
719    // sink, because the sink name matches a source-pattern segment.
720    if slot == usize::MAX && receiver_reads_a_source(ir, config, facts, key, name) {
721        return None;
722    }
723    // Slot layout is a property of the CALLED API, not of the shared
724    // method name. A package-scoped fact (declared for the call's import
725    // origin) carries the correct slots; without this lookup the engine
726    // arms a call through its scoped fact (`is_sink_call_scoped`) but then
727    // applies whatever bare-name signature shares the method name. The
728    // Go case is concrete: `http.Get(url)` (slot 0) and `db.Get(&dest, sql)`
729    // (slot 1, sqlx) both lower to `Get`, and the bare entry silently
730    // dropped the SSRF finding. `sink_signature_scoped` falls back to the
731    // universal tables when no scoped entry applies, so behaviour for
732    // unresolved origins is unchanged.
733    let signature = facts.sink_signature_scoped(name, origin_package, language);
734    // Receiver-taint rule: for a slot-restricted sink (dangerous_args
735    // non-empty), a tainted RECEIVER is the object being read (the DB
736    // handle, the request object), not data being consumed, no alert.
737    // Only all-args sinks keep receiver taint dangerous.
738    if slot == usize::MAX
739        && let Some(sig) = signature
740        && !sig.dangerous_args.is_empty()
741    {
742        return None;
743    }
744    // Sink-signature check: dangerous slot (or receiver) only.
745    if let Some(sig) = signature
746        && slot != usize::MAX
747        && !sig.is_dangerous(slot)
748    {
749        return None; // safe binding channel, no alert
750    }
751    // Identity-payload emission policy: an IDOR-class finder sink (vocabulary
752    // from the default pack or a `.frc` bundle) reports ONLY
753    // when the tainted argument IS an identity payload - a top-level key from
754    // the identity set answers "which record". Leaf values inside
755    // parameterized clauses (`{ where: { id: taint } }`), non-identity object
756    // fields and bare scalars are not provable access-control violations and
757    // are not reported at all (zero-FP: structurally unprovable findings
758    // never fire). The signature must be the same language-scoped one the
759    // match used: reading the shared table here would miss a language-keyed
760    // IDOR fact (a TypeScript `update`) and emit every write it guards.
761    if let Some(sig) = signature
762        && slot != usize::MAX
763        && !sig.idor_keys.is_empty()
764        && !arg_is_idor_shape(ir, var, &sig.idor_keys)
765    {
766        return None;
767    }
768    // Arg-shape classification: a tainted identity payload at an
769    // IDOR-class sink is an access-control query, not an injection - the
770    // driver parameterizes object values. Reported with the Idor class so
771    // consumers rank it below Critical. (Non-identity shapes at these sinks
772    // never reach here - the emission gate above returned already.)
773    let class = if let Some(sig) = signature
774        && !sig.idor_keys.is_empty()
775        && slot != usize::MAX
776        && arg_is_idor_shape(ir, var, &sig.idor_keys)
777    {
778        // Auth-guarded handlers: an identity query behind an early-return
779        // branch over an auth-vocab call only executes for authenticated
780        // callers - not an access-control violation (Idor class only;
781        // injection findings are never suppressed here).
782        if idor_suppressed_by_auth_guard(ir, facts, block) {
783            return None;
784        }
785        crate::analysis::taint::engine::FindingClass::Idor
786    } else {
787        crate::analysis::taint::engine::FindingClass::Injection
788    };
789    Some(SinkAlert {
790        sink: name.to_string(),
791        slot,
792        function: ir.name.clone(),
793        class,
794    })
795}

797/// Does the tainted var originate from an object literal whose pair keys
798/// intersect `idor_keys`? Follows plain `Assign` chains (composite merge
799/// vars, destructuring temps) but stops at anything else.
800fn arg_is_idor_shape(ir: &FunctionIR, mut var: VarId, idor_keys: &[String]) -> bool {
801    for _ in 0..8 {
802        if let Some(meta) = ir.var_metadata.get(&var)
803            && meta
804                .object_keys
805                .iter()
806                .any(|k| idor_keys.iter().any(|ik| ik == k))
807        {
808            return true;
809        }
810        // Follow one Assign hop back (composite merge chain).
811        let mut next = None;
812        for b in ir.blocks.values() {
813            for instr in &b.instructions {
814                if let Instruction::Assign { dest, src } = instr
815                    && *dest == var
816                    && let Operand::Var(v) = src
817                {
818                    next = Some(*v);
819                }
820            }
821        }
822        match next {
823            Some(v) => var = v,
824            None => return false,
825        }
826    }
827    false
828}

834/// Dotted path of a virtual call: receiver chain + method
835/// (`security.authenticatedUsers.from`). Used by the auth-guard gate and by
836/// policy checks that need more than the last segment.
837pub(crate) fn receiver_call_path(ir: &FunctionIR, recv: &Operand, method: &str) -> String {
838    let base = operand_member_path(ir, recv);
839    if base.is_empty() {
840        method.to_string()
841    } else {
842        format!("{base}.{method}")
843    }
844}

846/// Resolve an operand to a member path through its defining instructions.
847fn operand_member_path(ir: &FunctionIR, op: &Operand) -> String {
848    operand_member_path_at(ir, op, 0)
849}
850
851/// [`operand_member_path`] continuing an existing depth budget.
852///
853/// The budget must travel through EVERY edge of the walk. `Assign` used to
854/// call the entry point above, which restarts at depth 0, so a cycle of
855/// assignments (`v_a := v_b`, `v_b := v_a` - Python's `try`/`except` lowering
856/// emits these) recursed between `var_member_path` and
857/// `operand_member_path` forever and overflowed the stack instead of
858/// returning the empty path the depth guard promises.
859fn operand_member_path_at(ir: &FunctionIR, op: &Operand, depth: usize) -> String {
860    match op {
861        Operand::Var(v) => var_member_path(ir, *v, depth),
862        _ => String::new(),
863    }
864}
865
866/// Walk `LoadField`/`LoadGlobal`/`Assign` definitions to rebuild the dotted
867/// path a receiver expression denotes (`security.authenticatedUsers`).
868fn var_member_path(ir: &FunctionIR, v: VarId, depth: usize) -> String {
869    if depth > 16 {
870        return String::new();
871    }
872    for b in ir.blocks.values() {
873        for instr in &b.instructions {
874            match instr {
875                Instruction::Assign { dest, src } if *dest == v => {
876                    return operand_member_path_at(ir, src, depth + 1);
877                }
878                Instruction::LoadField {
879                    dest, base, field, ..
880                } if *dest == v => {
881                    let p = var_member_path(ir, *base, depth + 1);
882                    return if p.is_empty() {
883                        field.clone()
884                    } else {
885                        format!("{p}.{field}")
886                    };
887                }
888                Instruction::LoadGlobal { dest, name, .. } if *dest == v => {
889                    return name.clone();
890                }
891                _ => {}
892            }
893        }
894    }
895    ir.var_metadata
896        .get(&v)
897        .and_then(|m| m.source_name.clone())
898        .unwrap_or_default()
899}

901/// Push the operand onto the worklist when it is a variable reference.
902fn push_var(work: &mut Vec<VarId>, op: &Operand) {
903    if let Operand::Var(s) = op {
904        work.push(*s);
905    }
906}

908/// True when `path` names an auth call per the spec/bundle vocabulary
909/// (`facts.auth_guard_hints`, seeded by the default pack).
910fn auth_path_matches(path: &str, facts: &FactTable) -> bool {
911    let lower = path.to_ascii_lowercase();
912    facts.auth_guard_hints.iter().any(|h| lower.contains(h))
913}

915/// Does the branch condition derive (through unary/binary/assign/await/
916/// field chains) from a call to an auth-vocabulary callee?
917fn cond_derives_from_auth_call(ir: &FunctionIR, cond: &Operand, facts: &FactTable) -> bool {
918    let Operand::Var(start) = cond else {
919        return false;
920    };
921    let mut work = vec![*start];
922    let mut seen = FxHashSet::default();
923    while let Some(v) = work.pop() {
924        if !seen.insert(v) || seen.len() > 64 {
925            continue;
926        }
927        let Some(instr) = ir.def_instruction(v) else {
928            continue;
929        };
930        match instr {
931            Instruction::CallStatic { func, dest, .. }
932                if *dest == Some(v) && auth_path_matches(func, facts) =>
933            {
934                return true;
935            }
936            Instruction::CallVirtual {
937                method,
938                receiver,
939                dest,
940                ..
941            } if *dest == Some(v) => {
942                let path = receiver_call_path(ir, receiver, method);
943                if auth_path_matches(&path, facts) {
944                    return true;
945                }
946            }
947            Instruction::UnaryOp { dest, src, .. } if *dest == v => push_var(&mut work, src),
948            Instruction::BinaryOp { dest, lhs, rhs, .. } if *dest == v => {
949                push_var(&mut work, lhs);
950                push_var(&mut work, rhs);
951            }
952            Instruction::Assign { dest, src } if *dest == v => push_var(&mut work, src),
953            Instruction::Cast { dest, src, .. } if *dest == v => push_var(&mut work, src),
954            Instruction::Await { dest, promise, .. } if *dest == v => push_var(&mut work, promise),
955            Instruction::LoadField { dest, base, .. } if *dest == v => {
956                work.push(*base);
957            }
958            _ => {}
959        }
960    }
961    false
962}

964/// Can `start` reach a Return/Throw without ever entering `avoid`?
965fn exit_reachable_avoiding(ir: &FunctionIR, start: BlockId, avoid: BlockId) -> bool {
966    if start == avoid {
967        return false;
968    }
969    let mut stack = vec![start];
970    let mut seen = FxHashSet::default();
971    while let Some(b) = stack.pop() {
972        if b == avoid || !seen.insert(b) {
973            continue;
974        }
975        let Some(block) = ir.blocks.get(&b) else {
976            continue;
977        };
978        match &block.terminator {
979            Terminator::Return { .. } | Terminator::Throw { .. } => return true,
980            Terminator::Jump(t) => stack.push(*t),
981            Terminator::Branch {
982                true_block,
983                false_block,
984                ..
985            } => {
986                stack.push(*true_block);
987                stack.push(*false_block);
988            }
989            Terminator::Switch {
990                cases,
991                default_block,
992                ..
993            } => {
994                for (_, t) in cases {
995                    stack.push(*t);
996                }
997                stack.push(*default_block);
998            }
999            _ => {}
1000        }
1001    }
1002    false
1003}

1005/// Suppress an IDOR-class candidate at `sink_block` when the function has
1006/// an *auth guard*: a branch whose condition derives from an
1007/// auth-vocabulary call, which dominates the sink (the query is only
1008/// reachable through the check) and which has a rejecting path
1009/// (Return/Throw) that never reaches the sink. Vocabulary:
1010/// `facts.auth_guard_hints`.
1011pub(crate) fn idor_suppressed_by_auth_guard(
1012    ir: &FunctionIR,
1013    facts: &FactTable,
1014    sink_block: BlockId,
1015) -> bool {
1016    let doms = crate::ir::control::dominators(ir);
1017    for (bid, block) in &ir.blocks {
1018        let Terminator::Branch {
1019            cond,
1020            true_block,
1021            false_block,
1022        } = &block.terminator
1023        else {
1024            continue;
1025        };
1026        // The guard must be a distinct block: a branch in the sink's own
1027        // block comes after the sink instruction (terminators end blocks),
1028        // so it cannot gate it.
1029        if *bid == sink_block {
1030            continue;
1031        }
1032        if !cond_derives_from_auth_call(ir, cond, facts) {
1033            continue;
1034        }
1035        // `dominators[sink]` = blocks that dominate the sink: the guard
1036        // block must be one of them (reflexive sets, so the `bid ==
1037        // sink_block` case above already excluded).
1038        let Some(guard_doms) = doms.get(&sink_block) else {
1039            continue;
1040        };
1041        if !guard_doms.contains(bid) {
1042            continue;
1043        }
1044        if exit_reachable_avoiding(ir, *true_block, sink_block)
1045            || exit_reachable_avoiding(ir, *false_block, sink_block)
1046        {
1047            return true;
1048        }
1049    }
1050    false
1051}

1053/// True when `var` is (an alias or tuple field of) the result of a call the
1054/// pack declares a statement factory: the handle then executes text fixed at
1055/// creation, and its call arguments are bindings.
1056///
1057/// Which names prepare a statement is pack knowledge
1058/// ([`FactTable::prepared_statement_prefixes`], taught by the default pack or
1059/// a bundle); the engine only walks the def chain and matches the prefix. An
1060/// empty set recognises nothing, so a config that teaches nothing never
1061/// suppresses a flow.
1062fn receiver_is_prepared_handle(
1063    ir: &FunctionIR,
1064    var: crate::ir::function::VarId,
1065    facts: &FactTable,
1066) -> bool {
1067    let prefixes = &facts.prepared_statement_prefixes;
1068    if prefixes.is_empty() {
1069        return false;
1070    }
1071    let mut cur = var;
1072    for _ in 0..6 {
1073        let mut next = None;
1074        for blk in ir.blocks.values() {
1075            for instr in &blk.instructions {
1076                match instr {
1077                    Instruction::Assign {
1078                        dest,
1079                        src: Operand::Var(s),
1080                        ..
1081                    } if *dest == cur => next = Some(*s),
1082                    Instruction::ExtractValue {
1083                        dest,
1084                        tuple: Operand::Var(t),
1085                        ..
1086                    } if *dest == cur => next = Some(*t),
1087                    Instruction::CallVirtual {
1088                        dest: Some(d),
1089                        method,
1090                        ..
1091                    }
1092                    | Instruction::CallStatic {
1093                        dest: Some(d),
1094                        func: method,
1095                        ..
1096                    } if *d == cur => {
1097                        let seg = method.rsplit('.').next().unwrap_or(method);
1098                        let seg = seg.to_ascii_lowercase();
1099                        return prefixes.iter().any(|p| seg.starts_with(p.as_str()));
1100                    }
1101                    _ => {}
1102                }
1103            }
1104        }
1105        match next {
1106            Some(n) => cur = n,
1107            None => return false,
1108        }
1109    }
1110    false
