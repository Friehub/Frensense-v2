7/// Deterministic edge order (call-site key, then node position) with
8/// duplicate edges dropped; every edge-install pass ends with this.
9fn sort_dedup_edges(edges: &mut Vec<(usize, NodeKey)>) {
10    edges.sort_by_key(|(f, k)| (*f, k.block.0, k.instr_idx, k.var.0));
11    edges.dedup();
12}

19    pub(super) fn install_cross_edges(&mut self, facts: &FactTable) {
20        for fi in 0..self.functions.len() {
21            let bindings = self.functions[fi].bindings.clone();
22            for b in &bindings {
23                for &gi in &b.callees {
24                    let ge = &self.functions[gi];
25
26                    let callee_has_recv = ge.ir.parameters.first().is_some_and(|&p| {
27                        ge.ir
28                            .var_metadata
29                            .get(&p)
30                            .and_then(|m| m.source_name.as_deref())
31                            .is_some_and(|name| facts.is_receiver_param(name))
32                    });
33
34                    // Receiver passing: only if callee has a formal receiver parameter.
35                    if callee_has_recv
36                        && let Some(r_key) = b.receiver
37                        && let Some(&param_var) = ge.ir.parameters.first()
38                        && let Some(&param_key) = ge.def_site.get(&param_var)
39                    {
40                        self.cross_edges
41                            .entry((fi, r_key))
42                            .or_default()
43                            .push((gi, param_key));
44                    }
45
46                    // Parameter passing: ActualArg(use node) → FormalParam(slot).
47                    let slot_offset = if callee_has_recv { 1 } else { 0 };
48                    for (arg_key, slot) in &b.args {
49                        let target_slot = *slot + slot_offset;
50                        if let Some(&param_var) = ge.ir.parameters.get(target_slot)
51                            && let Some(&param_key) = ge.def_site.get(&param_var)
52                        {
53                            self.cross_edges
54                                .entry((fi, *arg_key))
55                                .or_default()
56                                .push((gi, param_key));
57                        }
58                    }
59
60                    // Return passing: every FormalRet node of the callee feeds the
61                    // caller's ActualRet def node. Keyed by the CALLEE (the from-node
62                    // lives there) so the map always stores edges in value-flow
63                    // direction: FormalRet → ActualRet.
64                    if let Some(ret) = b.ret_node {
65                        let mut formal_rets: Vec<NodeKey> = ge
66                            .svfg
67                            .nodes
68                            .values()
69                            .filter(|n| n.kind == NodeKind::FormalRet)
70                            .map(|n| n.key)
71                            .collect();
72                        formal_rets.sort_by_key(|k| (k.block.0, k.instr_idx, k.var.0));
73                        for fr in formal_rets {
74                            self.cross_edges
75                                .entry((gi, fr))
76                                .or_default()
77                                .push((fi, ret));
78                        }
79                    }
80                }
81            }
82        }
83
84        // Deterministic edge lists (sorted + deduped).
85        for edges in self.cross_edges.values_mut() {
86            sort_dedup_edges(edges);
87        }
88    }

90    /// Closure (captured free-variable) edges. A function extracted from a
91    /// nested position reads variables its own IR never defines; the value
92    /// lives in the innermost enclosing extracted function that DOES define
93    /// them, resolved through [`FunctionIR::enclosing_fn`]. The module-scope
94    /// boundary ends the walk - no analysed IR owns those names (matching
95    /// the pre-restructure behavior where inlining resolved captures only
96    /// inside a function's own IR).
97    ///
98    /// Value-flow direction `def(E, name) ──▶ use(F, v)`, keyed by the
99    /// DEFINING node (the map's convention: key = value source, as with
100    /// `FormalRet → ActualRet`), so the backward taint walk crosses into the
101    /// defining scope at each free use node - including multi-block free
102    /// vars (per-scope `VarId`s) and terminator uses, since every node whose
103    /// var has no def site is covered.
104    pub(super) fn install_closure_edges(&mut self) {
105        for fi in 0..self.functions.len() {
106            // Free use nodes grouped by source name: a node whose var has no
107            // definition (or formal parameter) in this function.
108            let mut free_by_name: FxHashMap<String, Vec<NodeKey>> = FxHashMap::default();
109            {
110                let fe = &self.functions[fi];
111                for &key in fe.svfg.nodes.keys() {
112                    if fe.def_site.contains_key(&key.var) {
113                        continue;
114                    }
115                    let Some(meta) = fe.ir.var_metadata.get(&key.var) else {
116                        continue;
117                    };
118                    if meta.is_memory_state {
119                        continue;
120                    }
121                    let Some(name) = meta.source_name.as_deref() else {
122                        continue;
123                    };
124                    if name.is_empty() {
125                        continue;
126                    }
127                    free_by_name.entry(name.to_string()).or_default().push(key);
128                }
129            }
130            if free_by_name.is_empty() {
131                continue;
132            }
133
134            for (name, mut use_nodes) in free_by_name {
135                // Walk the lexical chain to the first enclosing function
136                // that actually defines `name`.
137                let mut target: Option<usize> = None;
138                let mut cur = self.functions[fi].ir.enclosing_fn.clone();
139                let mut hops = 0;
140                let mut truncated = false;
141                while let Some(enc) = cur {
142                    if hops >= 32 {
143                        // Ran out of lexical chain with an enclosing
144                        // function still in hand: the name is unresolved.
145                        truncated = true;
146                        break;
147                    }
148                    hops += 1;
149                    let Some(&gi) = self.func_index.get(&enc) else {
150                        break;
151                    };
152                    if self.defines_name(gi, &name) {
153                        target = Some(gi);
154                        break;
155                    }
156                    cur = self.functions[gi].ir.enclosing_fn.clone();
157                }
158                if truncated {
159                    crate::telemetry::Cap::ForwardHops.record();
160                }
161                let Some(gi) = target else {
162                    continue; // module scope: no analysed IR owns the name
163                };
164
165                // All def nodes of `name` in the target (params, assignments,
166                // phis): conservative may-reach, like any other cross edge.
167                let mut def_nodes: Vec<NodeKey> = Vec::new();
168                for (&var, &dkey) in &self.functions[gi].def_site {
169                    if self.functions[gi]
170                        .ir
171                        .var_metadata
172                        .get(&var)
173                        .is_some_and(|m| m.source_name.as_deref() == Some(name.as_str()))
174                    {
175                        def_nodes.push(dkey);
176                    }
177                }
178                if def_nodes.is_empty() {
179                    continue;
180                }
181                def_nodes.sort_by_key(|k| (k.block.0, k.instr_idx, k.var.0));
182                def_nodes.dedup();
183                use_nodes.sort_by_key(|k| (k.block.0, k.instr_idx, k.var.0));
184                use_nodes.dedup();
185
186                for u in use_nodes {
187                    for &d in &def_nodes {
188                        // Key = value source (the def in the defining
189                        // scope), so reverse lookup at the free use node
190                        // finds its feeders.
191                        self.cross_edges.entry((gi, d)).or_default().push((fi, u));
192                    }
193                }
194            }
195        }
196
197        // Re-establish determinism over the whole map (closure edges join
198        // the parameter/return/heap edges).
199        for edges in self.cross_edges.values_mut() {
200            sort_dedup_edges(edges);
201        }
202    }
203}

210    /// Install heap value-flow edges *across* call boundaries:
211    ///
212    /// ```text
213    ///   caller: obj.data = src          callee: return p.data
214    ///           call f(obj)      ⟶            (load of p.data)
215    /// ```
216    ///
217    /// Scalar parameter/return edges (Step 4) carry direct values, but a
218    /// field *written into an object* before a call and *read from it*
219    /// (before or during the call) has no edge, the classic missed bug
220    /// `obj.data = source(); sink(obj.data)` split across two functions.
221    ///
222    /// Two directions are installed per call site, per matching field:
223    ///
224    ///  1. **caller-store → callee-load**: a `StoreField` in the caller whose
225    ///     base is the same object as an actual argument feeds every
226    ///     `LoadField` in the callee whose base is that callee parameter.
227    ///  2. **callee-store → caller-load**: a `StoreField` in the callee on
228    ///     its parameter object feeds every `LoadField` in the caller whose
229    ///     base is the corresponding actual argument (covers callee-side
230    ///     mutation the caller reads afterwards).
231    ///  3. **callee-store → sibling-load**: two callees that receive the
232    ///     *same argument object* (same arg variable at the same call-site
233    ///     root) are linked: a store in one feeds loads of the same field in
234    ///     the other. This is the write-then-read pipeline,
235    ///     `handler(obj, input)` calling `writer(obj, input)` then
236    ///     `reader(obj)`, where neither the caller nor either callee alone
237    ///     completes the flow.
238    ///
239    /// Object identity is anchored on SSA roots (`object_roots`): parameters
240    /// and allocation sites. A field matches on exact name, or when either
241    /// side is the element wildcard `"*"`. Edges enter the regular
242    /// `cross_edges` map, so both the forward engine and the demand-driven
243    /// backward engine (via its reverse index) traverse them unchanged.
244    pub(super) fn install_heap_cross_edges(&mut self) {
245        // Per-function caches, computed lazily.
246        let mut roots: Vec<Option<FxHashMap<VarId, ObjectRoot>>> = vec![None; self.functions.len()];
247        let mut field_ops: Vec<Option<(Vec<FieldStore>, Vec<FieldLoad>)>> =
248            vec![None; self.functions.len()];
249
250        let mut new_edges: Vec<((usize, NodeKey), (usize, NodeKey))> = Vec::new();
251
252        for fi in 0..self.functions.len() {
253            let (c_stores, c_loads) = {
254                if field_ops[fi].is_none() {
255                    field_ops[fi] = Some(collect_field_ops(self.functions[fi].ir));
256                }
257                field_ops[fi].as_ref().unwrap().clone()
258            };
259            // Sibling heap edges (direction 3) need only the callees' field
260            // ops, so an empty caller set must not skip the whole loop:
261            // directions 1-2 iterate empty vectors harmlessly.
262            if self.functions[fi].bindings.is_empty() {
263                continue;
264            }
265            let bindings = self.functions[fi].bindings.clone();
266            for b in &bindings {
267                for &gi in &b.callees {
268                    let (g_stores, g_loads) = {
269                        if field_ops[gi].is_none() {
270                            field_ops[gi] = Some(collect_field_ops(self.functions[gi].ir));
271                        }
272                        field_ops[gi].as_ref().unwrap().clone()
273                    };
274
275                    for (arg_key, slot) in &b.args {
276                        let arg_var = arg_key.var;
277                        self.ensure_roots(&mut roots, fi);
278                        let arg_root = match roots[fi].as_ref().unwrap().get(&arg_var).copied() {
279                            Some(r) => r,
280                            None => continue,
281                        };
282                        let param_root = ObjectRoot::Param(*slot);
283
284                        // 1. caller store → callee load.
285                        for st in &c_stores {
286                            if roots[fi].as_ref().unwrap().get(&st.base).copied() != Some(arg_root)
287                            {
288                                continue;
289                            }
290                            for ld in &g_loads {
291                                self.ensure_roots(&mut roots, gi);
292                                if roots[gi].as_ref().unwrap().get(&ld.base).copied()
293                                    != Some(param_root)
294                                {
295                                    continue;
296                                }
297                                if !fields_match(st.field.as_str(), ld.field.as_str()) {
298                                    continue;
299                                }
300                                new_edges.push(((fi, st.src_use), (gi, ld.dest)));
301                            }
302                        }
303
304                        // 2. callee store → caller load.
305                        for st in &g_stores {
306                            self.ensure_roots(&mut roots, gi);
307                            if roots[gi].as_ref().unwrap().get(&st.base).copied()
308                                != Some(param_root)
309                            {
310                                continue;
311                            }
312                            for ld in &c_loads {
313                                if roots[fi].as_ref().unwrap().get(&ld.base).copied()
314                                    != Some(arg_root)
315                                {
316                                    continue;
317                                }
318                                if !fields_match(st.field.as_str(), ld.field.as_str()) {
319                                    continue;
320                                }
321                                new_edges.push(((gi, st.src_use), (fi, ld.dest)));
322                            }
323                        }
324
325                        // 3. callee store → sibling callee load. Every OTHER
326                        // binding of this call site whose arg at the SAME
327                        // slot carries the same object root receives the
328                        // stores this callee makes on that parameter object.
329                        // Pairwise across all bindings of `fi`, so both
330                        // directions emerge from the (writer, reader) loop.
331                        for bj in self.functions[fi].bindings.iter() {
332                            if bj.call_site == b.call_site {
333                                continue; // same binding: directions 1+2 cover it
334                            }
335                            let Some(&(_, sj)) = bj
336                                .args
337                                .iter()
338                                .find(|(k, s)| *s == *slot && k.var == arg_var)
339                            else {
340                                continue; // sibling does not bind this object here
341                            };
342                            for &gj in &bj.callees {
343                                if gj == gi {
344                                    continue;
345                                }
346                                if field_ops[gj].is_none() {
347                                    field_ops[gj] = Some(collect_field_ops(self.functions[gj].ir));
348                                }
349                                self.ensure_roots(&mut roots, gj);
350                                let sibling_root = ObjectRoot::Param(sj);
351                                for st in &g_stores {
352                                    if roots[gi].as_ref().unwrap().get(&st.base).copied()
353                                        != Some(param_root)
354                                    {
355                                        continue;
356                                    }
357                                    for ld in &field_ops[gj].as_ref().unwrap().1 {
358                                        if roots[gj].as_ref().unwrap().get(&ld.base).copied()
359                                            != Some(sibling_root)
360                                        {
361                                            continue;
362                                        }
363                                        if !fields_match(st.field.as_str(), ld.field.as_str()) {
364                                            continue;
365                                        }
366                                        new_edges.push(((gi, st.src_use), (gj, ld.dest)));
367                                    }
368                                }
369                            }
370                        }
371                    }
372                }
373            }
374        }
375
376        for ((ff, fk), (tf, tk)) in new_edges {
377            self.cross_edges.entry((ff, fk)).or_default().push((tf, tk));
378        }
379
380        // Re-establish determinism over the whole map (heap edges join the
381        // parameter/return edges installed by Step 4).
382        for edges in self.cross_edges.values_mut() {
383            sort_dedup_edges(edges);
384        }
385    }
386
387    /// Fill the object-root cache of function `fi` on first use.
388    fn ensure_roots(&self, roots: &mut [Option<FxHashMap<VarId, ObjectRoot>>], fi: usize) {
389        if roots[fi].is_none() {
390            roots[fi] = Some(object_roots(self.functions[fi].ir));
391        }
392    }
393}

399/// SSA-anchored identity of an object: where it came from.
400#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
401pub(crate) enum ObjectRoot {
402    /// A formal parameter (slot order = parameter index).
403    Param(usize),
404    /// An `Allocate` instruction (block, idx), a locally created object.
405    Alloc(BlockId, usize),
406    /// The initial/global memory state (module-level objects).
407    Global,
408}
409
410/// A field write: `base.field = src` (or `base[field] = src`).
411#[derive(Debug, Clone)]
412pub(crate) struct FieldStore {
413    pub base: VarId,
414    pub field: String,
415    /// The SVFG *use* node of the stored value inside the store instruction,
416    /// the point where value-flow leaves the writer.
417    pub src_use: NodeKey,
418}
419
420/// A field read: `dest = base.field` (or `dest = base[field]`).
421#[derive(Debug, Clone)]
422pub(crate) struct FieldLoad {
423    pub base: VarId,
424    pub field: String,
425    /// The SVFG *def* node of the destination, the point where value-flow
426    /// enters the reader.
427    pub dest: NodeKey,
428}
429
430/// Do two field names denote the same heap slot? Exact match, or either side
431/// is the dynamic-element wildcard `"*"` (an unknown index can hit any slot).
432pub(crate) fn fields_match(a: &str, b: &str) -> bool {
433    a == b || a == "*" || b == "*"
434}
435
436/// Map every variable to the SSA root of the object it denotes (parameter,
437/// allocation site, or global state). Follows `Assign`/`Cast` chains; nothing
438/// else propagates object identity (a `LoadField` result is a *field value*,
439/// not the object itself, treating it as such would be unsound in the other
440/// direction only for receiver-chaining, which `member_access_path` already
441/// handles for sources).
442pub(crate) fn object_roots(ir: &FunctionIR) -> FxHashMap<VarId, ObjectRoot> {
443    let mut map: FxHashMap<VarId, ObjectRoot> = FxHashMap::default();
444
445    for (slot, &p) in ir.parameters.iter().enumerate() {
446        map.insert(p, ObjectRoot::Param(slot));
447    }
448    map.insert(ir.initial_memory_state, ObjectRoot::Global);
449
450    let mut block_ids: Vec<BlockId> = ir.blocks.keys().copied().collect();
451    block_ids.sort_by_key(|b| b.0);
452    for &b in &block_ids {
453        let blk = &ir.blocks[&b];
454        for (idx, instr) in blk.instructions.iter().enumerate() {
455            match instr {
456                Instruction::Allocate { dest, .. } => {
457                    map.insert(*dest, ObjectRoot::Alloc(b, idx));
458                }
459                Instruction::Assign {
460                    dest,
461                    src: Operand::Var(s),
462                }
463                | Instruction::Cast {
464                    dest,
465                    src: Operand::Var(s),
466                    ..
467                } => {
468                    if let Some(r) = map.get(s).copied() {
469                        // First writer wins (SSA: one def per var; multiple
470                        // entries can only come from the parameter seeding,
471                        // which is authoritative).
472                        map.entry(*dest).or_insert(r);
473                    }
474                }
475                // An object literal `{}` / `[]` lowers to `dest = Unknown`
476                // (or a literal) with no Allocate, it is still a *fresh*
477                // object. Anchor it as its own allocation site so field
478                // stores into it participate in heap flow.
479                Instruction::Assign {
480                    dest,
481                    src: Operand::Unknown,
482                }
483                | Instruction::Assign {
484                    dest,
485                    src: Operand::Null,
486                }
487                | Instruction::Assign {
488                    dest,
489                    src: Operand::StringLiteral(_),
490                }
491                | Instruction::Assign {
492                    dest,
493                    src: Operand::IntLiteral(_),
494                }
495                | Instruction::Assign {
496                    dest,
497                    src: Operand::BoolLiteral(_),
498                } => {
499                    map.entry(*dest).or_insert(ObjectRoot::Alloc(b, idx));
500                }
501                _ => {}
502            }
503        }
504    }
505    map
506}

508/// Collect all field stores/loads of a function with their SVFG attachment
509/// points (store: the src *use* node; load: the dest *def* node).
510pub(crate) fn collect_field_ops(ir: &FunctionIR) -> (Vec<FieldStore>, Vec<FieldLoad>) {
511    let mut stores = Vec::new();
512    let mut loads = Vec::new();
513
514    let mut block_ids: Vec<BlockId> = ir.blocks.keys().copied().collect();
515    block_ids.sort_by_key(|b| b.0);
516    for &b in &block_ids {
517        let blk = &ir.blocks[&b];
518        for (idx, instr) in blk.instructions.iter().enumerate() {
519            match instr {
520                Instruction::StoreField {
521                    base,
522                    field,
523                    src: Operand::Var(sv),
524                    ..
525                } => {
526                    stores.push(FieldStore {
527                        base: *base,
528                        field: field.clone(),
529                        src_use: NodeKey::instr(b, idx, *sv),
530                    });
531                }
532                Instruction::StoreElement {
533                    base, index, src, ..
534                } => {
535                    if let (Operand::Var(sv), Some(field)) = (src, elem_field(index)) {
536                        stores.push(FieldStore {
537                            base: *base,
538                            field,
539                            src_use: NodeKey::instr(b, idx, *sv),
540                        });
541                    }
542                }
543                Instruction::LoadField {
544                    dest, base, field, ..
545                } => {
546                    loads.push(FieldLoad {
547                        base: *base,
548                        field: field.clone(),
549                        dest: NodeKey::instr(b, idx, *dest),
550                    });
551                }
552                Instruction::LoadElement {
553                    dest, base, index, ..
554                } => {
555                    if let Some(field) = elem_field(index) {
556                        loads.push(FieldLoad {
557                            base: *base,
558                            field,
559                            dest: NodeKey::instr(b, idx, *dest),
560                        });
561                    }
562                }
563                _ => {}
564            }
565        }
566    }
567    (stores, loads)
568}

570/// Field name for an element index operand: string literal verbatim,
571/// anything dynamic is the wildcard.
572fn elem_field(index: &Operand) -> Option<String> {
573    Some(match index {
574        Operand::StringLiteral(s) => s.clone(),
575        _ => "*".to_string(),
576    })
