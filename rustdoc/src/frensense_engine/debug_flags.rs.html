5//! Debug-tracing flags, installed once per process by the binary.
6//!
7//! Analysis hot loops (BFS over the value-flow graph) consult these flags
8//! per node; [`DebugFlags::get`] is a struct copy (no env lookup, no lock
9//! on the read path). The engine never reads the process environment: the
10//! binary that owns the process calls [`DebugFlags::install_from`] at
11//! startup with its own environment lookup.
12//!
13//! Flags (all off by default):
14//! * `FRENSdbg_WALK`      - trace every BFS node visit in the backward engine
15//! * `FRENSdbg_DEADEND`   - trace dead-end classification decisions
16//! * `FRENSdbg_UNRES`     - trace unresolvable-node reasoning
17//! * `FRENSdbg_IS_SOURCE` - trace source-matching in the forward engine
18//! * `FRENSdbg_STATIC`    - harness-level static-analysis trace
19//! * `FRENSdbg_PTS`       - points-to fixed-point cost (passes, set sizes)
20//! * `FRENSdbg_MEM`       - phase boundary RSS + graph sizes (memory attribution)
21//! * `FRENSdbg_CAPS`      - announce each safety-cap firing once per scan
22//! * `FRENSdbg_CHECKS`    - announce learned-check path divergences
23
24use std::ffi::OsString;
25use std::sync::OnceLock;
26
27/// Cached per-process debug flags.
28#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
29pub struct DebugFlags {
30    /// `FRENSdbg_WALK`: trace every node visit in the backward engine.
31    pub walk: bool,
32    /// `FRENSdbg_DEADEND`: trace dead-end classification decisions.
33    pub deadend: bool,
34    /// `FRENSdbg_UNRES`: trace unresolvable-node reasoning.
35    pub unres: bool,
36    /// `FRENSdbg_IS_SOURCE`: trace source matching in the forward engine.
37    pub is_source: bool,
38    /// `FRENSdbg_STATIC`: harness-level static-analysis trace.
39    pub static_trace: bool,
40    /// `FRENSdbg_PTS`: points-to fixed-point cost (passes, set sizes).
41    pub pts: bool,
42    /// `FRENSdbg_MEM`: phase-boundary resident memory and graph sizes.
43    pub mem: bool,
44}
45
46static FLAGS: OnceLock<DebugFlags> = OnceLock::new();
47
48/// `FRENSdbg_CAPS`, kept out of [`DebugFlags`] so adding the flag does not
49/// change the shape of that public struct (an additive method instead).
50static CAPS: OnceLock<bool> = OnceLock::new();
51
52/// `FRENSdbg_CHECKS`, kept out of [`DebugFlags`] for the same reason.
53static CHECKS: OnceLock<bool> = OnceLock::new();
54
55impl DebugFlags {
56    /// Install the process-wide flags by querying the caller's lookup for
57    /// each `FRENSdbg_*` name. The first install wins; once [`Self::get`]
58    /// has cached a value (any earlier call), later installs are ignored -
59    /// binaries call this as the first statement of `main`.
60    pub fn install_from(lookup: impl Fn(&'static str) -> Option<OsString>) {
61        let flags = DebugFlags {
62            walk: lookup("FRENSdbg_WALK").is_some(),
63            deadend: lookup("FRENSdbg_DEADEND").is_some(),
64            unres: lookup("FRENSdbg_UNRES").is_some(),
65            is_source: lookup("FRENSdbg_IS_SOURCE").is_some(),
66            static_trace: lookup("FRENSdbg_STATIC").is_some(),
67            pts: lookup("FRENSdbg_PTS").is_some(),
68            mem: lookup("FRENSdbg_MEM").is_some(),
69        };
70        // The separate flags follow the same first-install-wins rule: do not
71        // set them when `get()` already initialized the shared flags.
72        if FLAGS.set(flags).is_ok() {
73            let _ = CAPS.set(lookup("FRENSdbg_CAPS").is_some());
74            let _ = CHECKS.set(lookup("FRENSdbg_CHECKS").is_some());
75        }
76    }
77
78    /// `FRENSdbg_CAPS`: announce each safety-cap firing once per scan.
79    ///
80    /// Kept separate from [`DebugFlags::get`] so the public struct keeps its
81    /// shape; like the other flags, the first [`Self::install_from`] wins.
82    #[must_use]
83    pub(crate) fn caps_enabled() -> bool {
84        *CAPS.get_or_init(|| false)
85    }
86
87    /// `FRENSdbg_CHECKS`: report findings that only one of the two
88    /// learned-check evaluation paths produced.
89    #[must_use]
90    pub(crate) fn checks_enabled() -> bool {
91        *CHECKS.get_or_init(|| false)
92    }
93
94    /// The installed flags (all-off default until [`Self::install_from`]
95    /// runs); subsequent calls are a cheap struct copy.
96    pub fn get() -> Self {
97        *FLAGS.get_or_init(DebugFlags::default)
98    }
99}

101/// Resident set size of the current process, in KiB (0 when unreadable).
102///
103/// Used only by the `FRENSdbg_MEM` phase probes: memory attribution for a
104/// whole-program scan has to come from the process itself, since no external
105/// profiler can attach in a restricted-ptrace environment.
106#[must_use]
107pub fn rss_kb() -> u64 {
108    std::fs::read_to_string("/proc/self/status")
109        .ok()
110        .and_then(|s| {
111            s.lines()
112                .find(|l| l.starts_with("VmRSS:"))
113                .and_then(|l| l.split_whitespace().nth(1))
114                .and_then(|p| p.parse::<u64>().ok())
115        })
116        .unwrap_or(0)
117}

119/// Trace helper: `dbg_trace!(flags.walk, "fmt args...")`.
120///
121/// The format arguments are only evaluated when the flag is on, so building
122/// trace strings costs nothing when tracing is off.
123#[macro_export]
124macro_rules! dbg_trace {
125    ($on:expr, $($arg:tt)*) => {
126        if $on {
127            eprintln!($($arg)*);
128        }
129    };
130}
