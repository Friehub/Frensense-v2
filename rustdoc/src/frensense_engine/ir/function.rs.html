5//! Phase 1: Intermediate Representation (IR)
6//!
7//! This module defines the 3-Address Code and Control Flow Graph structures.
8//! It serves as the foundation for modern SSA-based dataflow analysis across
9//! dynamic, object-oriented, and systems languages.
10//!
11//! NEW: Incorporates Memory SSA concepts. The heap is no longer a global state.
12//! Instead, memory mutations generate a new `MemoryState` (represented as a VarId),
13//! which perfectly tracks flow-sensitive object modifications over time.
14

44/// Metadata mapping a VarId back to its original source code context.
45#[derive(Debug, Clone)]
46#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
47pub struct VarMetadata {
48    /// Name of the variable as written in the source, when it has one. `None` for compiler temporaries.
49    pub source_name: Option<String>,
50    /// Declared or inferred type name from the source, when the language provides one.
51    pub type_name: Option<String>,
52    /// Byte range `(start, end)` of the defining source text, used to map findings back to source.
53    pub byte_range: Option<(usize, usize)>,
54    /// Flags if this VarId represents a hidden Memory State rather than a real code variable
55    pub is_memory_state: bool,
56    /// For a var produced by an object-literal composite: the keys whose
57    /// values were merged into it (e.g. `["where", "message"]`). Sinks use
58    /// this to classify the *shape* of the tainted argument (IDOR-style
59    /// object payload vs raw injection string).
60    pub object_keys: Vec<String>,
61    /// Bound by a declaration visited while lowering this function's body
62    /// (`char *p;`, `let x;`, `x = v` in Python, destructuring patterns) -
63    /// provably function-scoped. Vars created at first *expression* use are
64    /// left `false`: those are file/module/header globals (or undeclared
65    /// names), whose values are initialized elsewhere. Uninitialized-free
66    /// analysis fires only on `declared` vars.
67    #[cfg_attr(feature = "serialize", serde(default))]
68    pub declared: bool,
69}

101/// A single linear instruction in 3-Address Code form.
102#[derive(Debug, Clone)]
103#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
104pub enum Instruction {
105    /// v_dest = v_src
106    Assign {
107        /// Variable being defined.
108        dest: VarId,
109        /// Value assigned to `dest`.
110        src: Operand,
111    },
112
113    // --- Object & Array Memory Access (Memory SSA Enabled) ---
114    // Reads depend on a specific moment in time (mem_in).
115    // Writes consume a moment in time, and generate a new one (mem_out).
116    /// v_dest = base.field  @ mem_in
117    LoadField {
118        /// Variable that receives the loaded value.
119        dest: VarId,
120        /// Memory state the load reads from.
121        mem_in: VarId,
122        /// Variable holding the object whose field is read.
123        base: VarId,
124        /// Name of the field being read.
125        field: String,
126    },
127
128    /// base.field = src  -> generates mem_out
129    StoreField {
130        /// New memory state produced by the store.
131        mem_out: VarId,
132        /// Memory state the store applies to.
133        mem_in: VarId,
134        /// Variable holding the object being written.
135        base: VarId,
136        /// Name of the field being written.
137        field: String,
138        /// Value stored into the field.
139        src: Operand,
140        /// Byte range of the expression that performs the write (the
141        /// assignment, its member target, or the JSX attribute). The write is
142        /// the statement a property-sink finding must point at: the stored
143        /// value's variable is defined somewhere else. `None` for stores a
144        /// lowering synthesised without one source expression (object-literal
145        /// fields, test fixtures).
146        span: Option<(usize, usize)>,
147    },
148
149    /// `v_dest = base[index]  @ mem_in`
150    LoadElement {
151        /// Variable that receives the loaded element.
152        dest: VarId,
153        /// Memory state the load reads from.
154        mem_in: VarId,
155        /// Variable holding the array or collection.
156        base: VarId,
157        /// Index or key operand selecting the element.
158        index: Operand,
159    },
160
161    /// `base[index] = src` generates `mem_out`.
162    StoreElement {
163        /// New memory state produced by the store.
164        mem_out: VarId,
165        /// Memory state the store applies to.
166        mem_in: VarId,
167        /// Variable holding the array or collection being written.
168        base: VarId,
169        /// Index or key operand selecting the element.
170        index: Operand,
171        /// Value stored into the element.
172        src: Operand,
173    },
174
175    // --- Global Environment ---
176    /// v_dest = global_name
177    LoadGlobal {
178        /// Variable that receives the global's value.
179        dest: VarId,
180        /// Memory state the load reads from.
181        mem_in: VarId,
182        /// Name of the global being read.
183        name: String,
184    },
185
186    /// global_name = src
187    StoreGlobal {
188        /// New memory state produced by the store.
189        mem_out: VarId,
190        /// Memory state the store applies to.
191        mem_in: VarId,
192        /// Name of the global being written.
193        name: String,
194        /// Value stored into the global.
195        src: Operand,
196    },
197
198    // --- Functions & Calls ---
199    // Function calls can mutate memory, so they consume mem_in and return mem_out.
200    /// mem_out, dest = func(args) @ mem_in
201    CallStatic {
202        /// Variable receiving the call's return value, or `None` when the result is discarded.
203        dest: Option<VarId>,
204        /// Memory state after the call (calls may mutate memory).
205        mem_out: VarId,
206        /// Memory state the call executes in.
207        mem_in: VarId,
208        /// Name of the called function as written at the call site.
209        func: String,
210        /// Argument operands, in call order.
211        args: Vec<Operand>,
212        /// Keyword/keyed argument names, parallel to `args`: entry `i` is
213        /// the name of `args[i]`, or empty when that argument is
214        /// positional. Shorter than `args` (or empty) means the source
215        /// language, or the lowered node, carries no names - callers must
216        /// treat a missing entry as positional. See the `args` field documentation.
217        arg_names: Vec<String>,
218    },
219
220    /// mem_out, dest = receiver.method(args) @ mem_in
221    CallVirtual {
222        /// Variable receiving the call's return value, or `None` when the result is discarded.
223        dest: Option<VarId>,
224        /// Memory state after the call (calls may mutate memory).
225        mem_out: VarId,
226        /// Memory state the call executes in.
227        mem_in: VarId,
228        /// Name of the invoked method.
229        method: String,
230        /// The receiver the method is invoked on (`self`, `this`, or the object before the dot).
231        receiver: Operand,
232        /// Argument operands, in call order, not including the receiver.
233        args: Vec<Operand>,
234        /// Parallel to `args`; see [`Instruction::CallStatic`].
235        arg_names: Vec<String>,
236    },
237
238    /// mem_out, dest = func_ptr(args) @ mem_in
239    CallPointer {
240        /// Variable receiving the call's return value, or `None` when the result is discarded.
241        dest: Option<VarId>,
242        /// Memory state after the call (calls may mutate memory).
243        mem_out: VarId,
244        /// Memory state the call executes in.
245        mem_in: VarId,
246        /// Operand holding the callee value (a function pointer, closure or callable variable).
247        func_ptr: Operand,
248        /// Argument operands, in call order.
249        args: Vec<Operand>,
250        /// Parallel to `args`; see [`Instruction::CallStatic`].
251        arg_names: Vec<String>,
252    },
253
254    // --- Memory & Pointers (Systems Languages) ---
255    /// mem_out, v_dest = new Object/Array() @ mem_in
256    Allocate {
257        /// Variable that receives the new object or array.
258        dest: VarId,
259        /// Memory state after the allocation.
260        mem_out: VarId,
261        /// Memory state the allocation executes in.
262        mem_in: VarId,
263        /// What is being allocated; used by points-to analysis to name the abstract object.
264        kind: AllocationKind,
265    },
266
267    /// v_dest = &src
268    AddressOf {
269        /// Variable that receives the address.
270        dest: VarId,
271        /// Variable whose address is taken.
272        src: VarId,
273    },
274
275    /// v_dest = *ptr @ mem_in
276    Dereference {
277        /// Variable that receives the dereferenced value.
278        dest: VarId,
279        /// Memory state the load reads from.
280        mem_in: VarId,
281        /// Pointer operand being dereferenced.
282        ptr: Operand,
283    },
284
285    // --- Types & Multiple Returns ---
286    /// Converts `src` to `target_type`, defining `dest`.
287    Cast {
288        /// Variable that receives the converted value.
289        dest: VarId,
290        /// Value being converted.
291        src: Operand,
292        /// Name of the target type as written in the source.
293        target_type: String,
294    },
295    /// Extracts one component of a tuple or multiple-return value.
296    ExtractValue {
297        /// Variable that receives the extracted component.
298        dest: VarId,
299        /// The tuple (or multi-value result) being unpacked.
300        tuple: Operand,
301        /// Zero-based position of the component.
302        index: usize,
303    },
304
305    // --- Operations & Async ---
306    /// A binary operation, `dest = lhs op rhs`.
307    BinaryOp {
308        /// Variable that receives the result.
309        dest: VarId,
310        /// Operator as recorded by lowering: the token as written (for example `==`, `&&`), except that `+` is recorded as `concat`.
311        op: String,
312        /// Left operand.
313        lhs: Operand,
314        /// Right operand.
315        rhs: Operand,
316    },
317    /// A unary operation, `dest = op src`.
318    UnaryOp {
319        /// Variable that receives the result.
320        dest: VarId,
321        /// Operator token as written (for example `!`, `-`, `~`), or `neg` when none was found.
322        op: String,
323        /// Operand.
324        src: Operand,
325    },
326    /// Awaits a promise or future; `dest` receives the resolved value.
327    Await {
328        /// Variable that receives the resolved value.
329        dest: VarId,
330        /// Memory state after the await (other code may run while suspended).
331        mem_out: VarId,
332        /// Memory state the await starts in.
333        mem_in: VarId,
334        /// The promise or future being awaited.
335        promise: Operand,
336    },
337    /// Yields a value from a generator or coroutine; `dest` receives whatever the caller sends back.
338    Yield {
339        /// Variable that receives the value sent back into the generator, if it is used.
340        dest: Option<VarId>,
341        /// Memory state after the yield.
342        mem_out: VarId,
343        /// Memory state the yield starts in.
344        mem_in: VarId,
345        /// The yielded value, if any.
346        src: Option<Operand>,
347    },
348}
349
350impl Instruction {
351    /// The value variable this instruction defines, if any.
352    ///
353    /// Covers every instruction kind that produces a value: assignments,
354    /// loads, calls (when the result is used), allocations, address-of,
355    /// dereference, casts, aggregate extraction, unary/binary operations,
356    /// `await` and `yield`. Memory-state outputs are not included; see
357    /// `Instruction::mem_out`. This is the one definition of "what does this
358    /// instruction define"; do not re-match the variants elsewhere, because
359    /// a copy that forgets a variant silently drops findings on it.
360    #[must_use]
361    pub fn dest(&self) -> Option<VarId> {
362        match self {
363            Self::Assign { dest, .. }
364            | Self::LoadField { dest, .. }
365            | Self::LoadElement { dest, .. }
366            | Self::LoadGlobal { dest, .. }
367            | Self::Allocate { dest, .. }
368            | Self::AddressOf { dest, .. }
369            | Self::Dereference { dest, .. }
370            | Self::Cast { dest, .. }
371            | Self::ExtractValue { dest, .. }
372            | Self::BinaryOp { dest, .. }
373            | Self::UnaryOp { dest, .. }
374            | Self::Await { dest, .. } => Some(*dest),
375            Self::CallStatic { dest, .. }
376            | Self::CallVirtual { dest, .. }
377            | Self::CallPointer { dest, .. }
378            | Self::Yield { dest, .. } => *dest,
379            // No `_` arm on purpose: a new instruction kind must decide here
380            // whether it defines a value.
381            Self::StoreField { .. } | Self::StoreElement { .. } | Self::StoreGlobal { .. } => None,
382        }
383    }
384
385    /// The memory-state variable this instruction produces, if any.
386    #[must_use]
387    pub(crate) fn mem_out(&self) -> Option<VarId> {
388        match self {
389            Self::StoreField { mem_out, .. }
390            | Self::StoreElement { mem_out, .. }
391            | Self::StoreGlobal { mem_out, .. }
392            | Self::CallStatic { mem_out, .. }
393            | Self::CallVirtual { mem_out, .. }
394            | Self::CallPointer { mem_out, .. }
395            | Self::Allocate { mem_out, .. }
396            | Self::Await { mem_out, .. }
397            | Self::Yield { mem_out, .. } => Some(*mem_out),
398            Self::Assign { .. }
399            | Self::LoadField { .. }
400            | Self::LoadElement { .. }
401            | Self::LoadGlobal { .. }
402            | Self::AddressOf { .. }
403            | Self::Dereference { .. }
404            | Self::Cast { .. }
405            | Self::ExtractValue { .. }
406            | Self::BinaryOp { .. }
407            | Self::UnaryOp { .. } => None,
408        }
409    }
410
411    /// Every variable this instruction defines: the value result first, then
412    /// the memory-state output.
413    #[must_use]
414    pub(crate) fn defs(&self) -> Vec<VarId> {
415        self.dest().into_iter().chain(self.mem_out()).collect()
416    }
417}

419/// Determines how control flow exits a BasicBlock.
420#[derive(Debug, Clone)]
421#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
422pub enum Terminator {
423    /// Unconditional jump to the given block.
424    Jump(BlockId),
425    /// Two-way conditional branch on `cond`.
426    Branch {
427        /// Condition operand.
428        cond: Operand,
429        /// Block taken when the condition is truthy.
430        true_block: BlockId,
431        /// Block taken when the condition is falsy.
432        false_block: BlockId,
433    },
434    /// Multi-way branch on `cond`.
435    Switch {
436        /// Value being switched on.
437        cond: Operand,
438        /// `(case value, target block)` pairs in source order.
439        cases: Vec<(Operand, BlockId)>,
440        /// Block taken when no case matches.
441        default_block: BlockId,
442    },
443    /// Returns from the function.
444    Return {
445        /// Returned value, or `None` for a bare return.
446        src: Option<Operand>,
447    },
448    /// Raises an exception or panics with the given value.
449    Throw {
450        /// Value being thrown.
451        src: Operand,
452    },
453    /// No terminator yet: the block is still under construction (the state of every block created by `BasicBlock::new`).
454    None,
455}

457/// A straight-line sequence of instructions with a single entry and a single terminator.
458#[derive(Debug, Clone)]
459#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
460pub struct BasicBlock {
461    /// Identifier of this block within its function.
462    pub id: BlockId,
463    /// Phi nodes evaluated at block entry, before any instruction.
464    pub phis: Vec<Phi>,
465    /// Instructions in execution order.
466    pub instructions: Vec<Instruction>,
467    /// How control leaves the block.
468    pub terminator: Terminator,
469    /// Blocks with an edge into this one.
470    pub predecessors: Vec<BlockId>,
471    /// Blocks this one has an edge to.
472    pub successors: Vec<BlockId>,
473}

475impl BasicBlock {
476    /// Creates an empty block with the given id: no phis, no instructions, [`Terminator::None`] and no edges.
477    pub(crate) fn new(id: BlockId) -> Self {

479            id,
480            phis: Vec::new(),
481            instructions: Vec::new(),
482            terminator: Terminator::None,
483            predecessors: Vec::new(),
484            successors: Vec::new(),
485        }
486    }

489/// The IR of one function: its parameters, control-flow graph and per-variable metadata.
490///
491/// The heap is modelled explicitly. Every function owns a hidden memory-state parameter ([`FunctionIR::initial_memory_state`]) and every memory-affecting instruction consumes and produces a memory-state variable, so flow-sensitive heap reasoning is ordinary SSA dataflow.
492#[derive(Debug, Clone)]
493#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
494pub struct FunctionIR {
495    /// Program-level name of the function.
496    pub name: String,
497    /// Formal parameters in declaration order, not including the hidden memory-state parameter.
498    pub parameters: Vec<VarId>,
499    /// The initial Memory State parameter for the function (representing the heap upon entry)
500    pub initial_memory_state: VarId,
501    /// All basic blocks, keyed by their [`BlockId`].
502    pub blocks: FxHashMap<BlockId, BasicBlock>,
503    /// The block execution starts in.
504    pub(crate) entry_block: BlockId,
505    /// Source-level metadata for every variable, including memory states.
506    pub var_metadata: FxHashMap<VarId, VarMetadata>,
507    /// Language key this IR was lowered from (`spec.fact_language()`:
508    /// `"typescript"`,
509    /// `"rust"`, ...). Sink and sanitizer facts are language-keyed, so every
510    /// call site must know its own language: without this, a Rust
511    /// `File::create` signature would fire on a JavaScript `prisma.create`
512    /// in the same mixed-language scan. Empty when the IR was built by hand
513    /// (synthetic test IRs) or decoded from legacy data.
514    #[cfg_attr(feature = "serialize", serde(default))]
515    pub language: String,
516    /// Program name of the innermost enclosing extracted function, when this
517    /// function was extracted from a nested position (None for module scope).
518    /// Closure edges resolve free variables through this lexical chain.
519    #[cfg_attr(feature = "serialize", serde(default))]
520    pub enclosing_fn: Option<String>,
521    /// Decorator / attribute call names applied to this function's
522    /// definition (`@guard`, `#[guard]`). A decorator executes before ANY
523    /// statement of the body, so policy guard requirements are satisfied
524    /// by a matching entry guard without a guard call inside the body.
525    #[cfg_attr(feature = "serialize", serde(default))]
526    pub entry_guards: Vec<String>,
527    /// This IR is a module-level *data* binding, not a callable function:
528    /// `const tools = {...}`, `const client = require("pkg")`. The harness
529    /// lowers such initializers so non-dataflow checkers can assert policy
530    /// over static configuration, but a call written with the binding's name
531    /// (`client(req, res)`) is a call into the library, never into this IR.
532    /// Resolving it here would hand the call an empty parameter list and a
533    /// vacuous summary, which deleted the call site's pass-through taint
534    /// edges and made every call through a `require` binding read Unknown.
535    #[cfg_attr(feature = "serialize", serde(default))]
536    pub is_static_data: bool,
537    /// True when the per-function analysis budget ran out while preparing
538    /// this IR (SSA construction degraded to flow-insensitive form), so this
539    /// function was analyzed with less precision than usual. The scanner
540    /// reports every such function as `analysis_incomplete` instead of
541    /// silently producing weaker results for it.
542    #[cfg_attr(feature = "serialize", serde(default))]
543    pub analysis_incomplete: bool,
544    #[cfg_attr(feature = "serialize", serde(default))]
545    // legacy safety: recompute via max ids if absent
546    next_var_id: usize,
547    #[cfg_attr(feature = "serialize", serde(default))]
548    next_block_id: usize,
549}

551impl FunctionIR {
552    /// Creates an empty function with one empty entry block, and allocates the hidden memory-state parameter as [`FunctionIR::initial_memory_state`].
553    pub fn new(name: String) -> Self {
554        let entry_id = BlockId(0);
555        let mut blocks = FxHashMap::default();
556        blocks.insert(entry_id, BasicBlock::new(entry_id));
557
558        let mut f = Self {
559            name,
560            parameters: Vec::new(),
561            initial_memory_state: VarId(0), // Placeholder
562            blocks,
563            entry_block: entry_id,
564            var_metadata: FxHashMap::default(),
565            language: String::new(),
566            enclosing_fn: None,
567            entry_guards: Vec::new(),
568            is_static_data: false,
569            analysis_incomplete: false,
570            next_var_id: 1,
571            next_block_id: 1,
572        };
573
574        // Allocate the hidden Memory State parameter
575        f.initial_memory_state = f.new_var(VarMetadata {
576            source_name: Some(INITIAL_HEAP_STATE.into()),
577            type_name: None,
578            byte_range: None,
579            is_memory_state: true,
580            object_keys: Vec::new(),
581            declared: false,
582        });
583
584        f
585    }
586
587    /// Source name of `var`, or `v<id>` when lowering recorded none.
588    pub(crate) fn var_name(&self, var: VarId) -> String {
589        self.var_metadata
590            .get(&var)
591            .and_then(|m| m.source_name.clone())
592            .unwrap_or_else(|| format!("v{}", var.0))
593    }
594
595    /// Byte range of the expression that produced `var`, if lowering recorded one.
596    pub(crate) fn var_span(&self, var: VarId) -> Option<(usize, usize)> {
597        self.var_metadata.get(&var).and_then(|m| m.byte_range)
598    }
599
600    /// The instruction that defines `var`, if any. The IR is SSA, so there is
601    /// at most one.
602    pub(crate) fn def_instruction(&self, var: VarId) -> Option<&Instruction> {
603        self.blocks
604            .values()
605            .flat_map(|b| b.instructions.iter())
606            .find(|instr| instr.dest() == Some(var))
607    }
608
609    /// Byte range of a call instruction: the call result variable carries the
610    /// call expression's range. `None` for non-calls and calls with no result.
611    pub(crate) fn instr_span(&self, instr: &Instruction) -> Option<(usize, usize)> {
612        let dest = match instr {
613            Instruction::CallStatic { dest, .. }
614            | Instruction::CallVirtual { dest, .. }
615            | Instruction::CallPointer { dest, .. } => (*dest)?,
616            _ => return None,
617        };
618        self.var_span(dest)
619    }

621    /// Allocates a fresh variable id, records its `metadata` and returns the id.
622    pub fn new_var(&mut self, metadata: VarMetadata) -> VarId {
623        let id = VarId(self.next_var_id);
624        self.next_var_id += 1;
625        self.var_metadata.insert(id, metadata);
626        id
627    }

629    /// Allocates a fresh, empty, unconnected block and returns its id.
630    pub(crate) fn new_block(&mut self) -> BlockId {
631        let id = BlockId(self.next_block_id);
632        self.next_block_id += 1;
633        self.blocks.insert(id, BasicBlock::new(id));
634        id
635    }

659    /// Records a control-flow edge `from -> to` in both blocks' successor and predecessor lists. Idempotent: an edge that is already present is not added twice.
660    pub(crate) fn add_edge(&mut self, from: BlockId, to: BlockId) {
661        if let Some(f) = self.blocks.get_mut(&from)
662            && !f.successors.contains(&to)
663        {
664            f.successors.push(to);
665        }
666        if let Some(t) = self.blocks.get_mut(&to)
667            && !t.predecessors.contains(&from)
668        {
669            t.predecessors.push(from);
670        }
671    }

673    /// The receiver operand of the virtual call whose result is `var`, if
674    /// any: lets a chained call (`conn.cursor().execute(..)`) resolve to
675    /// the root of its receiver chain.
676    pub(crate) fn call_receiver_def(&self, var: VarId) -> Option<VarId> {
677        for b in self.blocks.values() {
678            for instr in &b.instructions {
679                if let Instruction::CallVirtual {
680                    dest: Some(d),
681                    receiver: Operand::Var(r),
682                    ..
683                } = instr
684                    && *d == var
685                {
686                    return Some(*r);
687                }
688            }
689        }
690        None
691    }
692
693    /// The `LoadField` instruction defining `var`, if any (first match - IR
694    /// is SSA-shaped): returns the base var and the loaded field. Shared
695    /// walk step for member-access path reconstruction.
696    pub(crate) fn loadfield_def(&self, var: VarId) -> Option<(VarId, String)> {
697        for b in self.blocks.values() {
698            for instr in &b.instructions {
699                if let Instruction::LoadField {
700                    dest, base, field, ..
701                } = instr
702                    && *dest == var
703                {
704                    return Some((*base, field.clone()));
705                }
706            }
707        }
708        None
709    }
710}

712/// Strips one pair of matching `'`, `"` or `` ` `` quotes from a string
713/// literal's source text, after trimming. Unmatched or lone quotes are left as is.
714pub(crate) fn strip_quotes(lit: &str) -> &str {
715    let lit = lit.trim();
716    let bytes = lit.as_bytes();
717    if let (Some(&first), Some(&last)) = (bytes.first(), bytes.last())
718        && bytes.len() >= 2
719        && matches!(first, b'\'' | b'"' | b'`')
720        && first == last
721    {
722        return &lit[1..bytes.len() - 1];
723    }
724    lit
725}
