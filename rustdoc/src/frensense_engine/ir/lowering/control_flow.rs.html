13    /// Lowers a loop into a header block, a body block and an exit block.
14    ///
15    /// The header evaluates the condition (the first child that is not a statement, block, declarator, update or initializer; unknown if there is none) and branches to the body or the exit. The body jumps back to the header unless it already ended. `break` and `continue` target the exit and the header while the body is lowered. Always returns `None`; afterwards lowering continues in the exit block.
16    pub(crate) fn visit_loop(&mut self, node: Node) -> Option<Operand> {
17        // Find condition and body children by grammar convention.
18        let mut cursor = node.walk();
19        let children: Vec<Node> = node
20            .children(&mut cursor)
21            .filter(|c| c.is_named())
22            .collect();
23
24        // Heuristic across grammars: the last named child that is a
25        // statement/block is the body; the condition is the operand
26        // expression just before it (for `for` loops this is coarse,
27        // acceptable: the condition only affects branch pruning).
28        let body = children
29            .iter()
30            .rev()
31            .find(|c| {
32                let k = c.kind();
33                self.spec.is_block_or_statement(k)
34            })
35            .copied();
36        let cond = children
37            .iter()
38            .find(|c| {
39                let k = c.kind();
40                self.spec.is_loop_condition_node(k)
41            })
42            .copied();
43
44        // For `for x in iterable` / `for x of iterable` loops the spec
45        // exposes both the binding and the iterable as separate named fields.
46        // The iterable must be visited in the header block so that taint from
47        // the iterable expression (e.g. a Flask source call) can flow into
48        // the loop variable and from there into any container the body builds.
49        let for_in = self.spec.for_loop_iterable(node);
50
51        let loop_header = self.ir.new_block();
52        let loop_body = self.ir.new_block();
53        let loop_exit = self.ir.new_block();
54        // The body may assign a variable that was constant on entry (and may
55        // run zero times), so no constant survives the loop unless the body
56        // never touches it.
57        let const_before = self.const_snapshot();
58
59        // A preceding `break`/`return` already terminated this block.
60        self.jump_if_open(loop_header);
61
62        // Header (Evaluate Condition)
63        self.current_block = loop_header;
64
65        // For `for x in iterable:` / `for x of iterable`: visit the iterable
66        // first to emit its IR (and capture taint), then assign the result to
67        // the loop variable binding so the SVFG carries the taint edge.
68        if let Some((binding, iterable)) = for_in {
69            let iter_op = self.visit_node(iterable).unwrap_or(Operand::Unknown);
70            // Assign the iterable's value to the loop variable so that any
71            // read of the loop variable has a taint path back to the iterable.
72            if let Some(LValue::Variable(dest)) = self.visit_lvalue(binding) {
73                self.mark_declared(dest);
74                self.ir.push_instruction(
75                    self.current_block,
76                    Instruction::Assign { dest, src: iter_op },
77                );
78                self.forget_const(dest);
79            }
80        }
81
82        let cond_op = if for_in.is_some() {
83            Operand::Unknown
84        } else {
85            cond.and_then(|c| self.visit_node(c))
86                .unwrap_or(Operand::Unknown)
87        };
88        self.ir.set_terminator(
89            self.current_block,
90            Terminator::Branch {
91                cond: cond_op,
92                true_block: loop_body,
93                false_block: loop_exit,
94            },
95        );
96        self.ir.add_edge(self.current_block, loop_body);
97        self.ir.add_edge(self.current_block, loop_exit);
98
99        self.loop_stack.push(LoopTarget {
100            break_target: loop_exit,
101            continue_target: Some(loop_header),
102        });
103
104        // Body
105        self.current_block = loop_body;
106        self.env.push(FxHashMap::default());
107        if let Some(body) = body {
108            self.visit_node(body);
109        }
110        // A body ending in `break`/`continue`/`return` owns its edge.
111        self.jump_if_open(loop_header);
112        self.env.pop();
113        self.loop_stack.pop();
114
115        self.const_join(&const_before);
116        self.current_block = loop_exit;
117        None
118    }

120    /// `switch (expr) { case v: ... default: ... }` lowers to a condition
121    /// chain: each `expr == v` dispatches to its case body, and an unmatched
122    /// value falls to `default` (or the merge when there is none). Case
123    /// bodies fall through to the next body in source order, matching C/JS
124    /// semantics; `break` jumps to the merge. The condition is evaluated
125    /// once, in the dispatch block.
126    pub(crate) fn visit_switch(&mut self, node: Node) -> Option<Operand> {
127        let mut cursor = node.walk();
128        let children: Vec<Node> = node
129            .children(&mut cursor)
130            .filter(|c| c.is_named())
131            .collect();
132
133        let is_body = |k: &str| self.spec.is_body_node(k);
134        let body = node
135            .child_by_field_name("body")
136            .or_else(|| children.iter().copied().find(|c| is_body(c.kind())));
137        let cond_node = node
138            .child_by_field_name("condition")
139            .or_else(|| children.iter().copied().find(|c| !is_body(c.kind())));
140
141        // Clause layout: `case_statement` (C), `switch_case`/`switch_default`
142        // (JS/TS). The value node (absent on `default`) is the first named
143        // child that is not itself a statement.
144        let mut clauses: Vec<(Option<Node>, Vec<Node>)> = Vec::new();
145        if let Some(body) = body {
146            let mut body_cursor = body.walk();
147            for clause in body.named_children(&mut body_cursor) {
148                let kind = clause.kind();
149                if !self.spec.is_case_clause(kind) {
150                    continue;
151                }
152                let mut clause_cursor = clause.walk();
153                let kids: Vec<Node> = clause.named_children(&mut clause_cursor).collect();
154                let value_pos = kids.iter().position(|n| {
155                    let k = n.kind();
156                    !self.spec.is_statement_node(k)
157                });
158                let value = value_pos.map(|i| kids[i]);
159                let stmts: Vec<Node> = kids
160                    .iter()
161                    .enumerate()
162                    .filter(|(i, _)| Some(*i) != value_pos)
163                    .map(|(_, n)| *n)
164                    .collect();
165                clauses.push((value, stmts));
166            }
167        }
168        if clauses.is_empty() {
169            // Not a recognized switch shape: keep the coarse generic walk.
170            return self.visit_children_generic(node);
171        }
172
173        // Case bodies may assign any variable; nothing stays constant across
174        // the dispatch unless every case agrees.
175        let const_before = self.const_snapshot();
176
177        let cond_op = cond_node
178            .and_then(|c| self.visit_node(c))
179            .unwrap_or(Operand::Unknown);
180
181        let merge_block = self.ir.new_block();
182        let bodies: Vec<BlockId> = clauses.iter().map(|_| self.ir.new_block()).collect();
183        let default_idx = clauses.iter().position(|(v, _)| v.is_none());
184        let value_idx: Vec<usize> = (0..clauses.len())
185            .filter(|&i| clauses[i].0.is_some())
186            .collect();
187
188        self.loop_stack.push(LoopTarget {
189            break_target: merge_block,
190            continue_target: None,
191        });
192        self.env.push(FxHashMap::default());
193
194        // Condition chain: entries[0] is the dispatch block that just
195        // evaluated `expr`; later entries are fresh blocks.
196        let mut entries: Vec<BlockId> = Vec::with_capacity(value_idx.len());
197        for n in 0..value_idx.len() {
198            entries.push(if n == 0 {
199                self.current_block
200            } else {
201                self.ir.new_block()
202            });
203        }
204        for (n, &ci) in value_idx.iter().enumerate() {
205            let entry = entries[n];
206            let false_next = if n + 1 < value_idx.len() {
207                entries[n + 1]
208            } else if let Some(d) = default_idx {
209                bodies[d]
210            } else {
211                merge_block
212            };
213            self.current_block = entry;
214            let value_op = clauses[ci]
215                .0
216                .and_then(|v| self.visit_node(v))
217                .unwrap_or(Operand::Unknown);
218            let eq = self.new_temp(node);
219            self.ir.push_instruction(
220                entry,
221                Instruction::BinaryOp {
222                    dest: eq,
223                    op: "==".to_string(),
224                    lhs: cond_op.clone(),
225                    rhs: value_op,
226                },
227            );
228            self.ir.set_terminator(
229                entry,
230                Terminator::Branch {
231                    cond: Operand::Var(eq),
232                    true_block: bodies[ci],
233                    false_block: false_next,
234                },
235            );
236            self.ir.add_edge(entry, bodies[ci]);
237            self.ir.add_edge(entry, false_next);
238        }
239        // A switch with only `default` (or no condition) always takes it.
240        if value_idx.is_empty()
241            && let Some(d) = default_idx
242        {
243            self.jump_if_open(bodies[d]);
244        }
245
246        // Case bodies, in source order: each falls through to the next.
247        for (i, (_, stmts)) in clauses.iter().enumerate() {
248            self.current_block = bodies[i];
249            for stmt in stmts {
250                self.visit_node(*stmt);
251            }
252            let fall = if i + 1 < clauses.len() {
253                bodies[i + 1]
254            } else {
255                merge_block
256            };
257            self.jump_if_open(fall);
258        }
259
260        self.env.pop();
261        self.loop_stack.pop();
262        self.const_join(&const_before);
263        self.current_block = merge_block;
264        None
265    }

267    /// Lowers a ternary expression without branching: the condition and both arms are evaluated in the current block.
268    ///
269    /// When both arms are variables, a fresh variable is assigned both of them, so it may hold either. When only one arm is a variable, that variable is the result. A node that does not have exactly three named children is lowered by walking its children. Because no branch is created, a guard inside an arm does not constrain the other arm.
270    pub(crate) fn visit_ternary_straight_line(&mut self, node: Node) -> Option<Operand> {
271        let mut cursor = node.walk();
272        let children: Vec<Node> = node
273            .children(&mut cursor)
274            .filter(|c| c.is_named() && !self.spec.is_comment(c.kind()))
275            .collect();
276        // JS grammar: [condition, consequence, alternative]
277        if children.len() != 3 {
278            // Malformed ternary: fall through to generic child walk.
279            return self.visit_children_generic(node);
280        }
281
282        // Condition and both arms evaluated in the current block.
283        // Side effects in the condition/arms still lower to their
284        // instructions; only control-flow precision is lost.
285        let const_before = self.const_snapshot();
286        let _ = self.visit_node(children[0]).unwrap_or(Operand::Unknown);
287        let true_val = self.visit_node(children[1]);
288        let false_val = self.visit_node(children[2]);
289
290        // Merge: prefer a fresh var assigned from the first var-valued
291        // arm (preserves one def site for SVFG edges). A literal arm
292        // carries no taint, so the var arm dominates the may-analysis.
293        let result = match (true_val, false_val) {
294            (Some(Operand::Var(tv)), Some(Operand::Var(fv))) => {
295                let dest = self.new_temp(node);
296                self.ir.push_instruction(
297                    self.current_block,
298                    Instruction::Assign {
299                        dest,
300                        src: Operand::Var(tv),
301                    },
302                );
303                self.ir.push_instruction(
304                    self.current_block,
305                    Instruction::Assign {
306                        dest,
307                        src: Operand::Var(fv),
308                    },
309                );
310                // Two arms merge into one variable: its value is not fixed.
311                self.forget_const(dest);
312                Some(Operand::Var(dest))
313            }
314            // One arm is a var, other is a constant: the var survives;
315            // only the tainted arm matters for alerts.
316            (Some(Operand::Var(v)), Some(_)) | (Some(_), Some(Operand::Var(v))) => {
317                Some(Operand::Var(v))
318            }
319            (Some(v), None) | (None, Some(v)) => Some(v),
320            (None, None) => None,
321            // Both constants: fold to the first, neither carries taint.
322            (a, b) => a.or(b),
323        };
324        // Only one arm executes at runtime, so a value set by exactly one of
325        // them is not constant after the expression.
326        self.const_join(&const_before);
327        result
328    }

330    /// Lowers a ternary conditional expression as a real branch.
331    ///
332    /// The condition block branches to a true block and a false block, each of which assigns its arm's value to one shared fresh variable and jumps to a merge block; the SSA pass later turns the two assignments into a phi. Which of the three children is the condition comes from the language spec. Nodes with fewer than three named children are lowered by walking their children. Returns the shared variable.
333    pub(crate) fn visit_conditional(&mut self, node: Node) -> Option<Operand> {
334        let mut cursor = node.walk();
335        let children: Vec<Node> = node
336            .children(&mut cursor)
337            .filter(|c| c.is_named())
338            .collect();
339        if children.len() < 3 {
340            // Degenerate: fall through the generic child walk.
341            return self.visit_children_generic(node);
342        }
343        let cond_idx = self.spec.ternary_cond_index().min(children.len() - 1);
344        let cond_node = children[cond_idx];
345        let (then_node, else_node) = if cond_idx == 0 {
346            // JS/C layout: [cond, then, else].
347            (children[1], children[2])
348        } else {
349            // Python layout: [then, cond, else].
350            (children[0], children[2])
351        };
352
353        let cond_op = self.visit_node(cond_node).unwrap_or(Operand::Unknown);
354        let const_before = self.const_snapshot();
355
356        let dest = self.new_temp(node);
357
358        let true_block = self.ir.new_block();
359        let false_block = self.ir.new_block();
360        let merge_block = self.ir.new_block();
361        self.ir.set_terminator(
362            self.current_block,
363            Terminator::Branch {
364                cond: cond_op,
365                true_block,
366                false_block,
367            },
368        );
369        self.ir.add_edge(self.current_block, true_block);
370        self.ir.add_edge(self.current_block, false_block);
371
372        // Then arm: bind the merge var.
373        self.env.push(FxHashMap::default());
374        self.current_block = true_block;
375        let then_op = self.visit_node(then_node).unwrap_or(Operand::Unknown);
376        self.ir.push_instruction(
377            self.current_block,
378            Instruction::Assign { dest, src: then_op },
379        );
380        self.ir
381            .set_terminator(self.current_block, Terminator::Jump(merge_block));
382        self.ir.add_edge(self.current_block, merge_block);
383        self.env.pop();
384
385        // Else arm: bind the same merge var.
386        self.env.push(FxHashMap::default());
387        self.current_block = false_block;
388        let else_op = self.visit_node(else_node).unwrap_or(Operand::Unknown);
389        self.ir.push_instruction(
390            self.current_block,
391            Instruction::Assign { dest, src: else_op },
392        );
393        self.ir
394            .set_terminator(self.current_block, Terminator::Jump(merge_block));
395        self.ir.add_edge(self.current_block, merge_block);
396        self.env.pop();
397
398        // The two arms merge into `dest`, whose value is whichever arm ran.
399        self.forget_const(dest);
400        self.const_join(&const_before);
401
402        self.current_block = merge_block;
403        Some(Operand::Var(dest))
404    }

406    /// Pop a branch-arm scope and hoist its *new* bindings into the
407    /// enclosing scope. Python/JS/C give assignments function scope, not
408    /// block scope: a name first bound inside an `if`/`else` arm must
409    /// survive to the merge so the SSA pass can place a phi for it.
410    /// Names already present in an enclosing scope are left alone (the
411    /// arm reused the same VarId, and the phi is driven by the def sites).
412    fn hoist_arm_scope(env: &mut Vec<FxHashMap<String, VarId>>) {
413        let arm = match env.pop() {
414            Some(m) => m,
415            None => return,
416        };
417        if let Some(parent) = env.last_mut() {
418            for (name, var) in arm {
419                parent.entry(name).or_insert(var);
420            }
421        }
422    }

424    /// C preprocessor conditionals (`#if` / `#ifdef` / `#elif` / `#else`).
425    ///
426    /// The alternatives of one chain are mutually exclusive build
427    /// configurations, never statements that all run. Lowering them as
428    /// sequential code put three `realloc` calls in a row in redis's
429    /// `ztryrealloc_usable_internal` (one per `#ifdef` branch) and reported
430    /// the first as leaked. Each alternative becomes an arm of an
431    /// unknown-condition branch that rejoins after the chain, so every
432    /// configuration is analysed as its own path and a `return` inside one
433    /// ends only that path. An `#ifdef` without `#else` forks against "not
434    /// compiled".
435    pub(crate) fn visit_preproc_conditional(&mut self, node: Node) -> Option<Operand> {
436        let mut cursor = node.walk();
437        let skip: Vec<usize> = ["name", "condition", "alternative"]
438            .iter()
439            .filter_map(|f| node.child_by_field_name(f).map(|n| n.id()))
440            .collect();
441        let body: Vec<Node> = node
442            .children(&mut cursor)
443            .filter(|c| c.is_named() && !skip.contains(&c.id()) && !self.spec.is_comment(c.kind()))
444            .collect();
445
446        // `#else` is not a fork of its own: it is the false arm's body.
447        if self.spec.is_preproc_else(node.kind()) {
448            for child in body {
449                self.visit_node(child);
450            }
451            return None;
452        }
453
454        let alt_node = node.child_by_field_name("alternative");
455        // The alternatives are exclusive configurations: a variable assigned
456        // in one is not constant after the chain.
457        let const_before = self.const_snapshot();
458        let true_block = self.ir.new_block();
459        let false_block = self.ir.new_block();
460        let merge_block = self.ir.new_block();
461        let false_target = if alt_node.is_some() {
462            false_block
463        } else {
464            merge_block
465        };
466        self.ir.set_terminator(
467            self.current_block,
468            Terminator::Branch {
469                cond: Operand::Unknown,
470                true_block,
471                false_block: false_target,
472            },
473        );
474        self.ir.add_edge(self.current_block, true_block);
475        self.ir.add_edge(self.current_block, false_target);
476
477        self.env.push(FxHashMap::default());
478        self.current_block = true_block;
479        for child in body {
480            self.visit_node(child);
481        }
482        self.close_arm(merge_block);
483        Self::hoist_arm_scope(&mut self.env);
484
485        if let Some(alt) = alt_node {
486            self.env.push(FxHashMap::default());
487            self.current_block = false_block;
488            self.visit_node(alt);
489            self.close_arm(merge_block);
490            Self::hoist_arm_scope(&mut self.env);
491        }
492
493        self.const_join(&const_before);
494        self.current_block = merge_block;
495        None
496    }

499    /// terminator (`return`, `goto`, ...).
500    fn close_arm(&mut self, merge: BlockId) {
501        self.jump_if_open(merge);
502    }

504    /// Lowers an `if`, and branch-role kinds whose name contains `switch` other than `switch_statement` (which has its own visitor): optional initializer, condition, consequence and alternative, joined at a merge block.
505    ///
506    /// When the condition is a known constant, only the arm that is taken is lowered, with no branch (not for `switch`). A `switch` pushes a `break` target for the merge block while its body is lowered. Always returns `None`.
507    pub(crate) fn visit_branch(&mut self, node: Node) -> Option<Operand> {
508        let mut cursor = node.walk();
509        let children: Vec<Node> = node
510            .children(&mut cursor)
511            .filter(|c| c.is_named())
512            .collect();
513
514        // Grammar fields first (every language we ship names these), so an
515        // `if` initializer is never mistaken for the condition and a
516        // grammar without an explicit `else_clause` node still finds its
517        // alternative. Go is the concrete case: `if v := get(); v != "" {...}
518        // else {...}` has an `initializer` field (`v := get()`) that the old
519        // first-child heuristic lowered as the condition - leaving the real
520        // `v != ""` comparison and the whole `else` arm unlowered, so taint
521        // assigned in the `else` arm (and the join) vanished.
522        let init_node = node.child_by_field_name("initializer");
523        let cond_node = node.child_by_field_name("condition").or_else(|| {
524            children
525                .iter()
526                .find(|c| {
527                    let k = c.kind();
528                    self.spec.is_branch_condition_node(k) && !self.spec.is_else_clause(k)
529                })
530                .copied()
531        });
532        let cons_node = node.child_by_field_name("consequence").or_else(|| {
533            children
534                .iter()
535                .find(|c| {
536                    let k = c.kind();
537                    self.spec.is_block_or_statement(k)
538                })
539                .copied()
540        });
541        let alt_node = node.child_by_field_name("alternative").or_else(|| {
542            children
543                .iter()
544                .find(|c| self.spec.is_else_clause(c.kind()))
545                .copied()
546        });
547
548        // An initializer runs on the fall-through path before the test; its
549        // bindings must be visible to the condition and both arms.
550        if let Some(init) = init_node {
551            self.visit_node(init);
552        }
553
554        // Go-style switches lower here as a branch; C/JS `switch` goes to
555        // `visit_switch`. An unlabeled `break` inside such a branch must
556        // exit the switch (jump to the merge), not the enclosing loop.
557        let is_switch = self.spec.is_switch_like_node(node.kind());
558
559        // A condition whose value is fixed before the program runs selects
560        // exactly one arm. Lowering the unreachable arm anyway puts its
561        // assignments into the join, and the backward walk then reports taint
562        // no execution can reach:
563        //
564        //     bar = "This should never happen"
565        //     if 'should' in bar:   bar = param   # dead, but merged into bar
566        //
567        // The condition expression is pure by construction when its value is
568        // known (a literal, a variable bound to one, or an operator over
569        // those), so skipping its lowering loses no side effect. The taken
570        // arm is lowered inline: it is the only path, so its bindings are not
571        // a join and stay constant.
572        if !is_switch && let Some(truth) = cond_node.and_then(|c| self.const_truth(c)) {
573            let taken = if truth { cons_node } else { alt_node };
574            if let Some(arm) = taken {
575                self.env.push(FxHashMap::default());
576                self.visit_node(arm);
577                Self::hoist_arm_scope(&mut self.env);
578            }
579            return None;
580        }
581
582        let cond_op = cond_node
583            .and_then(|c| self.visit_node(c))
584            .unwrap_or(Operand::Unknown);
585        let const_before = self.const_snapshot();
586
587        let true_block = self.ir.new_block();
588        let false_block = self.ir.new_block();
589        let merge_block = self.ir.new_block();
590
591        let false_target = if alt_node.is_some() {
592            false_block
593        } else {
594            merge_block
595        };
596
597        self.ir.set_terminator(
598            self.current_block,
599            Terminator::Branch {
600                cond: cond_op,
601                true_block,
602                false_block: false_target,
603            },
604        );
605        self.ir.add_edge(self.current_block, true_block);
606        self.ir.add_edge(self.current_block, false_target);
607
608        if is_switch {
609            self.loop_stack.push(LoopTarget {
610                break_target: merge_block,
611                continue_target: None,
612            });
613        }
614
615        // True Branch
616        self.env.push(FxHashMap::default());
617        self.current_block = true_block;
618        if let Some(cons) = cons_node {
619            self.visit_node(cons);
620        }
621        self.jump_if_open(merge_block);
622        Self::hoist_arm_scope(&mut self.env);
623
624        // False Branch
625        if let Some(alt) = alt_node {
626            self.env.push(FxHashMap::default());
627            self.current_block = false_block;
628            self.visit_node(alt);
629            self.jump_if_open(merge_block);
630            Self::hoist_arm_scope(&mut self.env);
631        }
632
633        if is_switch {
634            self.loop_stack.pop();
635        }
636
637        // Either arm may have assigned a variable, so a value is constant
638        // after the merge only if both paths left it alone.
639        self.const_join(&const_before);
640
641        self.current_block = merge_block;
642        None
643    }
644
645    /// `try` / `except` / `finally` lower as one straight-line region: every
646    /// body executes in source order. No constant survives it either - the
647    /// `try` body may be abandoned by an exception, and a handler may never
648    /// run - so the region is wrapped in a const join.
649    pub(crate) fn visit_try(&mut self, node: Node) -> Option<Operand> {
650        let const_before = self.const_snapshot();
651        let out = self.visit_children_generic(node);
652        self.const_join(&const_before);
653        out
654    }
655
656    /// Lowers a `return`: sets the current block's terminator to `Return`, carrying the value of the first non-comment child (none for a bare return). Returns `None`.
657    pub(crate) fn visit_return(&mut self, node: Node) -> Option<Operand> {
658        let mut cursor = node.walk();
659        let value: Option<Operand> = node
660            .named_children(&mut cursor)
661            .find(|c| !self.spec.is_comment(c.kind()))
662            .and_then(|c| self.visit_node(c));
663        self.ir
664            .set_terminator(self.current_block, Terminator::Return { src: value });
665        None
666    }

668    /// Lowers a `goto`: jumps from the current block (when it has no terminator yet) to the block for the named label, then continues in a fresh unreachable block. Returns `None`.
669    pub(crate) fn visit_goto(&mut self, node: Node) -> Option<Operand> {
670        let label_name = node
671            .child_by_field_name("label")
672            .or_else(|| {
673                let mut c = node.walk();
674                node.named_children(&mut c)
675                    .find(|n| self.spec.is_jump_target_name(n.kind()))
676            })
677            .map(|n| self.source[n.start_byte()..n.end_byte()].trim().to_string());
678
679        if let Some(name) = label_name {
680            let target_block = self.get_or_create_label_block(&name);
681            self.jump_to(target_block);
682        }
683        None
684    }
685
686    /// An unlabeled `break` leaves the innermost loop or switch. Labeled
687    /// `break label;` keeps the generic walk: labels lower as goto targets,
688    /// not break targets.
689    pub(crate) fn visit_break(&mut self, node: Node) -> Option<Operand> {
690        if has_break_label(node, self.spec) {
691            return None;
692        }
693        let target = self.loop_stack.last().map(|t| t.break_target)?;
694        self.jump_to(target);
695        None
696    }
697
698    /// An unlabeled `continue` jumps to the innermost loop header, skipping
699    /// the rest of the body. Switches on the stack contribute no continue
700    /// target, so a `continue` inside a switch still reaches the loop.
701    pub(crate) fn visit_continue(&mut self, node: Node) -> Option<Operand> {
702        if has_break_label(node, self.spec) {
703            return None;
704        }
705        let target = self
706            .loop_stack
707            .iter()
708            .rev()
709            .find_map(|t| t.continue_target)?;
710        self.jump_to(target);
711        None
712    }
713
714    /// Ends the current block with a jump to `target`, unless an earlier
715    /// `break`/`continue`/`return` already terminated it (that statement owns
716    /// the edge).
717    fn jump_if_open(&mut self, target: BlockId) {
718        if self
719            .ir
720            .blocks
721            .get(&self.current_block)
722            .is_none_or(|b| matches!(b.terminator, Terminator::None))
723        {
724            self.ir
725                .set_terminator(self.current_block, Terminator::Jump(target));
726            self.ir.add_edge(self.current_block, target);
727        }
728    }
729
730    /// Terminate the current block with a jump to `target`, then move to a
731    /// fresh block so any following statements lower as unreachable code.
732    fn jump_to(&mut self, target: BlockId) {
733        self.jump_if_open(target);
734        self.current_block = self.ir.new_block();
735    }

737    /// Lowers a labeled statement: jumps to the label's block (when the current block has no terminator yet), continues lowering there, and lowers the labelled statement into it. Returns `None`.
738    pub(crate) fn visit_labeled_statement(&mut self, node: Node) -> Option<Operand> {
739        let label_name = node
740            .child_by_field_name("label")
741            .or_else(|| {
742                let mut c = node.walk();
743                node.named_children(&mut c)
744                    .find(|n| self.spec.is_jump_target_name(n.kind()))
745            })
746            .map(|n| self.source[n.start_byte()..n.end_byte()].trim().to_string());
747
748        if let Some(name) = label_name {
749            let label_block = self.get_or_create_label_block(&name);
750            self.jump_if_open(label_block);
751            self.current_block = label_block;
752        }
753
754        let inner_stmt = node.child_by_field_name("statement").or_else(|| {
755            let mut c = node.walk();
756            // A comment between the label and its statement
757            // (`cleanup: /* ... */ *out = p;`) is the first named child
758            // after the label; picking it dropped the statement entirely.
759            node.named_children(&mut c).find(|n| {
760                !self.spec.is_jump_target_name(n.kind()) && !self.spec.is_comment(n.kind())
761            })
762        });
763
764        if let Some(stmt) = inner_stmt {
765            self.visit_node(stmt);
766        }
767        None
768    }

770    /// Lowers a `match` or `switch` into a dispatch chain, one block per arm and a merge block.
771    ///
772    /// Each arm is selected by a branch on an opaque temporary, not by a comparison against the scrutinee; the arm pattern is bound to the scrutinee inside its block. Each arm's value is assigned to one shared fresh variable, which is returned. For a `match_statement` over a known constant, only the selected arm is lowered. A node with no recognisable arms is lowered by walking its children.
773    pub(crate) fn visit_match(&mut self, node: Node) -> Option<Operand> {
774        let mut cursor = node.walk();
775        let children: Vec<Node> = node
776            .children(&mut cursor)
777            .filter(|c| c.is_named())
778            .collect();
779
780        let value_node = node
781            .child_by_field_name("value")
782            .or_else(|| node.child_by_field_name("subject"))
783            .or_else(|| {
784                children
785                    .iter()
786                    .copied()
787                    .find(|c| !self.spec.is_containment_node(c.kind()))
788            });
789
790        let scrutinee_op = value_node
791            .and_then(|v| self.visit_node(v))
792            .unwrap_or(Operand::Unknown);
793
794        let body_node = node.child_by_field_name("body").or_else(|| {
795            children
796                .iter()
797                .copied()
798                .find(|c| self.spec.is_containment_node(c.kind()))
799        });
800
801        let mut arms = Vec::new();
802        if let Some(body) = body_node {
803            let mut bc = body.walk();
804            for arm in body.children(&mut bc) {
805                if !arm.is_named() {
806                    continue;
807                }
808                let kind = arm.kind();
809                if self.spec.is_case_alternative(kind) {
810                    arms.push(arm);
811                }
812            }
813        }
814        if arms.is_empty() {
815            return self.visit_children_generic(node);
816        }
817
818        // A Python `match` on a value fixed before the program runs executes
819        // exactly one arm. Lowering every arm merged an unreachable tainted
820        // arm into the name read after the `match`; see `const_eval`.
821        // Restricted to the statement form: an expression-shaped `match`
822        // (Rust) produces a value its caller consumes, which the pruned path
823        // does not synthesise.
824        if self.spec.is_match_statement(node.kind())
825            && let Some(subject) = value_node.and_then(|v| self.const_value(v))
826            && let Some(selected) = self.select_const_arm(&arms, &subject)
827        {
828            if let Some(arm) = selected {
829                self.env.push(FxHashMap::default());
830                if let Some(body) = match_arm_body(arms[arm]) {
831                    self.visit_node(body);
832                }
833                Self::hoist_arm_scope(&mut self.env);
834            }
835            return None;
836        }
837
838        let const_before = self.const_snapshot();
839        let dispatch_block = self.current_block;
840        let merge_block = self.ir.new_block();
841        let dest = self.new_temp(node);
842
843        let arm_blocks: Vec<BlockId> = arms.iter().map(|_| self.ir.new_block()).collect();
844
845        let mut cur_dispatch = dispatch_block;
846        for (i, &arm_blk) in arm_blocks.iter().enumerate() {
847            let next_dispatch = if i + 1 < arm_blocks.len() {
848                self.ir.new_block()
849            } else {
850                merge_block
851            };
852
853            let arm_cond = self.new_temp(arms[i]);
854            self.ir.set_terminator(
855                cur_dispatch,
856                Terminator::Branch {
857                    cond: Operand::Var(arm_cond),
858                    true_block: arm_blk,
859                    false_block: next_dispatch,
860                },
861            );
862            self.ir.add_edge(cur_dispatch, arm_blk);
863            self.ir.add_edge(cur_dispatch, next_dispatch);
864            cur_dispatch = next_dispatch;
865        }
866
867        for (i, arm) in arms.iter().enumerate() {
868            self.current_block = arm_blocks[i];
869            self.env.push(FxHashMap::default());
870
871            if let Some(pat) = match_arm_pattern(*arm, self.spec) {
872                self.bind_match_pattern(pat, &scrutinee_op);
873            }
874
875            if let Some(body) = match_arm_body(*arm)
876                && let Some(res_op) = self.visit_node(body)
877            {
878                self.ir.push_instruction(
879                    self.current_block,
880                    Instruction::Assign { dest, src: res_op },
881                );
882            }
883
884            self.jump_if_open(merge_block);
885
886            // Python case blocks are function-scoped: a name first bound in
887            // one arm is visible after the `match`, exactly like an `if` arm.
888            // Popping the scope dropped those bindings, so a later read of the
889            // arm-assigned variable resolved to a fresh unknown var and a
890            // reachable tainted arm stopped feeding the sink.
891            Self::hoist_arm_scope(&mut self.env);
892        }
893
894        // An arm may have assigned a variable; only values every arm left
895        // alone (or set to the same literal) survive the join.
896        self.const_join(&const_before);
897
898        self.current_block = merge_block;
899        Some(Operand::Var(dest))
900    }
901
902    fn bind_match_pattern(&mut self, pat: Node, scrutinee_op: &Operand) {
903        let kind = pat.kind();
904        match kind {
905            kind if self.spec.is_binding_pattern(kind) => {
906                let name = self.source[pat.start_byte()..pat.end_byte()]
907                    .trim()
908                    .to_string();
909                if !name.is_empty()
910                    && name != "_"
911                    && !name.chars().next().is_some_and(|c| c.is_uppercase())
912                {
913                    let var = self.ir.new_var(VarMetadata {
914                        source_name: Some(name.clone()),
915                        type_name: None,
916                        byte_range: Some((pat.start_byte(), pat.end_byte())),
917                        is_memory_state: false,
918                        object_keys: Vec::new(),
919                        declared: true,
920                    });
921                    self.env.last_mut().unwrap().insert(name, var);
922                    self.ir.push_instruction(
923                        self.current_block,
924                        Instruction::Assign {
925                            dest: var,
926                            src: scrutinee_op.clone(),
927                        },
928                    );
929                }
930            }
931            kind if self.spec.is_structured_pattern(kind) => {
932                let type_node = pat.child_by_field_name("type");
933                let mut c = pat.walk();
934                for child in pat.children(&mut c) {
935                    if child.is_named() && Some(child) != type_node {
936                        self.bind_match_pattern(child, scrutinee_op);
937                    }
938                }
939            }
940            _ => {
941                let mut c = pat.walk();
942                for child in pat.children(&mut c) {
943                    if child.is_named() {
944                        self.bind_match_pattern(child, scrutinee_op);
945                    }
946                }
947            }
948        }
949    }
950}

952/// True for `break label;` / `continue label;` - the optional label is a
953/// named `statement_identifier`/`identifier` child (never present in C).
954fn has_break_label(node: Node, spec: &dyn frensense_lang::spec::LanguageSpec) -> bool {
955    if node.child_by_field_name("label").is_some() {
956        return true;
957    }
958    let mut cursor = node.walk();
959    node.named_children(&mut cursor)
960        .any(|n| spec.is_jump_target_name(n.kind()))
961}
