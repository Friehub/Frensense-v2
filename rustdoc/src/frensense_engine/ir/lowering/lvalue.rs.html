5use crate::ir::function::{Operand, VarId, strip_quotes};
6use crate::ir::lowering::LoweringContext;
7use frensense_lang::NodeRole;
8use tree_sitter::Node;
9
10/// Helper to differentiate Assigning to a Variable vs Mutating an Object
11#[derive(Debug, Clone, PartialEq)]
12pub(crate) enum LValue {
13    /// Assigning to a plain variable.
14    Variable(VarId),
15    /// Mutating a field of an object.
16    Field {
17        /// Variable holding the object.
18        base: VarId,
19        /// Name of the field.
20        field: String,
21    },
22    /// Mutating an element of an array or collection (also used for `*ptr`, with index 0).
23    Element {
24        /// Variable holding the array or collection.
25        base: VarId,
26        /// Index or key operand.
27        index: Operand,
28    },
29}
30
31impl<'a> LoweringContext<'a> {
32    /// Classifies the target of an assignment as a variable, field or element.
33    ///
34    /// An identifier is a variable and `*ptr` is an element at index 0. A member access is an element when its property evaluates to a variable and the language does not classify that node kind as a property name (`obj[i]`), and a field otherwise (`obj.name`). Returns `None` for any other shape, or when the object does not evaluate to a variable.
35    pub(crate) fn visit_lvalue(&mut self, node: Node) -> Option<LValue> {
36        let role = self.classify_node(node.kind());
37        match role {
38            NodeRole::Identifier => {
39                if let Operand::Var(v) = self.resolve_identifier(node) {
40                    Some(LValue::Variable(v))
41                } else {
42                    None
43                }
44            }
45            _ if self.spec.is_pointer_expression(node.kind()) => {
46                let arg = node
47                    .child_by_field_name("argument")
48                    .or_else(|| node.named_child(0))?;
49                if let Some(Operand::Var(base)) = self.visit_node(arg) {
50                    Some(LValue::Element {
51                        base,
52                        index: Operand::IntLiteral(0),
53                    })
54                } else {
55                    None
56                }
57            }
58            // MemberAccess covers both `obj.field` and `obj[i]`: JS maps
59            // subscript_expression here with property_field = "index". Distinguish
60            // by attempting to evaluate the property node as an operand.
61            NodeRole::MemberAccess {
62                object_field,
63                property_field,
64            } => {
65                let obj_node = node.child_by_field_name(object_field.as_ref())?;
66                let prop_node = node.child_by_field_name(property_field.as_ref())?;
67
68                let base_op = self.visit_node(obj_node)?;
69
70                if let Operand::Var(base_var) = base_op {
71                    match self.visit_node(prop_node) {
72                        // Index-like (identifier/expression resolving to a var):
73                        // prefer element semantics for lvalues. Property-name
74                        // decision is language-driven (spec.is_property_kind):
75                        // Python attributes are plain `identifier`, Go/Rust use
76                        // `field_identifier`, substring guessing broke all
77                        // three into LoadElement/CallPointer, losing sink names.
78                        Some(Operand::Var(index_var)) => {
79                            if self.is_property_kind(prop_node.kind()) {
80                                let field_name = strip_quotes(
81                                    &self.source[prop_node.start_byte()..prop_node.end_byte()],
82                                )
83                                .to_string();
84                                Some(LValue::Field {
85                                    base: base_var,
86                                    field: field_name,
87                                })
88                            } else {
89                                Some(LValue::Element {
90                                    base: base_var,
91                                    index: Operand::Var(index_var),
92                                })
93                            }
94                        }
95                        _ => {
96                            let field_name = strip_quotes(
97                                &self.source[prop_node.start_byte()..prop_node.end_byte()],
98                            )
99                            .to_string();
100                            Some(LValue::Field {
101                                base: base_var,
102                                field: field_name,
103                            })
104                        }
105                    }
106                } else {
107                    None
108                }
109            }
110            _ => None,
111        }
112    }
113}
