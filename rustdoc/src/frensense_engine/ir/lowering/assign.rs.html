5use crate::ir::function::*;

10impl<'a> LoweringContext<'a> {
11    /// Compound part of an assignment node's operator: `*=` → `*`, `+=` → `+`.
12    ///
13    /// Languages spell compound assignment as an anonymous token between the
14    /// lhs/rhs fields (`alloc *= 2` in C, `x += 1` in Python/Rust/Go/JS).
15    /// Lowering it as a plain `=` silently drops the read-modify-write and
16    /// const-folds the variable to the RHS literal - e.g. `alloc *= 2` became
17    /// `alloc = 2`, shrinking malloc capacities and firing false overflows.
18    /// Returns `None` for plain assignment (`=`, `:=`) or no operator token,
19    /// which keep the plain-store lowering.
20    fn compound_operator(&self, node: Node) -> Option<String> {
21        let mut cursor = node.walk();
22        for child in node.children(&mut cursor) {
23            if child.is_named() || self.spec.is_comment(child.kind()) {
24                continue;
25            }
26            let text = self.source[child.start_byte()..child.end_byte()].trim();
27            if text.is_empty() {
28                continue;
29            }
30            if text == "=" || text == ":=" {
31                return None;
32            }
33            let stripped = text.strip_suffix('=')?;
34            if stripped.is_empty() || stripped == ":" {
35                return None;
36            }
37            return Some(stripped.to_string());
38        }
39        None
40    }
41
42    /// `a, b` on the left of `=` / `:=`: a list of two or more targets.
43    pub(super) fn is_multi_target(&self, node: Node) -> bool {
44        self.spec.is_multi_target_node(node.kind()) && node.named_child_count() > 1
45    }

47    /// Bind `rhs` to every plain-variable element of a multi-target list;
48    /// `_` is skipped. The right side is a tuple, so any element may carry
49    /// the taint.
50    pub(super) fn bind_each_target(&mut self, list: Node, rhs: Operand) {
51        let mut cursor = list.walk();
52        let targets: Vec<Node> = list.named_children(&mut cursor).collect();
53        for target in targets {
54            if self.source[target.start_byte()..target.end_byte()].trim() == "_" {
55                continue;
56            }
57            if let Some(LValue::Variable(dest)) = self.visit_lvalue(target) {
58                self.mark_declared(dest);
59                self.ir.push_instruction(
60                    self.current_block,
61                    Instruction::Assign {
62                        dest,
63                        src: rhs.clone(),
64                    },
65                );
66                // A tuple element is not the whole RHS value.
67                self.forget_const(dest);
68            }
69        }
70    }

72    /// Value stored by a field or element assignment.
73    ///
74    /// For a plain assignment this is `rhs`. For a compound assignment
75    /// (`op` is `Some`) it first loads the current value with the
76    /// instruction `load(cur)` builds, then returns the temporary holding
77    /// `cur <op> rhs`.
78    fn compound_store_value(
79        &mut self,
80        node: Node,
81        op: Option<&str>,
82        rhs: Operand,
83        load: impl FnOnce(VarId) -> Instruction,
84    ) -> Operand {
85        let Some(op) = op else {
86            return rhs;
87        };
88        let (_, tmp) = self.read_modify(node, op, rhs, load);
89        Operand::Var(tmp)
90    }
91
92    /// Loads the current value with `load(cur)`, then defines a temporary
93    /// holding `cur <op> rhs`. Returns `(cur, result)`.
94    ///
95    /// The one lowering of a read-modify-write on a field or element: both
96    /// compound assignment (`o.f += 2`) and update expressions (`o.f++`,
97    /// which also needs `cur` for the postfix value) go through it.
98    pub(super) fn read_modify(
99        &mut self,
100        node: Node,
101        op: &str,
102        rhs: Operand,
103        load: impl FnOnce(VarId) -> Instruction,
104    ) -> (VarId, VarId) {
105        let cur = self.new_temp(node);
106        self.ir.push_instruction(self.current_block, load(cur));
107        let tmp = self.new_temp(node);
108        self.ir.push_instruction(
109            self.current_block,
110            Instruction::BinaryOp {
111                dest: tmp,
112                op: op.to_string(),
113                lhs: Operand::Var(cur),
114                rhs,
115            },
116        );
117        (cur, tmp)
118    }

120    /// Lowers an assignment `lhs = rhs`; the field names of the two sides come from the language spec's role for the node.
121    ///
122    /// A variable target becomes an `Assign`, a field target a `StoreField` and an element target a `StoreElement`. A compound operator (`+=`, `*=`) lowers as a read-modify-write: a `BinaryOp` of the target and the right side, then the store. A multi-target left side (`a, b = f()`) binds the right side to every plain-variable target.
123    ///
124    /// Returns the assigned variable when the target is a plain variable (so `a = b = c` and `if (!(p = malloc(n)))` see the variable), the right-hand operand for a multi-target assignment, and `None` for a field or element target.
125    pub(crate) fn visit_assignment(
126        &mut self,
127        node: Node,
128        lhs_field: &str,
129        rhs_field: &str,
130    ) -> Option<Operand> {
131        // Some grammars wrap the target and/or value in a list node: Go's
132        // `assignment_statement` gives `expression_list` for both sides, so
133        // `q = v` reached `visit_lvalue(expression_list)` (role Other) and the
134        // store was dropped entirely - taint assigned inside an `if`/loop
135        // body never reached the variable read after the join. Unwrap the
136        // single-target form the same way declarations already do
137        // (`short_var_declaration` was unaffected, which is why its binding
138        // survived while plain assignment did not).
139        let left_node = self
140            .spec
141            .unwrap_declarator_node(node.child_by_field_name(lhs_field)?);
142        let right_node = self
143            .spec
144            .unwrap_declarator_node(node.child_by_field_name(rhs_field)?);
145
146        let rhs_op = self.visit_node(right_node).unwrap_or(Operand::Unknown);
147        let compound = self.compound_operator(node);
148
149        // A pattern target (`({ a } = rhs)`, `[a, b] = rhs`) is not a
150        // variable, field or element: each name binds to a load off the
151        // value, exactly as the declaration form (`const { a } = rhs`) does.
152        // Without this the target had no lvalue, so no store was emitted and
153        // every name read an undefined variable afterwards.
154        if compound.is_none()
155            && self.is_destructuring_pattern(left_node.kind())
156            && let Operand::Var(base) = rhs_op.clone()
157        {
158            self.bind_pattern(left_node, base);
159            return Some(rhs_op);
160        }
161
162        // Multi-target form (`body, err := f(x)`, `a, b = g()`): the call's
163        // result is a tuple, and any element of it can carry the taint. The
164        // single-target path below saw an `expression_list` lvalue and
165        // dropped the binding, so every Go `v, err :=` and Python tuple
166        // unpack lost its value flow.
167        if compound.is_none() && self.is_multi_target(left_node) {
168            self.bind_each_target(left_node, rhs_op.clone());
169            return Some(rhs_op);
170        }
171        let lval = self.visit_lvalue(left_node);
172        // The value of an assignment expression is the assigned variable:
173        // `if (!(p = malloc(n)))`, `while ((c = next()) != EOF)`,
174        // `a = b = c`. It used to be Unknown, so a condition over an
175        // assignment hid its operand and null-guard proofs never matched
176        // (the leak check reported the failed-allocation path as a leak).
177        let result = match &lval {
178            Some(LValue::Variable(dest)) => Some(Operand::Var(*dest)),
179            _ => None,
180        };
181
182        match lval {
183            Some(LValue::Variable(dest)) => match &compound {
184                Some(op) => {
185                    // dest = dest <op> rhs
186                    let tmp = self.new_temp(node);
187                    self.ir.push_instruction(
188                        self.current_block,
189                        Instruction::BinaryOp {
190                            dest: tmp,
191                            op: op.clone(),
192                            lhs: Operand::Var(dest),
193                            rhs: rhs_op,
194                        },
195                    );
196                    self.ir.push_instruction(
197                        self.current_block,
198                        Instruction::Assign {
199                            dest,
200                            src: Operand::Var(tmp),
201                        },
202                    );
203                    self.forget_const(dest);
204                }
205                None => {
206                    self.ir.push_instruction(
207                        self.current_block,
208                        Instruction::Assign {
209                            dest,
210                            src: rhs_op.clone(),
211                        },
212                    );
213                    self.record_const(dest, right_node, &rhs_op);
214                }
215            },
216            Some(LValue::Field { base, field }) => {
217                let mem_in = self.memory_var;
218                let src = self.compound_store_value(node, compound.as_deref(), rhs_op, |cur| {
219                    Instruction::LoadField {
220                        dest: cur,
221                        mem_in,
222                        base,
223                        field: field.clone(),
224                    }
225                });
226                self.store_field(base, field, src, Some((node.start_byte(), node.end_byte())));
227            }
228            Some(LValue::Element { base, index }) => {
229                let mem_in = self.memory_var;
230                let src = self.compound_store_value(node, compound.as_deref(), rhs_op, |cur| {
231                    Instruction::LoadElement {
232                        dest: cur,
233                        mem_in,
234                        base,
235                        index: index.clone(),
236                    }
237                });
238                self.ir.push_instruction(
239                    self.current_block,
240                    Instruction::StoreElement {
241                        mem_out: self.memory_var,
242                        mem_in: self.memory_var,
243                        base,
244                        index,
245                        src,
246                    },
247                );
248            }
249            None => {}
250        }
251        result
252    }
