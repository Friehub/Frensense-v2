5//! Shared analysis harness: the ONE implementation of "source text → lowered,
6//! SSA'd, interprocedurally-linked program graph" used by the scanner, the
7//! e2e drivers, and the bundler's fact extractor.
8//!
9//! The bundler MUST lower corpus pairs through this same harness (not its own
10//! walker) so that facts are extracted from exactly the IR the engine scans.
11

20/// Lower a single source file into a set of SSA'd `FunctionIR`s keyed by
21/// function name.
22///
23/// Handles the module shapes that matter for app code:
24/// * `function_declaration` / `method_definition` bodies,
25/// * arrow/function-expression handlers registered via
26///   `app.get|post|put|delete|use|all(path, handler)`, the dominant
27///   Express-style shape,
28/// * nested arrows inside those handlers.
29///
30/// Handler names are synthetic and file-position-stable
31/// (`<path:handler@byte>`), so bundles derived from the same file hash to the
32/// same keys.
33///
34/// ```
35/// use frensense_engine::harness::lower_source;
36///
37/// let src = "declare const app: any\n\
38///            app.get('/x', (req: any, res: any) => { res.send(req.query.a) })\n\
39///            function f() { return 1 }";
40/// let fns = lower_source("t.ts", src, "ts").unwrap();
41/// // Ordinary declarations are keyed by their name.
42/// assert!(fns.contains_key("f"));
43/// // The route handler gets a synthetic name built from the path and the byte
44/// // offset of the handler, here the start of the arrow function's parameters.
45/// let offset = src.find("(req").unwrap();
46/// assert!(fns.contains_key(&format!("<t.ts:handler@{offset}>")));
47/// ```
48pub fn lower_source(
49    path: &str,
50    source: &str,
51    ext: &str,
52) -> Result<FxHashMap<String, FunctionIR>, String> {
53    lower_source_with_facts(path, source, ext, None)
54}

56/// Lower a single source file into SSA'd `FunctionIR`s with optional bundle facts
57/// that supply dynamic grammar classifications and features.
58pub fn lower_source_with_facts(
59    path: &str,
60    source: &str,
61    ext: &str,
62    facts: Option<&FactTable>,
63) -> Result<FxHashMap<String, FunctionIR>, String> {
64    Ok(lower_source_detailed(path, source, ext, facts)?.0)
65}

67/// Lower one file and report whether its parse tree contained errors.
68///
69/// tree-sitter does not fail on a syntax error: it hands back a tree full of
70/// `ERROR`/missing nodes and the lowering extracts whatever survived, so a
71/// file that is a third unparseable used to be analyzed as if it were intact
72/// and reported as clean. The second element carries that fact so the scan
73/// can say "partially analyzed" instead.
74pub(crate) fn lower_source_detailed(
75    path: &str,
76    source: &str,
77    ext: &str,
78    facts: Option<&FactTable>,
79) -> Result<(FxHashMap<String, FunctionIR>, bool), String> {
80    let spec = frensense_lang::spec_for_ext(ext)
81        .ok_or_else(|| format!("no language spec for extension '{ext}'"))?;
82
83    // Single-file-container formats (e.g. Vue SFCs) rewrite the source
84    // before parsing: the spec blanks non-code markup into same-length
85    // whitespace, so every byte offset, and every reported line/column,
86    // still points at the original file.
87    let rewritten = spec.preprocess(source);
88    let source: &str = rewritten.as_ref();
89
90    let mut parser = tree_sitter::Parser::new();
91    parser
92        .set_language(&spec.tree_sitter_language())
93        .map_err(|e| format!("parser init failed: {e}"))?;
94    let tree = parser
95        .parse(source, None)
96        .ok_or_else(|| format!("parse failed for {path}"))?;
97
98    let mut irs: FxHashMap<String, FunctionIR> = FxHashMap::default();
99    // Spans of function values already lowered under their binding name
100    // (lexical const / route registration). The generic walker must not
101    // lower the same node again under a positional `<fn@N>` name: the twin
102    // duplicates every finding and policy segment of the bound function.
103    let mut bound_fns: FxHashSet<usize> = FxHashSet::default();
104    let mut stack = vec![tree.root_node()];
105    while let Some(node) = stack.pop() {
106        // Language-driven function detection: JS/TS `function_declaration` /
107        // `method_definition`, Python `function_definition` (+ async),
108        // Rust `function_item`, Go `function_declaration` / `method_declaration`.
109        if spec.is_function_node(node.kind()) {
110            let name = function_name_of(node, source, spec);
111            // Skip only bindings whose lexical declaration or route
112            // registration already lowered this node under its binding
113            // name. EVERY other function - including nameless nested ones
114            // - is extracted: with inlining gone, extraction is its only
115            // home, and call sites resolve to it by name or by cross edge.
116            let is_twin = name.is_none() && bound_fns.contains(&node.start_byte());
117            if !is_twin {
118                let name = name.unwrap_or_else(|| format!("<fn@{}>", node.start_byte()));
119                let mut ir = lower_one(&name, node, source, spec, facts);
120                ir.enclosing_fn = enclosing_extracted_name(node, source, path, spec, facts);
121                insert_ir(&mut irs, ir, node.start_byte());
122            }
123        }
124        // Top-level arrow functions bound to consts/lets, the dominant
125        // shape for utility libraries (`export const hash = (d) => ...`).
126        if spec.is_variable_declaration_statement(node.kind()) {
127            let mut dc = node.walk();
128            for decl in node.named_children(&mut dc) {
129                if !spec.is_variable_declarator(decl.kind()) {
130                    continue;
131                }
132                crate::dbg_trace!(
133                    crate::debug_flags::DebugFlags::get().static_trace,
134                    "[static] decl kind={} value={:?}",
135                    decl.kind(),
136                    decl.child_by_field_name("value").map(|v| v.kind())
137                );
138                let Some(value) = decl.child_by_field_name("value") else {
139                    continue;
140                };
141                let name = decl
142                    .child_by_field_name("name")
143                    .map(|n| source[n.start_byte()..n.end_byte()].to_string())
144                    .unwrap_or_else(|| format!("<fn@{}>", decl.start_byte()));
145                if spec.is_function_value(value.kind()) {
146                    bound_fns.insert(value.start_byte());
147                    let mut ir = lower_one(&name, value, source, spec, facts);
148                    ir.enclosing_fn = enclosing_extracted_name(value, source, path, spec, facts);
149                    insert_ir(&mut irs, ir, value.start_byte());
150                } else if !inside_function(node, spec) {
151                    // Top-level non-function static initializer: allowlist
152                    // Sets, config objects, schema builders. Lowered as
153                    // pseudo-IR so the checker layer can assert policy over
154                    // them. A NESTED declaration is skipped: its value is
155                    // already lowered as part of the enclosing function's
156                    // body, and extracting it again created a phantom
157                    // top-level function attributed to no source construct
158                    // (`const r = runTool(cmd)` inside a handler fired
159                    // policy twice, once for `r`, once for the handler).
160                    let ir = lower_static(&name, value, source, spec, facts);
161                    insert_ir(&mut irs, ir, value.start_byte());
162                }
163            }
164        }
165        // Route-registration handlers: `app.get("/", (req, res) => ...)`.
166        if let NodeRole::Call {
167            callee_field,
168            args_field,
169        } = spec.classify(node.kind())
170            && let (Some(callee), Some(args)) = (
171                node.child_by_field_name(callee_field.as_ref()),
172                node.child_by_field_name(args_field.as_ref()),
173            )
174        {
175            let callee_text = &source[callee.start_byte()..callee.end_byte()];
176            if is_route_registration(callee_text, spec, facts) {
177                let mut ac = args.walk();
178                let arg_nodes: Vec<_> = args.named_children(&mut ac).collect();
179                for a in &arg_nodes {
180                    if spec.is_function_value(a.kind()) {
181                        let name = format!("<{}:handler@{}>", path, a.start_byte());
182                        bound_fns.insert(a.start_byte());
183                        let mut ir = lower_one(&name, *a, source, spec, facts);
184                        ir.enclosing_fn = enclosing_extracted_name(*a, source, path, spec, facts);
185                        insert_ir(&mut irs, ir, a.start_byte());
186                    }
187                }
188            }
189        }
190        // Named children only: the anonymous `function` keyword token is
191        // classified as a Function node by the JS spec, and visiting it
192        // spawned an empty `<fn@N>` pseudo-function for every declaration.
193        let mut c = node.walk();
194        let kids: Vec<_> = node.named_children(&mut c).collect();
195        for k in kids.into_iter().rev() {
196            stack.push(k);
197        }
198    }
199    // Top-level executable code. Taint and policy checks only ever saw
200    // function bodies, so the same two statements that were reported inside
201    // `def handler():` were silent at file scope - CLI tools, ETL/notebook
202    // scripts, Django `settings.py`, and an
203    // `if __name__ == "__main__": app.run(debug=True)` guard were invisible.
204    if let Some(ir) = lower_module_scope(tree.root_node(), source, spec, facts) {
205        insert_ir(&mut irs, ir, tree.root_node().start_byte());
206    }
207    Ok((irs, tree.root_node().has_error()))
208}

210/// Lower the executable statements that sit directly in a file's module scope
211/// into one pseudo-function IR.
212///
213/// Only *root-level* named children are visited:
214///
215/// * a function is its own IR (the extraction walk owns it) - lowering it
216///   here would report every finding twice under two names;
217/// * an import contributes no code;
218/// * a JS/TS `const`/`let`/`var` statement is already lowered by the binding
219///   walk (function values) or as a static pseudo-IR (everything else), which
220///   is where policy checks read module-level config today.
221///
222/// The last point is also the known limit: when a module-level binding feeds a
223/// later module-level sink (`const input = process.argv[2]; execSync(input)`),
224/// the two statements live in different IRs, so the source is not linked to
225/// that sink. That shape stays a function-only finding.
226///
227/// `None` when the file has no module-level code that produces an instruction,
228/// so a function-only file (the common case, and every corpus file) gains no
229/// empty pseudo-function.
230fn lower_module_scope(
231    root: tree_sitter::Node,
232    source: &str,
233    spec: &'static dyn frensense_lang::spec::LanguageSpec,
234    facts: Option<&FactTable>,
235) -> Option<FunctionIR> {
236    let mut ctx = LoweringContext::new_with_facts(spec, source, "<module>".to_string(), facts);
237    let mut cursor = root.walk();
238    let children: Vec<_> = root.named_children(&mut cursor).collect();
239    for child in children {
240        if module_scope_skip(child, spec) {
241            continue;
242        }
243        ctx.visit_node(child);
244    }
245    if ctx.ir.blocks.values().all(|b| b.instructions.is_empty()) {
246        return None;
247    }
248    let mut ir = SSABuilder::new(ctx.ir).build();
249    ir.language = spec.fact_language().to_string();
250    // The pseudo-function is a module's top level, never a call target:
251    // there is no source name a call site could resolve to it.
252    ir.is_static_data = true;
253    Some(ir)
254}

256/// Should the module-scope IR leave this root-level node to another lowering?
257fn module_scope_skip(
258    node: tree_sitter::Node<'_>,
259    spec: &dyn frensense_lang::spec::LanguageSpec,
260) -> bool {
261    let kind = node.kind();
262    if spec.is_function_node(kind) || spec.import_node_kinds().contains(&kind) {
263        return true;
264    }
265    if spec.is_variable_declaration_statement(kind) {
266        return true;
267    }
268    // `export const x = ...` / `export function f` wrap the same nodes; a
269    // default export of an expression (`export default runTool(cmd)`) does
270    // execute at import time and stays in the module IR.
271    if matches!(spec.classify(kind), NodeRole::Export) {
272        let mut cursor = node.walk();
273        return node.named_children(&mut cursor).any(|child| {
274            spec.is_function_node(child.kind())
275                || spec.is_variable_declaration_statement(child.kind())
276        });
277    }
278    false
279}

281/// Insert an IR keyed by its name, disambiguating on collision with the
282/// function node's byte offset. One file legitimately contains several
283/// same-named functions (four `set` object setters, several `handler`s):
284/// the old `entry().or_insert()` kept the first and silently erased every
285/// later twin's body from all analysis (the Juice Shop `weakPassword`
286/// miss). The kept name is the first one in walk order, so keys stay
287/// content-deterministic.
288fn insert_ir(irs: &mut FxHashMap<String, FunctionIR>, mut ir: FunctionIR, start: usize) {
289    if irs.contains_key(&ir.name) {
290        let base = ir.name.clone();
291        let mut key = format!("{base}@{start}");
292        let mut n = 0usize;
293        while irs.contains_key(&key) {
294            n += 1;
295            key = format!("{base}@{start}#{n}");
296        }
297        ir.name = key;
298    }
299    irs.insert(ir.name.clone(), ir);
300}

302/// True when `node` sits inside any function body (a named ancestor
303/// classifies as Function). Only used to skip NESTED non-function static
304/// initializers: their value is lowered as part of the enclosing function.
305fn inside_function(node: tree_sitter::Node, spec: &dyn frensense_lang::spec::LanguageSpec) -> bool {
306    let mut cur = node.parent();
307    while let Some(ancestor) = cur {
308        if spec.is_function_node(ancestor.kind()) {
309            return true;
310        }
311        cur = ancestor.parent();
312    }
313    false
314}

316/// The declaration-level name of a function node: the `name` field, or the
317/// grammar's declarator chain (`function_definition -> declarator -> ...`),
318/// which the language spec resolves.
319fn function_name_of(
320    node: tree_sitter::Node,
321    source: &str,
322    spec: &dyn frensense_lang::spec::LanguageSpec,
323) -> Option<String> {
324    spec.function_name_node(node)
325        .map(|n| source[n.start_byte()..n.end_byte()].to_string())
326}

328/// The program name under which `fn_node` was extracted, mirroring every
329/// extraction branch in [`lower_source_with_facts`]. Always `Some` for a
330/// function node - nameless, unbound ones get the positional `<fn@byte>`
331/// name. Used to resolve an extracted function's lexical parent.
332fn extracted_ancestor_name(
333    fn_node: tree_sitter::Node,
334    source: &str,
335    path: &str,
336    spec: &dyn frensense_lang::spec::LanguageSpec,
337    facts: Option<&FactTable>,
338) -> Option<String> {
339    // Direct declaration name (function_declaration, method_definition,
340    // named function expression) - same order as extraction.
341    if let Some(name) = function_name_of(fn_node, source, spec) {
342        return Some(name);
343    }
344    // Bound to a const/let: the lexical branch extracts it under the
345    // binding's name (`const helper = (x) => …`).
346    if let Some(parent) = fn_node.parent()
347        && spec.is_variable_declarator(parent.kind())
348        && parent.child_by_field_name("value") == Some(fn_node)
349        && spec.is_function_value(fn_node.kind())
350    {
351        let name = parent
352            .child_by_field_name("name")
353            .map(|n| source[n.start_byte()..n.end_byte()].to_string())
354            .unwrap_or_else(|| format!("<fn@{}>", parent.start_byte()));
355        return Some(name);
356    }
357    // Route-registration handler: `app.get("/", handler)`.
358    if let Some(args) = fn_node.parent()
359        && matches!(spec.classify(args.kind()), NodeRole::Arguments)
360        && let Some(call) = args.parent()
361        && let NodeRole::Call { callee_field, .. } = spec.classify(call.kind())
362        && let Some(callee) = call.child_by_field_name(callee_field.as_ref())
363        && is_route_registration(&source[callee.start_byte()..callee.end_byte()], spec, facts)
364    {
365        return Some(format!("<{}:handler@{}>", path, fn_node.start_byte()));
366    }
367    // Nameless, unbound function: extracted positionally by byte offset.
368    Some(format!("<fn@{}>", fn_node.start_byte()))
369}

371/// The program name of the innermost enclosing extracted function for a
372/// function node at `node`'s position, or None at module scope.
373fn enclosing_extracted_name(
374    node: tree_sitter::Node,
375    source: &str,
376    path: &str,
377    spec: &dyn frensense_lang::spec::LanguageSpec,
378    facts: Option<&FactTable>,
379) -> Option<String> {
380    let mut cur = node.parent();
381    while let Some(ancestor) = cur {
382        if spec.is_function_node(ancestor.kind()) {
383            return extracted_ancestor_name(ancestor, source, path, spec, facts);
384        }
385        cur = ancestor.parent();
386    }
387    None
388}

390/// Route-registration predicate: a call site is a route handler when the
391/// spec's hard verb suffixes match the callee, OR when pack vocabulary
392/// (`facts.route_patterns`) lists a registration shape for that spec's
393/// language.
394///
395/// Used by the harness's IR lowering to extract route handlers, and by
396/// the backward engine's per-call-site resolution (when wired via
397/// [`EngineBuilder::route_registration`]) so a framework origin arms
398/// only at route handlers (schema §5.1).
399pub(crate) fn is_route_registration(
400    callee_text: &str,
401    spec: &dyn frensense_lang::spec::LanguageSpec,
402    facts: Option<&FactTable>,
403) -> bool {
404    if spec
405        .known_route_verbs()
406        .iter()
407        .any(|s| callee_text.ends_with(s))
408    {
409        return true;
410    }
411    // Receiver-qualified patterns (`app.patch(`) match on their stem, so
412    // they extend the verb list wherever the spec is narrower than the
413    // source dialect. The patterns are pack vocabulary (Phase 6.4): the
414    // default pack installs them into `FactTable::route_patterns` keyed by
415    // language, so bare lowering without facts gets verbs only.
416    facts
417        .and_then(|f| f.route_patterns.get(spec.fact_language()))
418        .is_some_and(|pats| {
419            pats.iter()
420                .any(|p| callee_text.ends_with(p.trim_end_matches('(')))
421        })
422}

424fn lower_one(
425    name: &str,
426    fn_node: tree_sitter::Node,
427    source: &str,
428    spec: &'static dyn frensense_lang::spec::LanguageSpec,
429    facts: Option<&FactTable>,
430) -> FunctionIR {
431    let mut ctx = LoweringContext::new_with_facts(spec, source, name.to_string(), facts);
432
433    // The list lives wherever the grammar puts it: a `parameters` field for
434    // most languages, one declarator hop further for C (`function_definition →
435    // declarator → function_declarator → parameters`). `function_params`
436    // follows `NodeRole::Function::params_field` until it reaches a
437    // `Parameters` node, so the engine carries no C-specific walk.
438    if let Some(params) = spec.function_params(fn_node) {
439        bind_params(&mut ctx, spec, params, source);
440    }
441    if let Some(body) = fn_node.child_by_field_name("body") {
442        let has_implicit_ret = spec.implicit_return_node(fn_node, body).is_some();
443        let ret_op = ctx.visit_node(body);
444        if has_implicit_ret
445            && let Some(ret) = ret_op
446            && ctx
447                .ir
448                .blocks
449                .get(&ctx.current_block)
450                .is_none_or(|b| matches!(b.terminator, Terminator::None))
451        {
452            ctx.ir
453                .set_terminator(ctx.current_block, Terminator::Return { src: Some(ret) });
454        }
455    }
456    let mut ir = SSABuilder::new(ctx.ir).build();
457    ir.language = spec.fact_language().to_string();
458    ir.entry_guards = spec
459        .decorator_guard_calls(fn_node, source)
460        .into_iter()
461        .map(str::to_owned)
462        .collect();
463    ir
464}

466/// Lower a top-level non-function initializer (`const x = new Set([...])`,
467/// `const cfg = { ... }`, `const tools = { ... tool({...}) ... }`) into a
468/// pseudo-function IR so non-dataflow checkers can assert policy over
469/// static configuration, allowlist definitions, schema builders, feature
470/// flags, that no function body contains.
471fn lower_static(
472    name: &str,
473    value_node: tree_sitter::Node,
474    source: &str,
475    spec: &'static dyn frensense_lang::spec::LanguageSpec,
476    facts: Option<&FactTable>,
477) -> FunctionIR {
478    let mut ctx = LoweringContext::new_with_facts(spec, source, name.to_string(), facts);
479    let init = ctx.visit_node(value_node);
480    // Module-level alias: `const zip = unzipper;`. An identifier has no call
481    // or literal to lower, so the visit yields only the aliased value and the
482    // pseudo-function came out empty - invisible to receiver-origin
483    // resolution, because the consuming function's IR has no def for the
484    // module-level name to walk. Record the binding as an assignment, the
485    // module-level twin of the `Assign` the declaration path emits inside a
486    // function body.
487    if matches!(ctx.classify_node(value_node.kind()), NodeRole::Identifier)
488        && let Some(Operand::Var(src)) = init
489    {
490        let dest = ctx.ir.new_var(VarMetadata {
491            source_name: Some(name.to_string()),
492            type_name: None,
493            byte_range: Some((value_node.start_byte(), value_node.end_byte())),
494            is_memory_state: false,
495            object_keys: Vec::new(),
496            declared: true,
497        });
498        let block = ctx.current_block;
499        ctx.ir.push_instruction(
500            block,
501            Instruction::Assign {
502                dest,
503                src: Operand::Var(src),
504            },
505        );
506    }
507    let mut ir = SSABuilder::new(ctx.ir).build();
508    ir.language = spec.fact_language().to_string();
509    ir.is_static_data = true;
510    ir
511}

513fn bind_params(
514    ctx: &mut LoweringContext,
515    spec: &dyn frensense_lang::spec::LanguageSpec,
516    params: tree_sitter::Node,
517    source: &str,
518) {
519    let mut pc = params.walk();
520    for p in params.children(&mut pc) {
521        if spec.is_self_parameter(p.kind()) {
522            let name = "self".to_string();
523            let v = ctx.ir.new_var(VarMetadata {
524                source_name: Some(name.clone()),
525                type_name: None,
526                byte_range: Some((p.start_byte(), p.end_byte())),
527                is_memory_state: false,
528                object_keys: Vec::new(),
529                declared: true,
530            });
531            ctx.ir.parameters.push(v);
532            ctx.env.last_mut().unwrap().insert(name, v);
533            continue;
534        }
535        // Parameter TYPE annotation (`Query<P>`, `web::Query<P>`,
536        // `*http.Request`), normalized to its path. Source vocabulary that
537        // names a type (the axum/actix extractor family, Go's `*gin.Context`)
538        // matches this: the parameter's type - not the author's variable name -
539        // decides whether the argument is attacker-controlled. One Go
540        // declaration shares its type across every grouped name.
541        let type_name = p
542            .child_by_field_name("type")
543            .map(|t| normalize_type_path(&source[t.start_byte()..t.end_byte()]))
544            .filter(|t| !t.is_empty());
545        for binding in spec.param_bindings(p) {
546            let name = source[binding.start_byte()..binding.end_byte()].to_string();
547            let v = ctx.ir.new_var(VarMetadata {
548                source_name: Some(name.clone()),
549                type_name: type_name.clone(),
550                byte_range: Some((p.start_byte(), p.end_byte())),
551                is_memory_state: false,
552                object_keys: Vec::new(),
553                declared: true,
554            });
555            ctx.ir.parameters.push(v);
556            // A destructured parameter (`function run({ h }: Opts)`) names no
557            // variable of its own: the pattern's names read the parts of the
558            // parameter value, exactly like `const { h } = opts` inside the
559            // body. The slot still occupies the parameter position (so the
560            // call's argument maps to it), but the pattern text is not a name
561            // the body can resolve, so it is not entered into the scope.
562            if ctx.is_destructuring_pattern(binding.kind()) {
563                ctx.bind_pattern(binding, v);
564                continue;
565            }
566            ctx.env.last_mut().unwrap().insert(name, v);
567        }
568    }
569}

571/// Normalize a parameter type annotation to its path: strip references and
572/// mutability (`&mut Foo`), keep the path before the generic arguments
573/// (`web::Query<P>` → `web::Query`), drop whitespace.
574fn normalize_type_path(raw: &str) -> String {
575    // The borrow marker is PRESERVED: framework request extractors are
576    // consumed by value (`Path<P>`, `web::Query<Q>`, `Json<B>`), while a
577    // borrowed type is the language's own value (`&Path` is `std::path::Path`).
578    // Dropping `&` made them indistinguishable, so every helper taking
579    // `&Path` was treated as an axum extractor and became a taint source.
580    let borrowed = raw.trim_start().starts_with('&');
581    let t = raw.trim();
582    let t = t.trim_start_matches('&').trim();
583    // Lifetime parameter: `&'a Path` -> `Path`.
584    let t = t
585        .strip_prefix('\'')
586        .map_or(t, |rest| match rest.find(char::is_whitespace) {
587            Some(i) => rest[i..].trim(),
588            None => rest,
589        });
590    let t = t.trim_start_matches("mut").trim();
591    let t = t.split('<').next().unwrap_or(t);
592    let path: String = t.chars().filter(|c| !c.is_whitespace()).collect();
593    if borrowed {
594        let mut owned = String::with_capacity(path.len() + 1);
595        owned.push('&');
596        owned.push_str(&path);
597        owned
598    } else {
599        path
600    }
601}

890    fn taint_findings(src: &str) -> Vec<crate::analysis::taint::engine::SinkFinding> {
891        use crate::analysis::forward::ProgramSvfg;
892        use crate::analysis::taint::config::TaintConfig;
893        use crate::analysis::taint::engine::BackwardTaintEngine;
894        use crate::ir::function::FunctionIR;
895        use rustc_hash::FxHashMap;
896
897        let fns = lower_source("probe.vue", src, "vue").expect("sfc must lower");
898        let owned: Vec<FunctionIR> = fns.into_values().collect();
899        let mut statics: FxHashMap<String, &FunctionIR> = FxHashMap::default();
900        for ir in &owned {
901            statics.insert(ir.name.clone(), ir);
902        }
903        let mut config = TaintConfig::default();
904        config.sources.insert("location.hash".into());
905        config.sinks.insert("eval".into());
906        let prog = ProgramSvfg::new(&statics, &config);
907        let mut engine = BackwardTaintEngine::new(&prog, &config);
908        engine.run();
909        engine.findings
910    }

912    /// The positive twin: a `.vue` script block taints `eval` from
913    /// `location.hash`. The blanked template must not lose, duplicate or
914    /// shift the finding: the sink span must still point INTO the script
915    /// body of the ORIGINAL file (the position contract).
916    #[test]
917    fn vue_sfc_script_taint_is_found() {
918        use crate::analysis::taint::engine::BackwardVerdict;
919        let findings = taint_findings(VUE_POSITIVE);
920        let vuln: Vec<_> = findings
921            .iter()
922            .filter(|f| f.verdict == BackwardVerdict::Vulnerable)
923            .collect();
924        assert!(
925            !vuln.is_empty(),
926            "eval(location.hash) must alert: {findings:?}"
927        );
928
929        // Position contract: the sink span sits inside the `<script>` body
930        // of the original SFC, not in blanked-out template/style bytes.
931        let body = "eval(payload)";
932        let expected = VUE_POSITIVE.find(body).expect("fixture shape");
933        let span = vuln[0]
934            .sink_span
935            .expect("lowering must record the sink span");
936        assert!(
937            span.0 >= expected && span.0 <= expected + body.len(),
938            "sink span {:?} must cover the eval call at byte {expected}",
939            span
