5//! Missing release of memory after its effective lifetime (CWE-401).
6//!
7//! Path-sensitive may-leak accounting over the intraprocedural CFG: an
8//! allocation is reported when some concrete path reaches function exit
9//! with the allocation still frame-owned - never released, returned, or
10//! stored anywhere that outlives the call.
11//!
12//! Lang declares / corpus teaches / engine computes: which calls allocate
13//! fresh memory and which consume their arguments comes from the
14//! lang-declared memory vocabulary seeded into the
15//! [`crate::analysis::taint::facts::FactTable`] and
16//! merged into the registry (`MemorySummaryRegistry::from_facts` -
17//! spec vocab plus bundle-learned contracts), never from hardcoded names
18//! here. Stack allocators (`alloca` and friends) come from
19//! `FactTable::stack_allocators` (default-pack-seeded since Phase 6.5)
20//! and are never leak candidates: their storage dies with the frame. The
21//! analysis itself (join, phi merge, escape, null-guard skip) is the
22//! stable engine "how".
23//!
24//! Zero-FP by construction:
25//! - The live state is a flow-sensitive variable -> allocations map:
26//!   SSA copies bind the values that actually flow there and phi nodes
27//!   merge per-edge, so a `free(phi)` or `store(phi)` sees every
28//!   allocation the merge can carry.
29//! - Exit states are may-live (union join over predecessors): a finding
30//!   means an actual path to `return` exists on which nothing released
31//!   the allocation.
32//! - Ownership escapes remove the candidate: returned operands, values
33//!   stored to fields/elements/globals, ownership assigned to a
34//!   parameter, derived handles (`&p->field` / load chains) passed to
35//!   callees, and arguments the callee's contract consumes. Direct
36//!   pointer arguments to unknown callees keep frame ownership; a pointer
37//!   retained by an unsummarized callee may therefore be reported as a leak.
38//! - A branch whose condition compares an allocation against NULL
39//!   (`== NULL`, `!= NULL`, `!p`) kills it on the null edge: the failed
40//!   allocation path is not a live object. The proof covers every
41//!   version of that *storage* (SSA re-assignments and the phi that
42//!   merges them): a stale binding a loop join carried along cannot hold
43//!   a live object where the storage itself is NULL. Pointer identity
44//!   guards (`p != buf` taken false) likewise drop allocations bound to
45//!   one side only - `buf` (a stack buffer) holds none, so the edge
46//!   cannot carry the heap object.
47//! - A branch on a literal condition (`while (1)` lowering) propagates
48//!   only along the statically live edge; the placeholder block on the
49//!   dead edge is unreachable code, not a function exit.
50//! - The exit scan counts only *reaching* definitions: per storage cell
51//!   (SSA versions of one variable, grouped by copy/phi edges and shared
52//!   source name), only the deepest definition that dominates the exit is
53//!   live there. Stale versions a loop join carried into an earlier
54//!   block's state are superseded by the reaching key (or its phi), so
55//!   their lingering bindings are not separate leaks.
56//! - Unknown (unsummarized) callees consume nothing.
57

69/// One tracked allocation: the call instruction's result variable plus the

71#[derive(Clone)]
72struct Alloc {
73    name: String,

76/// Alias edge from `src` to `dest`: value-preserving copies
77/// (`Assign`/`Cast`/phi merges) and derived handles (`&p->field`,
78/// loads, dereferences) alike. Resolution decides directness by
79/// whether the live state binds the variable itself.
80#[derive(Clone, Copy)]
81struct Edge {
82    src: VarId,
83    dest: VarId,
84}

86/// Frame-owned allocations per variable at a program point. Direct
87/// bindings only: SSA copies and phi merges bind exactly the values that
88/// flow there, and releases/stores remove the allocation from every
89/// variable holding it.
90type PointsTo = FxHashMap<VarId, FxHashSet<VarId>>;

92/// Fresh allocations (vocab from `facts`) and the alias edges between
93/// variables that can denote them. Stack allocators are excluded: their
94/// storage is frame-local by definition and can never leak.
95fn collect(
96    ir: &FunctionIR,
97    summaries: &MemorySummaryRegistry,
98    facts: &crate::analysis::taint::facts::FactTable,
99) -> (FxHashMap<VarId, Alloc>, Vec<Edge>) {
100    let mut allocs: FxHashMap<VarId, Alloc> = FxHashMap::default();
101    let mut edges: Vec<Edge> = Vec::new();
102    for block in ir.blocks.values() {
103        for phi in &block.phis {
104            for (_, arg) in &phi.incoming {
105                edges.push(Edge {
106                    src: *arg,
107                    dest: phi.dest,
108                });
109            }
110        }
111        for instr in &block.instructions {
112            match instr {
113                Instruction::CallStatic {
114                    dest: Some(dest),
115                    func,
116                    ..
117                } if summaries.returns_fresh(func) && !facts.is_stack_allocator(func) => {
118                    allocs.insert(
119                        *dest,
120                        Alloc {
121                            name: last_segment(func).to_string(),
122                        },
123                    );
124                }
125                Instruction::CallVirtual {
126                    dest: Some(dest),
127                    method,
128                    ..
129                } if summaries.returns_fresh(method) && !facts.is_stack_allocator(method) => {
130                    allocs.insert(
131                        *dest,
132                        Alloc {
133                            name: last_segment(method).to_string(),
134                        },
135                    );
136                }
137                Instruction::Assign {
138                    dest,
139                    src: Operand::Var(src),
140                }
141                | Instruction::Cast {
142                    dest,
143                    src: Operand::Var(src),
144                    ..
145                } => edges.push(Edge {
146                    src: *src,
147                    dest: *dest,
148                }),
149                Instruction::AddressOf { dest, src } => edges.push(Edge {
150                    src: *src,
151                    dest: *dest,
152                }),
153                // `ptr + OFFSET` / `(char *)ptr + n` is a derived handle to
154                // the same object (allocators that hand back the address
155                // past a size header). C lowers the addition as `concat`.
156                Instruction::BinaryOp { dest, op, lhs, rhs }
157                    if matches!(op.as_str(), "+" | "-" | "add" | "sub" | "concat") =>
158                {
159                    for operand in [lhs, rhs] {
160                        if let Operand::Var(v) = operand {
161                            edges.push(Edge {
162                                src: *v,
163                                dest: *dest,
164                            });
165                        }
166                    }
167                }
168                Instruction::LoadField { dest, base, .. }
169                | Instruction::LoadElement { dest, base, .. } => edges.push(Edge {
170                    src: *base,
171                    dest: *dest,
172                }),
173                Instruction::Dereference {
174                    dest,
175                    ptr: Operand::Var(v),
176                    ..
177                }
178                | Instruction::ExtractValue {
179                    dest,
180                    tuple: Operand::Var(v),
181                    ..
182                } => edges.push(Edge {
183                    src: *v,
184                    dest: *dest,
185                }),
186                _ => {}
187            }
188        }
189    }
190    (allocs, edges)
191}

193/// Reverse alias graph: `dest -> [src, ...]` over every alias edge, for
194/// resolving a variable the live state does not bind directly (a derived
195/// handle, or a value whose binding was killed on this path).
196fn alias_parents(edges: &[Edge]) -> FxHashMap<VarId, Vec<VarId>> {
197    let mut parents: FxHashMap<VarId, Vec<VarId>> = FxHashMap::default();
198    for e in edges {
199        parents.entry(e.dest).or_default().push(e.src);
200    }
201    parents
202}

204/// Resolve `op` to the live frame-owned allocations it may denote.
205/// State-first: if the variable itself holds allocations, the callee
206/// receives that very pointer value (direct). Otherwise walk the reverse
207/// alias edges into the state - a derived handle or a stale copy, which
208/// hands the object off when passed along. Never resurrects an
209/// allocation the state no longer holds.
210fn resolve(
211    op: &Operand,
212    state: &PointsTo,
213    parents: &FxHashMap<VarId, Vec<VarId>>,
214) -> (FxHashSet<VarId>, bool) {
215    let Operand::Var(v) = op else {
216        return (FxHashSet::default(), false);
217    };
218    if let Some(set) = state.get(v) {
219        return (set.clone(), true);
220    }
221    let mut found: FxHashSet<VarId> = FxHashSet::default();
222    let mut seen: FxHashSet<VarId> = FxHashSet::default();
223    let mut stack: Vec<VarId> = vec![*v];
224    while let Some(u) = stack.pop() {
225        for p in parents.get(&u).into_iter().flatten() {
226            if !seen.insert(*p) {
227                continue;
228            }
229            if let Some(set) = state.get(p) {
230                found.extend(set.iter().copied());
231            } else {
232                stack.push(*p);
233            }
234        }
235    }
236    (found, false)
237}

239/// Drop an allocation from every variable that holds it.
240fn drop_alloc(state: &mut PointsTo, id: VarId) {
241    for set in state.values_mut() {
242        set.remove(&id);
243    }
244}

246/// Statically dead edge: a branch on a literal condition (`while (1)`
247/// lowering) never takes one side. Lowering parks post-loop code and
248/// empty placeholder blocks behind that dead edge - propagating state
249/// there would invent a function exit that no execution can reach.
250fn dead_edge(pred: &BasicBlock, succ: BlockId) -> bool {
251    let Terminator::Branch {
252        cond: Operand::IntLiteral(n),
253        true_block,
254        false_block,
255    } = &pred.terminator
256    else {
257        return false;
258    };
259    if true_block == false_block {
260        return false;
261    }
262    let taken = if *n != 0 { *true_block } else { *false_block };
263    taken != succ
264}

266/// Union-find root of `v`'s storage cell, with path compression. Cells
267/// are keyed by every member variable; an unregistered variable is its
268/// own root.
269fn cell_find(cell: &mut FxHashMap<VarId, VarId>, v: VarId) -> VarId {
270    let mut root = v;
271    while let Some(&p) = cell.get(&root) {
272        if p == root {
273            break;
274        }
275        root = p;
276    }
277    let mut cur = v;
278    while cur != root {
279        let Some(&p) = cell.get(&cur) else {
280            break;
281        };
282        cell.insert(cur, root);
283        cur = p;
284    }
285    root
286}

288/// `a` and `b` are versions of the same storage cell.
289fn cell_union(cell: &mut FxHashMap<VarId, VarId>, a: VarId, b: VarId) {
290    cell.entry(a).or_insert(a);
291    cell.entry(b).or_insert(b);
292    let (ra, rb) = (cell_find(cell, a), cell_find(cell, b));
293    if ra != rb {
294        cell.insert(rb, ra);
295    }
296}

298/// Definition block plus the two variable groupings the check needs:
299///
300/// * the exit-scan *cell* - union over value-preserving edges (copies,
301///   casts, phi merges) and shared source names, so SSA versions of one
302///   source variable form one cell (derived handles are excluded:
303///   `&p->field` is a different value, not another version of `p`);
304/// * the *storage* - phi merges and shared source names only, no plain
305///   copies: `q = p` gives two variables the same value, not the same
306///   storage, so a proof about `q` says nothing about `p`.
307fn reaching_index(
308    ir: &FunctionIR,
309) -> (
310    FxHashMap<VarId, BlockId>,
311    FxHashMap<VarId, VarId>,
312    FxHashMap<VarId, VarId>,
313) {
314    let mut def_block: FxHashMap<VarId, BlockId> = FxHashMap::default();
315    let mut cell: FxHashMap<VarId, VarId> = FxHashMap::default();
316    let mut storage: FxHashMap<VarId, VarId> = FxHashMap::default();
317    let mut blocks: Vec<BlockId> = ir.blocks.keys().copied().collect();
318    blocks.sort_by_key(|b| b.0);
319    for bid in blocks {
320        let Some(block) = ir.blocks.get(&bid) else {
321            continue;
322        };
323        for phi in &block.phis {
324            def_block.entry(phi.dest).or_insert(bid);
325            for (_, arg) in &phi.incoming {
326                cell_union(&mut cell, *arg, phi.dest);
327                cell_union(&mut storage, *arg, phi.dest);
328            }
329        }
330        for instr in &block.instructions {
331            let Some(dest) = instr.dest() else {
332                continue;
333            };
334            def_block.entry(dest).or_insert(bid);
335            match instr {
336                Instruction::Assign {
337                    dest,
338                    src: Operand::Var(src),
339                }
340                | Instruction::Cast {
341                    dest,
342                    src: Operand::Var(src),
343                    ..
344                } => cell_union(&mut cell, *src, *dest),
345                _ => {
346                    cell.entry(dest).or_insert(dest);
347                }
348            }
349        }
350    }
351    // A shared source name is the same storage cell even when no copy
352    // edge links the versions (declared, then re-assigned).
353    let mut by_name: FxHashMap<String, VarId> = FxHashMap::default();
354    for (v, meta) in &ir.var_metadata {
355        cell.entry(*v).or_insert(*v);
356        storage.entry(*v).or_insert(*v);
357        if let Some(name) = &meta.source_name {
358            if let Some(&first) = by_name.get(name) {
359                cell_union(&mut cell, first, *v);
360                cell_union(&mut storage, first, *v);
361            } else {
362                by_name.insert(name.clone(), *v);
363            }
364        }
365    }
366    let resolve = |mut map: FxHashMap<VarId, VarId>| {
367        let members: Vec<VarId> = map.keys().copied().collect();
368        let mut root_of: FxHashMap<VarId, VarId> = FxHashMap::default();
369        for v in members {
370            let root = cell_find(&mut map, v);
371            root_of.insert(v, root);
372        }
373        root_of
374    };
375    (def_block, resolve(cell), resolve(storage))
376}

378/// The keys of `state` whose definitions actually reach `exit`. Per
379/// storage cell only the deepest definition dominating the exit counts:
380/// older versions of the cell that survived a join are stale - the
381/// reaching key (or the phi at the join) carries their values - so an
382/// allocation still bound only to a stale key is not live here. With no
383/// reaching candidate the group is kept whole (parameters and lowering
384/// temps without an indexed definition).
385fn reaching_keys(
386    state: &PointsTo,
387    exit: BlockId,
388    doms: &DomSets,
389    def_block: &FxHashMap<VarId, BlockId>,
390    cell_root: &FxHashMap<VarId, VarId>,
391) -> Vec<VarId> {
392    let mut groups: FxHashMap<VarId, Vec<VarId>> = FxHashMap::default();
393    for (v, set) in state {
394        if set.is_empty() {
395            continue;
396        }
397        let root = cell_root.get(v).copied().unwrap_or(*v);
398        groups.entry(root).or_default().push(*v);
399    }
400    let mut keep = Vec::new();
401    for keys in groups.into_values() {
402        let mut deepest: Option<BlockId> = None;
403        for k in &keys {
404            let Some(d) = def_block.get(k).copied() else {
405                continue;
406            };
407            if !doms.get(&exit).is_some_and(|s| s.contains(&d)) {
408                continue;
409            }
410            // Dominators of one node are totally ordered: the deeper
411            // definition is the one the current candidate dominates.
412            deepest = Some(match deepest {
413                None => d,
414                Some(cur) if doms.get(&d).is_some_and(|s| s.contains(&cur)) => d,
415                Some(cur) => cur,
416            });
417        }
418        match deepest {
419            None => keep.extend(keys),
420            Some(best) => keep.extend(keys.into_iter().filter(|k| def_block.get(k) == Some(&best))),
421        }
422    }
423    keep
424}

426/// Add the allocations live at `exit` that reach it (see
427/// [`reaching_keys`]).
428fn extend_leaks(
429    leaked: &mut FxHashSet<VarId>,
430    state: &PointsTo,
431    exit: BlockId,
432    doms: &DomSets,
433    def_block: &FxHashMap<VarId, BlockId>,
434    cell_root: &FxHashMap<VarId, VarId>,
435) {
436    for k in reaching_keys(state, exit, doms, def_block, cell_root) {
437        if let Some(set) = state.get(&k) {
438            leaked.extend(set.iter().copied());
439        }
440    }
441}

443/// Run the allocation-lifetime check over one function.
444///
445/// ```
446/// use frensense_engine::checks::memory_summary::{CapacitySpec, MemorySummary, MemorySummaryRegistry};
447/// use frensense_engine::analysis::taint::facts::FactTable;
448/// use frensense_engine::harness::lower_source;
449///
450/// // `malloc` returns fresh memory, `free` consumes parameter 0.
451/// let mut summaries = MemorySummaryRegistry::default();
452/// summaries.summaries.insert("malloc".into(), MemorySummary {
453///     returns_fresh: true,
454///     return_capacity: CapacitySpec::Unknown,
455///     consumes_params: vec![],
456///     out_params_fresh: vec![],
457/// });
458/// summaries.summaries.insert("free".into(), MemorySummary {
459///     returns_fresh: false,
460///     return_capacity: CapacitySpec::Unknown,
461///     consumes_params: vec![0],
462///     out_params_fresh: vec![],
463/// });
464/// let rules = |src: &str| -> Vec<String> {
465///     let fns = lower_source("t.c", src, "c").unwrap();
466///     frensense_engine::checks::leak::check(&fns["f"], &summaries, &FactTable::default())
467///         .into_iter()
468///         .map(|f| f.rule)
469///         .collect()
470/// };
471///
472/// assert_eq!(rules("void f(void) { char *p = malloc(8); p[0] = 1; }"), ["memory_leak"]);
473/// // Released on the only path: no leak.
474/// assert!(rules("void f(void) { char *p = malloc(8); free(p); }").is_empty());
475/// ```
476pub fn check(
477    ir: &FunctionIR,
478    summaries: &MemorySummaryRegistry,
479    facts: &crate::analysis::taint::facts::FactTable,
480) -> Vec<CheckerFinding> {
481    let (allocs, edges) = collect(ir, summaries, facts);
482    if allocs.is_empty() {
483        return Vec::new();
484    }
485    let parents = alias_parents(&edges);
486
487    // Definition index for branch-condition inspection (null guards).
488    let mut defs: FxHashMap<VarId, &Instruction> = FxHashMap::default();
489    for block in ir.blocks.values() {
490        for instr in &block.instructions {
491            match instr {
492                Instruction::Assign { dest, .. }
493                | Instruction::Cast { dest, .. }
494                | Instruction::BinaryOp { dest, .. }
495                | Instruction::UnaryOp { dest, .. }
496                | Instruction::CallStatic {
497                    dest: Some(dest), ..
498                }
499                | Instruction::LoadField { dest, .. }
500                | Instruction::LoadElement { dest, .. }
501                | Instruction::AddressOf { dest, .. } => {
502                    defs.insert(*dest, instr);
503                }
504                _ => {}
505            }
506        }
507    }
508
509    // Destinations that outlive the call: parameters, and names that no
510    // declaration in this body binds (file-scope statics, `extern`
511    // globals, `environ`). Assigning an allocation to either hands it
512    // over, exactly like a store to a field.
513    let mut params: FxHashSet<VarId> = ir.parameters.iter().copied().collect();
514    for (var, meta) in &ir.var_metadata {
515        if meta.source_name.is_some() && !meta.declared && !meta.is_memory_state {
516            params.insert(*var);
517        }
518    }
519
520    // Reaching-definition index for the exit scan: dominators plus the
521    // storage-cell grouping of SSA versions.
522    let doms = dominators(ir);
523    let (def_block, cell_root, storage) = reaching_index(ir);
524
525    // Per-block exit state (variable -> live allocations), Jacobi rounds
526    // from the bottom, capped like the value analysis.
527    let mut out_state: FxHashMap<BlockId, PointsTo> = FxHashMap::default();
528    let max_iters = 64 * (ir.blocks.len() + 1);
529    for _ in 0..max_iters {
530        let mut next: FxHashMap<BlockId, PointsTo> = FxHashMap::default();
531        for (bid, block) in &ir.blocks {
532            let mut state = join_edges(ir, &defs, &parents, &storage, &out_state, *bid, facts);
533            transfer(summaries, &allocs, &parents, &params, block, &mut state);
534            next.insert(*bid, state);
535        }
536        if next == out_state {
537            break;
538        }
539        out_state = next;
540    }
541
542    // Final pass over converged states: leaks are allocations still live
543    // at a `return` after the return operand's own escape.
544    let mut leaked: FxHashSet<VarId> = FxHashSet::default();
545    for (bid, block) in &ir.blocks {
546        let mut state = join_edges(ir, &defs, &parents, &storage, &out_state, *bid, facts);
547        transfer(summaries, &allocs, &parents, &params, block, &mut state);
548        // A block that ends the process (`exit`, `abort`, ...) is not a
549        // function exit: nothing outlives it, so nothing leaks past it.
550        if crate::checks::noreturn::block_diverges(block) {
551            continue;
552        }
553        // Explicit returns and fall-off-the-end blocks (`Terminator::None`,
554        // void functions with no trailing `return`) are function exits.
555        match &block.terminator {
556            Terminator::Return { src } => {
557                if let Some(op) = src {
558                    let (ids, _) = resolve(op, &state, &parents);
559                    for id in ids {
560                        drop_alloc(&mut state, id);
561                    }
562                }
563                extend_leaks(&mut leaked, &state, *bid, &doms, &def_block, &cell_root);
564            }
565            Terminator::None => {
566                extend_leaks(&mut leaked, &state, *bid, &doms, &def_block, &cell_root);
567            }
568            _ => {}
569        }
570    }
571
572    let mut out = Vec::new();
573    let mut emitted: FxHashSet<VarId> = FxHashSet::default();
574    for id in leaked {
575        let Some(alloc) = allocs.get(&id) else {
576            continue;
577        };
578        if !emitted.insert(id) {
579            continue;
580        }
581        let span = ir.var_span(id);
582        out.push(CheckerFinding {
583            provenance: Provenance::Spec,
584            function: ir.name.clone(),
585            rule: frensense_lang::rules::MEMORY_LEAK.to_string(),
586            message: String::new(),
587            params: vec![("alloc", alloc.name.clone())],
588            span,
589            severity: String::new(),
590        });
591    }
592    out
593}

595/// Apply one block's instructions to the state: releases kill,
596/// ownership escapes remove, SSA copies rebind, fresh allocations gen.
597fn transfer(
598    summaries: &MemorySummaryRegistry,
599    allocs: &FxHashMap<VarId, Alloc>,
600    parents: &FxHashMap<VarId, Vec<VarId>>,
601    params: &FxHashSet<VarId>,
602    block: &BasicBlock,
603    state: &mut PointsTo,
604) {
605    for instr in &block.instructions {
606        match instr {
607            Instruction::CallStatic {
608                dest, func, args, ..
609            } => {
610                call_transfer(
611                    summaries,
612                    allocs,
613                    parents,
614                    params,
615                    state,
616                    Some(func.as_str()),
617                    args,
618                    *dest,
619                    &[],
620                );
621            }
622            Instruction::CallVirtual {
623                dest,
624                method,
625                receiver,
626                args,
627                ..
628            } => {
629                call_transfer(
630                    summaries,
631                    allocs,
632                    parents,
633                    params,
634                    state,
635                    Some(method.as_str()),
636                    args,
637                    *dest,
638                    std::slice::from_ref(receiver),
639                );
640            }
641            Instruction::CallPointer { dest, args, .. } => {
642                call_transfer(
643                    summaries,
644                    allocs,
645                    parents,
646                    params,
647                    state,
648                    None,
649                    args,
650                    *dest,
651                    &[],
652                );
653            }
654            Instruction::StoreField { src, .. }
655            | Instruction::StoreElement { src, .. }
656            | Instruction::StoreGlobal { src, .. } => {
657                let (ids, _) = resolve(src, state, parents);
658                for id in ids {
659                    drop_alloc(state, id);
660                }
661            }
662            Instruction::Assign { dest, src } | Instruction::Cast { dest, src, .. } => {
663                let held = match src {
664                    Operand::Var(s) => state.get(s).cloned(),
665                    _ => None,
666                };
667                match held {
668                    // Ownership reassigned to a parameter leaves the frame.
669                    Some(set) if params.contains(dest) => {
670                        for id in set {
671                            drop_alloc(state, id);
672                        }
673                    }
674                    Some(set) => {
675                        state.insert(*dest, set);
676                    }
677                    None => {
678                        state.remove(dest);
679                    }
680                }
681            }
682            _ => {}
683        }
684    }
685}

687#[allow(clippy::too_many_arguments)]
688fn call_transfer(
689    summaries: &MemorySummaryRegistry,
690    allocs: &FxHashMap<VarId, Alloc>,
691    parents: &FxHashMap<VarId, Vec<VarId>>,
692    params: &FxHashSet<VarId>,
693    state: &mut PointsTo,
694    func: Option<&str>,
695    args: &[Operand],
696    dest: Option<VarId>,
697    extra: &[Operand],
698) {
699    // Consumed arguments are released / ownership-transferred by contract.
700    let consumes: Vec<usize> = func
701        .and_then(|f| {
702            let c = summaries.consumes_params(f);
703            if c.is_empty() { None } else { Some(c) }
704        })
705        .unwrap_or_default();
706    for (i, arg) in args.iter().enumerate() {
707        if !consumes.contains(&i) {
708            continue;
709        }
710        let (ids, _) = resolve(arg, state, parents);
711        for id in ids {
712            drop_alloc(state, id);
713        }
714    }
715    // Derived handles passed to an unknown callee hand off the object;
716    // direct pointers keep frame ownership unless an explicit contract
717    // consumes the argument (handled in the first loop above).
718    for arg in args.iter().chain(extra.iter()) {
719        let (ids, direct) = resolve(arg, state, parents);
720        if direct {
721            continue;
722        }
723        for id in ids {
724            drop_alloc(state, id);
725        }
726    }
727    // Fresh result joins the frame's live set - unless the call writes it
728    // straight into a parameter, which belongs to the caller's frame.
729    if let Some(d) = dest
730        && allocs.contains_key(&d)
731        && !params.contains(&d)
732    {
733        let mut held = FxHashSet::default();
734        held.insert(d);
735        state.insert(d, held);
736    }
737}

739/// State at `bid`'s entry: union over predecessor exits, each with its
740/// own edge applied - a null-guard edge drops the allocation before the
741/// join (the failed-allocation path carries no live object), and phi
742/// nodes take the predecessor's value for their argument. Predecessors
743/// behind a statically dead edge contribute nothing.
744fn join_edges(
745    ir: &FunctionIR,
746    defs: &FxHashMap<VarId, &Instruction>,
747    parents: &FxHashMap<VarId, Vec<VarId>>,
748    storage: &FxHashMap<VarId, VarId>,
749    out_state: &FxHashMap<BlockId, PointsTo>,
750    bid: BlockId,
751    facts: &crate::analysis::taint::facts::FactTable,
752) -> PointsTo {
753    let mut live: PointsTo = PointsTo::default();
754    if bid == ir.entry_block {
755        return live;
756    }
757    let Some(block) = ir.blocks.get(&bid) else {
758        return live;
759    };
760    for pred in &block.predecessors {
761        let Some(pred_block) = ir.blocks.get(pred) else {
762            continue;
763        };
764        if dead_edge(pred_block, bid) {
765            continue;
766        }
767        let Some(mut p) = out_state.get(pred).cloned() else {
768            continue;
769        };
770        edge_guard_kill(ir, defs, parents, storage, *pred, bid, &mut p, facts);
771        // Phi nodes merge per-edge: this predecessor contributes its own
772        // value for the phi argument.
773        for phi in &block.phis {
774            if let Some((_, arg)) = phi.incoming.iter().find(|(b, _)| b == pred)
775                && let Some(set) = p.get(arg)
776            {
777                let carried: Vec<VarId> = set.iter().copied().collect();
778                p.entry(phi.dest).or_default().extend(carried);
779            }
780        }
781        for (v, set) in p {
782            live.entry(v).or_default().extend(set);
783        }
784    }
785    live
786}

788/// On edge `pred -> succ`, a guard condition proves facts that make the
789/// live state inconsistent with the edge: `== NULL`/`!= NULL`/`!p` says
790/// the storage is NULL on the null edge (a failed allocation, not a
791/// live object) - every version of it, so stale bindings a loop join
792/// carried along go too - and pointer identity (`p != buf` taken false,
793/// `p == buf` taken true) says both variables denote the same object -
794/// so an allocation bound to one but absent from the other cannot be
795/// live here (`buf` itself, a stack buffer or another pointer, never
796/// holds it).
797#[allow(clippy::too_many_arguments)] // guard plumbing, no meaningful grouping
798fn edge_guard_kill(
799    ir: &FunctionIR,
800    defs: &FxHashMap<VarId, &Instruction>,
801    parents: &FxHashMap<VarId, Vec<VarId>>,
802    storage: &FxHashMap<VarId, VarId>,
803    pred: BlockId,
804    succ: BlockId,
805    state: &mut PointsTo,
806    facts: &crate::analysis::taint::facts::FactTable,
807) {
808    let Some(block) = ir.blocks.get(&pred) else {
809        return;
810    };
811    let Terminator::Branch {
812        cond,
813        true_block,
814        false_block,
815    } = &block.terminator
816    else {
817        return;
818    };
819    let on_true = *true_block == succ;
820    let on_false = *false_block == succ;
821    let Operand::Var(c) = cond else {
822        return;
823    };
824    // The C lowering renders the NULL macro as either a Null operand or
825    // a string literal; which literals compare as the null pointer is
826    // spec-owned vocabulary (`FactTable::null_tokens`).
827    let is_nullish = |o: &Operand| match o {
828        Operand::Null | Operand::IntLiteral(0) => true,
829        Operand::StringLiteral(s) => facts.is_null_token(s),
830        _ => false,
831    };
832    let compared = match defs.get(c) {
833        Some(Instruction::BinaryOp { op, lhs, rhs, .. }) => {
834            let eq = matches!(op.as_str(), "==" | "===");
835            let ne = matches!(op.as_str(), "!=" | "!==");
836            if (eq && on_true) || (ne && on_false) {
837                match (lhs, rhs) {
838                    (Operand::Var(v), r) if is_nullish(r) => Some(*v),
839                    (l, Operand::Var(v)) if is_nullish(l) => Some(*v),
840                    (Operand::Var(a), Operand::Var(b)) => {
841                        identity_kill(state, *a, *b);
842                        None
843                    }
844                    _ => None,
845                }
846            } else {
847                None
848            }
849        }
850        Some(Instruction::UnaryOp {
851            op,
852            src: Operand::Var(v),
853            ..
854        }) if (op == "!" || op == "not") && on_true => Some(*v),
855        // Truthiness test, `if (p)` / `while (p)`: the false edge is the
856        // NULL edge. Without it `if (types) free(types);` kept the
857        // allocation live down the "types is NULL" side.
858        Some(Instruction::UnaryOp { .. }) => None,
859        _ if on_false => Some(*c),
860        _ => None,
861    };
862    if let Some(v) = compared {
863        let (ids, _) = resolve(&Operand::Var(v), state, parents);
864        for id in ids {
865            drop_alloc(state, id);
866        }
867        // The proof is about the storage, not one binding: where `v` is
868        // NULL, no version of the same storage can hold a live object.
869        for id in storage_held(state, storage, v) {
870            drop_alloc(state, id);
871        }
872    }
873}
874
875/// Allocations bound to any version of `v`'s storage (phi-linked
876/// versions and shared source names) in `state`.
877fn storage_held(state: &PointsTo, storage: &FxHashMap<VarId, VarId>, v: VarId) -> Vec<VarId> {
878    let root = storage.get(&v).copied().unwrap_or(v);
879    let mut out = Vec::new();
880    for (k, set) in state {
881        if !set.is_empty() && storage.get(k).copied().unwrap_or(*k) == root {
882            out.extend(set.iter().copied());
883        }
884    }
885    out
886}

888/// On an edge proven to hold `a == b`, allocations bound to exactly one
889/// of the two are dropped: one pointer value cannot be a live object on
890/// one side and a different object (or non-heap address) on the other.
891fn identity_kill(state: &mut PointsTo, a: VarId, b: VarId) {
892    let a_held = state.get(&a).cloned().unwrap_or_default();
893    let b_held = state.get(&b).cloned().unwrap_or_default();
894    let mut dead: Vec<VarId> = a_held.difference(&b_held).copied().collect();
895    dead.extend(b_held.difference(&a_held).copied());
896    for id in dead {
897        drop_alloc(state, id);
898    }
