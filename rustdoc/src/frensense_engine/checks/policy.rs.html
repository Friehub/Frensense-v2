5//! Co-occurrence policy checks: the generalized learned-check evaluator.
6//!
7//! A [`PolicyFact`] poses a question at every trigger call: do the policy's
8//! requirements hold in the trigger's scope? A function violating the policy
9//! (trigger present, at least one requirement unsatisfied) yields a
10//! [`CheckerFinding`]. This complements and operates alongside
11//! `LearnedCheckFact` - `unless_guard` is evaluated as a `GuardCall`
12//! requirement, `unless_range_check` as a `RangeCheck` - and adds the
13//! presence forms (`RequireCall`, `NotCall`) plus a cross-function scope,
14//! so bundles can express "privileged action without its audit log",
15//! "dangerous call without any of its sanctioned wrappers", or "trigger
16//! helper defined in a sibling module still counts as enforcement".
17

28/// A taught callable matches a written callee in one of two modes:
29///
30/// - bare fact (`issue_refund`): matched by the callee's last segment, the
31///   dominant learned-check rule that every sink vocabulary shares;
32/// - qualified fact (`payments.issue_refund`, `storage::ops::wipe_volume`):
33///   matched through the full dotted tail against the callee's written
34///   path, so a trigger pinned to one member no longer collides with an
35///   unrelated object that happens to share the verb
36///   (`billing.issue_refund` stays silent for a `payments.issue_refund`
37///   policy).
38///
39/// Presence-level requirements only see bare segment lists (see
40/// `call_segments`), so they keep bare matching; this narrowing applies
41/// exactly where instruction-letter callee paths are available.
42#[inline]
43fn callee_matches(taught: &str, written: &str) -> bool {
44    if !taught.contains('.') && !taught.contains("::") {
45        let seg = super::last_segment(written);
46        // A decorated mirror of the taught operation: a `extractAllTo`
47        // trigger also governs `extractAllToAsync`, the same call in its
48        // promise form.
49        return seg == taught
50            || crate::analysis::taint::facts::mirror_call_base(seg) == Some(taught);
51    }
52    let norm = |s: &str| -> String { s.replace("::", ".") };
53    let t = norm(taught);
54    let w = norm(written);
55    if w == t {
56        return true;
57    }
58    let mut dotted_tail = String::with_capacity(t.len() + 1);
59    dotted_tail.push('.');
60    dotted_tail.push_str(&t);
61    w.ends_with(dotted_tail.as_str())
62}

64/// Package-import context for policy scoping: which file each function
65/// came from, and which external packages each file imports. Policies
66/// carrying `package: Some(pkg)` only fire in functions whose file
67/// imports `pkg` (mirrors the sink/source package-scoping model).
68#[derive(Debug)]
69pub(crate) struct PackageScope<'a> {
70    /// Function name -> source file path.
71    pub fn_file: &'a FxHashMap<String, String>,
72    /// File path -> (local import name -> import record).
73    pub file_package_imports: &'a FxHashMap<String, FxHashMap<String, PackageImport>>,
74}
75
76impl<'a> PackageScope<'a> {
77    /// The import table that applies to `fn_name`, with the single-file fallback.
78    pub(crate) fn imports_for(
79        &self,
80        fn_name: &str,
81    ) -> Option<&'a FxHashMap<String, PackageImport>> {
82        crate::graph::module::imports_for_function(self.fn_file, self.file_package_imports, fn_name)
83    }
84
85    /// Does the file containing `fn_name` import `pkg`?
86    ///
87    /// Uses the taint engine's single-file fallback
88    /// ([`imports_for_function`](crate::graph::module::imports_for_function)).
89    pub(crate) fn fn_imports_package(&self, fn_name: &str, pkg: &str) -> bool {
90        self.imports_for(fn_name)
91            .is_some_and(|m| m.values().any(|i| i.package == pkg))
92    }
93}

95/// Evaluate every policy fact in `facts` over the scanned program.
96///
97/// Function-scoped policies evaluate per function; module-scoped policies
98/// evaluate against the union of all scanned functions (the check pipeline
99/// passes one scan's IR set here, which is the module the policy author
100/// refers to). Active `learned_checks` participate alongside native policies,
101/// ensuring both check representations fire harmoniously.
102///
103/// ```
104/// use frensense_engine::analysis::taint::facts::{FactTable, PolicyFact, PolicyRequirement, PolicyScope};
105/// use frensense_engine::checks::policy::check_program;
106/// use frensense_engine::checks::Provenance;
107/// use frensense_engine::harness::lower_source;
108/// use frensense_engine::ir::function::FunctionIR;
109///
110/// // Rule: a call to `deleteUser` requires an `authorize` call.
111/// let mut facts = FactTable::default();
112/// facts.policy_facts.push((
113///     PolicyFact {
114///         package: None,
115///         rule: "missing_authorize".into(),
116///         when_call: "deleteUser".into(),
117///         require: vec![PolicyRequirement::GuardCall { call: "authorize".into() }],
118///         scope: PolicyScope::Function,
119///         message: String::new(),
120///         severity: "warning".into(),
121///     },
122///     Provenance::Spec,
123/// ));
124/// let src = "declare function authorize(): void\n\
125///            declare function deleteUser(): void\n\
126///            function guarded() { authorize(); deleteUser() }\n\
127///            function unguarded() { deleteUser() }\n\
128///            function guard_after() { deleteUser(); authorize() }\n\
129///            function guard_on_one_path(c: boolean) { if (c) { authorize() } deleteUser() }";
130/// let fns = lower_source("t.ts", src, "ts").unwrap();
131/// let irs: Vec<&FunctionIR> = fns.values().collect();
132/// let mut flagged: Vec<String> = check_program(&irs, &facts).into_iter().map(|f| f.function).collect();
133/// flagged.sort();
134/// // Only a guard that runs before the trigger on every path counts.
135/// assert_eq!(flagged, ["guard_after", "guard_on_one_path", "unguarded"]);
136/// ```
137pub fn check_program(irs: &[&FunctionIR], facts: &FactTable) -> Vec<CheckerFinding> {
138    check_program_scoped(irs, facts, None)
139}

141/// [`check_program`] with package-scope filtering: policies carrying
142/// `package: Some(pkg)` only fire in functions whose file imports `pkg`.
143/// `scope = None` disables filtering (backward-compatible default).
144pub(crate) fn check_program_scoped(
145    irs: &[&FunctionIR],
146    facts: &FactTable,
147    scope: Option<&PackageScope<'_>>,
148) -> Vec<CheckerFinding> {
149    if facts.learned_checks.is_empty() && facts.policy_facts.is_empty() {
150        return Vec::new();
151    }
152
153    // Function index by name for stable ordering; segment sets per function
154    // (computed once per function, not per trigger).
155    let mut segsets: FxHashMap<&str, Vec<String>> = FxHashMap::default();
156    for ir in irs {
157        segsets.insert(ir.name.as_str(), call_segments(ir));
158    }
159    // Program-wide segment set for module-scoped requirements: every call
160    // segment anywhere PLUS every defined function name (a sibling module
161    // that DEFINES the enforcement helper counts as enforcement available;
162    // the definition is the strongest evidence a helper is in scope even if
163    // this scan never sees its call site).
164    let mut module_segs: Vec<String> = segsets.values().flatten().cloned().collect();
165    module_segs.extend(irs.iter().map(|ir| ir.name.clone()));
166    module_segs.sort();
167    module_segs.dedup();
168
169    // Learned checks evaluate alongside native policies (deduped by rule id:
170    // a bundle shipping both shapes of the same rule must not fire twice).
171    // Each entry keeps its own provenance so findings report the origin.
172    let mut native_rules: Vec<&str> = facts
173        .policy_facts
174        .iter()
175        .map(|(p, _)| p.rule.as_str())
176        .collect();
177    native_rules.sort();
178    native_rules.dedup();
179    let mut all_policies: Vec<(PolicyFact, Provenance)> = Vec::new();
180    all_policies.extend(facts.policy_facts.iter().cloned());
181    for (check, prov) in &facts.learned_checks {
182        if native_rules.contains(&check.rule.as_str()) {
183            continue;
184        }
185        all_policies.push((PolicyFact::from_legacy(check), *prov));
186    }
187
188    let mut findings = Vec::new();
189    // Function-scoped import-alias resolution (`from x import A as B`):
190    // a Trigger call spelled with the local alias `B` resolves to the
191    // original symbol `A`, so a check taught at the shipped name still
192    // fires on renamed callers (WI cross-cut: aliasing is callee
193    // resolution, engine-side mechanism, no per-language vocabulary).
194    let mut alias_maps: FxHashMap<&str, FxHashMap<String, String>> = FxHashMap::default();
195    if let Some(scope) = scope {
196        for ir in irs {
197            let imports = scope.imports_for(ir.name.as_str());
198            if let Some(imports) = imports {
199                let aliases: FxHashMap<String, String> = imports
200                    .iter()
201                    .filter_map(|(local, pi)| {
202                        pi.symbol
203                            .as_deref()
204                            .filter(|sym| *sym != local)
205                            .map(|sym| (local.to_string(), sym.to_string()))
206                    })
207                    .collect();
208                if !aliases.is_empty() {
209                    alias_maps.insert(ir.name.as_str(), aliases);
210                }
211            }
212        }
213    }
214    // Dominance for function-scoped guard/require presence: cached per
215    // function across policies (computed at most once per scanned IR).
216    let mut dom_cache: FxHashMap<&str, DomSets> = FxHashMap::default();
217    for (policy, policy_provenance) in &all_policies {
218        // The FULL taught callee goes into trigger matching: a bare fact
219        // matches by last segment, a qualified fact (`payments.issue_refund`)
220        // matches the written callee path (see `callee_matches`).
221        let trigger_call: &str = &policy.when_call;
222        let needs_dom = policy.scope == PolicyScope::Function
223            && policy.require.iter().any(|r| {
224                matches!(
225                    r,
226                    PolicyRequirement::GuardCall { .. } | PolicyRequirement::RequireCall { .. }
227                )
228            });
229        for ir in irs {
230            let segs = match segsets.get(ir.name.as_str()) {
231                Some(s) => s,
232                None => continue,
233            };
234            // Package-scope filter: a policy carrying `package: Some(pkg)`
235            // only fires in functions whose file imports `pkg` (mirrors the
236            // sink/source package-scoping model). When no scope context is
237            // supplied (unit tests, legacy callers) the filter is inert.
238            if let (Some(pkg), Some(scope)) = (&policy.package, scope)
239                && !scope.fn_imports_package(&ir.name, pkg)
240            {
241                continue;
242            }
243            let doms: Option<&DomSets> = if needs_dom {
244                let d: &DomSets = dom_cache
245                    .entry(ir.name.as_str())
246                    .or_insert_with(|| control::dominators(ir));
247                Some(d)
248            } else {
249                None
250            };
251            let val_info = crate::analysis::value::analyze(ir);
252            // Collect every trigger call site in this function with its args.
253            for (bid, idx, args, arg_names, span) in
254                trigger_sites(ir, trigger_call, alias_maps.get(ir.name.as_str()))
255            {
256                // A requirement violation is the finding, but a
257                // presence-only policy (`require` empty) alerts whenever
258                // the trigger call occurs at all: the corpus twins proved
259                // the call site itself is the vulnerability (the negative
260                // avoids the call, not a guard shape around it).
261                let satisfied = !policy.require.is_empty()
262                    && policy.require.iter().all(|req| {
263                        requirement_holds(
264                            req,
265                            ir,
266                            bid,
267                            idx,
268                            doms,
269                            &val_info,
270                            args,
271                            arg_names,
272                            segs,
273                            &module_segs,
274                            policy.scope,
275                        )
276                    });
277                if !satisfied {
278                    findings.push(CheckerFinding {
279                        function: ir.name.clone(),
280                        rule: policy.rule.clone(),
281                        message: policy.message.clone(),
282                        params: Vec::new(),
283                        span,
284                        severity: policy.severity.clone(),
285                        provenance: *policy_provenance,
286                    });
287                }
288            }
289        }
290    }
291    findings
292}

294/// Every call instruction in `ir` whose callee's last segment is `seg`,
295/// as (bid, instr index, args, keyword names, span) tuples. The index
296/// orders same-block guard/trigger pairs; `keyword names` is parallel to
297/// `args` and may be shorter (missing entry = positional).
298type TriggerSite<'a> = (
299    BlockId,
300    usize,
301    &'a [Operand],
302    &'a [String],
303    Option<(usize, usize)>,
304);
305
306fn trigger_sites<'a>(
307    ir: &'a FunctionIR,
308    seg: &str,
309    aliases: Option<&FxHashMap<String, String>>,
310) -> Vec<TriggerSite<'a>> {
311    let mut out = Vec::new();
312    for (&bid, block) in &ir.blocks {
313        for (idx, instr) in block.instructions.iter().enumerate() {
314            // Written callee path: static calls carry their full path
315            // (Rust `storage::ops::wipe_volume`); member calls are
316            // reconstructed from the receiver chain (`payments.x` ->
317            // `payments.x`), so qualified facts can match the exact
318            // member instead of every object sharing the bare verb.
319            let name = match instr {
320                Instruction::CallStatic { func, .. } => func.clone(),
321                Instruction::CallVirtual {
322                    method, receiver, ..
323                } => crate::analysis::forward::receiver_call_path(ir, receiver, method),
324                _ => continue,
325            };
326            // The callee's bare last segment, or an import alias resolving
327            // to `seg` (`refund` -> `issue_refund`), counts as the trigger.
328            // A qualified taught name matches the written callee path
329            // instead of the bare verb (see `callee_matches`).
330            let matches = callee_matches(seg, &name)
331                || aliases.is_some_and(|m| {
332                    m.get(super::last_segment(&name))
333                        .is_some_and(|sym| callee_matches(seg, sym))
334                });
335            if !matches {
336                continue;
337            }
338            let (args, arg_names) = call_args(instr).unwrap_or((&[], &[]));
339            out.push((bid, idx, args, arg_names, ir.instr_span(instr)));
340        }
341    }
342    out
343}

345/// A call instruction's argument list and its parallel keyword names, for
346/// site indexing outside the match arm that already destructures it. The
347/// name slice may be shorter than the argument slice; a missing entry is a
348/// positional argument.
349fn call_args(instr: &Instruction) -> Option<(&[Operand], &[String])> {
350    match instr {
351        Instruction::CallStatic {
352            args, arg_names, ..
353        } => Some((args.as_slice(), arg_names.as_slice())),
354        Instruction::CallVirtual {
355            args, arg_names, ..
356        } => Some((args.as_slice(), arg_names.as_slice())),
357        _ => None,
358    }
359}

361/// Does one requirement hold for this trigger site?
362#[allow(clippy::too_many_arguments)]
363fn requirement_holds(
364    req: &PolicyRequirement,
365    ir: &FunctionIR,
366    block: BlockId,
367    trigger_idx: usize,
368    doms: Option<&DomSets>,
369    val_info: &crate::analysis::value::ValueInfo,
370    args: &[Operand],
371    arg_names: &[String],
372    fn_segs: &[String],
373    module_segs: &[String],
374    scope: PolicyScope,
375) -> bool {
376    match req {
377        PolicyRequirement::GuardCall { call } => {
378            // Enforcement by a named helper. Function scope is spatial: the
379            // guard must EXECUTE before the trigger on every path (its block
380            // dominates the trigger's, or same block with the guard first) -
381            // a call that may be skipped enforces nothing. Module scope
382            // stays presence (the helper exists somewhere in the module).
383            let seg = super::last_segment(call);
384            if is_entry_guarded(ir, seg) {
385                return true;
386            }
387            if scope == PolicyScope::Module {
388                module_segs.iter().any(|s| s == seg)
389            } else {
390                // Spatial matching sees the full taught guard name so a
391                // qualified helper matches its member-call sites exactly.
392                guard_executes_before(ir, call, block, trigger_idx, doms)
393            }
394        }
395        PolicyRequirement::RequireCall { any_of } => {
396            if scope == PolicyScope::Module {
397                let wanted: Vec<&str> = any_of.iter().map(|c| super::last_segment(c)).collect();
398                return module_segs.iter().any(|s| wanted.contains(&s.as_str()));
399            }
400            any_of.iter().any(|c| {
401                is_entry_guarded(ir, super::last_segment(c))
402                    || guard_executes_before(ir, c, block, trigger_idx, doms)
403            })
404        }
405        PolicyRequirement::NotCall { call } => {
406            let seg = super::last_segment(call);
407            !scope_segs(scope, fn_segs, module_segs)
408                .iter()
409                .any(|s| s == seg)
410        }
411        PolicyRequirement::RangeCheck { ops } => args.iter().any(|a| match a {
412            Operand::Var(v) => is_range_guarded(ir, *v, ops, block, val_info),
413            _ => false,
414        }),
415        PolicyRequirement::BannedArgLiteral { slot, values } => {
416            if let Some(arg) = args.get(*slot)
417                && let Some(s) = extract_literal_string(arg, ir, val_info)
418            {
419                let s_lower = s.to_ascii_lowercase();
420                return !values.iter().any(|v| v.to_ascii_lowercase() == s_lower);
421            }
422            true
423        }
424        PolicyRequirement::RequiredArgLiteral { slot, values } => {
425            if let Some(arg) = args.get(*slot)
426                && let Some(s) = extract_literal_string(arg, ir, val_info)
427            {
428                let s_lower = s.to_ascii_lowercase();
429                return values.iter().any(|v| v.to_ascii_lowercase() == s_lower);
430            }
431            false
432        }
433        PolicyRequirement::RequireArgKey { slot, keys } => {
434            if *slot == usize::MAX {
435                args.iter().any(|a| match a {
436                    Operand::Var(v) => arg_has_key(ir, *v, keys),
437                    _ => false,
438                })
439            } else {
440                args.get(*slot).is_some_and(|arg| match arg {
441                    Operand::Var(v) => arg_has_key(ir, *v, keys),
442                    _ => false,
443                })
444            }
445        }
446        PolicyRequirement::RequireNamedArgLiteral { name, values } => {
447            let Some(idx) = arg_names.iter().position(|n| n.eq_ignore_ascii_case(name)) else {
448                // The caller never set the flag: the violation the policy
449                // exists to catch.
450                return false;
451            };
452            match args
453                .get(idx)
454                .and_then(|a| extract_literal_string(a, ir, val_info))
455            {
456                // A provable literal must be one of the required values.
457                Some(s) => values.iter().any(|v| v.eq_ignore_ascii_case(&s)),
458                // Unresolvable value: literal-only proof, so not a
459                // violation. This is the conservative direction - it can
460                // miss `secure=flag`, never invent it.
461                None => true,
462            }
463        }
464        PolicyRequirement::BannedNamedArgLiteral { name, values } => {
465            let Some(idx) = arg_names.iter().position(|n| n.eq_ignore_ascii_case(name)) else {
466                // The caller never set the flag: absent is NOT a violation.
467                return true;
468            };
469            match args
470                .get(idx)
471                .and_then(|a| extract_literal_string(a, ir, val_info))
472            {
473                // A provable literal must NOT match any of the banned values.
474                Some(s) => !values.iter().any(|v| v.eq_ignore_ascii_case(&s)),
475                // Unresolvable value: literal-only proof, conservative.
476                None => true,
477            }
478        }
479    }
480}

482/// True when the function's own definition carries a decorator/attribute
483/// whose call name is `seg` (`@guard`, `#[guard]`). A decorator executes
484/// before ANY statement of the body, so it dominates every trigger in the
485/// function by construction - the caller-side enforcement shape.
486fn is_entry_guarded(ir: &FunctionIR, seg: &str) -> bool {
487    ir.entry_guards
488        .iter()
489        .any(|g| super::last_segment(g) == seg)
490}

492/// The bundler's fact extractor consults the SAME guard model the scanner
493/// uses, so a corpus twin can teach a check whose negative suppresses the
494/// trigger dominantly (a bare `guard(x);` statement before the trigger,
495/// same block earlier or in a dominating block) even when no dataflow
496/// connects the guard's value to the trigger's arguments. Argument-nested
497/// calls (`new Date()` inside a `where: { lt: ... }` literal) are excluded,
498/// as their span lies inside the trigger's argument span. Mechanism only;
499/// conclusions stay bundle data.
500pub fn guard_dominates_trigger_call(ir: &FunctionIR, guard_seg: &str, trigger_seg: &str) -> bool {
501    let doms = control::dominators(ir);
502    let call_span = |bid: BlockId, idx: usize| -> Option<(usize, usize)> {
503        ir.blocks
504            .get(&bid)
505            .and_then(|b| b.instructions.get(idx))
506            .and_then(|instr| match instr {
507                Instruction::CallStatic { dest, .. }
508                | Instruction::CallVirtual { dest, .. }
509                | Instruction::CallPointer { dest, .. } => *dest,
510                _ => None,
511            })
512            .and_then(|d| ir.var_metadata.get(&d).and_then(|m| m.byte_range))
513    };
514    for (&tb, block) in &ir.blocks {
515        for (tidx, instr) in block.instructions.iter().enumerate() {
516            let name = match instr {
517                Instruction::CallStatic { func, .. } => func,
518                Instruction::CallVirtual { method, .. } => method,
519                _ => continue,
520            };
521            if super::last_segment(name) != trigger_seg {
522                continue;
523            }
524            // Guard sites that execute before this trigger, with the
525            // argument-nested coincidence excluded by span containment.
526            for (&gb, gblock) in &ir.blocks {
527                for (gi, ginstr) in gblock.instructions.iter().enumerate() {
528                    let gname = match ginstr {
529                        Instruction::CallStatic { func, .. } => func,
530                        Instruction::CallVirtual { method, .. } => method,
531                        _ => continue,
532                    };
533                    if super::last_segment(gname) != guard_seg {
534                        continue;
535                    }
536                    let prev = if gb == tb {
537                        gi < tidx
538                    } else {
539                        doms.get(&tb).is_some_and(|d| d.contains(&gb))
540                    };
541                    if !prev {
542                        continue;
543                    }
544                    // Exclude guards nested inside the trigger's own
545                    // argument lowering.
546                    if let (Some(gs), Some(ts)) = (call_span(gb, gi), call_span(tb, tidx))
547                        && gs.0 >= ts.0
548                        && gs.1 <= ts.1
549                    {
550                        continue;
551                    }
552                    return true;
553                }
554            }
555        }
556    }
557    false
558}

560/// Does `start` trace to an object construction carrying any of `keys`?
561/// Fully-keyed object literals lower to `Allocate(kind: Object)` plus one
562/// `StoreField { base, field }` per key, and the lowering records the pair
563/// keys on the composite var's metadata (shorthand/mixed composites carry
564/// the metadata without stores). Both paths are checked; plain `Assign`
565/// hops are followed like the shape classifier in `forward/predicates.rs`
566/// (`arg_is_idor_shape`), depth-bounded.
567fn arg_has_key(ir: &FunctionIR, start: VarId, keys: &[String]) -> bool {
568    let mut frontier = vec![start];
569    let mut seen: rustc_hash::FxHashSet<VarId> = rustc_hash::FxHashSet::default();
570    seen.insert(start);
571    for _ in 0..8 {
572        let mut next = Vec::new();
573        for v in frontier.drain(..) {
574            // (1) Metadata path: the composite var remembers its pair keys.
575            if ir
576                .var_metadata
577                .get(&v)
578                .is_some_and(|m| m.object_keys.iter().any(|k| keys.contains(k)))
579            {
580                return true;
581            }
582            for b in ir.blocks.values() {
583                for instr in &b.instructions {
584                    match instr {
585                        // (2) Store path: the literal's own per-key store
586                        // (`where: ...`), checked by field name.
587                        Instruction::StoreField {
588                            base, field, src, ..
589                        } if *base == v => {
590                            if keys.iter().any(|k| k == field) {
591                                return true;
592                            }
593                            // (3) Recurse into stored member objects:
594                            // a nested `{ lt: ... }` is a fresh Allocate
595                            // reached through the outer object's store.
596                            if let Operand::Var(s) = src
597                                && seen.insert(*s)
598                            {
599                                next.push(*s);
600                            }
601                        }
602                        // (4) Plain assign hop keeps the shape identity.
603                        Instruction::Assign {
604                            dest,
605                            src: Operand::Var(s),
606                        } if *dest == v && seen.insert(*s) => {
607                            next.push(*s);
608                        }
609                        _ => {}
610                    }
611                }
612            }
613        }
614        if next.is_empty() {
615            break;
616        }
617        frontier = next;
618    }
619    false
620}

622/// Extract constant literal string/bool/int representation from an operand.
623fn extract_literal_string(
624    op: &Operand,
625    ir: &FunctionIR,
626    val_info: &crate::analysis::value::ValueInfo,
627) -> Option<String> {
628    match op {
629        Operand::StringLiteral(s) => Some(strip_quotes(s).to_string()),
630        Operand::IntLiteral(i) => Some(i.to_string()),
631        Operand::BoolLiteral(b) => Some(b.to_string()),
632        Operand::Var(v) => {
633            if let Some(s) = val_info.const_str(*v) {
634                return Some(strip_quotes(s).to_string());
635            }
636            if let Some(c) = val_info.const_int(*v) {
637                return Some(c.to_string());
638            }
639            if let Some(b) = val_info.const_bool(*v) {
640                return Some(b.to_string());
641            }
642            for block in ir.blocks.values() {
643                for instr in &block.instructions {
644                    match instr {
645                        Instruction::Assign { dest, src } if dest == v => {
646                            return extract_literal_string(src, ir, val_info);
647                        }
648                        Instruction::Cast { dest, src, .. } if dest == v => {
649                            return extract_literal_string(src, ir, val_info);
650                        }
651                        _ => {}
652                    }
653                }
654            }
655            None
656        }
657        _ => None,
658    }
659}

673/// True when some call to `seg` provably executes before the trigger:
674/// its block dominates the trigger's block, or both share a block with the
675/// guard at a lower instruction index. Falls back to mere presence when no
676/// dominance information was computed (caller contract: function-scoped
677/// guard/require requirements always pass one).
678fn guard_executes_before(
679    ir: &FunctionIR,
680    seg: &str,
681    trigger_block: BlockId,
682    trigger_idx: usize,
683    doms: Option<&DomSets>,
684) -> bool {
685    let mut sites: Vec<(BlockId, usize)> = Vec::new();
686    for (&bid, block) in &ir.blocks {
687        for (idx, instr) in block.instructions.iter().enumerate() {
688            let matched = match instr {
689                Instruction::CallStatic { func, .. } => callee_matches(seg, func),
690                Instruction::CallVirtual {
691                    method, receiver, ..
692                } => {
693                    let path = crate::analysis::forward::receiver_call_path(ir, receiver, method);
694                    callee_matches(seg, &path)
695                }
696                _ => false,
697            };
698            if matched {
699                sites.push((bid, idx));
700            }
701        }
702    }
703    if sites.is_empty() {
704        return false;
705    }
706    let Some(doms) = doms else {
707        return true;
708    };
709    sites.iter().any(|&(gb, gi)| {
710        if gb == trigger_block {
711            gi < trigger_idx
712        } else {
713            doms.get(&trigger_block).is_some_and(|d| d.contains(&gb))
714        }
715    })
716}
