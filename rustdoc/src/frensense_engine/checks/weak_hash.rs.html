5//! The weak-hash / weak-crypto rule: non-dataflow policy assertions over lowered IR.
6//!
7//! Taint analysis answers "does attacker-controlled data reach a dangerous
8//! API?", but a whole class of vulnerabilities has no taint path at all:
9//! weak cryptographic primitives (MD5 for passwords), hardcoded credentials,
10//! insecure JWT algorithms (`{ algorithm: 'none' }`), disabled TLS
11//! verification. The bug is in *which API / which constant* the developer
12//! chose, not in how data flows.
13//!
14//! Each rule is a function over one [`FunctionIR`] plus the language spec's
15//! knowledge tables. A rule fires a [`CheckerFinding`] when its pattern
16//! matches; no sources, sinks or sanitizers are consulted. This keeps the
17//! checker immune to taint-config drift and free to run per function in
18//! parallel.
19

26// Weak-hash, key-size and insecure-selector policy tables are FACT DATA:
27// the engine reads only `FactTable` fields, seeded by the language spec
28// (the default pack) and extended by `.frc` bundles. No bootstrap
29// vocabulary is referenced from this module.

31/// Security-context qualification for the hash-wrapper rule: the
32/// enclosing function's name when it hints at security code
33/// (`facts.security_context_hints`), else `None`. A cheap static stand-in
34/// for the full receiver-chain walk, which needs cross-function
35/// information; the corpus replay gate learns wrapper-to-primitive
36/// mappings from pairs instead.
37fn receiver_path(ir: &FunctionIR, facts: &FactTable) -> Option<String> {
38    let n = ir.name.to_ascii_lowercase();
39    if facts.security_context_hints.iter().any(|h| n.contains(h)) {
40        return Some(ir.name.clone());
41    }
42    None
43}

45/// The weak-hash / weak-crypto rule: scan one function for security-weak
46/// primitives.
47pub fn check(ir: &FunctionIR, facts: &FactTable) -> Vec<CheckerFinding> {
48    check_function(ir, facts)
49}

51/// Scans one function for weak primitives: built-in and learned weak-crypto rules (including selector literals resolved through constant assignments) and suspicious hash wrappers. [`check`] is the entry point the check registry calls.
52pub(crate) fn check_function(ir: &FunctionIR, facts: &FactTable) -> Vec<CheckerFinding> {
53    // The value lattice lets selector rules see through constant
54    // assignments: `const alg = 'none'; jwt.sign(payload, secret, alg)`
55    // must fire even though the selector operand is a var, not a literal.
56    let values = crate::analysis::value::analyze(ir);
57    let mut findings = Vec::new();
58    for block in ir.blocks.values() {
59        for instr in &block.instructions {
60            match instr {
61                Instruction::CallStatic { func, args, .. } => {
62                    check_call(
63                        ir,
64                        func,
65                        func,
66                        args,
67                        ir.instr_span(instr),
68                        &values,
69                        facts,
70                        &mut findings,
71                    );
72                }
73                Instruction::CallVirtual {
74                    method,
75                    receiver,
76                    args,
77                    ..
78                } => {
79                    let path = crate::analysis::forward::receiver_call_path(ir, receiver, method);
80                    check_call(
81                        ir,
82                        method,
83                        &path,
84                        args,
85                        ir.instr_span(instr),
86                        &values,
87                        facts,
88                        &mut findings,
89                    );
90                }
91                _ => {}
92            }
93        }
94    }
95    findings
96}

98/// Resolve an argument operand to its string-literal value: direct literal,
99/// or a var whose lattice value is a provable string constant.
100/// The argument's literal token (a bool rendered as `true`/`false`, or a
101/// stripped string literal), lowercased: the form a pack-declared disclaimer
102/// takes.
103fn disclaimer_token(arg: &Operand, values: &crate::analysis::value::ValueInfo) -> Option<String> {
104    match arg {
105        Operand::BoolLiteral(b) => Some(if *b { "true" } else { "false" }.to_string()),
106        Operand::Var(v) => match values.const_bool(*v) {
107            Some(b) => Some(if b { "true" } else { "false" }.to_string()),
108            None => arg_str_literal(arg, values).map(str::to_ascii_lowercase),
109        },
110        other => arg_str_literal(other, values).map(str::to_ascii_lowercase),
111    }
112}
113
114fn arg_str_literal<'a>(
115    arg: &'a Operand,
116    values: &'a crate::analysis::value::ValueInfo,
117) -> Option<&'a str> {
118    match arg {
119        Operand::StringLiteral(s) => Some(strip_quotes(s)),
120        // Lattice constants carry the lowering's raw quote characters, same
121        // as direct string literals: strip before matching.
122        Operand::Var(v) => values.const_str(*v).map(strip_quotes),
123        _ => None,
124    }
125}
126
127/// Resolve an argument operand to its integer-literal value: direct literal,
128/// or a var whose lattice value is a provable integer constant.
129fn arg_int_literal(arg: &Operand, values: &crate::analysis::value::ValueInfo) -> Option<i64> {
130    match arg {
131        Operand::IntLiteral(n) => Some(*n),
132        Operand::Var(v) => values.const_int(*v),
133        _ => None,
134    }
135}

137#[allow(clippy::too_many_arguments)] // fact/lattice/out plumbing, no meaningful grouping
138fn check_call(
139    ir: &FunctionIR,
140    callee: &str,
141    full_path: &str,
142    args: &[Operand],
143    span: Option<(usize, usize)>,
144    values: &crate::analysis::value::ValueInfo,
145    facts: &FactTable,
146    out: &mut Vec<CheckerFinding>,
147) {
148    // Receiver chains are matched by last segment: `security.hash` and bare
149    // `hash` both hit the `hash` rule. This is deliberately over-approximate
150    // (a user object with an unrelated `.hash` method also fires); the
151    // message names the call, and false positives of this class are rare
152    // relative to the cost of tracking receiver types through the IR.
153    let callee_seg = last_segment(callee);
154    let callee_lower = callee_seg.to_ascii_lowercase();
155
156    if callee_lower == "ajv" {
157        for arg in args {
158            if let Operand::Var(v) = arg {
159                let has_data_true = ir.blocks.values().any(|b| {
160                    b.instructions.iter().any(|i| {
161                        if let Instruction::StoreField {
162                            base, field, src, ..
163                        } = i
164                        {
165                            if *base == *v && field == "$data" {
166                                match src {
167                                    Operand::BoolLiteral(true) => true,
168                                    Operand::StringLiteral(s) => strip_quotes(s) == "true",
169                                    Operand::Var(sv) => {
170                                        values.const_bool(*sv) == Some(true)
171                                            || values.const_str(*sv).map(strip_quotes)
172                                                == Some("true")
173                                    }
174                                    _ => false,
175                                }
176                            } else {
177                                false
178                            }
179                        } else {
180                            false
181                        }
182                    })
183                }) || ir
184                    .var_metadata
185                    .get(v)
186                    .is_some_and(|m| m.object_keys.contains(&"$data".to_string()));
187                if has_data_true {
188                    out.push(CheckerFinding {
189                        provenance: Provenance::Spec,
190                        function: ir.name.clone(),
191                        rule: "regex".to_string(),
192                        message: "Insecure Ajv configuration with `$data: true` enables catastrophic ReDoS via payload-supplied regex".to_string(),
193                        params: vec![("callee", callee_seg.to_string())],
194                        span,
195                        severity: "High".to_string(),
196                    });
197                    return;
198                }
199            }
200        }
201    }
202
203    // A weak-primitive call that carries an explicit disclaimer argument is
204    // the caller disclaiming security use (`has_md5(data, false)` marks the
205    // digest as a checksum / cache key): not a credential or integrity
206    // policy violation.
207    //
208    // Which tokens disclaim is pack knowledge
209    // ([`FactTable::weak_hash_disclaimer_tokens`]); the engine supplies only
210    // the mechanism - slot 0 is the data being hashed, so a disclaimer is
211    // looked for among the trailing flag positions, and languages lower a
212    // bare `false` either as a bool operand or as literal text.
213    let disclaims_security = !facts.weak_hash_disclaimer_tokens.is_empty()
214        && args.iter().skip(1).any(|a| {
215            disclaimer_token(a, values)
216                .is_some_and(|t| facts.weak_hash_disclaimer_tokens.contains(&t))
217        });
218    for rule in &facts.weak_hash_rules {
219        if disclaims_security {
220            break;
221        }
222        let provenance = Provenance::Spec;
223        // Selector shape: `createHash('md5')`. Rules with
224        // `requires_credential_context` fire only inside credential-named
225        // functions or with credential-named parameters (a general-purpose
226        // `createHash` doubles as a checksum API); rules without it fire
227        // wherever a weak algorithm is named explicitly.
228        if (!rule.requires_credential_context || in_credential_context(ir, facts))
229            && rule.selector_calls.iter().any(|c| {
230                let c = last_segment(c).to_ascii_lowercase();
231                c == callee_lower
232            })
233            && let Some(sel) = args
234                .get(rule.selector_slot)
235                .and_then(|a| arg_str_literal(a, values))
236                .map(str::to_ascii_lowercase)
237            && rule.weak_selectors.contains(&sel)
238        {
239            out.push(CheckerFinding {
240                provenance,
241                function: ir.name.clone(),
242                rule: rule.rule_id.to_string(),
243                message: rule.message.clone(),
244                params: vec![("callee", callee_seg.to_string()), ("sel", sel.to_string())],
245                span,
246                severity: rule.severity.to_string(),
247            });
248            return; // one finding per call site
249        }
250        // Bare shape: `md5(data)`.
251        if rule
252            .bare_calls
253            .iter()
254            .any(|c| last_segment(c).to_ascii_lowercase() == callee_lower)
255        {
256            out.push(CheckerFinding {
257                provenance,
258                function: ir.name.clone(),
259                rule: rule.rule_id.to_string(),
260                message: rule.message.clone(),
261                params: vec![("callee", callee_seg.to_string())],
262                span,
263                severity: rule.severity.to_string(),
264            });
265            return;
266        }
267    }
268
269    // Evaluate corpus-learned weak crypto rules
270    for fact in &facts.weak_crypto_rules {
271        let fact_call_seg = last_segment(&fact.call).to_ascii_lowercase();
272        if fact_call_seg == callee_lower {
273            if let Some(slot) = fact.selector_slot {
274                if let Some(sel) = args
275                    .get(slot)
276                    .and_then(|a| arg_str_literal(a, values))
277                    .map(str::to_ascii_lowercase)
278                    && fact
279                        .weak_selectors
280                        .iter()
281                        .any(|ws| ws.to_ascii_lowercase() == sel)
282                {
283                    out.push(CheckerFinding {
284                        provenance: Provenance::Learned,
285                        function: ir.name.clone(),
286                        rule: fact.rule_id.clone(),
287                        message: fact.message.clone(),
288                        params: vec![("callee", callee_seg.to_string()), ("sel", sel.to_string())],
289                        span,
290                        severity: fact.severity.clone(),
291                    });
292                    return;
293                }
294            } else {
295                out.push(CheckerFinding {
296                    provenance: Provenance::Learned,
297                    function: ir.name.clone(),
298                    rule: fact.rule_id.clone(),
299                    message: fact.message.clone(),
300                    params: vec![("callee", callee_seg.to_string())],
301                    span,
302                    severity: fact.severity.clone(),
303                });
304                return;
305            }
306        }
307    }
308
309    // Hash-wrapper heuristic: a call named `hash`/`hashPassword` whose
310    // receiver is a security-ish namespace. Matched by last segment of the
311    // receiver chain when available; bare `hash(x)` is too generic to flag
312    // on its own, so require a known crypto-receiver qualification.
313    let spec_wrapper = facts
314        .suspicious_hash_wrappers
315        .contains(callee_lower.as_str());
316    if spec_wrapper && let Some(path) = receiver_path(ir, facts) {
317        out.push(CheckerFinding {
318            provenance: Provenance::Spec,
319            function: ir.name.clone(),
320            rule: frensense_lang::rules::WEAK_HASH_WRAPPER.to_string(),
321            message: String::new(),
322            params: vec![("path", path.clone())],
323            span,
324            severity: String::new(),
325        });
326    }
327
328    // Weak key size: a key-generation/parameter call whose bit-length
329    // argument is a PROVABLE CONSTANT below the security floor. The value
330    // lattice resolves `const bits = 512; generateKey(bits)` the same as a
331    // direct literal, so wrapper indirection doesn't hide the weakness.
332    for rule in &facts.key_size_rules {
333        let provenance = Provenance::Spec;
334        if last_segment(rule.call.as_str()).to_ascii_lowercase() != callee_lower {
335            continue;
336        }
337        if let Some(slot) = args.get(rule.slot) {
338            let weak = arg_int_literal(slot, values)
339                .map(|bits| bits < rule.min_bits)
340                .unwrap_or(false);
341            if weak {
342                out.push(CheckerFinding {
343                    provenance,
344                    function: ir.name.clone(),
345                    rule: rule.rule_id.to_string(),
346                    message: rule.message.clone(),
347                    params: vec![
348                        ("callee", callee_seg.to_string()),
349                        ("min_bits", rule.min_bits.to_string()),
350                        ("kind", rule.kind.to_string()),
351                    ],
352                    span,
353                    severity: rule.severity.to_string(),
354                });
355                return;
356            }
357        }
358    }
359
360    // Insecure literal selectors (`{ algorithm: 'none' }` shapes land here
361    // once object literals are flattened; for now the string form).
362    for rule in &facts.insecure_config_selectors {
363        let provenance = Provenance::Spec;
364        // Match the full dotted call path (`jwt.verify`) or the callee
365        // (`jwtChallenge`); only a *verifier* accepting an insecure
366        // algorithm is the violation - token issuers that merely embed the
367        // literal (`jwtChallenge(id, req, 'none', ...)`) are harness code.
368        let path_lower = full_path.to_ascii_lowercase();
369        if !(path_lower.starts_with(rule.prefix.as_str())
370            || callee_lower.starts_with(rule.prefix.as_str()))
371        {
372            continue;
373        }
374        if !jwt_algorithm_context(&path_lower, &callee_lower, facts) {
375            continue;
376        }
377        for arg in args {
378            if let Some(lit) = arg_str_literal(arg, values)
379                && rule.selectors.contains(&lit.to_ascii_lowercase())
380            {
381                out.push(CheckerFinding {
382                    provenance,
383                    function: ir.name.clone(),
384                    rule: rule.rule_id.to_string(),
385                    message: rule.message.clone(),
386                    params: vec![("callee", callee_seg.to_string()), ("sel", lit.to_string())],
387                    span,
388                    severity: rule.severity.to_string(),
389                });
390                return;
391            }
392        }
393    }
394}

396/// Algorithm-operation qualification for `insecure_jwt_algorithm`: the
397/// callee path must hint at an operation whose algorithm choice matters
398/// (verify/decode/sign/...), vocabulary from the default pack
399/// (`FactTable::jwt_algorithm_hints`) - harness wrappers
400/// that merely embed the literal stay silent.
401fn jwt_algorithm_context(path: &str, callee: &str, facts: &FactTable) -> bool {
402    let matches = |s: &str| facts.jwt_algorithm_hints.iter().any(|h| s.contains(h));
403    matches(path) || matches(callee)
404}

406/// Credential-context qualification for the selector-shape weak-hash rule:
407/// the enclosing function name or a parameter name must hint at a
408/// credential context (password/secret/token/...). A generic digest
409/// utility (`const digest = (data) => createHash('md5')`) is not a KDF
410/// shape; credential-named wrappers (`hashPassword(clearText)`) are.
411/// Bare weak calls (`md5(data)`) stay unqualified - they are weak by
412/// themselves.
413fn in_credential_context(ir: &FunctionIR, facts: &FactTable) -> bool {
414    let hint_match = |s: &str| {
415        let l = s.to_ascii_lowercase();
416        facts.credential_context_hints.iter().any(|h| l.contains(h))
417    };
418    if hint_match(&ir.name) {
419        return true;
420    }
421    ir.parameters.iter().any(|p| {
422        ir.var_metadata
423            .get(p)
424            .and_then(|m| m.source_name.as_deref())
425            .is_some_and(hint_match)
426    })
427}

429#[cfg(test)]
430/// Assertions lifted from the doc examples of items that are now
431/// crate-private; a doctest cannot see them.
432mod doc_examples {
433    #[test]
434    fn doc_example_check_function() {
435        use crate::analysis::taint::facts::{FactTable, KeySizeRule};
436        use crate::checks::weak_hash::check_function;
437        use crate::harness::lower_source;
438
439        let mut facts = FactTable::default();
440        facts.key_size_rules.push(KeySizeRule {
441            rule_id: "weak_key_size".into(),
442            call: "generateKey".into(),
443            slot: 1,
444            min_bits: 2048,
445            kind: "rsa".into(),
446            severity: "high".into(),
447            message: String::new(),
448        });
449        let rules = |src: &str| -> Vec<String> {
450            let fns = lower_source("t.ts", src, "ts").unwrap();
451            check_function(&fns["f"], &facts)
452                .into_iter()
453                .map(|f| f.rule)
454                .collect()
455        };
456
457        // A constant below the floor is reported ...
458        assert_eq!(
459            rules(
460                "declare function generateKey(k: string, bits: number): void\nfunction f() { generateKey('rsa', 1024) }"
461            ),
462            ["weak_key_size"]
463        );
464        // ... also when it reaches the call through a constant assignment.
465        assert_eq!(
466            rules(
467                "declare function generateKey(k: string, bits: number): void\nfunction f() { const bits = 512; generateKey('rsa', bits) }"
468            ),
469            ["weak_key_size"]
470        );
471        // At or above the floor, or not a provable constant: nothing.
472        assert!(rules("declare function generateKey(k: string, bits: number): void\nfunction f() { generateKey('rsa', 2048) }").is_empty());
473        assert!(rules("declare function generateKey(k: string, bits: number): void\nfunction f(n: number) { generateKey('rsa', n) }").is_empty());
474    }
