5//! Structural health: functions whose control flow has outgrown review.
6//!
7//! This is the first "algorithm class" check that is not a security proof:
8//! it measures the lowered IR instead of matching vocabulary, so it needs
9//! no library or framework names and works for every language the
10//! providers lower. Findings are `Hygiene` tier (see
11//! `frensense_lang::severity::checker_rule_tier`), which keeps them out of
12//! the default and recommended profiles.
13//!
14//! Metric: McCabe cyclomatic complexity over the IR control-flow graph,
15//! `1 + (two-way branches) + (switch arms)`. Straight-line code and
16//! unconditional jumps add nothing, so the number is stable across
17//! providers that lower the same source differently.
18
19use crate::checks::{CheckerFinding, Provenance};
20use crate::ir::function::{FunctionIR, Terminator};
21
22/// Rule id emitted by this check.
23pub(crate) const FUNCTION_COMPLEXITY: &str = "function_complexity";
24
25/// Complexity above which a function is reported. 25 is deliberately high:
26/// the first goal is to surface the outliers, not to police every
27/// function.
28pub(crate) const MAX_COMPLEXITY: usize = 25;
29
30/// Cyclomatic complexity of one lowered function.
31pub(crate) fn cyclomatic_complexity(ir: &FunctionIR) -> usize {
32    let decisions: usize = ir
33        .blocks
34        .values()
35        .map(|b| match &b.terminator {
36            Terminator::Branch { .. } => 1,
37            Terminator::Switch { cases, .. } => cases.len(),
38            _ => 0,
39        })
40        .sum();
41    1 + decisions
42}
43
44/// Number of IR instructions, a size proxy that is independent of
45/// formatting.
46pub(crate) fn instruction_count(ir: &FunctionIR) -> usize {
47    ir.blocks.values().map(|b| b.instructions.len()).sum()
48}
49
50/// Report `ir` when its complexity exceeds `max`.
51pub(crate) fn check_with_limit(ir: &FunctionIR, max: usize) -> Vec<CheckerFinding> {
52    let complexity = cyclomatic_complexity(ir);
53    if complexity <= max {
54        return Vec::new();
55    }
56    // Anchor at the earliest known source byte of the function so the
57    // advisory lands at the top of the body.
58    let span = ir
59        .var_metadata
60        .values()
61        .filter_map(|m| m.byte_range)
62        .min_by_key(|r| r.0);
63    vec![CheckerFinding {
64        provenance: Provenance::Spec,
65        function: ir.name.clone(),
66        rule: FUNCTION_COMPLEXITY.to_string(),
67        message: String::new(),
68        params: vec![
69            ("complexity", complexity.to_string()),
70            ("limit", max.to_string()),
71            ("ops", instruction_count(ir).to_string()),
72        ],
73        span,
74        severity: "low".to_string(),
75    }]
76}
77
78/// Default-threshold entry point used by the registry.
79///
80/// ```
81/// use frensense_engine::checks::health::check;
82/// use frensense_engine::harness::lower_source;
83///
84/// let fns = lower_source("t.ts", "function g() { return 0 }", "ts").unwrap();
85/// assert!(check(&fns["g"]).is_empty());
86/// ```
87pub fn check(ir: &FunctionIR) -> Vec<CheckerFinding> {
88    check_with_limit(ir, MAX_COMPLEXITY)
89}

95#[cfg(test)]
96/// Assertions lifted from the doc examples of items that are now
97/// crate-private; a doctest cannot see them.
98mod doc_examples {
99    #[test]
100    fn doc_example_check_with_limit() {
101        use crate::checks::health::{FUNCTION_COMPLEXITY, check_with_limit};
102        use crate::harness::lower_source;
103
104        let src =
105            "function f(a: number) { if (a > 1) { return 1 } if (a > 2) { return 2 } return 3 }";
106        let fns = lower_source("t.ts", src, "ts").unwrap();
107        // Complexity 3 is not above a limit of 3 ...
108        assert!(check_with_limit(&fns["f"], 3).is_empty());
109        // ... but is above a limit of 2.
110        let found = check_with_limit(&fns["f"], 2);
111        assert_eq!(found.len(), 1);
112        assert_eq!(found[0].rule, FUNCTION_COMPLEXITY);
113        assert_eq!(found[0].function, "f");
114    }
115    #[test]
116    fn doc_example_instruction_count() {
117        use crate::checks::health::instruction_count;
118        use crate::harness::lower_source;
119
120        let fns = lower_source("t.ts", "function g() { return 0 }", "ts").unwrap();
121        assert_eq!(instruction_count(&fns["g"]), 0);
122    }
123    #[test]
124    fn doc_example_cyclomatic_complexity() {
125        use crate::checks::health::cyclomatic_complexity;
126        use crate::harness::lower_source;
127
128        let src = "function f(a: number) { if (a > 1) { return 1 } if (a > 2) { return 2 } return 3 }\nfunction g() { return 0 }";
129        let fns = lower_source("t.ts", src, "ts").unwrap();
130        // One plus one per two-way branch.
131        assert_eq!(cyclomatic_complexity(&fns["f"]), 3);
132        // A straight-line function has complexity 1.
133        assert_eq!(cyclomatic_complexity(&fns["g"]), 1);
134    }
