9/// A violation of the object lifecycle.
10///
11/// `Free` is the subject of each variant, not a decorative suffix: the three
12/// cases are the ways a release can go wrong, so trimming it would leave
13/// `UseAfter`, `Double` and `Uninitialized`.
14#[allow(clippy::enum_variant_names)]
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub(crate) enum Violation {
17    /// Memory used after free (read or write through an aliasing var).
18    UseAfterFree,
19    /// The same object freed twice.
20    DoubleFree,
21    /// Memory freed without guaranteed initialization on all reachable paths.
22    UninitializedFree,
23}

25/// One site that releases the object held by `var`.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub(crate) struct FreeSite {
28    /// Block containing the freeing instruction.
29    pub block: BlockId,
30    /// Index of the freeing instruction within its block.
31    pub idx: usize,
32    /// Variable holding the object that is released.
33    pub var: VarId,
34}

36/// How a discovered (free, use) pair must be validated before firing.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub(crate) enum PairCheck {
39    /// Free and use are both in the scanned function: must-reach in this CFG.
40    Local,
41    /// Free lives in the callee entered at `call`; the call must precede the
42    /// use AND every normal exit of `callee_fi` must pass a free.
43    Callee {
44        /// Block and instruction index of the call in the scanned function.
45        call: (BlockId, usize),
46        /// Index of the callee in the program's function list.
47        callee_fi: usize,
48    },
49    /// Free lives in the caller that entered *this* function at `call`; the
50    /// free must precede the call in the caller's CFG.
51    Caller {
52        /// Block and instruction index of the call, in the caller's CFG.
53        call: (BlockId, usize),
54        /// Index of the caller in the program's function list.
55        caller_fi: usize,
56    },
57}
58
59/// One discovered pair awaiting must-validation.
60#[derive(Debug, Clone, Copy)]
61pub(crate) struct Pair {
62    /// Kind of violation this pair would be.
63    pub violation: Violation,
64    /// How the pair must be validated before it is reported.
65    pub check: PairCheck,
66    /// Location of the free (its block / instruction index).
67    pub free_block: BlockId,
68    /// Index of the freeing instruction within `free_block`.
69    pub free_idx: usize,
70}

72/// Which function a walk step currently sits in, relative to the start.
73#[derive(Debug, Clone, Copy, PartialEq, Eq)]
74pub(crate) enum CrossCtx {
75    /// Inside the scanned (start) function.
76    None,
77    /// Inside a callee entered from the start function at `call`.
78    InCallee {
79        /// Block and instruction index of the call in the start function.
80        call: (BlockId, usize),
81        /// Index of the callee in the program's function list.
82        callee_fi: usize,
83    },
84    /// Inside the caller that entered the start function at `call`.
85    InCaller {
86        /// Block and instruction index of the call, in the caller's CFG.
87        call: (BlockId, usize),
88        /// Index of the caller in the program's function list.
89        caller_fi: usize,
90    },
91}

93/// Builds the checker finding for `violation` on `var`.
94///
95/// The variable is named from its source metadata (or `v<id>` when it has none), and the span is the byte range of `span_var` when one is given.
96pub(crate) fn finding(
97    ir: &FunctionIR,
98    violation: Violation,
99    var: &VarId,
100    span_var: Option<VarId>,
101) -> CheckerFinding {
102    let name = ir
103        .var_metadata
104        .get(var)
105        .and_then(|m| m.source_name.clone())
106        .unwrap_or_else(|| format!("v{}", var.0));
107    let span = span_var.and_then(|v| ir.var_metadata.get(&v).and_then(|m| m.byte_range));
108    let rule = match violation {
109        Violation::UseAfterFree => frensense_lang::rules::USE_AFTER_FREE,
110        Violation::DoubleFree => frensense_lang::rules::DOUBLE_FREE,
111        Violation::UninitializedFree => frensense_lang::rules::UNINITIALIZED_FREE,
112    };
113    CheckerFinding {
114        function: ir.name.clone(),
115        rule: rule.to_string(),
116        message: String::new(),
117        params: vec![("name", name)],
118        span,
119        severity: String::new(),
120        provenance: Provenance::Spec,
121    }
122}
