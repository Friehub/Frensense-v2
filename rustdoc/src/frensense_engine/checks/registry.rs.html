5//! The non-dataflow check seam: one trait, one registry, one driver.
6//!
7//! Every non-dataflow check declares *how often it runs* and *how to run*;
8//! the driver in `super::check_all_with_telemetry` supplies everything
9//! that is common to all of them - the IR set, the whole-program value-flow
10//! graph, the fact table, the package scope, the interprocedural memory
11//! summaries - and owns the single finding-identity/dedup rule.
12//!
13//! Adding a check class is a new module plus one line in
14//! `default_registry`. The driver does not change, `scan.rs` does not
15//! change, and every existing check keeps its own mechanism (a proof over
16//! the IR), while the *vocabulary* it consumes stays in [`FactTable`].
17//!
18//! Layering: the context deliberately carries no [`TaintConfig`], so a
19//! non-dataflow check cannot reach into the taint engine's configuration.
20//! Checks see lowered IR and facts only.
21//!
22//! [`TaintConfig`]: crate::analysis::taint::config::TaintConfig
23

31/// How often a registered check runs.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub(crate) enum CheckScope {
34    /// Once per function, with [`CheckContext::current`] set.
35    Function,
36    /// Once over the whole IR set, for checks that correlate definitions
37    /// and guards across function boundaries.
38    Program,
39}
40
41/// Everything a non-dataflow check may consult.
42///
43/// All fields are shared references owned by the driver for the duration
44/// of one `check_all*` call; a check returns findings and never mutates
45/// the context.
46#[derive(Clone, Copy)]
47pub(crate) struct CheckContext<'a> {
48    /// Every function in the scanned program, in driver order.
49    pub irs: &'a [&'a FunctionIR],
50    /// The function under check: `Some` exactly when the running check is
51    /// [`CheckScope::Function`].
52    pub current: Option<&'a FunctionIR>,
53    /// Whole-program value-flow graph, when the scan built one.
54    pub prog: Option<&'a ProgramSvfg<'a>>,
55    /// Built-in (pack-seeded) plus bundle-learned knowledge.
56    pub facts: &'a FactTable,
57    /// Package-import filter for policy facts; `None` disables filtering.
58    pub pkg_scope: Option<&'a PackageScope<'a>>,
59    /// Interprocedural memory summaries, inferred once per scan.
60    pub summaries: &'a MemorySummaryRegistry,
61}
62
63impl std::fmt::Debug for CheckContext<'_> {
64    /// Summarizes the context: the program size and the function under
65    /// check. The shared knowledge tables and the diagnostic sink are left out.
66    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
67        f.debug_struct("CheckContext")
68            .field("functions", &self.irs.len())
69            .field("current", &self.current.map(|ir| ir.name.as_str()))
70            .field("has_program_graph", &self.prog.is_some())
71            .finish_non_exhaustive()
72    }
73}
74
75impl<'a> CheckContext<'a> {
76    /// The function under check.
77    ///
78    /// # Panics
79    ///
80    /// Panics when a [`CheckScope::Program`] check asks for a single
81    /// subject: the scope method and `run` disagree, which is a
82    /// programming error, not a property of the scanned code.
83    pub(crate) fn ir(&self) -> &'a FunctionIR {
84        self.current
85            .expect("function-scoped check ran without a current function")
86    }
87}

89/// One non-dataflow check class.
90///
91/// Implementations live in [`default_registry`] as thin adapters over the
92/// mechanism modules, so a mechanism module stays a plain function over IR
93/// and never needs to know that a registry exists.
94///
95/// `Sync` is required because the default registry is a `static`: every
96/// implementation is stateless.
97pub(crate) trait NonDataflowCheck: Sync {
98    /// Stable id for this check class, used by diagnostics, per-class
99    /// stats, and (future) bundle capability declarations.
100    fn id(&self) -> &'static str;
101
102    /// How often [`Self::run`] should be called by the driver.
103    fn scope(&self) -> CheckScope;
104
105    /// Run the check and return its findings. The driver deduplicates and
106    /// orders the union of every check's output; a check itself makes no
107    /// ordering promises and must not dedup against other checks.
108    fn run(&self, ctx: &CheckContext<'_>) -> Vec<CheckerFinding>;
109}
110
111/// An ordered set of check classes.
112///
113/// Registry order is part of the output contract: the driver's final sort
114/// is stable, so findings sharing a `(function, span)` key keep their
115/// insertion order, which follows this list on function-scoped runs.
116pub(crate) struct CheckRegistry {
117    checks: &'static [&'static dyn NonDataflowCheck],
118}
119
120impl std::fmt::Debug for CheckRegistry {
121    /// Lists the check ids in registry order.
122    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
123        f.debug_list()
124            .entries(self.checks.iter().map(|c| c.id()))
125            .finish()
126    }
127}
128
129impl CheckRegistry {
130    /// Construct a registry from a static slice of check references.
131    #[cfg(test)]
132    pub(crate) const fn from_slice(checks: &'static [&'static dyn NonDataflowCheck]) -> Self {
133        Self { checks }
134    }
135
136    /// The registered check classes, in run order.
137    pub(crate) fn checks(&self) -> impl Iterator<Item = &'static dyn NonDataflowCheck> + '_ {
138        self.checks.iter().copied()
139    }
140
141    /// Number of registered check classes.
142    #[cfg(test)]
143    pub(crate) fn len(&self) -> usize {
144        self.checks.len()
145    }
146
147    /// Whether no check is registered (only constructible by a caller
148    /// building its own registry; [`default_registry`] is never empty).
149    #[cfg(test)]
150    pub(crate) fn is_empty(&self) -> bool {
151        self.checks.is_empty()
152    }
153}

216/// Free-then-use / double-free / uninitialized free.
217struct UseAfterFree;
218impl NonDataflowCheck for UseAfterFree {
219    fn id(&self) -> &'static str {
220        "use_after_free"
221    }
222    fn scope(&self) -> CheckScope {
223        CheckScope::Function
224    }
225    fn run(&self, ctx: &CheckContext<'_>) -> Vec<CheckerFinding> {
226        // The program graph upgrades discovery (callee-internal frees,
227        // factory provenance); without it, parameter-consumption summaries
228        // still recognise frees at call sites.
229        match ctx
230            .prog
231            .and_then(|p| p.function_index(&ctx.ir().name).map(|fi| (p, fi)))
232        {
233            Some((p, fi)) => super::uaf::check_with_prog(fi, p, ctx.summaries),
234            None => super::uaf::check_with_summaries(ctx.ir(), ctx.summaries),
235        }
236    }
237}

344/// The default check set, in run order.
345///
346/// Function-scoped checks are listed in the order the driver used to chain
347/// them per function; program-scoped checks are listed first because they
348/// run once, before any per-function pass.
349pub(crate) fn default_registry() -> CheckRegistry {
350    static CHECKS: &[&dyn NonDataflowCheck] = &[
351        &AllowlistDefinitions,
352        &Policies,
353        &WeakHash,
354        &GuardBypass,
355        &CredentialLeak,
356        &SchemaPolicy,
357        &UseAfterFree,
358        &OutOfBounds,
359        &IntegerOverflow,
360        &MemoryLeak,
361        &LearnedChecks,
362        &FunctionComplexity,
363    ];
364    CheckRegistry { checks: CHECKS }
365}

372    /// Ids identify classes: two checks sharing one would make per-class
373    /// stats and gating ambiguous.
374    #[test]
375    fn ids_are_unique_and_non_empty() {
376        let registry = default_registry();
377        let mut seen: FxHashSet<&'static str> = FxHashSet::default();
378        for check in registry.checks() {
379            let id = check.id();
380            assert!(!id.is_empty(), "check id must not be empty");
381            assert!(seen.insert(id), "duplicate check id: {id}");
382        }
383    }

385    /// The driver has two passes; a registry missing either scope silently
386    /// drops that whole class of findings.
387    #[test]
388    fn registry_covers_both_scopes() {
389        let registry = default_registry();
390        assert!(
391            registry.checks().any(|c| c.scope() == CheckScope::Program),
392            "no program-scoped check registered"
393        );
394        assert!(
395            registry.checks().any(|c| c.scope() == CheckScope::Function),
396            "no function-scoped check registered"
397        );
398    }

461    #[test]
462    fn check_failure_is_isolated_and_recorded() {
463        static PANICKING: PanickingFunctionCheck = PanickingFunctionCheck;
464        static HEALTHY: HealthyFunctionCheck = HealthyFunctionCheck;
465        static CHECKS: &[&'static dyn NonDataflowCheck] = &[&PANICKING, &HEALTHY];
466
467        let registry = CheckRegistry::from_slice(CHECKS);
468        let ir = FunctionIR::new("target_fn".into());
469        let irs = vec![&ir];
470        let facts = FactTable::default();
471
472        let (findings, telemetry) = crate::checks::check_all_with_registry_and_telemetry(
473            &registry, irs, &facts, None, None,
474        );
475
476        // Healthy check finding should still be emitted:
477        assert_eq!(findings.len(), 1);
478        assert_eq!(findings[0].rule, "HEALTHY_RULE");
479
480        // Telemetry should record both checks, panicking check having Panicked outcome:
481        assert_eq!(telemetry.len(), 2);
482        let panic_tel = telemetry
483            .iter()
484            .find(|t| t.id == "panicking_check")
485            .unwrap();
486        match &panic_tel.outcome {
487            crate::telemetry::CheckOutcome::Panicked(msg) => {
488                assert!(msg.contains("intentional check panic"));
489            }
490            _ => panic!("expected Panicked outcome"),
491        }
492        let healthy_tel = telemetry.iter().find(|t| t.id == "healthy_check").unwrap();
493        assert!(matches!(
494            healthy_tel.outcome,
495            crate::telemetry::CheckOutcome::Ok
496        ));
497        assert_eq!(healthy_tel.findings, 1);
498    }
