5//! Non-dataflow checks: policy assertions over the lowered IR that need no
6//! taint analysis.
7//!
8//! Two tiers, deliberately separated:
9//!
10//! **Built-in seed checks** (`weak_hash` and future structural rules) supply
11//! only *mechanism*, how to look at the IR. They are the bootstrapping
12//! oracle, kept minimal so shape-thinking cannot creep back in.
13//!
14//! **Corpus-learned checks** (`LearnedCheckFact` in the fact table) supply
15//! the *conclusions*, which calls are violations, with bundle-authored
16//! advisory text. They arrive through `.frc` bundles, were verified by the
17//! bundler's replay gate (fire on positives, silent on negatives), and are
18//! the scale path: nobody writes 45k rules.
19//!
20//! Line drawn: **built-in = how to look; learned = what to conclude.**

35/// Text of a caught panic payload (`&str` and `String` payloads; anything else
36/// is reported as `unknown panic`).
37fn panic_message(payload: &(dyn std::any::Any + Send)) -> String {
38    if let Some(s) = payload.downcast_ref::<&str>() {
39        (*s).to_string()
40    } else if let Some(s) = payload.downcast_ref::<String>() {
41        s.clone()
42    } else {
43        "unknown panic".to_string()
44    }
45}

76/// Last segment of a call reference: everything after the last `.`, then
77/// after the last `::` (Rust/C++ paths), so `security.hash`, `sha2::digest`
78/// and `ns::mod.fn` all reduce to the bare method name the check
79/// vocabularies are keyed by.
80pub(crate) fn last_segment(call: &str) -> &str {
81    let s = call.rsplit('.').next().unwrap_or(call);
82    s.rsplit("::").next().unwrap_or(s)
83}

85/// One non-dataflow policy finding.
86#[derive(Debug, Clone, PartialEq, Eq)]
87pub struct CheckerFinding {
88    /// Function containing the violating instruction.
89    pub function: String,
90    /// Stable rule id (e.g. `"weak_hash_md5"`, or a bundle's rule id).
91    /// Built-in rules use `&'static str` ids; learned rules carry their
92    /// bundle-supplied id, interned here.
93    pub rule: String,
94    /// Short human description of the violation. Carried verbatim when the
95    /// driving knowledge (bundle fact, authored policy) supplied prose.
96    /// Spec checks leave it empty and emit `params` instead; the consumer
97    /// then renders the observation from a `frensense-lang` template
98    /// keyed by `rule`.
99    pub message: String,
100    /// Structured observation parameters `(template key, value)` for
101    /// findings whose `message` is empty. Empty when `message` is set.
102    pub params: Vec<(&'static str, String)>,
103    /// Byte range of the violating call in the source file, when known.
104    pub span: Option<(usize, usize)>,
105    /// Advisory severity declared by the driving fact (learned check,
106    /// authored policy, or spec rule), copied verbatim. Empty when the
107    /// driving rule declares none; consumers then resolve severity from
108    /// their own rule registry.
109    pub severity: String,
110    /// Knowledge source: spec seed, authored policy, or bundle-learned fact.
111    pub provenance: Provenance,
112}

114/// The single finding identity: two findings from the same function with
115/// the same rule at the same span are the same finding. This is the only
116/// dedup key - `check_all` is the one place findings are deduplicated.
117fn finding_key(f: &CheckerFinding) -> (String, String, usize) {
118    (
119        f.function.clone(),
120        f.rule.clone(),
121        f.span.map(|s| s.0).unwrap_or(usize::MAX),
122    )
123}

125/// Name each learned-check finding that only one evaluator produced.
126///
127/// `LearnedChecks` (function-scoped: last-segment matching, a guard anywhere
128/// in the function suppresses) and `Policies` (program-scoped: the same
129/// facts via `PolicyFact::from_legacy`, import-alias resolution, the guard
130/// must dominate the trigger) both consume `facts.learned_checks`. The
131/// driver keeps the union, so a disagreement silently widens or narrows the
132/// reported set. `FRENSdbg_CHECKS` makes that visible without changing the
133/// result, which is the prerequisite for choosing one evaluator deliberately.
134fn trace_learned_divergence(
135    irs: &[&FunctionIR],
136    facts: &FactTable,
137    pkg_scope: Option<&policy::PackageScope<'_>>,
138) {
139    if !crate::debug_flags::DebugFlags::checks_enabled() || facts.learned_checks.is_empty() {
140        return;
141    }
142    let (legacy_only, policy_only) = learned_divergence(irs, facts, pkg_scope);
143    for (path, only) in [("legacy", legacy_only), ("policy", policy_only)] {
144        for (function, rule, span) in only {
145            crate::dbg_trace!(
146                true,
147                "frensense: learned-check divergence ({path} only): {function} {rule} @{span}"
148            );
149        }
150    }
151}

153/// Findings that only one of the two learned-check evaluators produced.
154///
155/// Returns `(legacy_only, policy_only)`, each sorted by
156/// `(function, rule, span)`. Exposed to the test module so a change to either
157/// path's semantics shows up as a fixture failure rather than a silent shift
158/// in the reported union.
159type FindingKey = (String, String, usize);
160
161fn learned_divergence(
162    irs: &[&FunctionIR],
163    facts: &FactTable,
164    pkg_scope: Option<&policy::PackageScope<'_>>,
165) -> (Vec<FindingKey>, Vec<FindingKey>) {
166    let key = |f: &CheckerFinding| {
167        (
168            f.function.clone(),
169            f.rule.clone(),
170            f.span.map(|s| s.0).unwrap_or(usize::MAX),
171        )
172    };
173    let legacy: std::collections::BTreeSet<_> = irs
174        .iter()
175        .flat_map(|ir| learned::check(ir, facts))
176        .map(|f| key(&f))
177        .collect();
178    let learned_rules: FxHashSet<&str> = facts
179        .learned_checks
180        .iter()
181        .map(|(fact, _)| fact.rule.as_str())
182        .collect();
183    let policy: std::collections::BTreeSet<_> = policy::check_program_scoped(irs, facts, pkg_scope)
184        .into_iter()
185        .filter(|f| learned_rules.contains(f.rule.as_str()))
186        .map(|f| key(&f))
187        .collect();
188    (
189        legacy.difference(&policy).cloned().collect(),
190        policy.difference(&legacy).cloned().collect(),
191    )
192}

194/// Push a finding unless an earlier check already reported the same
195/// `(function, rule, span)` identity.
196fn push_unique(
197    all: &mut Vec<CheckerFinding>,
198    seen: &mut FxHashSet<(String, String, usize)>,
199    f: CheckerFinding,
200) {
201    if seen.insert(finding_key(&f)) {
202        all.push(f);
203    }
204}

206/// Run every registered check over every function, deduped and ordered by
207/// (function, span) so output is deterministic.
208///
209/// `facts` carries the bundle-learned checks; pass `&FactTable::default()`
210/// when scanning without a bundle.
211#[cfg(test)]
212pub(crate) fn check_all<'a>(
213    irs: impl IntoIterator<Item = &'a FunctionIR>,
214    facts: &FactTable,
215) -> Vec<CheckerFinding> {
216    check_all_with_telemetry(irs, facts, None, None).0
217}

219/// `check_all` with the program value-flow graph and package scope, while
220/// also collecting execution telemetry (timings, run counts, findings, and
221/// outcomes with catch_unwind failure isolation).
222///
223/// `prog` lets the UAF checker walk interprocedural free/use edges (callee
224/// parameter frees, caller-side frees, factory provenance) that intraprocedural
225/// IR alone cannot see. `pkg_scope`: policies carrying `package: Some(pkg)`
226/// only fire in functions whose file imports `pkg`; `None` disables filtering.
227pub(crate) fn check_all_with_telemetry<'a>(
228    irs: impl IntoIterator<Item = &'a FunctionIR>,
229    facts: &FactTable,
230    prog: Option<&ProgramSvfg<'_>>,
231    pkg_scope: Option<&policy::PackageScope<'_>>,
232) -> (Vec<CheckerFinding>, Vec<crate::telemetry::CheckTelemetry>) {
233    let registry = registry::default_registry();
234    check_all_with_registry_and_telemetry(&registry, irs, facts, prog, pkg_scope)
235}

237/// [`check_all_with_telemetry`] with an explicit [`registry::CheckRegistry`].
238pub(crate) fn check_all_with_registry_and_telemetry<'a>(
239    registry: &registry::CheckRegistry,
240    irs: impl IntoIterator<Item = &'a FunctionIR>,
241    facts: &FactTable,
242    prog: Option<&ProgramSvfg<'_>>,
243    pkg_scope: Option<&policy::PackageScope<'_>>,
244) -> (Vec<CheckerFinding>, Vec<crate::telemetry::CheckTelemetry>) {
245    let irs: Vec<&FunctionIR> = irs.into_iter().collect();
246    trace_learned_divergence(&irs, facts, pkg_scope);
247    // One memory-summary registry per scan: program summaries are inferred
248    // from the whole IR set once, then shared by every memory check.
249    let summaries =
250        memory_summary::MemorySummaryRegistry::from_facts(facts).infer_program_summaries_into(&irs);
251    let program_ctx = CheckContext {
252        irs: &irs,
253        current: None,
254        prog,
255        facts,
256        pkg_scope,
257        summaries: &summaries,
258    };
259
260    let mut seen: FxHashSet<(String, String, usize)> = FxHashSet::default();
261    let mut all = Vec::new();
262    let mut telemetries = Vec::new();
263
264    // Program-scoped checks run once over the whole IR set (they correlate
265    // guards in one function with definitions in another).
266    for check in registry
267        .checks()
268        .filter(|c| c.scope() == CheckScope::Program)
269    {
270        let id = check.id().to_string();
271        let start = std::time::Instant::now();
272        let res =
273            std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| check.run(&program_ctx)));
274        let ms = start.elapsed().as_millis();
275        let mut count = 0;
276        let outcome = match res {
277            Ok(findings) => {
278                count = findings.len();
279                for f in findings {
280                    push_unique(&mut all, &mut seen, f);
281                }
282                crate::telemetry::CheckOutcome::Ok
283            }
284            Err(payload) => {
285                let msg = panic_message(payload.as_ref());
286                eprintln!("frensense: check `{id}` panicked: {msg}");
287                crate::telemetry::CheckOutcome::Panicked(msg)
288            }
289        };
290        telemetries.push(crate::telemetry::CheckTelemetry {
291            id,
292            scope: "program".to_string(),
293            ran: 1,
294            findings: count,
295            millis: ms,
296            outcome,
297        });
298    }
299
300    // Function-scoped checks are grouped by check so we time and isolate per-check:
301    for check in registry
302        .checks()
303        .filter(|c| c.scope() == CheckScope::Function)
304    {
305        let id = check.id().to_string();
306        let start = std::time::Instant::now();
307        let mut ran = 0;
308        let mut count = 0;
309        let mut outcome = crate::telemetry::CheckOutcome::Ok;
310
311        for ir in &irs {
312            ran += 1;
313            let ctx = CheckContext {
314                current: Some(*ir),
315                ..program_ctx
316            };
317            let res = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| check.run(&ctx)));
318            match res {
319                Ok(findings) => {
320                    count += findings.len();
321                    for f in findings {
322                        push_unique(&mut all, &mut seen, f);
323                    }
324                }
325                Err(payload) => {
326                    let msg = panic_message(payload.as_ref());
327                    eprintln!(
328                        "frensense: check `{id}` panicked on function `{}`: {msg}",
329                        ir.name
330                    );
331                    if matches!(outcome, crate::telemetry::CheckOutcome::Ok) {
332                        outcome = crate::telemetry::CheckOutcome::Panicked(msg);
333                    }
334                }
335            }
336        }
337        let ms = start.elapsed().as_millis();
338        telemetries.push(crate::telemetry::CheckTelemetry {
339            id,
340            scope: "function".to_string(),
341            ran,
342            findings: count,
343            millis: ms,
344            outcome,
345        });
346    }
347
348    all.sort_by_key(|f| {
349        (
350            f.function.clone(),
351            f.span.map(|s| s.0).unwrap_or(usize::MAX),
352        )
353    });
354    (all, telemetries)
355}

357/// The learned-check rule: apply corpus-verified checks from the fact
358/// table. Mechanism identical to the seed checks (call matching over the
359/// IR); the conclusions are the bundle's.
360pub(crate) mod learned {
361    use super::CheckerFinding;
362    use crate::analysis::taint::facts::FactTable;
363    use crate::ir::function::{FunctionIR, Instruction, Operand};
364
365    /// Last-segment names of every call in the function (one pass).
366    pub(crate) fn call_segments(ir: &FunctionIR) -> Vec<String> {
367        let mut segs = Vec::new();
368        for block in ir.blocks.values() {
369            for instr in &block.instructions {
370                match instr {
371                    Instruction::CallStatic { func, .. } => {
372                        segs.push(super::last_segment(func).to_string());
373                    }
374                    Instruction::CallVirtual { method, .. } => {
375                        segs.push(super::last_segment(method).to_string());
376                    }
377                    _ => {}
378                }
379            }
380        }
381        segs
382    }
383
384    /// True when `var` (or its def chain) is compared against a literal
385    /// with one of `ops` in this function, inline range enforcement
386    /// (`if (d < 0 || d > MAX) …`). Walks one BinaryOp hop up from `var`
387    /// and one down (comparisons may be written in either order).
388    pub(crate) fn has_range_check(
389        ir: &FunctionIR,
390        var: crate::ir::function::VarId,
391        ops: &[String],
392    ) -> bool {
393        // Collect BinaryOps involving `var` on either side.
394        let involved = |lhs: &Operand, rhs: &Operand| {
395            matches!(lhs, Operand::Var(v) if *v == var)
396                || matches!(rhs, Operand::Var(v) if *v == var)
397        };
398        for block in ir.blocks.values() {
399            for instr in &block.instructions {
400                if let Instruction::BinaryOp { op, lhs, rhs, .. } = instr
401                    && ops.contains(op)
402                    && involved(lhs, rhs)
403                {
404                    // The OTHER side must be a literal (the bound).
405                    let other = match (lhs, rhs) {
406                        (Operand::Var(v), r) if *v == var => r,
407                        (l, Operand::Var(v)) if *v == var => l,
408                        _ => continue,
409                    };
410                    if matches!(
411                        other,
412                        Operand::StringLiteral(_)
413                            | Operand::IntLiteral(_)
414                            | Operand::FloatLiteral(_)
415                    ) {
416                        return true;
417                    }
418                }
419            }
420        }
421        false
422    }
423
424    /// Check whether `var` is guarded by a range check at `block`.
425    /// First evaluates concrete intervals via `ValueInfo::range_at(block, var)`.
426    /// If an interval exists, verifies that it satisfies the bounds implied by `ops`.
427    /// If no interval exists (e.g. non-integer or complex unmodeled flow), falls
428    /// back to syntactic presence `has_range_check`.
429    pub(crate) fn is_range_guarded(
430        ir: &FunctionIR,
431        var: crate::ir::function::VarId,
432        ops: &[String],
433        block: crate::ir::function::BlockId,
434        val_info: &crate::analysis::value::ValueInfo,
435    ) -> bool {
436        if let Some((lo, hi)) = val_info.range_at(block, var) {
437            let requires_upper = ops.iter().any(|op| op == "<" || op == "<=");
438            let requires_lower = ops.iter().any(|op| op == ">" || op == ">=");
439            let satisfies_upper = !requires_upper || hi < i64::MAX;
440            let satisfies_lower = !requires_lower || lo > i64::MIN;
441            if satisfies_upper && satisfies_lower && (requires_upper || requires_lower) {
442                return true;
443            }
444            return false;
445        }
446        has_range_check(ir, var, ops)
447    }
448
449    /// Apply every learned check to one function.
450    pub(super) fn check(ir: &FunctionIR, facts: &FactTable) -> Vec<CheckerFinding> {
451        if facts.learned_checks.is_empty() {
452            return Vec::new();
453        }
454        let segs = call_segments(ir);
455        let val_info = crate::analysis::value::analyze(ir);
456        let mut findings = Vec::new();
457        for (&bid, block) in &ir.blocks {
458            for instr in &block.instructions {
459                let (callee, args) = match instr {
460                    Instruction::CallStatic { func, args, .. } => (func, args),
461                    Instruction::CallVirtual {
462                        method: func, args, ..
463                    } => (func, args),
464                    _ => continue,
465                };
466                let last = super::last_segment(callee);
467                for (fact, provenance) in facts.learned_checks_for(last) {
468                    // Guard qualification: a rule with `unless_guard` is
469                    // silent when the guard call is present in the same
470                    // function, the corpus says "trigger without
471                    // enforcement" is the violation.
472                    if let Some(guard) = &fact.unless_guard {
473                        let gseg = super::last_segment(guard);
474                        if segs.iter().any(|s| s == gseg) {
475                            continue;
476                        }
477                    }
478                    // Range qualification: silent when any trigger argument
479                    // var is compared against a literal bound, inline
480                    // enforcement without a named helper.
481                    if let Some(ops) = &fact.unless_range_check {
482                        let guarded = args.iter().any(|a| match a {
483                            Operand::Var(v) => is_range_guarded(ir, *v, ops, bid, &val_info),
484                            _ => false,
485                        });
486                        if guarded {
487                            continue;
488                        }
489                    }
490                    findings.push(CheckerFinding {
491                        function: ir.name.clone(),
492                        rule: fact.rule.clone(),
493                        message: fact.message.clone(),
494                        params: Vec::new(),
495                        span: ir.instr_span(instr),
496                        severity: fact.severity.clone(),
497                        provenance,
498                    });
499                }
500            }
501        }
502        findings
503    }
504}

506#[cfg(test)]
507mod panic_message_tests {
508    use super::panic_message;
509    use std::panic::{catch_unwind, panic_any};
510
511    #[test]
512    fn extracts_str_string_and_unknown_payloads() {
513        let str_payload = catch_unwind(|| panic!("plain")).unwrap_err();
514        assert_eq!(panic_message(str_payload.as_ref()), "plain");
515        let string_payload = catch_unwind(|| panic!("n={}", 3)).unwrap_err();
516        assert_eq!(panic_message(string_payload.as_ref()), "n=3");
517        let other = catch_unwind(|| panic_any(7_u8)).unwrap_err();
518        assert_eq!(panic_message(other.as_ref()), "unknown panic");
519    }
