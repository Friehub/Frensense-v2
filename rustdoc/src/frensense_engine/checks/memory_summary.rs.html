5//! Interprocedural memory semantics and function summaries (Limitation M1).
6//!
7//! Tracks allocation and deallocation contracts across function boundaries:
8//! - `@returns_fresh`: Functions returning fresh heap memory (factory functions).
9//! - `@consumes(param_idx)`: Functions deallocating a pointer passed to a parameter slot.
10//!
11//! Maintains strict 0-FP: only functions provably returning fresh allocations
12//! or executing deallocations on parameters receive contracts. Pointers passed
13//! for reading/writing without deallocation are never marked consumed.
14
15use rustc_hash::{FxHashMap, FxHashSet};
16
17use crate::analysis::value;
18use crate::checks::last_segment;
19use crate::ir::function::{FunctionIR, Instruction, Operand, Terminator, VarId};
20
21/// Buffer capacity specification for an allocation contract.
22///
23/// Derives serde so it can travel inside a `.frc` bundle's bincode payload.
24#[derive(Debug, Clone, PartialEq, Eq)]
25#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
26#[derive(Default)]
27pub enum CapacitySpec {
28    /// Exact constant byte/element capacity (e.g. malloc(64)).
29    Exact(i64),
30    /// Capacity matches the value passed to argument at index `usize`.
31    Param(usize),
32    /// Capacity matches `args[p1] * args[p2]` (e.g. calloc(n, sz)).
33    ParamProduct(usize, usize),
34    /// Fresh allocation, but capacity is dynamic or unconstrained.
35    #[default]
36    Unknown,
37}
38
39/// Summary contract describing a function's memory semantics.
40///
41/// Derives serde so it can travel inside a `.frc` bundle's bincode payload.
42#[derive(Debug, Clone, PartialEq, Eq, Default)]
43#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
44pub struct MemorySummary {
45    /// True if the function returns a newly allocated object.
46    pub returns_fresh: bool,
47    /// Capacity specification for the returned object.
48    pub return_capacity: CapacitySpec,
49    /// Parameter indices consumed (freed/deallocated) by this function.
50    pub consumes_params: Vec<usize>,
51    /// Parameter indices through which freshly allocated objects are written (out-parameters).
52    #[cfg_attr(feature = "serialize", serde(default))]
53    pub out_params_fresh: Vec<usize>,
54}
55
56/// Registry storing interprocedural memory contracts for library primitives
57/// and program functions.
58#[derive(Debug, Clone, Default)]
59pub struct MemorySummaryRegistry {
60    /// Contracts keyed by function or method name.
61    pub summaries: FxHashMap<String, MemorySummary>,
62}
63
64impl MemorySummaryRegistry {
65    /// Build a registry from a memory-function vocabulary (seeded into a
66    /// [`crate::analysis::taint::facts::FactTable`] by the default pack):
67    /// one summary per entry, mapping the vocabulary's capacity shape
68    /// onto [`CapacitySpec`].
69    fn from_vocabulary(vocab: &[crate::analysis::taint::facts::MemoryFuncSpec]) -> Self {
70        use crate::analysis::taint::facts::AllocCapacity;
71        let mut summaries = FxHashMap::default();
72        for f in vocab {
73            summaries.insert(
74                f.name.to_string(),
75                MemorySummary {
76                    returns_fresh: f.returns_fresh,
77                    return_capacity: match f.capacity {
78                        AllocCapacity::Unknown => CapacitySpec::Unknown,
79                        AllocCapacity::Param(n) => CapacitySpec::Param(n),
80                        AllocCapacity::ParamProduct(a, b) => CapacitySpec::ParamProduct(a, b),
81                    },
82                    consumes_params: f.consumes_params.to_vec(),
83                    out_params_fresh: Vec::new(),
84                },
85            );
86        }
87        Self { summaries }
88    }
89
90    /// Seed a registry from the spec's memory vocabulary in `facts`
91    /// (seeded via the default pack), then layer
92    /// bundle-learned contracts on top. Learned contracts override the
93    /// vocabulary on name collision (more specific corpus knowledge wins).
94    ///
95    /// Call this instead of `default()` when a `.frc` bundle is loaded, then
96    /// follow up with `infer_program_summaries_into` for the current project's
97    /// per-file fixpoint.
98    pub fn from_facts(facts: &crate::analysis::taint::facts::FactTable) -> Self {
99        let mut reg = Self::from_vocabulary(&facts.memory_functions);
100        for contract in &facts.memory_contracts {
101            reg.summaries.insert(
102                contract.name.clone(),
103                MemorySummary {
104                    returns_fresh: contract.returns_fresh,
105                    return_capacity: contract.return_capacity.clone(),
106                    consumes_params: contract.consumes_params.clone(),
107                    out_params_fresh: Vec::new(),
108                },
109            );
110        }
111        for alloc in &facts.custom_allocators {
112            reg.summaries
113                .entry(alloc.clone())
114                .or_insert_with(|| MemorySummary {
115                    returns_fresh: true,
116                    return_capacity: CapacitySpec::Unknown,
117                    consumes_params: Vec::new(),
118                    out_params_fresh: Vec::new(),
119                });
120        }
121        for dealloc in &facts.custom_deallocators {
122            reg.summaries
123                .entry(dealloc.clone())
124                .or_insert_with(|| MemorySummary {
125                    returns_fresh: false,
126                    return_capacity: CapacitySpec::Unknown,
127                    consumes_params: vec![0],
128                    out_params_fresh: Vec::new(),
129                });
130        }
131        reg
132    }

134    /// Run the fixpoint inference in-place on `self` (which was already seeded
135    /// from the bundle via `from_facts`).  This is the per-project pass: it
136    /// discovers project-specific wrapper chains on top of whatever the bundle
137    /// already taught.
138    pub fn infer_program_summaries_into(mut self, irs: &[&FunctionIR]) -> Self {
139        self.run_fixpoint(irs);
140        self
141    }
142
143    /// Retrieve summary for a function by exact or short segment name.
144    pub(crate) fn get(&self, name: &str) -> Option<&MemorySummary> {
145        self.summaries
146            .get(name)
147            .or_else(|| self.summaries.get(last_segment(name)))
148    }
149
150    /// Returns true if calling `name` returns a freshly allocated object.
151    /// Vocabulary-driven: the registry was seeded from the default pack
152    /// through `FactTable::memory_functions`.
153    pub fn returns_fresh(&self, name: &str) -> bool {
154        self.get(name).is_some_and(|s| s.returns_fresh)
155    }
156
157    /// Returns the capacity specification for a fresh allocation returned by `name`.
158    pub fn return_capacity(&self, name: &str) -> Option<&CapacitySpec> {
159        self.get(name).and_then(|s| {
160            if s.returns_fresh {
161                Some(&s.return_capacity)
162            } else {
163                None
164            }
165        })
166    }
167
168    /// Returns the indices of parameters consumed (deallocated) by `name`.
169    pub fn consumes_params(&self, name: &str) -> Vec<usize> {
170        self.get(name)
171            .map(|s| s.consumes_params.clone())
172            .unwrap_or_default()
173    }
174
175    /// Returns the indices of parameters through which fresh allocations are written by `name`.
176    pub(crate) fn out_params_fresh(&self, name: &str) -> Vec<usize> {
177        self.get(name)
178            .map(|s| s.out_params_fresh.clone())
179            .unwrap_or_default()
180    }

182    /// Internal fixpoint pass shared by `infer_program_summaries` and
183    /// `infer_program_summaries_into`. Mutates `self` in place.
184    fn run_fixpoint(&mut self, irs: &[&FunctionIR]) {
185        // Trace a variable back to a parameter index through SSA assignments / casts.
186        fn trace_to_param(ir: &FunctionIR, var: VarId) -> Option<usize> {
187            if let Some(pos) = ir.parameters.iter().position(|&p| p == var) {
188                return Some(pos);
189            }
190            let mut cur = var;
191            let mut visited = FxHashSet::default();
192            while visited.insert(cur) {
193                let mut found_next = None;
194                for blk in ir.blocks.values() {
195                    for instr in &blk.instructions {
196                        match instr {
197                            Instruction::Assign {
198                                dest,
199                                src: Operand::Var(src),
200                            }
201                            | Instruction::Cast {
202                                dest,
203                                src: Operand::Var(src),
204                                ..
205                            } if *dest == cur => {
206                                if let Some(pos) = ir.parameters.iter().position(|&p| p == *src) {
207                                    return Some(pos);
208                                }
209                                found_next = Some(*src);
210                                break;
211                            }
212                            _ => {}
213                        }
214                    }
215                    if found_next.is_some() {
216                        break;
217                    }
218                }
219                match found_next {
220                    Some(n) => cur = n,
221                    None => break,
222                }
223            }
224            None
225        }
226
227        // `realloc`-family call that resizes an object this function did not
228        // allocate: its first argument does not trace back to an allocation
229        // in the same body (a load through a parameter, a global, a phi of
230        // such loads). The result is the caller's object moved, not a new
231        // one, so a function returning it is not a factory. redis
232        // `ziplistMerge(&first, &second)` does `target = zrealloc(*first or
233        // *second, n); return target;` and every caller that stored the
234        // result back into its own list was reported as leaking.
235        fn resizes_borrowed_object(ir: &FunctionIR, func: &str, args: &[Operand]) -> bool {
236            if !last_segment(func).to_ascii_lowercase().contains("realloc") {
237                return false;
238            }
239            let Some(Operand::Var(v)) = args.first() else {
240                return false;
241            };
242            let mut cur = *v;
243            let mut seen = FxHashSet::default();
244            while seen.insert(cur) {
245                match ir.def_instruction(cur) {
246                    Some(Instruction::CallStatic { .. } | Instruction::Allocate { .. }) => {
247                        return false;
248                    }
249                    Some(Instruction::Assign {
250                        src: Operand::Var(next),
251                        ..
252                    })
253                    | Some(Instruction::Cast {
254                        src: Operand::Var(next),
255                        ..
256                    }) => cur = *next,
257                    _ => return true,
258                }
259            }
260            true
261        }
262
263        // Fixed-point inference loop over program functions.
264        let mut changed = true;
265        let mut iterations = 0;
266        while changed && iterations < 10 {
267            changed = false;
268            iterations += 1;
269
270            for &ir in irs {
271                let mut consumes: Vec<usize> = Vec::new();
272                let mut returns_fresh = false;
273                let mut return_capacity = CapacitySpec::Unknown;
274
275                // 1. Infer consumed parameters
276                for blk in ir.blocks.values() {
277                    for instr in &blk.instructions {
278                        if let Instruction::CallStatic { func, args, .. } = instr {
279                            let consumed_slots = self.consumes_params(func.as_str());
280                            for slot in consumed_slots {
281                                if let Some(Operand::Var(v)) = args.get(slot)
282                                    && let Some(param_idx) = trace_to_param(ir, *v)
283                                    && !consumes.contains(&param_idx)
284                                {
285                                    consumes.push(param_idx);
286                                }
287                            }
288                        }
289                    }
290                }
291                consumes.sort();
292
293                // 2. Infer returned allocation provenance
294                let mut return_vars: Vec<VarId> = Vec::new();
295                for blk in ir.blocks.values() {
296                    if let Terminator::Return {
297                        src: Some(Operand::Var(v)),
298                    } = &blk.terminator
299                    {
300                        return_vars.push(*v);
301                    }
302                }
303
304                if !return_vars.is_empty() {
305                    let val_info = value::analyze(ir);
306
307                    // Chase through Assign/Cast chains to find the root allocation
308                    // instruction for a given var (handles SSA copies like `p = malloc(n);
309                    // tmp = p; return tmp;`).
310                    let find_alloc_root = |start: VarId| -> Option<&Instruction> {
311                        let mut cur = start;
312                        let mut seen = FxHashSet::default();
313                        loop {
314                            if !seen.insert(cur) {
315                                break;
316                            }
317                            match ir.def_instruction(cur) {
318                                Some(
319                                    instr @ Instruction::CallStatic { .. }
320                                    | instr @ Instruction::Allocate { .. },
321                                ) => return Some(instr),
322                                Some(Instruction::Assign {
323                                    src: Operand::Var(next),
324                                    ..
325                                })
326                                | Some(Instruction::Cast {
327                                    src: Operand::Var(next),
328                                    ..
329                                }) => {
330                                    cur = *next;
331                                }
332                                _ => break,
333                            }
334                        }
335                        None
336                    };
337
338                    for ret_var in return_vars {
339                        if let Some(def) = find_alloc_root(ret_var) {
340                            match def {
341                                Instruction::CallStatic { func, args, .. }
342                                    if self.returns_fresh(func)
343                                        && !resizes_borrowed_object(ir, func, args) =>
344                                {
345                                    returns_fresh = true;
346                                    if let Some(cap_spec) = self.return_capacity(func) {
347                                        match cap_spec {
348                                            CapacitySpec::Param(p_idx) => {
349                                                if let Some(arg_op) = args.get(*p_idx) {
350                                                    match arg_op {
351                                                        Operand::IntLiteral(k) => {
352                                                            return_capacity =
353                                                                CapacitySpec::Exact(*k);
354                                                        }
355                                                        Operand::Var(v) => {
356                                                            if let Some(param_idx) =
357                                                                trace_to_param(ir, *v)
358                                                            {
359                                                                return_capacity =
360                                                                    CapacitySpec::Param(param_idx);
361                                                            } else if let Some(c) =
362                                                                val_info.const_int(*v)
363                                                            {
364                                                                return_capacity =
365                                                                    CapacitySpec::Exact(c);
366                                                            }
367                                                        }
368                                                        _ => {}
369                                                    }
370                                                }
371                                            }
372                                            CapacitySpec::ParamProduct(p1, p2) => {
373                                                let param1 =
374                                                    args.get(*p1).and_then(|op| match op {
375                                                        Operand::Var(v) => trace_to_param(ir, *v),
376                                                        _ => None,
377                                                    });
378                                                let param2 =
379                                                    args.get(*p2).and_then(|op| match op {
380                                                        Operand::Var(v) => trace_to_param(ir, *v),
381                                                        _ => None,
382                                                    });
383                                                if let (Some(a), Some(b)) = (param1, param2) {
384                                                    return_capacity =
385                                                        CapacitySpec::ParamProduct(a, b);
386                                                }
387                                            }
388                                            CapacitySpec::Exact(k) => {
389                                                return_capacity = CapacitySpec::Exact(*k);
390                                            }
391                                            CapacitySpec::Unknown => {}
392                                        }
393                                    }
394                                }
395                                // Object/array literals (`Allocate`) are not
396                                // freshness roots: inferring factories from
397                                // them marks every wrapper of a literal fresh,
398                                // pulling GC-managed objects (whose lifetime
399                                // needs no release) into allocation-lifetime
400                                // candidates. Freshness roots at the declared
401                                // allocator vocabulary - a `CallStatic`
402                                // returning a call that is already fresh.
403                                _ => {}
404                            }
405                        }
406                    }
407                }
408
409                // 3. Infer out-parameter allocation provenance
410                let mut out_params: Vec<usize> = Vec::new();
411                for blk in ir.blocks.values() {
412                    for instr in &blk.instructions {
413                        match instr {
414                            Instruction::StoreElement {
415                                base,
416                                src: Operand::Var(src),
417                                ..
418                            }
419                            | Instruction::StoreField {
420                                base,
421                                src: Operand::Var(src),
422                                ..
423                            } => {
424                                if let Some(param_idx) = trace_to_param(ir, *base) {
425                                    let mut cur = *src;
426                                    let mut is_fresh = false;
427                                    let mut seen = FxHashSet::default();
428                                    while seen.insert(cur) {
429                                        match ir.def_instruction(cur) {
430                                            Some(Instruction::CallStatic { func, .. }) => {
431                                                if self.returns_fresh(func) {
432                                                    is_fresh = true;
433                                                }
434                                                break;
435                                            }
436                                            Some(Instruction::Allocate { .. }) => {
437                                                is_fresh = true;
438                                                break;
439                                            }
440                                            Some(Instruction::Assign {
441                                                src: Operand::Var(next),
442                                                ..
443                                            })
444                                            | Some(Instruction::Cast {
445                                                src: Operand::Var(next),
446                                                ..
447                                            }) => {
448                                                cur = *next;
449                                            }
450                                            _ => break,
451                                        }
452                                    }
453                                    if is_fresh && !out_params.contains(&param_idx) {
454                                        out_params.push(param_idx);
455                                    }
456                                }
457                            }
458                            Instruction::CallStatic { func, args, .. } => {
459                                let callee_out = self.out_params_fresh(func);
460                                for &slot in &callee_out {
461                                    if let Some(Operand::Var(arg_var)) = args.get(slot) {
462                                        let target = match ir.def_instruction(*arg_var) {
463                                            Some(Instruction::AddressOf { src, .. }) => *src,
464                                            _ => *arg_var,
465                                        };
466                                        if let Some(param_idx) = trace_to_param(ir, target)
467                                            && !out_params.contains(&param_idx)
468                                        {
469                                            out_params.push(param_idx);
470                                        }
471                                    }
472                                }
473                            }
474                            _ => {}
475                        }
476                    }
477                }
478                out_params.sort();
479
480                // If any contract was discovered, register it.
481                if returns_fresh || !consumes.is_empty() || !out_params.is_empty() {
482                    let summary = MemorySummary {
483                        returns_fresh,
484                        return_capacity,
485                        consumes_params: consumes,
486                        out_params_fresh: out_params,
487                    };
488                    let prev = self.summaries.get(&ir.name);
489                    if prev != Some(&summary) {
490                        self.summaries.insert(ir.name.clone(), summary);
491                        changed = true;
492                    }
493                }
494            }
495        }
496    }
