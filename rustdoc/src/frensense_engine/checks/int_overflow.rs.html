5//! Integer overflow in allocation-size arithmetic (CWE-190 -> CWE-680):
6//! `malloc(count * elem_size)` where the multiplication can wrap past the
7//! platform's allocation-size limit yields an undersized buffer that the
8//! subsequent fill overruns. Modeled on the 2026 CVE class (llama.cpp
9//! CVE-2026-27940, Perl CVE-2026-8376, glibc CVE-2026-0861).
10//!
11//! Built-in = how to look; learned = what to conclude: the prover below is
12//! the stable, language-agnostic "how" (provable ranges, wrap threshold,
13//! allocation-capacity provenance). Which rule fires - id, threshold, and
14//! advisory - comes from `FactTable::integer_overflow_rules`: seeded by
15//! the default pack and extended by a corpus bundle from a family's
16//! `[frensense] check-rule:` declaration
17//! (`LearnedFactEntry::IntegerOverflowRule`). New rules of this class come
18//! from the pack or a corpus bundle; the engine does not change.
19//!
20//! Zero-FP by construction:
21//! - Both operand intervals must be provable and non-negative at the
22//!   multiplication's block (guards narrow via branch sharpening;
23//!   unknown operands skip the site).
24//! - The product corner (computed in `i128`, never wrapped) must exceed
25//!   the rule's wrap threshold (u64::MAX == SIZE_MAX on LP64/LLP64).
26//! - The product must flow into an allocation's capacity argument, whose
27//!   index comes from the lang-declared capacity spec. `calloc`-style
28//!   `ParamProduct` allocators multiply internally and are excluded, as
29//!   are deallocations (no capacity spec).
30

41/// Conservative inclusive upper bound for an unboundedly dynamic size
42/// operand under the unbounded-operand modality: the largest size the
43/// value lattice can denote. Two such operands can only exceed ANY
44/// realistic allocation-size threshold at their worst-case corner.
45const DYNAMIC_MAX: i64 = i64::MAX;
46
47/// Conservative inclusive upper-bound test for an unboundedly dynamic size
48/// operand under the unbounded-operand modality: a var the value lattice
49/// cannot bound at all (no interval, no provable constant) anywhere in the
50/// function - by definition `malloc(rows * cols)` operator-provided data.
51fn operand_is_dynamic(op: &Operand, val_info: &ValueInfo) -> bool {
52    match op {
53        Operand::Var(v) => {
54            val_info.range(*v).is_none()
55                && val_info.range_at(BlockId(0), *v).is_none()
56                && val_info.const_int(*v).is_none()
57        }
58        _ => false,
59    }
60}

62/// Evaluate an operand's inclusive integer interval at a block entry,
63/// mirroring the spatial checker's range lookup (block-entry sharpened
64/// state first, then the merged per-definition value).
65fn range_of(op: &Operand, block: BlockId, val_info: &ValueInfo) -> Option<(i64, i64)> {
66    match op {
67        Operand::IntLiteral(k) => Some((*k, *k)),
68        Operand::Var(v) => {
69            if let Some(r) = val_info.range_at(block, *v) {
70                return Some(r);
71            }
72            if let Some(c) = val_info.const_int_at(block, *v) {
73                return Some((c, c));
74            }
75            if let Some(r) = val_info.range(*v) {
76                return Some(r);
77            }
78            val_info.const_int(*v).map(|c| (c, c))
79        }
80        _ => None,
81    }
82}

84/// Argument index carrying a single caller-computed capacity for `func`,
85/// when the callee allocates fresh memory whose size is that argument.
86/// Capacity shapes come from the lang-declared vocabulary: `Param(i)` is
87/// the caller-computed size; `ParamProduct` multiplies internally
88/// (calloc-style, overflow-checked there); `Exact`/`Unknown` have no
89/// caller size expression. Non-allocators (no fresh return) yield `None`.
90fn alloc_capacity_arg(func: &str, summaries: &MemorySummaryRegistry) -> Option<usize> {
91    match summaries.return_capacity(func)? {
92        CapacitySpec::Param(i) => Some(*i),
93        CapacitySpec::ParamProduct(..) | CapacitySpec::Exact(_) | CapacitySpec::Unknown => None,
94    }
95}

97/// Forward closure of `start` through value-preserving copies
98/// (`Assign`/`Cast`), then check whether any copy reaches an allocation
99/// call's capacity argument. Returns the allocator's name when it does.
100fn flows_into_alloc_capacity(
101    ir: &FunctionIR,
102    start: VarId,
103    summaries: &MemorySummaryRegistry,
104) -> Option<String> {
105    let mut set: FxHashSet<VarId> = FxHashSet::default();
106    set.insert(start);
107    loop {
108        let mut grew = false;
109        for block in ir.blocks.values() {
110            for instr in &block.instructions {
111                let flow = match instr {
112                    Instruction::Assign {
113                        dest,
114                        src: Operand::Var(v),
115                    }
116                    | Instruction::Cast {
117                        dest,
118                        src: Operand::Var(v),
119                        ..
120                    } => Some((*dest, *v)),
121                    _ => None,
122                };
123                if let Some((dest, v)) = flow {
124                    grew |= set.contains(&v) && set.insert(dest);
125                }
126            }
127        }
128        if !grew {
129            break;
130        }
131    }
132    for block in ir.blocks.values() {
133        for instr in &block.instructions {
134            if let Instruction::CallStatic { func, args, .. } = instr
135                && let Some(idx) = alloc_capacity_arg(func, summaries)
136                && let Some(Operand::Var(v)) = args.get(idx)
137                && set.contains(v)
138            {
139                return Some(last_segment(func).to_string());
140            }
141        }
142    }
143    None
144}

146fn operand_label(ir: &FunctionIR, op: &Operand) -> String {
147    match op {
148        Operand::Var(v) => ir.var_name(*v),
149        Operand::IntLiteral(k) => k.to_string(),
150        other => format!("{:?}", other),
151    }
152}

154/// Run the allocation-size integer overflow check over one function for
155/// every active rule in the spec-seeded, bundle-extended rule table
156/// (`FactTable::integer_overflow_rules`), deduplicated by id + threshold.
157pub(crate) fn check(
158    ir: &FunctionIR,
159    summaries: &MemorySummaryRegistry,
160    facts: &FactTable,
161) -> Vec<CheckerFinding> {
162    let val_info = value::analyze(ir);
163    let mut out = Vec::new();
164    let mut seen: FxHashSet<(String, u128)> = FxHashSet::default();
165    for (rule, provenance) in &facts.integer_overflow_rules {
166        if !seen.insert((rule.rule_id.clone(), rule.wrap_threshold)) {
167            continue;
168        }
169        check_rule(ir, &val_info, summaries, rule, *provenance, &mut out);
170    }
171    out
172}

174/// The prover for one rule: find wrap-capable size multiplications whose
175/// product flows into an allocation capacity argument.
176fn check_rule(
177    ir: &FunctionIR,
178    val_info: &ValueInfo,
179    summaries: &MemorySummaryRegistry,
180    rule: &crate::analysis::taint::facts::IntegerOverflowRule,
181    provenance: Provenance,
182    out: &mut Vec<CheckerFinding>,
183) {
184    let threshold = i128::try_from(rule.wrap_threshold).unwrap_or(i128::MAX);
185    for (bid, block) in &ir.blocks {
186        for instr in &block.instructions {
187            let Instruction::BinaryOp { dest, op, lhs, rhs } = instr else {
188                continue;
189            };
190            if op != "*" {
191                continue;
192            }
193            let lhs_range = range_of(lhs, *bid, val_info);
194            let rhs_range = range_of(rhs, *bid, val_info);
195            // Default contract: both operand intervals must be PROVABLE
196            // and non-negative; an unprovable operand skips the site
197            // (zero-FP by construction). The corpus-taught unbounded
198            // modality additionally accepts the plain
199            // `malloc(rows * cols)` shape: BOTH operands unboundedly
200            // dynamic (no provable interval on either side and no
201            // constant), modeled at the platform worst case. Mixed shapes
202            // (`strlen(s) * 2`, one proven bounded side) and provably
203            // dynamic operands stay under the default contract.
204            let (al, ah, bl, bh, lhs_unb, rhs_unb) = match (lhs_range, rhs_range) {
205                (Some(l), Some(r)) => (l.0, l.1, r.0, r.1, false, false),
206                (None, None)
207                    if rule.unbounded_product
208                        && operand_is_dynamic(lhs, val_info)
209                        && operand_is_dynamic(rhs, val_info) =>
210                {
211                    (0, DYNAMIC_MAX, 0, DYNAMIC_MAX, true, true)
212                }
213                _ => continue,
214            };
215            // Only non-negative sizes participate (signed/unknown ranges skip).
216            if al < 0 || bl < 0 {
217                continue;
218            }
219            let corners = [
220                i128::from(al) * i128::from(bl),
221                i128::from(al) * i128::from(bh),
222                i128::from(ah) * i128::from(bl),
223                i128::from(ah) * i128::from(bh),
224            ];
225            let max_corner = corners.iter().max().copied().unwrap_or(0);
226            if max_corner <= threshold {
227                continue;
228            }
229            let Some(allocator) = flows_into_alloc_capacity(ir, *dest, summaries) else {
230                continue;
231            };
232            out.push(CheckerFinding {
233                provenance,
234                function: ir.name.clone(),
235                rule: rule.rule_id.clone(),
236                severity: rule.severity.clone(),
237                message: String::new(),
238                params: vec![
239                    ("message", rule.message.clone()),
240                    ("lhs", operand_label(ir, lhs)),
241                    (
242                        "al",
243                        if lhs_unb {
244                            "unbounded".to_string()
245                        } else {
246                            al.to_string()
247                        },
248                    ),
249                    (
250                        "ah",
251                        if lhs_unb {
252                            "unbounded".to_string()
253                        } else {
254                            ah.to_string()
255                        },
256                    ),
257                    ("rhs", operand_label(ir, rhs)),
258                    (
259                        "bl",
260                        if rhs_unb {
261                            "unbounded".to_string()
262                        } else {
263                            bl.to_string()
264                        },
265                    ),
266                    (
267                        "bh",
268                        if rhs_unb {
269                            "unbounded".to_string()
270                        } else {
271                            bh.to_string()
272                        },
273                    ),
274                    ("threshold", rule.wrap_threshold.to_string()),
275                    ("allocator", allocator),
276                ],
277                span: ir.var_span(*dest).or_else(|| match lhs {
278                    Operand::Var(v) => ir.var_span(*v),
279                    _ => None,
280                }),
281            });
282        }
283    }
284}

286#[cfg(test)]
287/// Assertions lifted from the doc examples of items that are now
288/// crate-private; a doctest cannot see them.
289mod doc_examples {
290    #[test]
291    fn doc_example_check() {
292        use crate::analysis::taint::facts::{FactTable, IntegerOverflowRule};
293        use crate::checks::Provenance;
294        use crate::checks::int_overflow::check;
295        use crate::checks::memory_summary::{CapacitySpec, MemorySummary, MemorySummaryRegistry};
296        use crate::harness::lower_source;
297
298        // `malloc` takes its size from argument 0.
299        let mut summaries = MemorySummaryRegistry::default();
300        summaries.summaries.insert(
301            "malloc".into(),
302            MemorySummary {
303                returns_fresh: true,
304                return_capacity: CapacitySpec::Param(0),
305                consumes_params: vec![],
306                out_params_fresh: vec![],
307            },
308        );
309        // A deliberately small threshold keeps the arithmetic obvious.
310        let mut facts = FactTable::default();
311        facts.integer_overflow_rules.push((
312            IntegerOverflowRule {
313                rule_id: "int_overflow".into(),
314                wrap_threshold: 1000,
315                severity: "high".into(),
316                message: String::new(),
317                unbounded_product: false,
318            },
319            Provenance::Spec,
320        ));
321        let rules = |src: &str| -> Vec<String> {
322            let fns = lower_source("t.c", src, "c").unwrap();
323            check(&fns["f"], &summaries, &facts)
324                .into_iter()
325                .map(|f| f.rule)
326                .collect()
327        };
328
329        // 100 * 100 = 10000 exceeds the threshold and feeds an allocation size.
330        assert_eq!(
331            rules("void f(void) { int a = 100; int b = 100; char *p = malloc(a * b); }"),
332            ["int_overflow"]
333        );
334        // 10 * 10 = 100 does not.
335        assert!(
336            rules("void f(void) { int a = 10; int b = 10; char *p = malloc(a * b); }").is_empty()
337        );
338        // Operands that are not provable ranges skip the site.
339        assert!(rules("void f(int a, int b) { char *p = malloc(a * b); }").is_empty());
340    }
