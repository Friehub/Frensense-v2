5//! Cross-file module import & export alias resolution (§8.10).
6//!
7//! Provides language-agnostic module path resolution and alias chaining:
8//! tracks import aliases (`import { a as b }`), local export aliases (`export { a as b }`),
9//! and re-export chains (`export { a as b } from "./mod"`) across multi-file project graphs.

15fn concat2(a: &str, b: &str) -> String {
16    let mut s = String::with_capacity(a.len() + b.len());
17    s.push_str(a);
18    s.push_str(b);
19    s
20}
21
22fn concat3(a: &str, b: &str, c: &str) -> String {
23    let mut s = String::with_capacity(a.len() + b.len() + c.len());
24    s.push_str(a);
25    s.push_str(b);
26    s.push_str(c);
27    s
28}

30/// Resolves a module specifier (e.g. `"./sink_raw"`, `".utils"`) relative to `from_file`
31/// against the known file paths in the program.
32pub(crate) fn resolve_module_path(
33    from_file: &str,
34    specifier: &str,
35    known_files: &[String],
36) -> Option<String> {
37    let from_norm = from_file.replace('\\', "/");
38    let parent = std::path::Path::new(&from_norm)
39        .parent()
40        .unwrap_or(std::path::Path::new(""));
41
42    let spec_clean = specifier.trim_matches(['"', '\'']);
43    let mut parts: Vec<&str> = parent
44        .iter()
45        .filter_map(|s| s.to_str())
46        .filter(|s| !s.is_empty() && *s != ".")
47        .collect();
48
49    if spec_clean.starts_with('.') {
50        // Python-style leading dots or standard relative path
51        if spec_clean.starts_with("./") || spec_clean.starts_with("../") {
52            for segment in spec_clean.split('/') {
53                if segment.is_empty() || segment == "." {
54                    continue;
55                } else if segment == ".." {
56                    parts.pop();
57                } else {
58                    parts.push(segment);
59                }
60            }
61        } else {
62            // Python relative import: e.g. `.utils` or `..views`
63            let dot_count = spec_clean.chars().take_while(|c| *c == '.').count();
64            let rest = &spec_clean[dot_count..];
65            for _ in 1..dot_count {
66                parts.pop();
67            }
68            if !rest.is_empty() {
69                for seg in rest.split('.') {
70                    if !seg.is_empty() {
71                        parts.push(seg);
72                    }
73                }
74            }
75        }
76    } else {
77        // Package / module name: e.g. `sink_module` or `crate::sink`
78        let clean = spec_clean
79            .trim_start_matches("crate::")
80            .trim_start_matches("super::");
81        for seg in clean.split('/') {
82            for sub in seg.split("::") {
83                if !sub.is_empty() {
84                    parts.push(sub);
85                }
86            }
87        }
88    }
89
90    let candidate_base = parts.join("/");
91    let candidate_stem = candidate_base.rsplit('/').next().unwrap_or(&candidate_base);
92
93    // 1. Direct path matches
94    for kf in known_files {
95        let kf_norm = kf.replace('\\', "/");
96        if kf_norm == candidate_base {
97            return Some(kf.clone());
98        }
99        for ext in &[
100            ".ts", ".js", ".tsx", ".jsx", ".mts", ".cts", ".vue", ".py", ".rs", ".go", ".c",
101            ".mjs", ".cjs",
102        ] {
103            if kf_norm == concat2(&candidate_base, ext) {
104                return Some(kf.clone());
105            }
106        }
107        for index in &[
108            "/index.ts",
109            "/index.js",
110            "/index.tsx",
111            "/index.jsx",
112            "/mod.rs",
113            "/__init__.py",
114        ] {
115            if kf_norm == concat2(&candidate_base, index) {
116                return Some(kf.clone());
117            }
118        }
119    }
120
121    // 2. Stem match fallback (single match across known files)
122    let mut stem_matches = Vec::new();
123    for kf in known_files {
124        let kf_norm = kf.replace('\\', "/");
125        let kf_stem = std::path::Path::new(&kf_norm)
126            .file_stem()
127            .and_then(|s| s.to_str())
128            .unwrap_or("");
129        if kf_stem == candidate_stem {
130            stem_matches.push(kf.clone());
131        }
132    }
133    if stem_matches.len() == 1 {
134        return Some(stem_matches.remove(0));
135    }
136
137    None
138}

140/// Maximum recursion depth when resolving re-exports and barrel exports.
141///
142/// Deep barrel chains (e.g. `entry -> barrel1 -> barrel2 -> ... -> target`)
143/// resolve transitively across multiple hops up to this limit. The depth
144/// bound prevents infinite recursion on cyclic re-export topologies.
145pub const MAX_REEXPORT_RESOLVE_DEPTH: usize = 12;
146
147#[derive(Debug, Clone)]
148enum ExportTarget {
149    Local(String),
150    ReExport { from_file: String, symbol: String },
151    Barrel(String),
152}
153
154type FileExportMap = FxHashMap<String, FxHashMap<String, ExportTarget>>;
155type FileBarrelMap = FxHashMap<String, Vec<String>>;
156
157/// The package-import table that applies to `fn_name`: its own file's, or, when
158/// the function's file is untracked and the program has exactly one file's
159/// imports, that single table. `None` otherwise (unknown origin stays inert).
160///
161/// This is the one definition of the single-file fallback; the taint engine,
162/// its guard map and the policy scope all resolve imports through it.
163pub(crate) fn imports_for_function<'m>(
164    fn_file: &FxHashMap<String, String>,
165    file_package_imports: &'m FxHashMap<String, FxHashMap<String, PackageImport>>,
166    fn_name: &str,
167) -> Option<&'m FxHashMap<String, PackageImport>> {
168    fn_file
169        .get(fn_name)
170        .and_then(|file| file_package_imports.get(file))
171        .or_else(|| {
172            if file_package_imports.len() == 1 {
173                file_package_imports.values().next()
174            } else {
175                None
176            }
177        })
178}

180/// An imported package symbol: package name and optional imported symbol.
181#[derive(Debug, Clone, PartialEq, Eq)]
182pub(crate) struct PackageImport {
183    /// Name of the imported package.
184    pub package: String,
185    /// The imported symbol, or `None` for a whole-package import.
186    pub symbol: Option<String>,
187}

189/// Result of whole-program module resolution.
190///
191/// Alias resolution is what taint needs; the other three maps are the
192/// surface it had to parse to get there, kept because every module-level
193/// question (what does this file declare, what does it export, who imports
194/// it) reads them, and re-parsing would make each consumer a second source
195/// of truth about the same source text.
196#[derive(Debug, Clone, Default)]
197pub(crate) struct ModuleResolution {
198    /// Per file, the local names that an import binds to a program function.
199    pub file_aliases: FxHashMap<String, FxHashMap<String, String>>,
200    /// Per file, the local names that an import binds to an external package.
201    pub file_package_imports: FxHashMap<String, FxHashMap<String, PackageImport>>,
202    /// Import declarations per file, aliases included.
203    pub file_imports: FxHashMap<String, Vec<Import>>,
204    /// Export and re-export declarations per file, with visibility and
205    /// declaration kind.
206    pub file_exports: FxHashMap<String, Vec<Export>>,
207    /// Every top-level declaration per file, exported or not.
208    pub file_module_items: FxHashMap<String, Vec<ModuleItem>>,
209}

211/// Computes module aliases and external package imports per file.
212#[derive(Debug)]
213pub(crate) struct ModuleAliasResolver;
214
215impl ModuleAliasResolver {
216    /// Resolves the imports and exports of `files` (`(path, source, extension)`
217    /// rows) against the program's functions `ir_names`, using `fn_file` to
218    /// place each function in its file. Borrows the caller's strings instead
219    /// of copying the corpus, so a per-file frontend never duplicates every
220    /// source file's text.
221    fn build_export_maps(
222        file_exports: &FxHashMap<String, Vec<Export>>,
223        known_files: &[String],
224        fn_file: &FxHashMap<String, String>,
225    ) -> (FileExportMap, FileBarrelMap) {
226        let mut export_map: FileExportMap = FxHashMap::default();
227        let mut barrel_map: FileBarrelMap = FxHashMap::default();
228
229        for (path, exps) in file_exports {
230            let m = export_map.entry(path.clone()).or_default();
231            for exp in exps {
232                if let Some(from_mod) = &exp.from_module {
233                    if let Some(target_file) = resolve_module_path(path, from_mod, known_files) {
234                        if exp.exported_name == "*" {
235                            m.insert("*".to_string(), ExportTarget::Barrel(target_file.clone()));
236                            let list = barrel_map.entry(path.clone()).or_default();
237                            if !list.contains(&target_file) {
238                                list.push(target_file);
239                            }
240                        } else {
241                            m.insert(
242                                exp.exported_name.clone(),
243                                ExportTarget::ReExport {
244                                    from_file: target_file,
245                                    symbol: exp.local_name.clone(),
246                                },
247                            );
248                        }
249                    }
250                } else {
251                    m.insert(
252                        exp.exported_name.clone(),
253                        ExportTarget::Local(exp.local_name.clone()),
254                    );
255                }
256            }
257        }
258
259        for (func_name, file_path) in fn_file {
260            let bare = func_name.rsplit("::").next().unwrap_or(func_name);
261            export_map
262                .entry(file_path.clone())
263                .or_default()
264                .entry(bare.to_string())
265                .or_insert_with(|| ExportTarget::Local(bare.to_string()));
266        }
267
268        (export_map, barrel_map)
269    }

271    pub(crate) fn build_from_refs(
272        files: &[(&str, &str, &str)], // (path, source, ext)
273        ir_names: &FxHashSet<String>,
274        fn_file: &FxHashMap<String, String>,
275    ) -> ModuleResolution {
276        let known_files: Vec<String> = files.iter().map(|(p, _, _)| (*p).to_owned()).collect();
277        let mut file_imports: FxHashMap<String, Vec<Import>> = FxHashMap::default();
278        let mut file_exports: FxHashMap<String, Vec<Export>> = FxHashMap::default();
279        let mut file_module_items: FxHashMap<String, Vec<ModuleItem>> = FxHashMap::default();
280
281        for &(path, source, ext) in files {
282            let Some(spec) = frensense_lang::spec_for_ext(ext) else {
283                continue;
284            };
285            let mut parser = tree_sitter::Parser::new();
286            if parser.set_language(&spec.tree_sitter_language()).is_err() {
287                continue;
288            }
289            let Some(tree) = parser.parse(source, None) else {
290                continue;
291            };
292
293            let imps = spec.extract_imports(tree.root_node(), source);
294            let exps = spec.extract_exports(tree.root_node(), source);
295            let items = spec.extract_module_items(tree.root_node(), source);
296            file_imports.insert(path.to_owned(), imps);
297            file_exports.insert(path.to_owned(), exps);
298            file_module_items.insert(path.to_owned(), items);
299        }
300
301        let (export_map, barrel_map) =
302            Self::build_export_maps(&file_exports, &known_files, fn_file);
303
304        // Resolve imports for each file
305        let mut file_aliases: FxHashMap<String, FxHashMap<String, String>> = FxHashMap::default();
306        let mut file_package_imports: FxHashMap<String, FxHashMap<String, PackageImport>> =
307            FxHashMap::default();
308
309        for (importer_path, imps) in &file_imports {
310            for imp in imps {
311                let local_name = &imp.local_name;
312                let package = &imp.package;
313                let symbol = imp.symbol.as_deref().unwrap_or(local_name.as_str());
314
315                let Some(target_file) = resolve_module_path(importer_path, package, &known_files)
316                else {
317                    let normalized_pkg = package
318                        .strip_prefix("node:")
319                        .unwrap_or(package.as_str())
320                        .to_string();
321                    file_package_imports
322                        .entry(importer_path.clone())
323                        .or_default()
324                        .insert(
325                            local_name.clone(),
326                            PackageImport {
327                                package: normalized_pkg,
328                                symbol: imp.symbol.clone(),
329                            },
330                        );
331                    continue;
332                };
333
334                if let Some(canonical) = Self::resolve_symbol(
335                    &target_file,
336                    symbol,
337                    &export_map,
338                    &barrel_map,
339                    ir_names,
340                    fn_file,
341                    0,
342                ) {
343                    file_aliases
344                        .entry(importer_path.clone())
345                        .or_default()
346                        .insert(local_name.clone(), canonical);
347                }
348            }
349        }
350
351        ModuleResolution {
352            file_aliases,
353            file_package_imports,
354            file_imports,
355            file_exports,
356            file_module_items,
357        }
358    }
359
360    /// Recursively resolve an imported `symbol` from `file` across re-exports and barrel files.
361    ///
362    /// Resolution order:
363    /// 1. Directly declared local export in `file` (`export function foo()`)
364    /// 2. Explicit named re-export (`export { foo } from "./other"`)
365    /// 3. Wildcard barrel exports (`export * from "./submodule"`), searching all barrels declared in `file`
366    /// 4. Direct match in `ir_names` qualified or bare for the file
367    ///
368    /// The search recurses across multi-hop re-export chains up to [`MAX_REEXPORT_RESOLVE_DEPTH`].
369    fn resolve_symbol(
370        file: &str,
371        symbol: &str,
372        export_map: &FileExportMap,
373        barrel_map: &FileBarrelMap,
374        ir_names: &FxHashSet<String>,
375        fn_file: &FxHashMap<String, String>,
376        depth: usize,
377    ) -> Option<String> {
378        if depth > MAX_REEXPORT_RESOLVE_DEPTH {
379            return None;
380        }
381
382        if let Some(target) = export_map.get(file).and_then(|m| m.get(symbol)) {
383            match target {
384                ExportTarget::Local(local_sym) => {
385                    let qual = concat3(file, "::", local_sym);
386                    if ir_names.contains(&qual) {
387                        return Some(qual);
388                    }
389                    if ir_names.contains(local_sym) {
390                        return Some(local_sym.clone());
391                    }
392                    return Some(local_sym.clone());
393                }
394                ExportTarget::ReExport {
395                    from_file,
396                    symbol: target_sym,
397                } => {
398                    return Self::resolve_symbol(
399                        from_file,
400                        target_sym,
401                        export_map,
402                        barrel_map,
403                        ir_names,
404                        fn_file,
405                        depth + 1,
406                    );
407                }
408                ExportTarget::Barrel(from_file) => {
409                    return Self::resolve_symbol(
410                        from_file,
411                        symbol,
412                        export_map,
413                        barrel_map,
414                        ir_names,
415                        fn_file,
416                        depth + 1,
417                    );
418                }
419            }
420        }
421
422        // Check all wildcard barrel exports for this file (multi-barrel support)
423        if let Some(barrels) = barrel_map.get(file) {
424            for barrel_file in barrels {
425                if let Some(resolved) = Self::resolve_symbol(
426                    barrel_file,
427                    symbol,
428                    export_map,
429                    barrel_map,
430                    ir_names,
431                    fn_file,
432                    depth + 1,
433                ) {
434                    return Some(resolved);
435                }
436            }
437        }
438
439        // Check if directly in ir_names
440        let qual = concat3(file, "::", symbol);
441        if ir_names.contains(&qual) {
442            return Some(qual);
443        }
444        if ir_names.contains(symbol) && fn_file.get(symbol).map(|f| f == file).unwrap_or(false) {
445            return Some(symbol.to_string());
446        }
447
448        None
449    }
450}

464    /// The surface the resolver parses must survive into the prepared
465    /// program: alias resolution is only one of its four readers.
466    #[test]
467    fn prepared_program_keeps_the_module_surface() {
468        let program = files(&[
469            (
470                "src/server.ts",
471                "export function handler() {}\nexport const LIMIT = 1;\nfunction local() {}\n",
472                "ts",
473            ),
474            (
475                "src/other.ts",
476                "import { handler } from \"./server\";\nhandler();\n",
477                "ts",
478            ),
479        ]);
480        let prepared = scan::prepare(&program).expect("prepare");
481
482        let exports = prepared.exports_of("src/server.ts");
483        assert_eq!(exports.len(), 2, "{exports:?}");
484        assert!(exports.iter().any(|e| e.exported_name == "handler"));
485        assert!(exports.iter().any(|e| e.exported_name == "LIMIT"));
486
487        let items = prepared.module_items_of("src/server.ts");
488        let local = items.iter().find(|i| i.name == "local").expect("local");
489        assert_eq!(local.visibility, Visibility::Private);
490
491        let imports = prepared.imports_of("src/other.ts");
492        assert!(imports.iter().any(|i| i.local_name == "handler"));
493
494        // Alias resolution still works off the same parse.
495        assert!(
496            prepared
497                .file_aliases
498                .get("src/other.ts")
499                .and_then(|m| m.get("handler"))
500                .is_some(),
501            "{:?}",
502            prepared.file_aliases
503        );
504
505        assert_eq!(prepared.files(), vec!["src/other.ts", "src/server.ts"]);
506    }

572            (
573                // Importer at 4th hop
574                "src/app.ts",
575                "import { coreHandler, otherHandler } from \"./barrel3\";\ncoreHandler();\notherHandler();\n",
576                "ts",
577            ),
578        ]);
579        let prepared = scan::prepare(&program).expect("prepare");
580
581        let aliases = prepared
582            .file_aliases
583            .get("src/app.ts")

585        assert!(
586            aliases.get("coreHandler").is_some(),

588        );
589        assert!(
590            aliases.get("otherHandler").is_some(),

600    #[test]
601    fn doc_example_resolve_module_path() {
602        use crate::graph::module::resolve_module_path;
603
604        let known = vec![
605            "src/a.ts".to_string(),
606            "src/lib/util.ts".to_string(),
607            "src/lib/index.ts".to_string(),
608            "pkg/mod.py".to_string(),
609        ];
610        // Relative specifier, extension filled in.
611        assert_eq!(
612            resolve_module_path("src/a.ts", "./lib/util", &known).as_deref(),
613            Some("src/lib/util.ts")
614        );
615        // `..` climbs one directory.
616        assert_eq!(
617            resolve_module_path("src/lib/util.ts", "../a", &known).as_deref(),
618            Some("src/a.ts")
619        );
620        // A directory resolves to its index file.
621        assert_eq!(
622            resolve_module_path("src/a.ts", "./lib", &known).as_deref(),
623            Some("src/lib/index.ts")
624        );
625        // Python-style leading dot.
626        assert_eq!(
627            resolve_module_path("pkg/x.py", ".mod", &known).as_deref(),
628            Some("pkg/mod.py")
629        );
630        // Unknown files and bare package names do not resolve.
631        assert_eq!(resolve_module_path("src/a.ts", "./missing", &known), None);
632        assert_eq!(resolve_module_path("src/a.ts", "react", &known), None);
633    }
