5//! Call-graph construction and dispatch resolution (§8.1)
6//!
7//! `ProgramSvfg::discover_bindings` resolves callees by
8//! **exact string match** against a flat function-name index. That breaks on
9//! the three constructs real code is made of:
10//!
11//! 1. **Aliased callees**, `const h = helper; h(x)`, re-exports. The call
12//!    site says `h`, the function is named `helper`.
13//! 2. **Method calls**, `db.query(...)` lowers to `CallVirtual { method:
14//!    "query", receiver }`. The callee is `query` *on the receiver's class*,
15//!    `query` alone matches every class in the program.
16//! 3. **Higher-order calls**, `CallPointer { func_ptr: Var(v) }` where `v`
17//!    holds a function reference or a callback parameter
18//!    (`router.get(path, handler)` lowers the handler reference to a Var).
19//!
20//! This module resolves all three into `CallTarget`s per call site, using
21//! only information in the lowered IRs:
22//!
23//! * **Alias chains**, SSA `Assign` chains over function-reference Vars.
24//! * **Class method tables**, `Allocate { kind: ClassInstance(class) }` ties
25//!   a receiver Var to its class; `this`-receivers resolve through the
26//!   enclosing method's qualified name (`Class.method`).
27//! * **Callback (higher-order) edges**, two-step Andersen-style treatment at
28//!   function granularity: (a) find parameters that are *called* anywhere,
29//!   (b) for each call site passing a function reference into such a
30//!   parameter slot, add an edge caller → referenced function.
31//!
32//! Unresolved sites stay unresolved: the interprocedural layer keeps its
33//! sound conservative pass-through edges for them. Resolution only ever adds
34//! precision, it never removes the sound fallback.
35//!
36//! ## Determinism
37//!
38//! All iterations run over sorted keys; candidate lists are sorted+deduped.
39//! The graph is byte-stable across runs (the corpus store hashes it).
40

50/// A resolved callee. One call site may resolve to several targets (dynamic
51/// dispatch, callbacks); the interprocedural layer installs cross edges for
52/// every target, sound (any target may run) and precise enough for taint.
53#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
54pub(crate) enum CallTarget {
55    /// A function in the analysed program, by its program-level key.
56    Function(String),
57    /// `Class.method` dispatch on a receiver of known class.
58    Method {
59        /// Class of the receiver.
60        class: String,
61        /// Method name.
62        method: String,
63    },
64}
65
66impl CallTarget {
67    /// The program-function key this target resolves to, if internal.
68    /// Method targets resolve under the `Class.method` key convention (see
69    /// [`method_key`]).
70    pub(crate) fn internal_key(&self) -> Option<String> {
71        match self {
72            CallTarget::Function(n) => Some(n.clone()),
73            CallTarget::Method { class, method } => Some(method_key(class, method)),
74        }
75    }
76}
77
78/// Program-index key for a method implementation. The project lowering
79/// registers methods qualified as `Class.method`; bare-named duplicates are
80/// matched by the caller (interprocedural layer) as a fallback.
81pub(crate) fn method_key(class: &str, method: &str) -> String {
82    format!("{class}.{method}")
83}

89/// Resolved call graph: per-call-site targets plus program-level edges.
90#[derive(Debug, Default)]
91pub(crate) struct CallGraph {
92    /// Program functions, sorted (the IR map's keys).
93    pub functions: Vec<String>,
94    /// `(caller_fn, (block, instr_idx)) → resolved targets` for every call
95    /// instruction encountered. Sites whose *only* target is external are
96    /// recorded in [`CallGraph::unresolved`] instead.
97    pub sites: FxHashMap<(String, (usize, usize)), Vec<CallTarget>>,
98    /// Internal edges: caller → sorted callee keys.
99    pub edges: FxHashMap<String, Vec<String>>,
100    /// Reverse internal edges: callee → sorted callers.
101    pub reverse_edges: FxHashMap<String, Vec<String>>,
102    /// Sites with no internal target, mapped to their best-known external
103    /// name (for the fact layer and diagnostics).
104    pub unresolved: FxHashMap<(String, (usize, usize)), String>,
105}
106
107impl CallGraph {
108    /// Targets for one call instruction inside `fn`.
109    pub(crate) fn targets_at(
110        &self,
111        fn_name: &str,
112        block: usize,
113        instr_idx: usize,
114    ) -> &[CallTarget] {
115        self.sites
116            .get(&(fn_name.to_string(), (block, instr_idx)))
117            .map(|v| v.as_slice())
118            .unwrap_or(&[])
119    }
120}

126/// Per-function value facts gathered in pass 1.
127#[derive(Default)]
128struct FnFacts {
129    /// `dest → src` copy assignments (alias chains).
130    aliases: FxHashMap<VarId, VarId>,
131    /// Vars known to hold a reference to a program function.
132    fn_refs: FxHashMap<VarId, String>,
133    /// Vars known to be instances of a class (`new Class()`).
134    classes: FxHashMap<VarId, String>,
135}
136
137/// Input: every function's IR, keyed by program-level name (the same map the
138/// `ProgramSvfg` builder consumes).
139#[derive(Debug)]
140pub(crate) struct CallGraphBuilder<'a> {
141    irs: &'a FxHashMap<String, &'a FunctionIR>,
142    aliases: Option<&'a FxHashMap<String, FxHashMap<String, String>>>,
143    fn_file: Option<&'a FxHashMap<String, String>>,
144}
145
146impl<'a> CallGraphBuilder<'a> {
147    /// Creates a builder over `irs`, keyed by program-level function name, with no import aliases.
148    pub(crate) fn new(irs: &'a FxHashMap<String, &'a FunctionIR>) -> Self {
149        Self {
150            irs,
151            aliases: None,
152            fn_file: None,
153        }
154    }
155
156    /// Supplies per-file import aliases and the function-to-file map, so a call through an imported symbol resolves to the function it names.
157    pub(crate) fn with_aliases(
158        mut self,
159        aliases: Option<&'a FxHashMap<String, FxHashMap<String, String>>>,
160        fn_file: Option<&'a FxHashMap<String, String>>,
161    ) -> Self {
162        self.aliases = aliases;
163        self.fn_file = fn_file;
164        self
165    }
166
167    fn resolve_call_alias<'b>(&'b self, fname: &str, symbol: &'b str) -> &'b str {
168        if let Some(fn_file) = self.fn_file
169            && let Some(aliases) = self.aliases
170            && let Some(file) = fn_file.get(fname)
171            && let Some(file_map) = aliases.get(file)
172            && let Some(target) = file_map.get(symbol)
173        {
174            return target.as_str();
175        }
176        symbol
177    }
178
179    /// Resolves every call site in every function and returns the call graph.
180    pub(crate) fn build(&self) -> CallGraph {
181        let mut functions: Vec<String> = self.irs.keys().cloned().collect();
182        functions.sort();
183        let fn_set: FxHashSet<&str> = functions.iter().map(|s| s.as_str()).collect();
184
185        // ---- Pass 1: per-function value facts ------------------------------
186        let mut facts: FxHashMap<&str, FnFacts> = FxHashMap::default();
187        for (fname, ir) in self.irs {
188            let mut f = FnFacts::default();
189            for (&v, meta) in &ir.var_metadata {
190                if let Some(name) = &meta.source_name
191                    && fn_set.contains(name.as_str())
192                {
193                    f.fn_refs.insert(v, name.clone());
194                }
195            }
196            for block in ir.blocks.values() {
197                for instr in &block.instructions {
198                    match instr {
199                        Instruction::Assign { dest, src, .. } => match src {
200                            Operand::Var(s) => {
201                                f.aliases.insert(*dest, *s);
202                            }
203                            Operand::StringLiteral(name) if fn_set.contains(name.as_str()) => {
204                                f.fn_refs.insert(*dest, name.clone());
205                            }
206                            _ => {}
207                        },
208                        Instruction::AddressOf { dest, src, .. } => {
209                            // `&func`, the address-of a parameter can't be a
210                            // function; a real fn-ref lowering path lands here
211                            // when IR gains named function constants. Track it
212                            // as an alias for now (conservative no-op unless
213                            // src is later known to be a fn ref).
214                            f.aliases.insert(*dest, *src);
215                        }
216                        Instruction::Allocate {
217                            dest,
218                            kind: AllocationKind::ClassInstance(class),
219                            ..
220                        } => {
221                            f.classes.insert(*dest, class.clone());
222                        }
223                        _ => {}
224                    }
225                }
226            }
227            facts.insert(fname.as_str(), f);
228        }
229
230        // ---- Pass 2: which formal parameters are CALLED? -------------------
231        // `(fn_name, param_slot)`, the two-step higher-order treatment.
232        let mut called_params: FxHashSet<(String, usize)> = FxHashSet::default();
233        for (fname, ir) in self.irs {
234            for block in ir.blocks.values() {
235                for instr in &block.instructions {
236                    let func_ptr: Option<&Operand> = match instr {
237                        Instruction::CallPointer { func_ptr, .. } => Some(func_ptr),
238                        _ => None,
239                    };
240                    if let Some(Operand::Var(v)) = func_ptr {
241                        for (slot, &pv) in ir.parameters.iter().enumerate() {
242                            if pv == *v {
243                                called_params.insert(((*fname).clone(), slot));
244                            }
245                        }
246                    }
247                }
248            }
249        }
250
251        // ---- Pass 3: resolve every call site -------------------------------
252        let mut graph = CallGraph {
253            functions,
254            ..CallGraph::default()
255        };
256
257        for (fname, ir) in self.irs {
258            let f = &facts[fname.as_str()];
259
260            for block in ir.blocks.values() {
261                for (idx, instr) in block.instructions.iter().enumerate() {
262                    let site_key = ((*fname).clone(), (block.id.0, idx));
263                    let mut targets: Vec<CallTarget> = Vec::new();
264                    let mut external_name: Option<String> = None;
265
266                    match instr {
267                        Instruction::CallStatic { func, .. } => {
268                            let resolved_func = self.resolve_call_alias(fname, func);
269                            if self.irs.contains_key(resolved_func) {
270                                targets.push(CallTarget::Function(resolved_func.to_string()));
271                            } else {
272                                external_name = Some(resolved_func.to_string());
273                            }
274                        }
275
276                        Instruction::CallVirtual {
277                            method, receiver, ..
278                        } => {
279                            let resolved_method = self.resolve_call_alias(fname, method);
280                            // Receiver → class:
281                            let class: Option<String> = match receiver {
282                                Operand::Var(r) => f.classes.get(r).cloned().or_else(|| {
283                                    // Follow alias chains to an allocation.
284                                    self.follow_alias_to_class(*r, f)
285                                }),
286                                _ => None,
287                            };
288                            // `this` receiver: the enclosing function's own
289                            // name may be a qualified `Class.method`.
290                            let class = class.or_else(|| this_class_of(fname));
291
292                            match class {
293                                Some(c) => {
294                                    let key = method_key(&c, resolved_method);
295                                    if self.irs.contains_key(&key) {
296                                        targets.push(CallTarget::Method {
297                                            class: c,
298                                            method: resolved_method.to_string(),
299                                        });
300                                    } else if self.irs.contains_key(resolved_method) {
301                                        // Unqualified registration fallback. The
302                                        // push makes the site internal, so no
303                                        // external name can be observed here.
304                                        targets.push(CallTarget::Function(
305                                            resolved_method.to_string(),
306                                        ));
307                                    } else {
308                                        external_name = Some(resolved_method.to_string());
309                                    }
310                                }
311                                None => {
312                                    // Unknown receiver (library object or
313                                    // unresolved import). Record the site as
314                                    // external ONLY: binding it to a program
315                                    // function that merely shares the bare
316                                    // method name (`client.get` -> a local
317                                    // `get` helper) fabricates interprocedural
318                                    // edges, pulling unrelated same-named
319                                    // functions into the taint walk. Bare-name
320                                    // binding here was the dominant
321                                    // false-positive source (tests/mre audit).
322                                    external_name = Some(resolved_method.to_string());
323                                }
324                            }
325                        }
326
327                        Instruction::CallPointer { func_ptr, .. } => {
328                            match func_ptr {
329                                Operand::Var(v) => {
330                                    match self.follow_alias_to_fn(*v, f) {
331                                        Some(fun) => {
332                                            let resolved_fun = self.resolve_call_alias(fname, &fun);
333                                            targets.push(CallTarget::Function(
334                                                resolved_fun.to_string(),
335                                            ));
336                                        }
337                                        None => {
338                                            // Formal parameter called as a
339                                            // function pointer: resolved by
340                                            // the callback pass (step 4) when
341                                            // callers pass fn refs into this
342                                            // slot. Nothing to record here.
343                                            let is_called_param =
344                                                ir.parameters.iter().enumerate().any(
345                                                    |(slot, &pv)| {
346                                                        pv == *v
347                                                            && called_params
348                                                                .contains(&((*fname).clone(), slot))
349                                                    },
350                                                );
351                                            if !is_called_param {
352                                                external_name = Some("<indirect>".to_string());
353                                            }
354                                        }
355                                    }
356                                }
357                                Operand::StringLiteral(name) => {
358                                    let resolved_name = self.resolve_call_alias(fname, name);
359                                    if self.irs.contains_key(resolved_name) {
360                                        targets
361                                            .push(CallTarget::Function(resolved_name.to_string()));
362                                    } else {
363                                        external_name = Some(resolved_name.to_string());
364                                    }
365                                }
366                                _ => {
367                                    external_name = Some("<indirect>".to_string());
368                                }
369                            }
370                        }
371
372                        _ => continue,
373                    }
374
375                    // ---- Step 4 (callback edges): every call that passes a
376                    // function reference into a CALLED parameter slot of an
377                    // internal callee gains an edge to the referenced fn.
378                    // Handled below in `install_callback_edges`.
379
380                    targets.sort();
381                    targets.dedup();
382
383                    let has_internal = !targets.is_empty();
384                    match (has_internal, external_name) {
385                        (true, _) => {
386                            graph.sites.insert(site_key, targets);
387                        }
388                        (false, Some(name)) => {
389                            graph.unresolved.insert(site_key, name);
390                        }
391                        (false, None) => {
392                            graph.sites.insert(site_key, targets);
393                        }
394                    }
395                }
396            }
397        }
398
399        // ---- Step 4: callback (higher-order) edges --------------------------
400        // For each internal call site passing a fn-ref Var at argument slot
401        // `s`, where slot `s` of the callee is a called parameter, add an
402        // edge to the referenced function, as an extra target on the site.
403        let mut extra_targets: FxHashMap<(String, (usize, usize)), Vec<CallTarget>> =
404            FxHashMap::default();
405
406        for (fname, ir) in self.irs {
407            let f = &facts[fname.as_str()];
408            for block in ir.blocks.values() {
409                for (idx, instr) in block.instructions.iter().enumerate() {
410                    let args: &[Operand] = match instr {
411                        Instruction::CallStatic { args, .. }
412                        | Instruction::CallVirtual { args, .. }
413                        | Instruction::CallPointer { args, .. } => args,
414                        _ => continue,
415                    };
416
417                    // Argument slots: no receiver here (receiver occupies
418                    // slot 0 in the interprocedural layer, but the callee's
419                    // *formal* slots line up with the explicit args only for
420                    // static calls; virtual calls shift by one, handled by
421                    // using explicit-arg position + 1 for virtual).
422                    let is_virtual = matches!(instr, Instruction::CallVirtual { .. });
423                    let slot_offset = if is_virtual { 1 } else { 0 };
424
425                    for (pos, a) in args.iter().enumerate() {
426                        // Args carry fn refs either directly (a string literal
427                        // naming a program function) or via a Var whose alias
428                        // chain resolves to one. Both shapes occur in lowered
429                        // code; treat them identically.
430                        let target_fn: String = match a {
431                            Operand::Var(av) => match self.follow_alias_to_fn(*av, f) {
432                                Some(t) => t,
433                                None => continue,
434                            },
435                            Operand::StringLiteral(name) => {
436                                if !self.irs.contains_key(name) {
437                                    continue;
438                                }
439                                name.clone()
440                            }
441                            _ => continue,
442                        };
443                        // Which internal function will receive it?
444                        let callee_key: Option<String> = match instr {
445                            Instruction::CallStatic { func, .. } => {
446                                if self.irs.contains_key(func) {
447                                    Some(func.clone())
448                                } else {
449                                    None
450                                }
451                            }
452                            Instruction::CallVirtual {
453                                method, receiver, ..
454                            } => {
455                                let class = match receiver {
456                                    Operand::Var(r) => f.classes.get(r).cloned(),
457                                    _ => None,
458                                }
459                                .or_else(|| this_class_of(fname));
460                                match class {
461                                    Some(c) => {
462                                        let k = method_key(&c, method);
463                                        if self.irs.contains_key(&k) {
464                                            Some(k)
465                                        } else if self.irs.contains_key(method) {
466                                            Some(method.clone())
467                                        } else {
468                                            None
469                                        }
470                                    }
471                                    None => {
472                                        // Unknown receiver: never bind the
473                                        // callee by bare method name (same FP
474                                        // class as the pass-3 unknown-receiver
475                                        // arm). The callback edge still lands
476                                        // via the fn-ref argument when the
477                                        // receiving slot is known.
478                                        None
479                                    }
480                                }
481                            }
482                            _ => None, // CallPointer target resolution above
483                        };
484                        let is_higher_order_method = match instr {
485                            Instruction::CallVirtual { method, .. } => {
486                                matches!(
487                                    method.as_str(),
488                                    "map"
489                                        | "and_then"
490                                        | "then"
491                                        | "flatMap"
492                                        | "flat_map"
493                                        | "forEach"
494                                        | "for_each"
495                                        | "filter"
496                                )
497                            }
498                            _ => false,
499                        };
500
501                        if is_higher_order_method {
502                            extra_targets
503                                .entry(((*fname).clone(), (block.id.0, idx)))
504                                .or_default()
505                                .push(CallTarget::Function(target_fn.clone()));
506                        } else if let Some(callee_key) = callee_key {
507                            let formal_slot = pos + slot_offset;
508                            if called_params.contains(&(callee_key, formal_slot)) {
509                                extra_targets
510                                    .entry(((*fname).clone(), (block.id.0, idx)))
511                                    .or_default()
512                                    .push(CallTarget::Function(target_fn));
513                            }
514                        }
515                    }
516                }
517            }
518        }
519
520        for (site, mut tgts) in extra_targets {
521            tgts.sort();
522            tgts.dedup();
523            graph.unresolved.remove(&site);
524            let entry = graph.sites.entry(site).or_default();
525            for t in tgts {
526                if !entry.contains(&t) {
527                    entry.push(t);
528                }
529            }
530            entry.sort();
531            entry.dedup();
532        }
533
534        // ---- Finalize edges --------------------------------------------------
535        let mut edges: FxHashMap<String, BTreeSet<String>> = Default::default();
536        let mut redges: FxHashMap<String, BTreeSet<String>> = Default::default();
537        let fn_names: FxHashSet<&str> = graph.functions.iter().map(|s| s.as_str()).collect();
538        for ((caller, _site), targets) in &graph.sites {
539            for t in targets {
540                if let Some(key) = t.internal_key()
541                    && fn_names.contains(key.as_str())
542                {
543                    edges.entry(caller.clone()).or_default().insert(key.clone());
544                    redges.entry(key).or_default().insert(caller.clone());
545                }
546            }
547        }
548        graph.edges = edges
549            .into_iter()
550            .map(|(k, v)| (k, v.into_iter().collect()))
551            .collect();
552        graph.reverse_edges = redges
553            .into_iter()
554            .map(|(k, v)| (k, v.into_iter().collect()))
555            .collect();
556
557        graph
558    }

560    /// Follow `v`'s alias chain to a function reference, if any.
561    fn follow_alias_to_fn(&self, v: VarId, f: &FnFacts) -> Option<String> {
562        let mut cur = v;
563        let mut hops = 0;
564        loop {
565            if let Some(fun) = f.fn_refs.get(&cur) {
566                return Some(fun.clone());
567            }
568            let next = f.aliases.get(&cur)?;
569            if *next == cur {
570                return None; // cycle guard
571            }
572            if hops > 32 {
573                crate::telemetry::Cap::CallgraphHops.record();
574                return None;
575            }
576            cur = *next;
577            hops += 1;
578        }
579    }
580
581    /// Follow `v`'s alias chain to a class allocation, if any.
582    fn follow_alias_to_class(&self, v: VarId, f: &FnFacts) -> Option<String> {
583        let mut cur = v;
584        let mut hops = 0;
585        loop {
586            if let Some(c) = f.classes.get(&cur) {
587                return Some(c.clone());
588            }
589            let next = f.aliases.get(&cur)?;
590            if *next == cur {
591                return None;
592            }
593            if hops > 32 {
594                crate::telemetry::Cap::CallgraphHops.record();
595                return None;
596            }
597            cur = *next;
598            hops += 1;
599        }
600    }
601}

603/// If `fname` is a qualified method name (`Class.method`), return `Class`.
604fn this_class_of(fname: &str) -> Option<String> {
605    fname.rsplit_once('.').map(|(c, _)| c.to_string())
606}
