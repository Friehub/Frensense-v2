5//! R4 phase 1: Steensgaard-style unification points-to (near-linear).
6//!
7//! Steensgaard's algorithm treats every assignment as an *equality*
8//! constraint and merges the pointees' equivalence classes with union-find.
9//! It is an over-approximation of Andersen's inclusion-based result (some
10//! sets that are separate will be merged) but runs in almost-linear time,
11//! the standard industrial first phase for scaling points-to to large
12//! programs.
13//!
14//! The output is a partition of variables into **pointer equivalence
15//! classes**. Consumers (two-phase pipeline) use it to decide where the
16//! expensive Andersen phase must run at full precision: only classes that
17//! can contain taint-relevant variables. Classes that cannot matter keep
18//! the cheap unified answer.
19
20use rustc_hash::FxHashMap;
21
22use crate::ir::function::{FunctionIR, Instruction, Operand, VarId};
23
24/// A pointer-equivalence class produced by unification.
25#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
26pub(crate) struct ClassId(pub usize);
27
28/// Union-find over variable ids with path compression + union by size.
29#[derive(Debug, Default)]
30struct UnionFind {
31    parent: FxHashMap<usize, usize>,
32    size: FxHashMap<usize, usize>,
33}
34
35impl UnionFind {
36    fn find(&mut self, v: usize) -> usize {
37        // Make sure the node exists.
38        self.parent.entry(v).or_insert(v);
39        self.size.entry(v).or_insert(1);
40        // Path compression (iterative).
41        let mut root = v;
42        while self.parent[&root] != root {
43            root = self.parent[&root];
44        }
45        let mut cur = v;
46        while self.parent[&cur] != cur {
47            let next = self.parent[&cur];
48            self.parent.insert(cur, root);
49            cur = next;
50        }
51        root
52    }
53
54    fn union(&mut self, a: usize, b: usize) {
55        let ra = self.find(a);
56        let rb = self.find(b);
57        if ra == rb {
58            return;
59        }
60        let (small, large) = if self.size[&ra] < self.size[&rb] {
61            (ra, rb)
62        } else {
63            (rb, ra)
64        };
65        self.parent.insert(small, large);
66        *self.size.get_mut(&large).unwrap() += self.size[&small];
67    }
68}

70/// The result of Steensgaard unification over one function: variable →
71/// equivalence class, plus the class membership lists.
72#[derive(Debug, Default)]
73pub(crate) struct Steensgaard {
74    /// Class of each participating variable.
75    pub class_of: FxHashMap<VarId, ClassId>,
76    /// Members per class (for phase-2 restricted Andersen).
77    pub members: FxHashMap<ClassId, Vec<VarId>>,
78}
79
80impl Steensgaard {
81    /// Unify over one function's assignment-like constraints.
82    ///
83    /// Constraints collected (all equalities, the Steensgaard premise):
84    /// - `Assign { dest, src: Var }`, `Cast { dest, src: Var }`
85    /// - Phi incoming edges (`dest = incoming`)
86    /// - `LoadField { dest, base, field }` unifies dest with the
87    ///   representative of `base.field` via a synthetic field-var class so
88    ///   field structure survives: two loads of the same base.field land in
89    ///   the same class, loads of different fields stay separate at this
90    ///   stage (phase 2 refines).
91    pub(crate) fn analyze(ir: &FunctionIR) -> Self {
92        let mut uf = UnionFind::default();
93        let mut field_var: FxHashMap<(VarId, String), VarId> = FxHashMap::default();
94        let mut next_synthetic = ir
95            .var_metadata
96            .keys()
97            .copied()
98            .map(|v| v.0)
99            .max()
100            .unwrap_or(0)
101            + 1000;
102
103        // Seed: every parameter and allocated var is its own class root.
104        for &p in &ir.parameters {
105            uf.find(p.0);
106        }
107        for block in ir.blocks.values() {
108            for instr in &block.instructions {
109                if let Instruction::Allocate { dest, .. } = instr {
110                    uf.find(dest.0);
111                }
112            }
113        }
114
115        // Collect equality constraints.
116        for block in ir.blocks.values() {
117            for phi in &block.phis {
118                for &(_, inc) in &phi.incoming {
119                    uf.union(phi.dest.0, inc.0);
120                }
121            }
122            for instr in &block.instructions {
123                match instr {
124                    Instruction::Assign {
125                        dest,
126                        src: Operand::Var(src),
127                    }
128                    | Instruction::Cast {
129                        dest,
130                        src: Operand::Var(src),
131                        ..
132                    } => uf.union(dest.0, src.0),
133                    Instruction::StoreField {
134                        base,
135                        field,
136                        src: Operand::Var(sv),
137                        ..
138                    } => {
139                        let rep =
140                            field_rep(&mut uf, &mut field_var, &mut next_synthetic, *base, field);
141                        uf.union(rep.0, sv.0);
142                    }
143                    Instruction::LoadField {
144                        dest, base, field, ..
145                    } => {
146                        let rep =
147                            field_rep(&mut uf, &mut field_var, &mut next_synthetic, *base, field);
148                        uf.union(dest.0, rep.0);
149                    }
150                    Instruction::StoreElement {
151                        base, index, src, ..
152                    } => {
153                        let f = match index {
154                            Operand::StringLiteral(s) => s.clone(),
155                            _ => "*".to_string(),
156                        };
157                        if let Operand::Var(sv) = src {
158                            let rep =
159                                field_rep(&mut uf, &mut field_var, &mut next_synthetic, *base, &f);
160                            uf.union(rep.0, sv.0);
161                        }
162                    }
163                    Instruction::LoadElement {
164                        dest, base, index, ..
165                    } => {
166                        let f = match index {
167                            Operand::StringLiteral(s) => s.clone(),
168                            _ => "*".to_string(),
169                        };
170                        let rep =
171                            field_rep(&mut uf, &mut field_var, &mut next_synthetic, *base, &f);
172                        uf.union(dest.0, rep.0);
173                    }
174                    _ => {}
175                }
176            }
177        }
178
179        // Materialize classes.
180        let mut class_of: FxHashMap<VarId, ClassId> = FxHashMap::default();
181        let mut members: FxHashMap<ClassId, Vec<VarId>> = FxHashMap::default();
182        let mut root_to_class: FxHashMap<usize, ClassId> = FxHashMap::default();
183        let mut next_class = 0usize;
184
185        let mut roots: Vec<usize> = uf
186            .parent
187            .keys()
188            .copied()
189            .filter(|v| !is_synthetic(*v, next_synthetic))
190            .collect();
191        roots.sort_unstable();
192        for v in roots {
193            let root = uf.find(v);
194            let class = *root_to_class.entry(root).or_insert_with(|| {
195                let c = ClassId(next_class);
196                next_class += 1;
197                c
198            });
199            let vid = VarId(v);
200            class_of.insert(vid, class);
201            members.entry(class).or_default().push(vid);
202        }
203
204        Steensgaard { class_of, members }
205    }
206
207    /// Equivalence class of `v`, if it participates in any constraint.
208    pub(crate) fn class_of(&self, v: VarId) -> Option<ClassId> {
209        self.class_of.get(&v).copied()
210    }
211}

213/// The synthetic variable representing a `(base, field)` access, created on
214/// first use so equality constraints on field reads and writes unify.
215fn field_rep(
216    uf: &mut UnionFind,
217    field_var: &mut FxHashMap<(VarId, String), VarId>,
218    next: &mut usize,
219    base: VarId,
220    field: &str,
221) -> VarId {
222    let key = (base, field.to_string());
223    if let Some(&v) = field_var.get(&key) {
224        return v;
225    }
226    let v = VarId(*next);
227    *next += 1;
228    field_var.insert(key, v);
229    uf.find(v.0);
230    // The field access points into the base's class neighborhood: unify with
231    // nothing by default; constraints drive merging.
232    v
233}
234
235/// Synthetic field-representation vars live above the real var-id space;
236/// they never appear in the emitted classes.
237fn is_synthetic(v: usize, synthetic_base: usize) -> bool {
238    v >= synthetic_base
239}

241#[cfg(test)]
242/// Assertions lifted from the doc examples of items that are now
243/// crate-private; a doctest cannot see them.
244mod doc_examples {
245    #[test]
246    fn doc_example_analyze() {
247        use crate::graph::steensgaard::Steensgaard;
248        use crate::harness::lower_source;
249
250        let fns = lower_source(
251            "t.ts",
252            "function g(a: any) { const b = a; const c = 5; return b }",
253            "ts",
254        )
255        .unwrap();
256        let g = &fns["g"];
257        let var = |name: &str| {
258            *g.var_metadata
259                .iter()
260                .find(|(_, m)| m.source_name.as_deref() == Some(name))
261                .map(|(v, _)| v)
262                .unwrap()
263        };
264        let classes = Steensgaard::analyze(g);
265        // `const b = a` unifies `b` with `a`.
266        assert_eq!(classes.class_of[&var("a")], classes.class_of[&var("b")]);
267        // `c` is assigned a literal, so it is not unified with them.
268        assert_ne!(
269            classes.class_of.get(&var("c")),
270            classes.class_of.get(&var("a"))
271        );
272    }
