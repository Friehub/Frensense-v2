5//! Phase 3.5: Sparse Value-Flow Graph (SVFG)
6//!
7//! The SVFG is a directed graph `G = (N, E)` where:
8//!   - Each node represents an SSA instruction that defines or uses a variable.
9//!   - A directed edge `A → B` means: a value defined at `A` is used at `B`.
10//!
11//! Because our IR is already in SSA form, every variable has exactly one static
12//! definition site. Def-use chains are therefore explicit and exact, the SVFG
13//! is simply those chains made into a first-class graph structure.
14//!
15//! ## Why this matters for taint analysis
16//!
17//! The historical fixed-point `TaintEngine` ran a loop over *all* instructions
18//! in *all* blocks, repeating until nothing changed. On a large codebase this
19//! visits huge amounts of irrelevant IR. The SVFG changes the cost model:
20//!
21//!   * Taint propagation = BFS/DFS from **source nodes**, following edges.
22//!   * No fixed-point loop. No visiting blocks that aren't reachable from a source.
23//!   * Cost is proportional to the number of actual taint paths, not program size.
24//!
25//! ## Node kinds
26//!
27//! Each `SvfgNode` carries a `NodeKind` that describes what kind of IR event it
28//! represents:
29//!
30//! | Kind              | Meaning                                                   |
31//! |-------------------|-----------------------------------------------------------|
32//! | `Phi`             | SSA φ-node merging two or more incoming values            |
33//! | `InstrDef`        | Instruction that **defines** a variable                   |
34//! | `InstrUse`        | Instruction that **uses** a variable as an operand        |
35//! | `FormalParam`     | Formal parameter of this function                         |
36//! | `ActualArg`       | Argument passed at a call site (interprocedural attachment point) |
37//! | `ActualRet`       | The destination variable that receives a call's return    |
38//! | `FormalRet`       | The source of a `Return` terminator (interprocedural stub)|
39//!
40//! ## Edge model (the critical part)
41//!
42//! For an instruction `J` that reads `v1` and writes `v2`, we model **three**
43//! nodes and two edges:
44//!
45//! ```text
46//!   def(v1) ──▶ use(J, v1) ──▶ def(J, v2)
47//! ```
48//!
49//! The `use(J, v1) → def(J, v2)` edge is the *intra-instruction flow* edge: it
50//! is what makes taint survive an assignment, a binary op, or a call
51//! destination. Without it, BFS dies at the first use node. This mirrors
52//! exactly what the fixed-point engine's `process_instruction` did when it
53//! moved taint from operands to dests, the graph just makes it explicit and
54//! reusable for demand-driven backward traversal (task 4.4).
55//!
56//! Memory states flow the same way: every instruction consuming `mem_in` and
57//! producing `mem_out` gets a `mem_in-use → mem_out-def` edge, so memory taint
58//! (from a `StoreField` whose `src` was tainted) keeps flowing forward through
59//! the mem-state chain even though the SVFG engine currently tracks taint on
60//! plain variables only.

70/// A stable, cheap key identifying a node in the SVFG.
71///
72/// We use the *instruction index* within a block rather than a pointer so the
73/// graph is fully serialisable and deterministic (important for task 4.5, the
74/// corpus graph store).
75#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
76#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
77pub struct NodeKey {
78    /// Block the node belongs to.
79    pub block: BlockId,
80    /// `None` for Phi nodes, `Some(idx)` for regular instructions.
81    pub instr_idx: Option<usize>,
82    /// Variable the node defines or uses.
83    pub var: VarId,
84}
85
86impl NodeKey {
87    pub(crate) fn phi(block: BlockId, var: VarId) -> Self {
88        Self {
89            block,
90            instr_idx: None,
91            var,
92        }
93    }
94
95    /// The instruction this node defines or uses, or `None` for phi nodes,
96    /// parameter sentinels and out-of-range indices.
97    pub(crate) fn instruction<'a>(&self, ir: &'a FunctionIR) -> Option<&'a Instruction> {
98        ir.blocks
99            .get(&self.block)?
100            .instructions
101            .get(self.instr_idx?)
102    }
103
104    pub(crate) fn instr(block: BlockId, idx: usize, var: VarId) -> Self {
105        Self {
106            block,
107            instr_idx: Some(idx),
108            var,
109        }
110    }
111}

117/// `arg_index` recorded on an [`NodeKind::ActualArg`] node created for a
118/// call's *receiver* operand (`x.map(..)`): receivers are not positional
119/// arguments, so they carry no slot number.
120pub const RECEIVER_ARG_INDEX: usize = usize::MAX;

122/// Does this method write its arguments into the receiver object, so a later
123/// read from that object sees the stored value?
124///
125/// Two families share the shape: collection mutators (`list.append`,
126/// `set.add`, `array.push`) and library-state setters (`ConfigParser.set`,
127/// `StringIO.write`). Without the store edge the argument's taint never
128/// reaches the object, and the reader that returns the accumulated state
129/// (`ConfigParser.get`, `StringIO.getvalue`) reads a clean value.
130///
131/// Both passes that model container writes (the whole-container fill edge and
132/// the field-sensitive store list) consult this one predicate so the two
133/// models cannot drift apart.
134fn mutates_receiver_object(method: &str) -> bool {
135    matches!(
136        method,
137        "append" | "push" | "add" | "extend" | "unshift" | "insert" | "set" | "write"
138    )
139}

141/// Separator joining a keyed library call's literal key parts into one field
142/// name. `set(section, key)` and `get(section, key)` both map to
143/// `section\u{1f}key`, so the halves of a keyed pair agree while any other
144/// key stays distinct.
145const KEYED_FIELD_SEP: char = '\u{1f}';
146
147/// Join the literal key parts of a keyed library call into one field name.
148fn join_keyed_field(parts: &[&str]) -> String {
149    let mut field = String::with_capacity(parts.iter().map(|p| p.len() + 1).sum());
150    for (i, part) in parts.iter().enumerate() {
151        if i > 0 {
152            field.push(KEYED_FIELD_SEP);
153        }
154        field.push_str(part);
155    }
156    field
157}
158
159/// The field a keyed library store writes, from its literal key arguments.
160///
161/// `conf.set('s', 'k', value)` writes `s\u{1f}k`; the two-argument form
162/// (`m.set('k', value)`) writes `k`. A non-literal key leaves the store
163/// unkeyed: it keeps the wildcard path so a caller passing a variable key
164/// still sees the value.
165fn keyed_store_field(args: &[Operand]) -> Option<String> {
166    let literal = |i: usize| match args.get(i) {
167        Some(Operand::StringLiteral(s)) => Some(s.as_str()),
168        _ => None,
169    };
170    match (literal(0), literal(1)) {
171        (Some(a), Some(b)) if args.len() >= 3 => Some(join_keyed_field(&[a, b])),
172        (Some(a), _) if args.len() == 2 => Some(a.to_string()),
173        _ => None,
174    }
175}
176
177/// True when this node is the receiver operand of a literal-keyed library
178/// read (`conf.get('s', 'k')`, `m.get('k')`).
179///
180/// Such a read takes its value from the Pass-3 keyed store edge for that exact
181/// key, so the blanket "external method call passes its receiver to the
182/// result" fallback must not run for it: that fallback would taint the read
183/// with every value the object ever stored, whichever key it asked for.
184/// Unkeyed reads (`d.get(variable_key)`) keep the fallback.
185pub(crate) fn is_keyed_library_read(ir: &FunctionIR, key: &NodeKey) -> bool {
186    let receiver = match key.instruction(ir) {
187        Some(Instruction::CallVirtual {
188            receiver: Operand::Var(r),
189            method,
190            args,
191            ..
192        }) if *r == key.var && method == "get" && keyed_load_field(args).is_some() => *r,
193        _ => return false,
194    };
195    let Some(name) = ir
196        .var_metadata
197        .get(&receiver)
198        .and_then(|m| m.source_name.as_deref())
199    else {
200        return false;
201    };
202    keyed_store_base_names(ir).contains(name)
203}
204
205/// Names of the objects this function stores into under a literal key.
206///
207/// The keyed read model applies only to a state object the same function
208/// populates with `set(section, key, value)` or `set(key, value)`. A framework
209/// accessor such as `request.args.get("name")` has no such store - it reads a
210/// request parameter - and cutting its receiver edge would delete the source.
211fn keyed_store_base_names(ir: &FunctionIR) -> FxHashSet<&str> {
212    let mut names = FxHashSet::default();
213    for block in ir.blocks.values() {
214        for instr in &block.instructions {
215            if let Instruction::CallVirtual {
216                receiver: Operand::Var(r),
217                method,
218                args,
219                ..
220            } = instr
221                && method == "set"
222                && keyed_store_field(args).is_some()
223                && let Some(name) = ir
224                    .var_metadata
225                    .get(r)
226                    .and_then(|m| m.source_name.as_deref())
227            {
228                names.insert(name);
229            }
230        }
231    }
232    names
233}
234
235/// The field a keyed library read loads, from its literal key arguments.
236///
237/// `conf.get('s', 'k')` reads `s\u{1f}k`, matching the store above; the
238/// one-argument form (`m.get('k')`) reads `k`, matching `m['k']` element
239/// stores. Returns `None` when any key part is not a literal, which keeps the
240/// read conservative: the receiver's own taint flows to the result.
241fn keyed_load_field(args: &[Operand]) -> Option<String> {
242    if args.is_empty() {
243        return None;
244    }
245    let mut parts = Vec::with_capacity(args.len());
246    for arg in args {
247        match arg {
248            Operand::StringLiteral(s) => parts.push(s.as_str()),
249            _ => return None,
250        }
251    }
252    Some(join_keyed_field(&parts))
253}

276/// What role a node plays for its variable.
277#[derive(Debug, Clone, PartialEq, Eq)]
278#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
279pub enum NodeKind {
280    /// SSA φ-function merging incoming values for `var`.
281    Phi,
282    /// This instruction *defines* `var` (dest or mem_out).
283    InstrDef,
284    /// This instruction *uses* `var` as an input operand or `mem_in`.
285    InstrUse,
286    /// `var` is a formal parameter of the enclosing function.
287    FormalParam,
288    /// `var` is passed as an argument at a call site (interprocedural attachment point).
289    ActualArg {
290        /// The call instruction's node.
291        call_site: NodeKey,
292        /// Argument position, or [`RECEIVER_ARG_INDEX`] for a call's receiver.
293        arg_index: usize,
294    },
295    /// `var` receives the return value of a call (interprocedural attachment point).
296    ActualRet {
297        /// The call instruction's node.
298        call_site: NodeKey,
299    },
300    /// `var` is the value returned by a `Return` terminator.
301    FormalRet,
302}

308/// A single node in the SVFG.
309#[derive(Debug, Clone)]
310#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
311pub struct SvfgNode {
312    /// Identity of the node.
313    pub key: NodeKey,
314    /// Role the node plays for its variable.
315    pub kind: NodeKind,
316    /// Edges leaving this node (i.e. nodes that use the value defined here).
317    successors: FxHashSet<NodeKey>,
318    /// Edges entering this node (for reverse traversal / debugging).
319    predecessors: FxHashSet<NodeKey>,
320}
321
322impl SvfgNode {
323    /// Successor keys in deterministic (sorted) order.
324    pub(crate) fn successors(&self) -> Vec<NodeKey> {
325        let mut v: Vec<NodeKey> = self.successors.iter().copied().collect();
326        v.sort_by_key(|k| (k.block.0, k.instr_idx, k.var.0));
327        v
328    }
329
330    /// Predecessor keys in deterministic (sorted) order.
331    pub(crate) fn predecessors(&self) -> Vec<NodeKey> {
332        let mut v: Vec<NodeKey> = self.predecessors.iter().copied().collect();
333        v.sort_by_key(|k| (k.block.0, k.instr_idx, k.var.0));
334        v
335    }
336}

342/// The Sparse Value-Flow Graph for a single function.
343///
344/// Build it with `SvfgBuilder::build`, then query it with the helper
345/// methods.
346#[derive(Debug, Default, Clone)]
347pub struct Svfg {
348    /// All nodes, keyed by their [`NodeKey`].
349    pub nodes: FxHashMap<NodeKey, SvfgNode>,
350}
351
352// Custom serde impls (task 4.5): serde_json cannot use struct-typed map keys
353// ("key must be a string"), so the node map serialises as a *sorted* list of
354// `(NodeKey, SvfgNode)` pairs, which is also byte-deterministic for the
355// corpus store's content hashing.
356#[cfg(feature = "serialize")]
357impl serde::Serialize for Svfg {
358    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
359        let mut v: Vec<(&NodeKey, &SvfgNode)> = self.nodes.iter().collect();
360        v.sort_by_key(|(k, _)| (k.block.0, k.instr_idx, k.var.0));
361        v.serialize(serializer)
362    }
363}
364
365#[cfg(feature = "serialize")]
366impl<'de> serde::Deserialize<'de> for Svfg {
367    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
368        let v = Vec::<(NodeKey, SvfgNode)>::deserialize(deserializer)?;
369        Ok(Self {
370            nodes: v.into_iter().collect(),
371        })
372    }
373}
374
375impl Svfg {
376    /// Returns the node for the given key, if it exists.
377    #[inline]
378    pub fn node(&self, key: &NodeKey) -> Option<&SvfgNode> {
379        self.nodes.get(key)
380    }
381
382    /// Number of value-flow edges leaving every node in this function's
383    /// graph, for the scan telemetry's graph size.
384    pub(crate) fn edge_count(&self) -> usize {
385        self.nodes.values().map(|n| n.successors.len()).sum()
386    }
387
388    /// Add a directed edge `from → to`. O(1); dedup via hash set.
389    fn add_edge(&mut self, from: NodeKey, to: NodeKey) {
390        if let Some(f) = self.nodes.get_mut(&from) {
391            f.successors.insert(to);
392        }
393        if let Some(t) = self.nodes.get_mut(&to) {
394            t.predecessors.insert(from);
395        }
396    }
397
398    /// Ensure a node exists; insert it if missing. Returns `true` if a *new*
399    /// node was created (callers use this to decide whether to overwrite the
400    /// default kind of a pre-existing node).
401    fn ensure_node(&mut self, key: NodeKey, kind: NodeKind) -> bool {
402        match self.nodes.entry(key) {
403            std::collections::hash_map::Entry::Occupied(_) => false,
404            std::collections::hash_map::Entry::Vacant(e) => {
405                e.insert(SvfgNode {
406                    key,
407                    kind,
408                    successors: FxHashSet::default(),
409                    predecessors: FxHashSet::default(),
410                });
411                true
412            }
413        }
414    }
415}

421/// Builds a `Svfg` from an already-SSA-converted `FunctionIR`.
422///
423/// The algorithm is a direct implementation of the spec:
424///
425/// ```text
426/// 1. For every instruction I that defines variable v  → create node(I, v)
427/// 2. For every instruction J that uses variable v     → create node(J, v)
428/// 3. For every use of v in J where v was defined at I → add edge node(I,v) → node(J,v)
429///    3b. Intra-instruction flow: use(J, v) → def(J, w) for every w that J defines
430/// 4. Call sites: actual-arg → formal-param edges, formal-ret → actual-ret edges
431///    (attachment points created here; cross-function linking lands in task 4.2)
432/// ```
433///
434/// Step 3 is O(1) per use because SSA guarantees exactly one definition for
435/// every variable, we just look up the pre-built `def_site` table.
436#[derive(Debug)]
437pub(crate) struct SvfgBuilder<'a> {
438    ir: &'a FunctionIR,
439    graph: Svfg,
440    buffer_builtins: Vec<crate::analysis::taint::facts::BufferBuiltinSpec>,
441    /// Maps each VarId to the SVFG node that *defines* it.
442    def_site: FxHashMap<VarId, NodeKey>,
443    /// Shared points-to analysis for this function, built lazily on first
444    /// alias query and reused for the rest of the build (R4: one heap per
445    /// function instead of one per query).
446    heap_cache: Option<std::rc::Rc<PointsToAnalysis>>,
447}
448
449impl<'a> SvfgBuilder<'a> {
450    /// Creates a builder for the function `ir`.
451    pub(crate) fn new(ir: &'a FunctionIR) -> Self {
452        Self {
453            ir,
454            graph: Svfg::default(),
455            def_site: FxHashMap::default(),
456            heap_cache: None,
457            buffer_builtins: Vec::new(),
458        }
459    }
460
461    /// Consume the builder and return the finished graph plus the def-site table.
462    pub(crate) fn build(self) -> (Svfg, FxHashMap<VarId, NodeKey>) {
463        self.build_inner()
464    }
465
466    /// Like [`SvfgBuilder::build`], but also models the given buffer-manipulation builtins, so that a call whose name matches a spec adds an edge from its source argument to later reads of its destination object (the base variable, with pointer arithmetic stripped). The edges are added only when the function's language is `c` or `cpp` and `specs` is non-empty; otherwise this behaves like [`SvfgBuilder::build`].
467    pub(crate) fn build_with_buffer_builtins(
468        mut self,
469        specs: &[crate::analysis::taint::facts::BufferBuiltinSpec],
470    ) -> (Svfg, FxHashMap<VarId, NodeKey>) {
471        self.buffer_builtins = specs.to_vec();
472        self.build_inner()
473    }
474
475    /// Pass 0: register formal parameters and the initial memory state as def
476    /// nodes. Neither is "inside" an instruction, so each gets a synthetic
477    /// key: real instructions occupy `[0, n)` and terminators use `MAX-1`, so
478    /// parameters live at `MAX - 2 - i` (one distinct slot per parameter).
479    fn register_formals(&mut self) {
480        let entry = self.ir.entry_block;
481        let param_slot_base = usize::MAX - 2;
482        for (idx, &param_var) in self.ir.parameters.iter().enumerate() {
483            let key = NodeKey::instr(entry, param_slot_base - idx, param_var);
484            self.graph.ensure_node(key, NodeKind::FormalParam);
485            self.def_site.insert(param_var, key);
486        }
487        // The initial memory state is also a "parameter" of sorts.
488        let mem = self.ir.initial_memory_state;
489        let key = NodeKey::instr(entry, param_slot_base, mem);
490        self.graph.ensure_node(key, NodeKind::FormalParam);
491        self.def_site.insert(mem, key);
492    }
493
494    fn build_inner(mut self) -> (Svfg, FxHashMap<VarId, NodeKey>) {
495        self.register_formals();
496
497        // -----------------------------------------------------------------
498        // Pass 1: Walk every block and register def nodes + def sites.
499        // -----------------------------------------------------------------
500        // We walk blocks in a stable order so the graph is deterministic.
501        let mut block_ids: Vec<BlockId> = self.ir.blocks.keys().copied().collect();
502        block_ids.sort_by_key(|b| b.0);
503
504        // The state objects this function fills with a literal-keyed `set`:
505        // only a `get` on one of them is resolved through the keyed store edge
506        // instead of the receiver's own taint.
507        let keyed_bases = keyed_store_base_names(self.ir);
508
509        for &block_id in &block_ids {
510            let block = &self.ir.blocks[&block_id];
511
512            // --- Phi nodes ---
513            for phi in &block.phis {
514                let key = NodeKey::phi(block_id, phi.dest);
515                self.graph.ensure_node(key, NodeKind::Phi);
516                self.def_site.insert(phi.dest, key);
517            }
518
519            // --- Regular instructions ---
520            for (idx, instr) in block.instructions.iter().enumerate() {
521                for def_var in instr.defs() {
522                    let key = NodeKey::instr(block_id, idx, def_var);
523                    self.graph.ensure_node(key, NodeKind::InstrDef);
524                    self.def_site.insert(def_var, key);
525                }
526            }
527        }
528
529        // -----------------------------------------------------------------
530        // Pass 2: Walk again to create use nodes and connect edges.
531        // -----------------------------------------------------------------
532        for &block_id in &block_ids {
533            let block = &self.ir.blocks[&block_id];
534
535            // --- Phi uses: each incoming (pred_block, incoming_var) is a use ---
536            for phi in &block.phis {
537                let phi_def_key = NodeKey::phi(block_id, phi.dest);
538                for &(_, inc_var) in &phi.incoming {
539                    // def(inc_var) → Phi(inc_var), value flows into the phi...
540                    self.connect_def_to_use(inc_var, phi_def_key);
541                }
542            }
543
544            // --- Regular instructions ---
545            for (idx, instr) in block.instructions.iter().enumerate() {
546                let defs = instr.defs();
547
548                // Memory-state handling: mem_in is consumed and (if the
549                // instruction produces mem_out) the memory flows onward.
550                let uses_mem = Self::instruction_uses_mem(instr);
551                let has_mem_out = defs.iter().any(|v| {
552                    self.ir
553                        .var_metadata
554                        .get(v)
555                        .is_some_and(|m| m.is_memory_state)
556                });
557
558                // a) Operand uses: def(v) → use(J, v) → def(J, w) for each defined w.
559                for use_var in Self::instruction_uses(instr) {
560                    let use_key = NodeKey::instr(block_id, idx, use_var);
561
562                    // A field read whose base is a fresh allocation is a
563                    // *container identity* use, not a value use: the field's
564                    // value arrives through the Pass-3 store→load edges (or
565                    // is absent = clean). Linking the base def here would let
566                    // the backward walk climb from the load into the
567                    // container's other stores and union every property back
568                    // together - the exact pollution the allocation lowering
569                    // exists to remove.
570                    if self.is_container_identity_use(instr, use_var, &keyed_bases) {
571                        continue;
572                    }
573
574                    self.graph.ensure_node(use_key, NodeKind::InstrUse);
575                    self.connect_def_to_use(use_var, use_key);
576
577                    // Intra-instruction flow: the consumed value contributes to
578                    // everything this instruction defines. This is the edge that
579                    // lets taint cross Assign / BinaryOp / Cast / Call-dest etc.
580                    //
581                    // Cut for boolean-producing operations (`a == b`, `a < b`,
582                    // `k in obj`, `instanceof`, `!a`): their destination holds a
583                    // BOOLEAN, whose printable payload is "true"/"false" -
584                    // operand bytes cannot survive into it, so linking
585                    // operand→dest manufactures source→sink paths that do not
586                    // exist (`flag = req.body == "admin"; execute(flag)`), and
587                    // the value lattice cannot see them (comparisons over
588                    // tainted operands fold to Top). Value-returning operators
589                    // keep their edges: `a || b` and `a + b` yield operands.
590                    if Self::defines_boolean_payload(instr) {
591                        continue;
592                    }
593                    for &def_var in &defs {
594                        let def_key = NodeKey::instr(block_id, idx, def_var);
595                        self.graph.add_edge(use_key, def_key);
596                    }
597                }
598
599                // a2) A store writes `src` INTO its container: the container's
600                // def node must reach `src` when walked backward, so a whole-
601                // container consumer (`sink(obj)`, `p = obj`) sees every
602                // stored value. Field reads are unaffected: they no longer
603                // climb through the container def (see the identity-use cut
604                // above) and resolve through Pass-3 edges instead.
605                //
606                // The edge always targets the ALLOCATION's own def node
607                // (following binding copies), never a copy's def: the
608                // backward engine stops field-demand walks exactly at
609                // allocation defs, and stores on non-allocation containers
610                // (parameters, unknown bases) keep their pre-existing edge
611                // structure - fill edges there would re-open the
612                // field-mismatch paths Pass 3 exists to filter.
613                if let Instruction::StoreField { base, src, .. }
614                | Instruction::StoreElement { base, src, .. } = instr
615                    && let Operand::Var(sv) = src
616                    && let Some(base_def) = self.allocation_def(*base)
617                {
618                    self.graph
619                        .add_edge(NodeKey::instr(block_id, idx, *sv), base_def);
620                }
621
622                if let Instruction::CallVirtual {
623                    receiver: Operand::Var(base),
624                    method,
625                    args,
626                    ..
627                } = instr
628                    && mutates_receiver_object(method)
629                {
630                    // The target is the container's own definition. Literal
631                    // containers (`lst = []`) resolve to their allocation, where
632                    // field-demand walks are designed to stop; a library object
633                    // built by a call (`ConfigParser()`, `StringIO()`) resolves
634                    // to the call's destination. Either way the mutation's
635                    // argument must be reachable from the object, or the reader
636                    // that returns the accumulated state (`get`, `getvalue`)
637                    // walks back into a clean object.
638                    let target = self
639                        .allocation_def(*base)
640                        .or_else(|| self.def_site.get(base).copied());
641                    if let Some(base_def) = target {
642                        for arg in args {
643                            if let Operand::Var(sv) = arg {
644                                self.graph
645                                    .add_edge(NodeKey::instr(block_id, idx, *sv), base_def);
646                            }
647                        }
648                    }
649                }
650
651                // b) Memory flow: mem_in-use → mem_out-def (pass-through /
652                //    weak update semantics at the graph level).
653                if uses_mem && has_mem_out {
654                    let mem_out = *defs
655                        .iter()
656                        .find(|v| {
657                            self.ir
658                                .var_metadata
659                                .get(v)
660                                .is_some_and(|m| m.is_memory_state)
661                        })
662                        .unwrap();
663                    let mem_in = Self::instruction_mem_in(instr).expect("uses_mem implies mem_in");
664                    let mem_in_key = NodeKey::instr(block_id, idx, mem_in);
665                    self.graph.ensure_node(mem_in_key, NodeKind::InstrUse);
666                    self.connect_def_to_use(mem_in, mem_in_key);
667                    let mem_out_key = NodeKey::instr(block_id, idx, mem_out);
668                    self.graph.add_edge(mem_in_key, mem_out_key);
669                }
670
671                // c) Call-site interprocedural attachment points.
672                self.register_call_stubs(block_id, idx, instr);
673            }
674
675            // --- Terminator uses (Return / Throw / Branch) ---
676            for use_var in Self::terminator_uses(&block.terminator) {
677                // Terminators don't define variables. We reserve `usize::MAX - 1`
678                // as the terminator's instr_idx (parameters live at MAX-2-i and
679                // below, so there is no collision).
680                let use_key = NodeKey::instr(block_id, usize::MAX - 1, use_var);
681                let kind = if matches!(&block.terminator, Terminator::Return { .. }) {
682                    NodeKind::FormalRet
683                } else {
684                    NodeKind::InstrUse
685                };
686                self.graph.ensure_node(use_key, kind);
687                self.connect_def_to_use(use_var, use_key);
688            }
689        }
690
691        // -----------------------------------------------------------------
692        // Pass 3: field-sensitive heap edges (store.src → load.dest).
693        // -----------------------------------------------------------------
694        self.connect_heap_edges(&block_ids);
695        // C's out-parameter writes (`strcpy(dst, src)`, `read(fd, dst, n)`,
696        // etc.) mutate their destination instead of returning its contents.
697        // Connect the source argument to subsequent reads through the same
698        // destination object, using the existing field-sensitive heap edges.
699        self.connect_buffer_builtin_edges(&block_ids);
700
701        (self.graph, self.def_site)
702    }

704    /// Resolve literal element reads of a locally built list by POSITION.
705    ///
706    /// `lst = []; lst.append('safe'); lst.append(t); lst.append('more');
707    /// lst.pop(0); bar = lst[1]` reads `'more'`. The generic model records every
708    /// `append` argument under the wildcard field, so a read at one index
709    /// matches the tainted element stored at another and a clean read is
710    /// reported.
711    ///
712    /// A position is claimed only where the shape is provable: every reference
713    /// to the container sits in ONE block, that block cannot run twice, the
714    /// container is freshly allocated inside it, and every mutation is
715    /// positional (`append(x)`, `pop(<literal>)`, `container[<literal>] = x`).
716    /// A loop, an escape to a call, an `insert`/`extend`/`remove`, or a
717    /// variable index leaves the container on the wildcard path, which is the
718    /// conservative direction.
719    fn positional_lists(&self, block_ids: &[BlockId]) -> PositionalLists {
720        let mut out = PositionalLists::default();
721        // Where each variable is mentioned, and which mutation instructions
722        // touch it in a form the shape model does not handle.
723        let mut blocks_of: FxHashMap<VarId, FxHashSet<BlockId>> = FxHashMap::default();
724        let mut mutations: FxHashMap<VarId, Vec<(BlockId, usize)>> = FxHashMap::default();
725        let mut unmodelled: FxHashSet<VarId> = FxHashSet::default();
726
727        for &block_id in block_ids {
728            let Some(block) = self.ir.blocks.get(&block_id) else {
729                continue;
730            };
731            for (idx, instr) in block.instructions.iter().enumerate() {
732                for var in Self::instruction_uses(instr)
733                    .into_iter()
734                    .chain(instr.defs())
735                {
736                    blocks_of.entry(var).or_default().insert(block_id);
737                }
738                let Instruction::CallVirtual {
739                    receiver: Operand::Var(r),
740                    method,
741                    ..
742                } = instr
743                else {
744                    continue;
745                };
746                if !changes_list_shape(method) {
747                    continue;
748                }
749                mutations.entry(*r).or_default().push((block_id, idx));
750                if !matches!(method.as_str(), "append" | "pop") {
751                    unmodelled.insert(*r);
752                }
753            }
754            for var in Self::terminator_uses(&block.terminator) {
755                blocks_of.entry(var).or_default().insert(block_id);
756            }
757        }
758
759        for (base, sites) in mutations {
760            if unmodelled.contains(&base) {
761                continue;
762            }
763            let Some(blocks) = blocks_of.get(&base) else {
764                continue;
765            };
766            // One block means the shape is a straight line: no other block can
767            // append to or read the container.
768            if blocks.len() != 1 {
769                continue;
770            }
771            let Some(&(block_id, _)) = sites.first() else {
772                continue;
773            };
774            if !blocks.contains(&block_id) || self.runs_more_than_once(block_id) {
775                continue;
776            }
777            let created_here = match self.allocation_def(base) {
778                Some(key) => key.block == block_id,
779                None => self.fresh_container_def(base, block_id),
780            };
781            if !created_here {
782                continue;
783            }
784            let Some(block) = self.ir.blocks.get(&block_id) else {
785                continue;
786            };
787
788            // Index -> the store instruction that currently holds it. A `pop`
789            // shifts the entries above it down, so the store answering a read
790            // is resolved at the read's own point in the block, not at the
791            // store's.
792            let mut slots: Vec<Option<(BlockId, usize, VarId)>> = Vec::new();
793            let mut writes: Vec<(BlockId, usize)> = Vec::new();
794            let mut reads: Vec<((BlockId, usize), Option<(BlockId, usize, VarId)>)> = Vec::new();
795            let mut consistent = true;
796            for (idx, instr) in block.instructions.iter().enumerate() {
797                match instr {
798                    Instruction::CallVirtual {
799                        receiver: Operand::Var(r),
800                        method,
801                        args,
802                        ..
803                    } if *r == base => match method.as_str() {
804                        "append" if args.len() == 1 => {
805                            slots.push(match args.first() {
806                                Some(Operand::Var(v)) => Some((block_id, idx, *v)),
807                                _ => None,
808                            });
809                            writes.push((block_id, idx));
810                        }
811                        "pop" => match args.first() {
812                            Some(Operand::IntLiteral(k))
813                                if *k >= 0 && (*k as usize) < slots.len() =>
814                            {
815                                slots.remove(*k as usize);
816                            }
817                            _ => {
818                                consistent = false;
819                                break;
820                            }
821                        },
822                        _ => {}
823                    },
824                    Instruction::StoreElement {
825                        base: b,
826                        index: Operand::IntLiteral(k),
827                        src,
828                        ..
829                    } if *b == base => {
830                        if *k < 0 {
831                            consistent = false;
832                            break;
833                        }
834                        let k = *k as usize;
835                        if slots.len() <= k {
836                            slots.resize(k + 1, None);
837                        }
838                        slots[k] = match src {
839                            Operand::Var(v) => Some((block_id, idx, *v)),
840                            _ => None,
841                        };
842                        writes.push((block_id, idx));
843                    }
844                    Instruction::LoadField { base: b, field, .. } if *b == base => {
845                        if let Ok(k) = field.parse::<usize>() {
846                            reads.push(((block_id, idx), slots.get(k).copied().flatten()));
847                        }
848                    }
849                    Instruction::LoadElement {
850                        base: b,
851                        index: Operand::IntLiteral(k),
852                        ..
853                    } if *b == base && *k >= 0 => {
854                        reads.push(((block_id, idx), slots.get(*k as usize).copied().flatten()));
855                    }
856                    _ => {}
857                }
858            }
859            if !consistent {
860                continue;
861            }
862            out.writes.extend(writes);
863            out.reads.extend(reads);
864        }
865        out
866    }

900    /// Can this block be entered more than once in one call? A loop body's
901    /// shape is not a straight line: positions computed from a single pass are
902    /// wrong when the block repeats, so the position model is not claimed.
903    fn runs_more_than_once(&self, block: BlockId) -> bool {
904        let Some(data) = self.ir.blocks.get(&block) else {
905            return true;
906        };
907        let mut stack: Vec<BlockId> = data.successors.clone();
908        let mut seen = FxHashSet::default();
909        while let Some(next) = stack.pop() {
910            if next == block {
911                return true;
912            }
913            if !seen.insert(next) {
914                continue;
915            }
916            if let Some(d) = self.ir.blocks.get(&next) {
917                stack.extend(d.successors.iter().copied());
918            }
919        }
920        false
921    }

923    /// Pass 3: field-sensitive heap edges.
924    ///
925    /// `StoreField { base, field, src }` writes a value into an object field;
926    /// `LoadField { base, field, dest }` reads it back. The generic Pass 2
927    /// edges only connect `src → store` and `base → load`, so taint written
928    /// into `obj.data` dies in the opaque memory state and a later
929    /// `x = obj.data` reads clean, the classic missed-bug shape
930    /// `obj.data = source(); sink(obj.data)`.
931    ///
932    /// This pass links them explicitly:
933    ///
934    /// ```text
935    ///   def(src) ──▶ use(store.src) ──▶ def(load.dest)
936    /// ```
937    ///
938    /// whenever the store's and load's *bases* may denote the same object
939    /// (checked with the points-to analysis: intersecting base location
940    /// sets, or both sides resolving to the same named parameter) and the
941    /// field names match. `StoreElement`/`LoadElement` participate through
942    /// the same mechanism with the wildcard field `"*"`.
943    ///
944    /// The pass runs on the SSA form, so a strong store kills the flow from
945    /// earlier same-field stores only implicitly: the load simply also links
946    /// to the later store, and both edges exist (over-approximation, never
947    /// under-approximation, sound for taint).
948    fn connect_heap_edges(&mut self, block_ids: &[BlockId]) {
949        // Containers whose whole shape is visible: their stores and literal
950        // index reads are resolved by position below instead of by the
951        // wildcard match every other container uses.
952        let positional = self.positional_lists(block_ids);
953        // Collect (block, idx, base, field, src_use_key) for every field store.
954        let mut stores: Vec<(VarId, String, NodeKey)> = Vec::new();
955        for &block_id in block_ids {
956            let block = &self.ir.blocks[&block_id];
957            for (idx, instr) in block.instructions.iter().enumerate() {
958                // The positional model owns this write: keeping the wildcard
959                // entry here would let a read at another index match it again.
960                if positional.writes.contains(&(block_id, idx)) {
961                    continue;
962                }
963                let (base, field, src) = match instr {
964                    Instruction::StoreField {
965                        base, field, src, ..
966                    } => (*base, field.clone(), src),
967                    Instruction::StoreElement {
968                        base, index, src, ..
969                    } => (
970                        *base,
971                        match index {
972                            Operand::StringLiteral(s) => s.clone(),
973                            _ => "*".to_string(),
974                        },
975                        src,
976                    ),
977                    Instruction::CallVirtual {
978                        receiver: Operand::Var(r),
979                        method,
980                        args,
981                        ..
982                    } if mutates_receiver_object(method) => {
983                        // A literal-keyed store (`conf.set('s', 'k', v)`,
984                        // `m.set('k', v)`) records the value under that key
985                        // only: the last argument is what gets stored, and a
986                        // wildcard entry here would let every later read match
987                        // it whatever key it asked for. Without a literal key
988                        // the store stays wildcard, so a variable-key caller
989                        // is not under-approximated.
990                        match keyed_store_field(args) {
991                            Some(field) => {
992                                if let Some(Operand::Var(sv)) = args.last() {
993                                    stores.push((*r, field, NodeKey::instr(block_id, idx, *sv)));
994                                }
995                            }
996                            None => {
997                                for arg in args {
998                                    if let Operand::Var(sv) = arg {
999                                        stores.push((
1000                                            *r,
1001                                            "*".to_string(),
1002                                            NodeKey::instr(block_id, idx, *sv),
1003                                        ));
1004                                    }
1005                                }
1006                            }
1007                        }
1008                        continue;
1009                    }
1010                    _ => continue,
1011                };
1012                if let Operand::Var(sv) = src {
1013                    stores.push((base, field, NodeKey::instr(block_id, idx, *sv)));
1014                }
1015            }
1016        }
1017        if stores.is_empty() && positional.reads.is_empty() {
1018            return;
1019        }
1020
1021        // For every field load, link to every matching store.
1022        for &block_id in block_ids {
1023            let block = &self.ir.blocks[&block_id];
1024            for (idx, instr) in block.instructions.iter().enumerate() {
1025                // A positional read resolves to the one store holding its index
1026                // at this point in the block (or to nothing, when the slot is
1027                // empty) - never to every value the container ever held.
1028                if let Some(answer) = positional.reads.get(&(block_id, idx)) {
1029                    if let Some((store_block, store_idx, src_var)) = *answer {
1030                        let dest = match instr {
1031                            Instruction::LoadField { dest, .. }
1032                            | Instruction::LoadElement { dest, .. } => *dest,
1033                            _ => continue,
1034                        };
1035                        self.graph.add_edge(
1036                            NodeKey::instr(store_block, store_idx, src_var),
1037                            NodeKey::instr(block_id, idx, dest),
1038                        );
1039                    }
1040                    continue;
1041                }
1042                let (base, field) = match instr {
1043                    Instruction::LoadField { base, field, .. } => (*base, field.clone()),
1044                    Instruction::LoadElement { base, index, .. } => (
1045                        *base,
1046                        match index {
1047                            Operand::StringLiteral(s) => s.clone(),
1048                            _ => "*".to_string(),
1049                        },
1050                    ),
1051                    // `conf.get('s', 'k')` / `m.get('k')` is the keyed read
1052                    // paired with the keyed store above. Without literal keys
1053                    // there is nothing to match on and the receiver keeps its
1054                    // own taint (see the identity-use cut).
1055                    Instruction::CallVirtual {
1056                        receiver: Operand::Var(r),
1057                        method,
1058                        args,
1059                        ..
1060                    } if method == "get" => match keyed_load_field(args) {
1061                        Some(field) => (*r, field),
1062                        None => continue,
1063                    },
1064                    _ => continue,
1065                };
1066                let load_dest_key = NodeKey::instr(
1067                    block_id,
1068                    idx,
1069                    match instr {
1070                        Instruction::LoadField { dest, .. }
1071                        | Instruction::LoadElement { dest, .. } => *dest,
1072                        Instruction::CallVirtual {
1073                            dest: Some(dest), ..
1074                        } => *dest,
1075                        _ => continue,
1076                    },
1077                );
1078                for (sbase, sfield, src_use_key) in &stores {
1079                    if sfield != &field && field != "*" && sfield != "*" {
1080                        continue; // field-sensitive: exact match or a wildcard side
1081                    }
1082                    if !self.may_alias_cached(*sbase, base) {
1083                        continue;
1084                    }
1085                    self.graph.add_edge(*src_use_key, load_dest_key);
1086                }
1087            }
1088        }
1089    }

1091    /// Reachable blocks from a given block, including the starting block.
1092    fn reachable_blocks(&self, start: BlockId) -> FxHashSet<BlockId> {
1093        let mut seen = FxHashSet::default();
1094        let mut work = vec![start];
1095        while let Some(block) = work.pop() {
1096            if !seen.insert(block) {
1097                continue;
1098            }
1099            if let Some(data) = self.ir.blocks.get(&block) {
1100                work.extend(data.successors.iter().copied());
1101            }
1102        }
1103        seen
1104    }

1106    /// Model C buffer-mutator calls as writes into their destination object.
1107    /// The slots come from the generated/default pack so the IR mechanism
1108    /// stays vocabulary-free. This is deliberately restricted to known
1109    /// buffer builtins and local post-call reads, rather than connecting every
1110    /// call argument to every variable in the function.
1111    fn connect_buffer_builtin_edges(&mut self, block_ids: &[BlockId]) {
1112        if !matches!(self.ir.language.as_str(), "c" | "cpp") || self.buffer_builtins.is_empty() {
1113            return;
1114        }
1115
1116        let mut writes: Vec<(BlockId, usize, VarId, NodeKey)> = Vec::new();
1117        for &block_id in block_ids {
1118            let block = &self.ir.blocks[&block_id];
1119            for (idx, instr) in block.instructions.iter().enumerate() {
1120                let (name, args) = match instr {
1121                    Instruction::CallStatic { func, args, .. } => (func.as_str(), args),
1122                    Instruction::CallVirtual { method, args, .. } => (method.as_str(), args),
1123                    _ => continue,
1124                };
1125                let last = name.rsplit(['.', ':']).next().unwrap_or(name);
1126                let Some(spec) = self.buffer_builtins.iter().find(|spec| spec.name == last) else {
1127                    continue;
1128                };
1129                let Some(dst_idx) = spec.dst_arg else {
1130                    continue;
1131                };
1132                let Some(Operand::Var(dst)) = args.get(dst_idx) else {
1133                    continue;
1134                };
1135                // `strcpy(buf + len, ...)`, `memcpy(&buf[0], ...)`, `(char *)buf`:
1136                // the written object is the base variable, not the
1137                // pointer-arithmetic temporary the argument lowered to.
1138                let dst = Self::strip_pointer_arithmetic(self.ir, *dst);
1139                let dst = &dst;
1140                let Some(src_idx) = spec.src_arg else {
1141                    continue;
1142                };
1143                let Some(Operand::Var(src)) = args.get(src_idx) else {
1144                    continue;
1145                };
1146                // Reuse the call argument use node as the source of the
1147                // mutation edge. The call's regular argument→return edge
1148                // remains unchanged; only destination reads receive this
1149                // write's source taint.
1150                writes.push((block_id, idx, *dst, NodeKey::instr(block_id, idx, *src)));
1151            }
1152        }
1153        if writes.is_empty() {
1154            return;
1155        }
1156
1157        // Cache reachability from each call block so branches after the
1158        // write are included without adding a whole-program O(writes*blocks)
1159        // traversal for each use.
1160        let mut reachable: FxHashMap<BlockId, FxHashSet<BlockId>> = FxHashMap::default();
1161        for &block_id in block_ids {
1162            let block = &self.ir.blocks[&block_id];
1163            for (idx, instr) in block.instructions.iter().enumerate() {
1164                let args = match instr {
1165                    Instruction::CallStatic { args, .. }
1166                    | Instruction::CallVirtual { args, .. }
1167                    | Instruction::CallPointer { args, .. } => args,
1168                    _ => continue,
1169                };
1170                for (write_block, write_idx, written_dst, src_use) in &writes {
1171                    let ordered_after = if block_id == *write_block {
1172                        idx > *write_idx
1173                    } else {
1174                        reachable
1175                            .entry(*write_block)
1176                            .or_insert_with(|| self.reachable_blocks(*write_block))
1177                            .contains(&block_id)
1178                    };
1179                    if !ordered_after {
1180                        continue;
1181                    }
1182                    for arg in args {
1183                        let Operand::Var(arg_var) = arg else {
1184                            continue;
1185                        };
1186                        if self.may_alias_cached(*written_dst, *arg_var) {
1187                            self.graph
1188                                .add_edge(*src_use, NodeKey::instr(block_id, idx, *arg_var));
1189                        }
1190                    }
1191                }
1192            }
1193        }
1194    }

1196    /// Follow `p + n`, `p - n`, `&p[i]` and casts from a temporary back to the
1197    /// variable naming the object. Stops at the first definition that is not
1198    /// one of these, so a pointer loaded from memory stays itself.
1199    fn strip_pointer_arithmetic(ir: &FunctionIR, mut v: VarId) -> VarId {
1200        for _ in 0..8 {
1201            let mut next = None;
1202            'search: for blk in ir.blocks.values() {
1203                for instr in &blk.instructions {
1204                    match instr {
1205                        Instruction::BinaryOp {
1206                            dest,
1207                            op,
1208                            lhs: Operand::Var(base),
1209                            ..
1210                        } if *dest == v && matches!(op.as_str(), "+" | "-" | "concat") => {
1211                            next = Some(*base);
1212                            break 'search;
1213                        }
1214                        Instruction::Cast {
1215                            dest,
1216                            src: Operand::Var(base),
1217                            ..
1218                        }
1219                        | Instruction::AddressOf { dest, src: base }
1220                            if *dest == v =>
1221                        {
1222                            next = Some(*base);
1223                            break 'search;
1224                        }
1225                        _ => {}
1226                    }
1227                }
1228            }
1229            match next {
1230                Some(n) => v = n,
1231                None => break,
1232            }
1233        }
1234        v
1235    }

1237    /// Alias query against the per-function cached heap analysis (R4:
1238    /// analyze once per function, not once per query; phase 2: the cached
1239    /// analysis is the *two-phase* Andersen restricted to taint-relevant
1240    /// Steensgaard classes, so its cost tracks the relevant cluster size).
1241    fn may_alias_cached(&mut self, a: VarId, b: VarId) -> bool {
1242        if a == b {
1243            return true;
1244        }
1245        // Fill the cache on first use.
1246        if self.heap_cache.is_none() {
1247            let mut heap = PointsToAnalysis::new();
1248            // Two-phase pays off only when the fixed point has real work:
1249            // below the threshold, plain Andersen is faster than paying for
1250            // Steensgaard + the relevance gate.
1251            const TWO_PHASE_MIN_INSTRS: usize = 4096;
1252            let instr_count: usize = self
1253                .ir
1254                .blocks
1255                .values()
1256                .map(|b| b.instructions.len() + b.phis.len())
1257                .sum();
1258            if instr_count >= TWO_PHASE_MIN_INSTRS {
1259                heap.analyze_two_phase(self.ir);
1260            } else {
1261                heap.analyze(self.ir);
1262            }
1263            self.heap_cache = Some(std::rc::Rc::new(heap));
1264        }
1265        let heap = self.heap_cache.as_ref().unwrap();
1266        let pa = heap.pts.get(&a).cloned().unwrap_or_default();
1267        let pb = heap.pts.get(&b).cloned().unwrap_or_default();
1268        if !pa.is_empty() && !pb.is_empty() {
1269            return !pa.is_disjoint(&pb);
1270        }
1271        Self::same_param_root(self.ir, a, b)
1272    }

1274    /// True if `a` and `b` both trace back (through Assign/Cast SSA chains)
1275    /// to the same formal parameter or the same `Allocate` site.
1276    fn same_param_root(ir: &FunctionIR, a: VarId, b: VarId) -> bool {
1277        fn root_of(ir: &FunctionIR, mut v: VarId) -> Option<VarId> {
1278            for _ in 0..32 {
1279                let mut found = None;
1280                'outer: for blk in ir.blocks.values() {
1281                    for instr in &blk.instructions {
1282                        match instr {
1283                            Instruction::Assign {
1284                                dest,
1285                                src: Operand::Var(s),
1286                            }
1287                            | Instruction::Cast {
1288                                dest,
1289                                src: Operand::Var(s),
1290                                ..
1291                            } if *dest == v => {
1292                                found = Some(*s);
1293                                break 'outer;
1294                            }
1295                            Instruction::Allocate { dest, .. } if *dest == v => {
1296                                return Some(v);
1297                            }
1298                            _ => {}
1299                        }
1300                    }
1301                }
1302                v = found?;
1303            }
1304            None
1305        }
1306        match (root_of(ir, a), root_of(ir, b)) {
1307            (Some(ra), Some(rb)) => ra == rb,
1308            _ => false,
1309        }
1310    }

1328    /// Create `ActualArg` and `ActualRet` attachment nodes for call instructions.
1329    ///
1330    /// These run *after* the generic use-node creation within the same
1331    /// instruction, so we can't collide with `InstrUse` nodes of the same
1332    /// `(block, idx, var)` key, instead of re-creating them we *upgrade* the
1333    /// existing node's kind in place (the node is the same value-flow point
1334    /// either way; the kind is just metadata for later interprocedural linking).
1335    fn register_call_stubs(&mut self, block: BlockId, idx: usize, instr: &Instruction) {
1336        let call_site_key = NodeKey::instr(block, idx, VarId(usize::MAX));
1337
1338        let (args, receiver, dest): (&Vec<Operand>, Option<&Operand>, Option<VarId>) = match instr {
1339            Instruction::CallStatic { dest, args, .. } => (args, None, *dest),
1340            Instruction::CallVirtual {
1341                dest,
1342                receiver,
1343                args,
1344                ..
1345            } => (args, Some(receiver), *dest),
1346            Instruction::CallPointer { dest, args, .. } => (args, None, *dest),
1347            _ => return,
1348        };
1349
1350        // Actual-arg attachment points (upgrade kind if the node already
1351        // exists as an InstrUse).
1352        for (arg_idx, arg) in args.iter().enumerate() {
1353            if let Operand::Var(use_var) = arg {
1354                let arg_key = NodeKey::instr(block, idx, *use_var);
1355                self.upgrade_node_kind(arg_key, || NodeKind::ActualArg {
1356                    call_site: call_site_key,
1357                    arg_index: arg_idx,
1358                });
1359            }
1360        }
1361        if let Some(Operand::Var(recv_var)) = receiver {
1362            let arg_key = NodeKey::instr(block, idx, *recv_var);
1363            self.upgrade_node_kind(arg_key, || NodeKind::ActualArg {
1364                call_site: call_site_key,
1365                arg_index: RECEIVER_ARG_INDEX,
1366            });
1367        }
1368
1369        // Actual-ret attachment point: the call's dest variable, if any. Its
1370        // def node already exists from Pass 1; just tag it.
1371        if let Some(ret_var) = dest {
1372            let ret_key = NodeKey::instr(block, idx, ret_var);
1373            self.upgrade_node_kind(ret_key, || NodeKind::ActualRet {
1374                call_site: call_site_key,
1375            });
1376        }
1377    }

1379    /// If `key` exists, overwrite its kind with `make_kind()`; if not, create
1380    /// the node with that kind directly.
1381    fn upgrade_node_kind<F: FnOnce() -> NodeKind>(&mut self, key: NodeKey, make_kind: F) {
1382        if let Some(node) = self.graph.nodes.get_mut(&key) {
1383            node.kind = make_kind();
1384        } else {
1385            self.graph.ensure_node(key, make_kind());
1386        }

1395    fn instruction_uses(instr: &Instruction) -> Vec<VarId> {
1396        let mut v: Vec<VarId> = Vec::with_capacity(4);
1397
1398        let push_op = |op: &Operand, v: &mut Vec<VarId>| {
1399            if let Operand::Var(var) = op {
1400                v.push(*var);
1401            }
1402        };
1403
1404        match instr {
1405            Instruction::Assign { src, .. }
1406            | Instruction::Cast { src, .. }
1407            | Instruction::UnaryOp { src, .. } => push_op(src, &mut v),
1408
1409            Instruction::BinaryOp { lhs, rhs, .. } => {
1410                push_op(lhs, &mut v);
1411                push_op(rhs, &mut v);
1412            }
1413
1414            Instruction::LoadField {
1415                mem_in: _, base, ..
1416            } => {
1417                v.push(*base);
1418            }
1419
1420            Instruction::StoreField { base, src, .. } => {
1421                v.push(*base);
1422                push_op(src, &mut v);
1423            }
1424
1425            Instruction::LoadElement { base, index, .. } => {
1426                v.push(*base);
1427                push_op(index, &mut v);
1428            }
1429
1430            Instruction::StoreElement {
1431                base, index, src, ..
1432            } => {
1433                v.push(*base);
1434                push_op(index, &mut v);
1435                push_op(src, &mut v);
1436            }
1437
1438            Instruction::LoadGlobal { .. } => {}
1439
1440            Instruction::StoreGlobal { src, .. } => push_op(src, &mut v),
1441
1442            Instruction::CallStatic { args, .. } => {
1443                args.iter().for_each(|a| push_op(a, &mut v));
1444            }
1445
1446            Instruction::CallVirtual { receiver, args, .. } => {
1447                push_op(receiver, &mut v);
1448                args.iter().for_each(|a| push_op(a, &mut v));
1449            }
1450
1451            Instruction::CallPointer { func_ptr, args, .. } => {
1452                push_op(func_ptr, &mut v);
1453                args.iter().for_each(|a| push_op(a, &mut v));
1454            }
1455
1456            Instruction::Allocate { .. } => {}
1457
1458            Instruction::AddressOf { src, .. } => v.push(*src),
1459
1460            Instruction::Dereference { ptr, .. } => push_op(ptr, &mut v),
1461
1462            Instruction::ExtractValue { tuple, .. } => push_op(tuple, &mut v),
1463
1464            Instruction::Await { promise, .. } => push_op(promise, &mut v),
1465
1466            Instruction::Yield { src, .. } => {
1467                if let Some(s) = src {
1468                    push_op(s, &mut v);
1469                }
1470            }
1471        }
1472        v
1473    }

1475    /// The def node of the allocation that `var` resolves to through
1476    /// binding copies, or `None` when the chain doesn't end in one.
1477    ///
1478    /// `const o = {...}` emits `Allocate` into a temporary and binds `o`
1479    /// with an `Assign`, so both the temp and the bound name count.
1480    fn allocation_def(&self, mut var: VarId) -> Option<NodeKey> {
1481        for _ in 0..8 {
1482            let key = *self.def_site.get(&var)?;
1483            let NodeKey {
1484                block,
1485                instr_idx: Some(i),
1486                ..
1487            } = key
1488            else {
1489                return None;
1490            };
1491            let ins = self.ir.blocks.get(&block)?.instructions.get(i)?;
1492            match ins {
1493                Instruction::Allocate { .. } => return Some(key),
1494                Instruction::Assign {
1495                    src: Operand::Var(v),
1496                    ..
1497                } => var = *v,
1498                _ => return None,
1499            }
1500        }
1501        None
1502    }

1509    /// Is `use_var` the container-identity operand of a field/element read
1510    /// whose base resolves to a fresh allocation? Such reads must not
1511    /// climb into the container's definition: their field's value arrives
1512    /// through the Pass-3 store→load edges only.
1513    fn is_container_identity_use(
1514        &self,
1515        instr: &Instruction,
1516        use_var: VarId,
1517        keyed_bases: &FxHashSet<&str>,
1518    ) -> bool {
1519        match instr {
1520            Instruction::LoadField { base, .. } | Instruction::LoadElement { base, .. } => {
1521                *base == use_var
1522                    && (self.resolves_to_allocation(*base) || self.is_fresh_empty_literal(*base))
1523            }
1524            // A literal-keyed `get` on an object this function fills with
1525            // `set(...)` resolves its value through the Pass-3 keyed store edge
1526            // alone. Letting the receiver's own taint flow here would taint
1527            // every key of the object, which is the missed-precision case
1528            // `set('s', 'kA', param); get('s', 'kB')`. A `get` on any other
1529            // receiver (`request.args.get('c')`) keeps its edge: it is a
1530            // parameter lookup, not object state.
1531            Instruction::CallVirtual {
1532                receiver: Operand::Var(r),
1533                method,
1534                args,
1535                ..
1536            } => {
1537                method == "get"
1538                    && *r == use_var
1539                    && keyed_load_field(args).is_some()
1540                    && self
1541                        .ir
1542                        .var_metadata
1543                        .get(r)
1544                        .and_then(|m| m.source_name.as_deref())
1545                        .is_some_and(|name| keyed_bases.contains(name))
1546            }
1547            _ => false,
1548        }
1549    }

1551    /// Does this instruction define a boolean *payload* rather than a
1552    /// transformed copy of its operands' bytes?
1553    ///
1554    /// Comparisons (`==`, `<`, `in`, `instanceof`, ...) and boolean
1555    /// negation always print as `"true"`/`"false"`, so taint on the
1556    /// operands cannot reach a consumer of the destination. Operators that
1557    /// yield an operand's value (`||`, `&&`, arithmetic/string `+`) are
1558    /// deliberately excluded - their destinations do carry payload bytes.
1559    fn defines_boolean_payload(instr: &Instruction) -> bool {
1560        match instr {
1561            Instruction::BinaryOp { op, .. } => matches!(
1562                op.as_str(),
1563                "==" | "==="
1564                    | "!="
1565                    | "!=="
1566                    | "<"
1567                    | "<="
1568                    | ">"
1569                    | ">="
1570                    | "in"
1571                    | "not in"
1572                    | "instanceof"
1573                    | "is"
1574                    | "is not"
1575            ),
1576            Instruction::UnaryOp { op, .. } => matches!(op.as_str(), "!" | "not"),
1577            _ => false,
1578        }
1579    }
