5//! Throwaway diagnostic: count learned-fact entry variants in a `.frc`
6//! bundle and dump any fact (Policy or Check) whose trigger matches the
7//! `when_call` argument, so the serialized requirement shape is visible.
8
9use frensense_bundler::format::load_bundle;
10use frensense_engine::analysis::taint::facts::LearnedFactEntry;
11
12fn main() {
13    let path = std::env::args()
14        .nth(1)
15        .unwrap_or_else(|| "bundle.frc".into());
16    let needle = std::env::args()
17        .nth(2)
18        .unwrap_or_else(|| "updateMany".into());
19    let bytes = std::fs::read(&path).expect("bundle read");
20    let bundle = load_bundle(&bytes).expect("bundle decode");
21
22    println!("bundle: {path}  entries: {}", bundle.learned_facts.len());
23    let mut counts: std::collections::BTreeMap<&str, usize> = Default::default();
24    for entry in &bundle.learned_facts {
25        let kind = match entry {
26            LearnedFactEntry::Policy { .. } => "Policy",
27            LearnedFactEntry::Check { .. } => "Check",
28            _ => "other",
29        };
30        *counts.entry(kind).or_insert(0) += 1;
31        match entry {
32            LearnedFactEntry::Policy {
33                rule,
34                when_call,
35                require,
36                scope,
37                message,
38                severity,
39                ..
40            } if when_call.contains(&needle) => {
41                println!("rule: {rule}");
42                println!("when_call: {when_call}");
43                println!("scope: {scope:?}");
44                println!("severity: {severity}");
45                println!("message: {message}");
46                println!("requirements ({}):", require.len());
47                for r in require {
48                    println!("  {r:?}");
49                }
50                println!("---");
51            }
52            LearnedFactEntry::Check {
53                rule,
54                call,
55                message,
56                severity,
57                unless_guard,
58                unless_range_check,
59            } if call.contains(&needle) => {
60                println!("(check) rule: {rule}");
61                println!("(check) call: {call}");
62                println!("(check) severity: {severity}");
63                println!("(check) message: {message}");
64                println!("(check) unless_guard: {unless_guard:?}");
65                println!("(check) unless_range_check: {unless_range_check:?}");
66                println!("---");
67            }
68            _ => {
69                let dbg = format!("{entry:?}");
70                if dbg.contains(&needle) {
71                    let trimmed: Vec<&str> = dbg.lines().take(24).collect();
72                    println!("[other entry containing target predicate]");
73                    for l in trimmed {
74                        println!("  {l}");
75                    }
76                    println!("---");
77                }
78            }
79        }
80    }
81    if let Some(patterns) = Some(&bundle.patterns) {
82        for p in patterns {
83            if format!("{p:?}").contains(&needle) {
84                println!("[bundle pattern containing target predicate]");
85                let dbg = format!("{p:?}");
86                for l in dbg.lines().take(30) {
87                    println!("  {l}");
88                }
89                println!("---");
90            }
91        }
92    }
93    for (k, v) in counts {
94        println!("variant counts: {k}: {v}");
95    }
96}
