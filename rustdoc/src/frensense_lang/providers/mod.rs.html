5//! Built-in [`LanguageSpec`](crate::spec::LanguageSpec) providers.
6//!
7//! One module per grammar family: [`c_lang`], [`go`], [`javascript`],
8//! [`python`] and [`rust_lang`]. Each implements the trait for its own node
9//! vocabulary and leaves the cross-language defaults in place where the
10//! grammar agrees with them; the JavaScript module additionally covers the
11//! TypeScript/TSX dialects and single-file containers through
12//! [`javascript::JsFamily`].
13//!
14//! [`crate::registry`] owns the extension -> spec lookup, so nothing here
15//! decides which provider handles which file.
16pub mod c_lang;
17pub mod go;
18pub mod javascript;
19pub mod python;
20pub mod rust_lang;
