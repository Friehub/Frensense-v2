5//! Shared AST classification for JS/TS (single tree-sitter grammar).
6
7use crate::spec::NodeRole;
8
9// ── Shared AST logic ──────────────────────────────────────────────────────────
10
11pub(super) fn classify_js(kind: &str) -> NodeRole {
12    match kind {
13        // ── Functions ────────────────────────────────────────────────────
14        "function_declaration"
15        | "function"
16        | "async_function_declaration"
17        | "generator_function_declaration" => NodeRole::Function {
18            is_method: false,
19            name_field: Some("name".into()),
20            params_field: "parameters".into(),
21            body_field: "body".into(),
22        },
23        "function_expression" | "async_function_expression" => NodeRole::Function {
24            is_method: false,
25            name_field: None,
26            params_field: "parameters".into(),
27            body_field: "body".into(),
28        },
29        "arrow_function" => NodeRole::Function {
30            is_method: false,
31            name_field: None,
32            params_field: "parameters".into(),
33            body_field: "body".into(),
34        },
35        "method_definition" => NodeRole::Function {
36            is_method: true,
37            name_field: Some("name".into()),
38            params_field: "parameters".into(),
39            body_field: "body".into(),
40        },
41
42        // ── Declarations / assignments ───────────────────────────────────
43        "variable_declarator" | "lexical_declarator" => NodeRole::Declaration {
44            name_field: "name".into(),
45            value_field: "value".into(),
46        },
47        "assignment_expression" => NodeRole::Assignment {
48            lhs_field: "left".into(),
49            rhs_field: "right".into(),
50        },
51        // `x += 1`, `x **= 2`, ... - JS uses a distinct node kind with the
52        // compound operator; unmapped it was dropped (x never changed).
53        "augmented_assignment_expression" => NodeRole::Assignment {
54            lhs_field: "left".into(),
55            rhs_field: "right".into(),
56        },
57
58        // ── Calls ────────────────────────────────────────────────────────
59        "call_expression" => NodeRole::Call {
60            callee_field: "function".into(),
61            args_field: "arguments".into(),
62        },
63        // tree-sitter-javascript names the callee of `new X(...)` field
64        // "constructor", not "function", sharing the call_expression arm
65        // made every `new X(...)` expression lower to nothing.
66        "new_expression" => NodeRole::Call {
67            callee_field: "constructor".into(),
68            args_field: "arguments".into(),
69        },
70        "member_expression" => NodeRole::MemberAccess {
71            object_field: "object".into(),
72            property_field: "property".into(),
73        },
74        "subscript_expression" => NodeRole::MemberAccess {
75            object_field: "object".into(),
76            property_field: "index".into(),
77        },
78
79        // ── Control flow ─────────────────────────────────────────────────
80        "if_statement" | "switch_statement" => NodeRole::Branch,
81        "ternary_expression" => NodeRole::Conditional,
82        "for_statement" | "for_in_statement" | "for_of_statement" | "while_statement"
83        | "do_statement" => NodeRole::Loop,
84        "return_statement" => NodeRole::Return,
85        "try_statement" => NodeRole::Try,
86        "catch_clause" => NodeRole::Catch,
87        "finally_clause" => NodeRole::Finally,
88        "throw_statement" => NodeRole::Throw,
89        "await_expression" => NodeRole::Await,
90
91        // ── Structural ───────────────────────────────────────────────────
92        "statement_block" => NodeRole::Block,
93        // Value-context object/array literals: `{ key: val }`, `[a, b]`, and
94        // spread payloads. NOT statement blocks, all children are
95        // value-producers, and taint in any child taints the whole composite.
96        "object" | "array" | "parenthesized_expression" => NodeRole::Composite,
97        "import_statement" => NodeRole::Import,
98        "export_statement" => NodeRole::Export,
99        "identifier" | "property_identifier" | "shorthand_property_identifier" => {
100            NodeRole::Identifier
101        }
102        "string" | "template_string" | "number" | "true" | "false" | "null" | "undefined"
103        | "regex" => NodeRole::Literal,
104
105        // ── Supplementary structural roles ───────────────────────────────
106        "formal_parameters" | "parameters" => NodeRole::Parameters,
107        "arguments" => NodeRole::Arguments,
108        "class_declaration" | "interface_declaration" | "enum_declaration" => NodeRole::ClassDef,
109        "binary_expression" => NodeRole::BinaryOp,
110        "unary_expression" | "update_expression" => NodeRole::UnaryOp,
111        "switch_expression" => NodeRole::Match,
112
113        _ => NodeRole::Other,
114    }
115}
116
117pub(super) fn is_js_cast(kind: &str) -> bool {
118    matches!(kind, "as_expression" | "cast_expression" | "type_assertion")
119}
120
121pub(super) fn is_js_template_string(kind: &str) -> bool {
122    matches!(kind, "template_string" | "string" | "binary_expression")
123}
124
125pub(super) fn is_js_template_literal_fragment(kind: &str) -> bool {
126    kind == "string_fragment"
127}
128
129pub(super) fn is_js_destructuring_pattern(kind: &str) -> bool {
130    matches!(kind, "object_pattern" | "array_pattern")
131}
132
133pub(super) fn is_js_pair_pattern(kind: &str) -> bool {
134    kind == "pair_pattern"
135}
136
137pub(super) fn is_js_pair_entry(kind: &str) -> bool {
138    kind == "pair"
139}
140
141pub(super) fn is_js_ternary_straight_line(kind: &str) -> bool {
142    matches!(
143        kind,
144        "ternary_expression" | "conditional_expression" | "conditional_type"
145    )
146}
