5//! One [`LanguageSpec`] implementation for the whole JavaScript family.
6//!
7//! JavaScript, TypeScript and a Vue SFC's script blocks are the *same*
8//! language to this crate: one tree-sitter grammar (TS is a superset, and a
9//! `.vue` script block is parsed with the TSX grammar after
10//! [`VueSpec`](super::VueSpec)'s `preprocess` blanks the markup), the same
11//! node classification, the same import/export extraction, the same package
12//! and parameter knowledge.
13//!
14//! So the family is written **once**, here, and each member states only what
15//! makes it itself — its [`JsDialect`]:
16//!
17//! ```ignore
18//! impl JsFamily for JavaScriptSpec {
19//!     const DIALECT: JsDialect = JsDialect::JavaScript;
20//! }
21//! ```
22//!
23//! [`JsFamily`] has a blanket [`LanguageSpec`] implementation, so a new
24//! single-file container (Svelte, Astro) needs a [`JsDialect`] arm and a
25//! `preprocess` override — not a fourth copy of the delegate ladder.
26//!
27//! Security vocabulary (sinks, sources, sanitizers) lives in
28//! `frensense-packgen`; this file is mechanism only.
29

50/// Definitions JavaScript's grammar spells with `identifier` names.
51const JS_SYMBOL_QUERY: &str = r#"
52            (function_declaration name: (identifier) @name)
53            (method_definition name: (property_identifier) @name)
54            (class_declaration name: (identifier) @name)
55            (variable_declarator name: (identifier) @name)
56            (call_expression arguments: (arguments (arrow_function) @name))
57            (call_expression arguments: (arguments (function_expression) @name))
58        "#;
59
60/// As [`JS_SYMBOL_QUERY`], plus TypeScript's `type_identifier` names and
61/// `interface_declaration`.
62const TS_SYMBOL_QUERY: &str = r#"
63            (function_declaration name: (identifier) @name)
64            (method_definition name: (property_identifier) @name)
65            (class_declaration name: (type_identifier) @name)
66            (interface_declaration name: (type_identifier) @name)
67            (variable_declarator name: (identifier) @name)
68            (call_expression arguments: (arguments (arrow_function) @name))
69            (call_expression arguments: (arguments (function_expression) @name))
70        "#;
71
72/// Call sites inside function bodies. Shared by every family member: the
73/// capture names (`@caller`, `@call`) and their node kinds are identical in
74/// the JavaScript and TypeScript grammars.
75const CALL_QUERY: &str = r#"
76            (function_declaration name: (identifier) @caller
77                body: (statement_block
78                    (expression_statement
79                        (call_expression function: (identifier) @call))))
80            (function_declaration name: (identifier) @caller
81                body: (statement_block
82                    (expression_statement
83                        (call_expression
84                            function: (member_expression
85                                property: (property_identifier) @call)))))
86            (method_definition name: (property_identifier) @caller
87                body: (statement_block
88                    (expression_statement
89                        (call_expression function: (identifier) @call))))
90        "#;

92// ── Dialect: the only thing that differs ─────────────────────────────────────
93
94/// Which member of the JavaScript family a spec is.
95///
96/// Every value here is a difference the grammar actually forces: the
97/// extensions a file lands on, which tree-sitter grammar parses it, how the
98/// language's names are spelled in its symbol query, and how a bare snippet
99/// must be wrapped to re-parse as a function body.
100#[derive(Debug, Clone, Copy, PartialEq, Eq)]
101#[non_exhaustive]
102pub enum JsDialect {
103    /// Plain JavaScript, parsed by tree-sitter-javascript.
104    JavaScript,
105    /// TypeScript, parsed by the TSX grammar.
106    TypeScript,
107    /// A Vue SFC. Its script blocks are TypeScript, parsed by the TSX grammar
108    /// after [`super::VueSpec`] blanks the markup in place.
109    Vue,
110}
111
112impl JsDialect {
113    /// Canonical short name used in fingerprints and the FRC bundle.
114    fn name(self) -> &'static str {
115        match self {
116            Self::JavaScript => "javascript",
117            Self::TypeScript => "typescript",
118            Self::Vue => "vue",
119        }
120    }
121
122    /// File extensions, lowercase and without the dot.
123    fn extensions(self) -> &'static [&'static str] {
124        match self {
125            Self::JavaScript => &["js", "jsx", "mjs", "cjs"],
126            Self::TypeScript => &["ts", "tsx", "mts", "cts"],
127            Self::Vue => &["vue"],
128        }
129    }
130
131    /// The grammar this dialect parses with.
132    fn language(self) -> tree_sitter::Language {
133        match self {
134            Self::JavaScript => javascript_grammar(),
135            Self::TypeScript | Self::Vue => typescript_grammar(),
136        }
137    }
138
139    /// The `FactTable` / bundle section this dialect's knowledge lives in.
140    ///
141    /// A Vue script block *is* TypeScript, so its facts belong in the
142    /// `typescript` section; a `vue` section is filled by nothing.
143    fn fact_language(self) -> &'static str {
144        match self {
145            Self::JavaScript => "javascript",
146            Self::TypeScript => "typescript",
147            Self::Vue => "typescript",
148        }
149    }
150
151    /// Symbol-definition query for this dialect's grammar.
152    fn symbol_query(self) -> &'static str {
153        match self {
154            Self::JavaScript => JS_SYMBOL_QUERY,
155            Self::TypeScript | Self::Vue => TS_SYMBOL_QUERY,
156        }
157    }
158
159    /// Wrap a snippet so it re-parses as a function body in this dialect.
160    fn wrap_region(self, code: &str) -> String {
161        match self {
162            Self::JavaScript => format!("function _region() {{\n{code}\n}}"),
163            Self::TypeScript | Self::Vue => format!("function _region(): void {{\n{code}\n}}"),
164        }
165    }
166
167    /// Is this decorator name an HTTP route decorator?
168    ///
169    /// NestJS's `@Get`-family decorators are TypeScript-only; plain
170    /// JavaScript has no decorator route vocabulary here.
171    fn http_route_decorator(self, decorator_name: &str) -> bool {
172        match self {
173            Self::JavaScript => false,
174            Self::TypeScript | Self::Vue => matches!(
175                decorator_name,
176                "Get"
177                    | "Post"
178                    | "Put"
179                    | "Delete"
180                    | "Patch"
181                    | "All"
182                    | "Controller"
183                    | "Route"
184                    | "HttpGet"
185                    | "HttpPost"
186                    | "UseGuards"
187                    | "UseInterceptors"
188            ),
189        }
190    }
191}

193// ── The family contract ──────────────────────────────────────────────────────
194
195/// A member of the JavaScript family.
196///
197/// Implement this instead of [`LanguageSpec`]: the whole grammar-facing
198/// contract is provided by the blanket implementation below, and a member
199/// only names its [`JsDialect`]. Override
200/// [`preprocess_source`](Self::preprocess_source) when the file is a
201/// container around script code (as Vue SFCs are).
202pub trait JsFamily: Send + Sync + 'static {
203    /// Which member of the family this spec is.
204    const DIALECT: JsDialect;
205
206    /// Rewrite the source before parsing. Defaults to the identity.
207    ///
208    /// Same contract as [`LanguageSpec::preprocess`]: the rewrite must
209    /// preserve every byte offset so findings keep pointing at the original
210    /// file.
211    fn preprocess_source<'a>(&self, source: &'a str) -> Cow<'a, str> {
212        Cow::Borrowed(source)
213    }
214}

216// ── One implementation, every family member ──────────────────────────────────
217
218impl<T: JsFamily> LanguageSpec for T {
219    fn name(&self) -> &'static str {
220        T::DIALECT.name()
221    }
222
223    fn extensions(&self) -> &'static [&'static str] {
224        T::DIALECT.extensions()
225    }
226
227    fn tree_sitter_language(&self) -> tree_sitter::Language {
228        T::DIALECT.language()
229    }
230
231    fn preprocess<'a>(&self, source: &'a str) -> Cow<'a, str> {
232        self.preprocess_source(source)
233    }
234
235    fn fact_language(&self) -> &'static str {
236        T::DIALECT.fact_language()
237    }
238
239    fn classify(&self, kind: &str) -> NodeRole {
240        classify_js(kind)
241    }
242
243    fn is_cast(&self, kind: &str) -> bool {
244        is_js_cast(kind)
245    }
246
247    fn is_template_string(&self, kind: &str) -> bool {
248        is_js_template_string(kind)
249    }
250
251    fn is_template_literal_fragment(&self, kind: &str) -> bool {
252        is_js_template_literal_fragment(kind)
253    }
254
255    fn is_destructuring_pattern(&self, kind: &str) -> bool {
256        is_js_destructuring_pattern(kind)
257    }
258
259    fn is_pair_pattern(&self, kind: &str) -> bool {
260        is_js_pair_pattern(kind)
261    }
262
263    fn is_pair_entry(&self, kind: &str) -> bool {
264        is_js_pair_entry(kind)
265    }
266
267    fn is_ternary_straight_line(&self, kind: &str) -> bool {
268        is_js_ternary_straight_line(kind)
269    }
270
271    fn is_variable_declaration_statement(&self, kind: &str) -> bool {
272        matches!(kind, "lexical_declaration" | "variable_declaration")
273    }
274
275    fn is_variable_declarator(&self, kind: &str) -> bool {
276        kind == "variable_declarator"
277    }
278
279    // The two anonymous function spellings the extraction walk binds to a
280    // name. `function_declaration` / `method_definition` are excluded on
281    // purpose: they carry their own name.
282    fn is_function_value(&self, kind: &str) -> bool {
283        matches!(kind, "arrow_function" | "function_expression")
284    }
285
286    fn is_array_composite(&self, kind: &str) -> bool {
287        kind == "array"
288    }
289
290    fn is_object_composite(&self, kind: &str) -> bool {
291        kind == "object"
292    }
293
294    fn is_array_pattern(&self, kind: &str) -> bool {
295        kind == "array_pattern"
296    }
297
298    fn is_assignment_pattern(&self, kind: &str) -> bool {
299        kind == "object_assignment_pattern"
300    }
301
302    fn is_rest_pattern(&self, kind: &str) -> bool {
303        kind == "rest_pattern"
304    }
305
306    fn is_switch_statement(&self, kind: &str) -> bool {
307        kind == "switch_statement"
308    }
309
310    fn is_case_clause(&self, kind: &str) -> bool {
311        matches!(kind, "switch_case" | "switch_default")
312    }
313
314    fn jsx_part(&self, kind: &str) -> Option<JsxPart> {
315        match kind {
316            "jsx_element" => Some(JsxPart::Element),
317            "jsx_self_closing_element" => Some(JsxPart::SelfClosingElement),
318            "jsx_fragment" => Some(JsxPart::Fragment),
319            "jsx_opening_element" => Some(JsxPart::OpeningElement),
320            "jsx_children" => Some(JsxPart::Children),
321            "jsx_attribute" => Some(JsxPart::Attribute),
322            _ => None,
323        }
324    }
325
326    fn is_jsx_attribute_value(&self, kind: &str) -> bool {
327        matches!(
328            kind,
329            "jsx_expression"
330                | "string"
331                | "jsx_element"
332                | "jsx_self_closing_element"
333                | "jsx_fragment"
334        )
335    }
336
337    fn implicit_return_node<'a>(&self, fn_node: Node<'a>, body: Node<'a>) -> Option<Node<'a>> {
338        if fn_node.kind() == "arrow_function" && body.kind() != "statement_block" {
339            Some(body)
340        } else {
341            None
342        }
343    }
344
345    fn wrap_region(&self, code: &str) -> String {
346        T::DIALECT.wrap_region(code)
347    }
348
349    fn import_node_kinds(&self) -> &'static [&'static str] {
350        &["import_statement"]
351    }
352
353    fn extract_imports<'tree>(&self, root: Node<'tree>, source: &str) -> Vec<Import> {
354        extract_js_imports(root, source)
355    }
356
357    fn extract_exports<'tree>(&self, root: Node<'tree>, source: &str) -> Vec<Export> {
358        extract_js_exports(root, source)
359    }
360
361    fn extract_module_items<'tree>(&self, root: Node<'tree>, source: &str) -> Vec<ModuleItem> {
362        extract_js_module_items(root, source)
363    }
364
365    fn symbol_query(&self) -> Option<&'static str> {
366        Some(T::DIALECT.symbol_query())
367    }
368
369    fn call_query(&self) -> Option<&'static str> {
370        Some(CALL_QUERY)
371    }
372
373    fn package_category(&self, pkg: &str) -> Option<PackageCategory> {
374        js_package_category(pkg)
375    }
376
377    fn classify_param_taint(
378        &self,
379        name: Option<&str>,
380        type_annotation: Option<&str>,
381    ) -> Option<TaintOrigin> {
382        classify_js_param(name, type_annotation)
383    }
384
385    fn is_http_route_decorator(&self, decorator_name: &str) -> bool {
386        T::DIALECT.http_route_decorator(decorator_name)
387    }
388
389    /// JS/TS `for … in` and `for … of` loops expose the iterable via the
390    /// `right` field and the loop binding via the `left` field.
391    ///
392    /// A `for (x in obj)` or `for (x of iterable)` has no meaningful runtime
393    /// iteration sink matching the Python model, but emitting the iterable as
394    /// IR still ensures taint from the `right` expression reaches the loop
395    /// body, which is the conservative-and-correct behaviour.
396    fn for_loop_iterable<'t>(
397        &self,
398        node: tree_sitter::Node<'t>,
399    ) -> Option<(tree_sitter::Node<'t>, tree_sitter::Node<'t>)> {
400        if !matches!(node.kind(), "for_in_statement" | "for_of_statement") {
401            return None;
402        }
403        let binding = node.child_by_field_name("left")?;
404        let iterable = node.child_by_field_name("right")?;
405        Some((binding, iterable))
406    }
407}

409// ── Feature-gated grammars ───────────────────────────────────────────────────

411#[cfg(feature = "javascript")]
412fn javascript_grammar() -> tree_sitter::Language {
413    tree_sitter_javascript::LANGUAGE.into()
414}
415
416#[cfg(not(feature = "javascript"))]
417fn javascript_grammar() -> tree_sitter::Language {
418    panic!("frensense-lang: 'javascript' feature not enabled")
419}
420
421#[cfg(feature = "typescript")]
422fn typescript_grammar() -> tree_sitter::Language {
423    tree_sitter_typescript::LANGUAGE_TSX.into()
424}
425
426#[cfg(not(feature = "typescript"))]
427fn typescript_grammar() -> tree_sitter::Language {
428    panic!("frensense-lang: 'typescript' feature not enabled")
