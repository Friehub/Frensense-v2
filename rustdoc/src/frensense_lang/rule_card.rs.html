5//! Rule cards: the one place a finding's identity, title, CWE and advice
6//! live. Terminal, SARIF, JSON and `explain` all render from here, so the
7//! surfaces cannot drift and a built-in rule cannot ship without a CWE
8//! (see the coverage tests here and in `frensense-packgen`).
9//!
10//! Taint findings resolve a card from the sink label slug
11//! (`SinkLabel::slug`); non-dataflow checkers resolve one from the checker
12//! rule id. Bundle-learned and user policy rules are not listed: they carry
13//! their own prose, and their CWE comes from the bundle pattern overlay.
14
15use crate::severity::SinkRole;
16
17/// Static description of one built-in rule.
18#[derive(Debug, PartialEq, Eq)]
19pub struct RuleCard {
20    /// Stable id, `taint.<class>` or `check.<class>`. Never contains
21    /// whitespace, backticks, or finding-specific text.
22    pub id: &'static str,
23    /// Short class name: "SQL injection".
24    pub title: &'static str,
25    /// CWE ids, primary first. Never empty for a built-in rule.
26    pub cwe: &'static [&'static str],
27    /// One sentence: why this is a problem.
28    pub why: &'static str,
29    /// One sentence: how to fix it.
30    pub fix: &'static str,
31}
32
33const fn card(
34    id: &'static str,
35    title: &'static str,
36    cwe: &'static [&'static str],
37    why: &'static str,
38    fix: &'static str,
39) -> RuleCard {
40    RuleCard {
41        id,
42        title,
43        cwe,
44        why,
45        fix,
46    }
47}
48
49/// Fallback card ids: the ones that name no vulnerability class.
50///
51/// A finding that resolves to one of these says only "untrusted data reached
52/// a sensitive operation": the sink's class is unknown, so the report gives
53/// the reader nothing to fix, suppress or trend. The scanner's reporting
54/// policy drops those findings instead of printing them.
55const GENERIC_IDS: &[&str] = &["taint.untrusted-data", "taint.injection"];
56
57impl RuleCard {
58    /// Does this card name no vulnerability class (a fallback card)?
59    #[must_use]
60    pub fn is_generic(&self) -> bool {
61        GENERIC_IDS.contains(&self.id)
62    }
63}

244/// Resolve the card for a taint finding.
245///
246/// `label` is the sink label slug (`SinkLabel::slug`), `role` the sink's
247/// role, `idor` whether the finding's class is an access-control shape.
248/// Always returns a card: unlabeled sinks fall back by role to the
249/// generic injection / untrusted-data cards.
250#[must_use]
251pub fn taint_card(label: Option<&str>, role: SinkRole, idor: bool) -> &'static RuleCard {
252    if idor {
253        return taint_by_slug("idor");
254    }
255    let slug = match (label, role) {
256        // "eval" is the historic slug for code-execution sinks.
257        (Some("eval"), _) => "execution",
258        (Some(l), _) if find(TAINT, l).is_some() => l,
259        (_, SinkRole::Xss) => "xss",
260        (_, SinkRole::Execution) => "injection",
261        _ => "other",
262    };
263    taint_by_slug(slug)
264}
265
266fn taint_by_slug(slug: &str) -> &'static RuleCard {
267    find(TAINT, slug).unwrap_or_else(|| find(TAINT, "other").expect("fallback card"))
268}

270/// The taint class a CWE id names, as `(role, label slug)`.
271///
272/// Corpus families declare `cwe:` in their `[frensense]` block and the bundler
273/// records it on the published advisory pattern, so this is class information
274/// every `.frc` has always carried. Deriving a learned sink's role and label
275/// from it classifies the sink **without a bundle format change**, which is
276/// what keeps previously published bundles loadable.
277///
278/// Only the TAINT cards are considered: a CWE that only a checker rule names
279/// (weak hash, TOCTOU) says nothing about what a sink does with its input.
280#[must_use]
281pub fn taint_class_for_cwe(cwe: &str) -> Option<(SinkRole, &'static str)> {
282    let want = cwe.trim();
283    // Parent CWEs describe "improper validation" in general, not what a sink
284    // does with its input, and several class cards list them. Deriving a
285    // class from one would invent a verdict the corpus never made.
286    const BROAD: &[&str] = &["cwe-20", "cwe-74", "cwe-693"];
287    let lower = want.to_ascii_lowercase();
288    if BROAD.contains(&lower.as_str()) {
289        return None;
290    }
291    // CWE variants an advisory files a defect under that our cards name by
292    // their canonical id: Django's SQL injection is CWE-564, argument
293    // injection is CWE-88. Aliased here rather than appended to the cards so
294    // a finding's own CWE tags keep the canonical id.
295    const ALIASES: &[(&str, &str)] = &[("cwe-564", "sql"), ("cwe-88", "command")];
296    if let Some((_, slug)) = ALIASES.iter().find(|(alias, _)| *alias == lower) {
297        return Some((taint_role_for_slug(slug), *slug));
298    }
299    TAINT
300        .iter()
301        // A generic card ("untrusted data reaches a sensitive operation")
302        // names no class, so landing on one would classify nothing. CWE-20
303        // and CWE-74 are attached to those catch-all cards.
304        .filter(|(_, c)| !c.is_generic())
305        .find(|(_, c)| c.cwe.iter().any(|id| id.eq_ignore_ascii_case(want)))
306        .map(|(slug, _)| (taint_role_for_slug(slug), *slug))
307}

309/// The role a taint card's class implies. Classes with a dedicated role keep
310/// it; every other class is code/SQL/command/deserialization, i.e.
311/// `Execution`.
312fn taint_role_for_slug(slug: &str) -> SinkRole {
313    match slug {
314        "traversal" => SinkRole::Resource,
315        "storage" => SinkRole::Storage,
316        "xss" => SinkRole::Xss,
317        "response" => SinkRole::Response,
318        "validation" => SinkRole::Validation,
319        _ => SinkRole::Execution,
320    }
321}

323/// Does a taint finding on this `(label, role)` pair name a vulnerability
324/// class a consumer can act on?
325///
326/// Single source of truth for "is this taint finding reportable". The bundler's
327/// replay gate and the CLI's advisory policy both answer it here, because they
328/// answered it separately before: the gate asked only whether a finding
329/// existed, the CLI asked whether it named a class, and the gate came to
330/// approve learned sinks no surface would ever report.
331#[must_use]
332pub fn names_vulnerability_class(label: Option<&str>, role: SinkRole, idor: bool) -> bool {
333    !taint_card(label, role, idor).is_generic()
334}

336/// Card for a built-in checker rule id, if one exists.
337#[must_use]
338pub fn checker_card(rule: &str) -> Option<&'static RuleCard> {
339    find(CHECK, rule)
340}

342/// The checker rule key (`weak_hash`) a card id (`check.weak-hash`) belongs
343/// to, for callers that hold a card id but need the registry key (tier,
344/// suppression, per-rule scoring).
345#[must_use]
346pub fn checker_key(card_id: &str) -> Option<&'static str> {
347    CHECK.iter().find(|(_, c)| c.id == card_id).map(|(k, _)| *k)
348}

350/// Card by its stable id (`taint.sql-injection`, `check.weak-hash`).
351#[must_use]
352pub fn card_by_id(id: &str) -> Option<&'static RuleCard> {
353    all_cards().find(|c| c.id == id)
354}

356/// Every built-in card: taint first, then checkers.
357pub fn all_cards() -> impl Iterator<Item = &'static RuleCard> {
358    TAINT.iter().chain(CHECK.iter()).map(|(_, c)| c)
359}
360
361/// Sink-label slugs that have a card (for coverage tests in crates that
362/// own the label enum).
363pub fn taint_slugs() -> impl Iterator<Item = &'static str> {
364    TAINT.iter().map(|(s, _)| *s)
365}
366
367fn find(table: &'static [(&'static str, RuleCard)], key: &str) -> Option<&'static RuleCard> {
368    table.iter().find(|(k, _)| *k == key).map(|(_, c)| c)
369}

371/// Reference URL for a CWE id such as `CWE-89`.
372#[must_use]
373pub fn cwe_url(cwe: &str) -> String {
374    let n = cwe.trim_start_matches("CWE-");
375    format!("https://cwe.mitre.org/data/definitions/{n}.html")
376}

383    fn every_card_has_cwe_and_clean_id() {
384        for c in all_cards() {
385            assert!(!c.cwe.is_empty(), "{} has no CWE", c.id);
386            for w in c.cwe {
387                assert!(
388                    w.starts_with("CWE-") && w[4..].chars().all(|d| d.is_ascii_digit()),
389                    "{}: bad CWE {w}",
390                    c.id

392            }

396                c.id

398            assert!(
399                c.id.chars()
400                    .all(|ch| ch.is_ascii_lowercase() || ch == '.' || ch == '-'),
401                "{} must be lowercase kebab",
402                c.id
403            );
404            assert!(!c.title.is_empty() && !c.why.is_empty() && !c.fix.is_empty());
405        }
