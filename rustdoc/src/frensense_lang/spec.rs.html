4//!
5//! The [`LanguageSpec`] trait is the single contract every language must
6//! satisfy. All engine subsystems (fingerprint, CFG, def-use, flow-fingerprint,
7//! import resolver, semantic provider) call *this* trait instead of containing
8//! their own hardcoded `match kind { "call_expression" | … }` arms.
9//!
10//! # Why one trait, not many?
11//!
12//! Previous analysis showed nine separate hardcoding sites. Splitting the fix
13//! across nine small traits creates nine places a new language author can
14//! forget to implement one. A single trait with good defaults makes the
15//! "forgot to implement" case a compile error, not a silent empty result.
16//!
17//! # The three layers of grammar knowledge
18//!
19//! A subsystem asks the spec at whichever layer matches its question, from
20//! coarsest to finest:
21//!
22//! 1. **Roles** - [`LanguageSpec::classify`] answers "what is this node?" with a
23//!    [`NodeRole`], field names included. Use it when one answer drives one
24//!    branch (`Call`, `Return`, `Composite`, ...).
25//! 2. **Predicates** - `is_*` queries answer a yes/no question the analysis
26//!    asks about a node it already holds: [`LanguageSpec::is_statement_node`],
27//!    [`LanguageSpec::is_block_node`], [`LanguageSpec::is_case_clause`],
28//!    [`LanguageSpec::is_jump_target_name`],
29//!    [`LanguageSpec::is_function_value`],
30//!    [`LanguageSpec::is_discard_pattern`], and the rest. They exist because a
31//!    role is sometimes too narrow (one grammar spells one concept several
32//!    ways) or too broad, while a lowering site only needs the boolean. Every
33//!    predicate defaults to the cross-language superset, so a provider overrides
34//!    only what its grammar spells differently.
35//! 3. **Tables** - [`LanguageSpec::jump_kind`],
36//!    [`LanguageSpec::literal_shape`] and [`LanguageSpec::param_bindings`] map a
37//!    kind to structured data (which jump, which literal shape, which binding),
38//!    for walkers that dispatch on the answer instead of branching on a
39//!    boolean.
40//!
41//! The rule for contributors: **the engine may not spell a node kind.** When a
42//! decision needs one, add a layer-1/2/3 query here with a default that
43//! reproduces today's behaviour, override it per grammar where the spelling
44//! differs, and pin both in `providers/node_vocabulary_tests.rs`.

52/// What role does a tree-sitter node play in the language?
53///
54/// Variants carry the **field names** needed to walk child nodes so callers
55/// never need a second lookup. All `*_field` values are `Cow<'static, str>`:
56/// `Borrowed` for spec vocabulary (tree-sitter grammar constants, no
57/// allocation), `Owned` for bundle-learned roles (allocated once per fact,
58/// never leaked).
59#[derive(Debug, Clone, PartialEq, Eq)]
60pub enum NodeRole {
61    // ── Definitions ──────────────────────────────────────────────────────
62    /// A named or anonymous function / method / closure.
63    Function {
64        /// True when the grammar distinguishes methods from top-level functions
65        /// (TS `method_definition`, Go `method_declaration`).
66        is_method: bool,
67        /// Field holding the function's name identifier, if any.
68        /// `None` for arrow functions, lambdas, anonymous closures.
69        name_field: Option<Cow<'static, str>>,
70        /// Field holding the parameter list node.
71        params_field: Cow<'static, str>,
72        /// Field holding the body block node.
73        body_field: Cow<'static, str>,
74    },
75
76    // ── Assignments / declarations ────────────────────────────────────────
77    /// A variable declaration with an initializer: `let x = expr` (JS/Rust),
78    /// `x := expr` (Go), and Python's `x = expr`, which has no separate
79    /// declaration form.
80    Declaration {
81        /// Field holding the declared name (or binding pattern).
82        name_field: Cow<'static, str>,
83        /// Field holding the initializer expression.
84        value_field: Cow<'static, str>,
85    },
86    /// A mutation of an existing binding (`x = expr`, `x += expr`).
87    ///
88    /// Only the grammars that spell mutation as its own node kind use this;
89    /// Python's plain `x = expr` is a [`NodeRole::Declaration`].
90    Assignment {
91        /// Field holding the assigned target.
92        lhs_field: Cow<'static, str>,
93        /// Field holding the assigned value expression.
94        rhs_field: Cow<'static, str>,
95    },
96
97    // ── Calls ─────────────────────────────────────────────────────────────
98    /// Any function / method invocation.
99    Call {
100        /// Field holding the callee expression.
101        callee_field: Cow<'static, str>,
102        /// Field holding the argument list node.
103        args_field: Cow<'static, str>,
104    },
105    /// Member / field access producing a value (not a call).
106    /// `obj.field`, `obj->field`, `obj.attribute`.
107    MemberAccess {
108        /// Field holding the accessed object.
109        object_field: Cow<'static, str>,
110        /// Field holding the accessed property name.
111        property_field: Cow<'static, str>,
112    },
113
114    // ── Control flow ──────────────────────────────────────────────────────
115    /// An `if` / `else if` / `else` chain, or a C-style `switch_statement`.
116    ///
117    /// Grammars whose switch/match dispatches on one subject are
118    /// [`NodeRole::Match`] instead, and a ternary expression is
119    /// [`NodeRole::Conditional`].
120    Branch,
121    /// Ternary conditional EXPRESSION (`a ? b : c`, Python `a if c else b`).
122    /// Separate from Branch: both arms are expressions producing a value,
123    /// lowered as branch + Phi, not as a statement-level fork.
124    Conditional,
125    /// A loop (`for`, `while`, `do`, `loop`): the CFG adds a back-edge.
126    Loop,
127    /// A `return` statement or expression.
128    Return,
129    /// A `try` block: the exception edge starts here.
130    Try,
131    /// An exception handler (`catch` clause, Python `except` clause).
132    Catch,
133    /// A `finally` block.
134    Finally,
135    /// A `throw` statement or expression (Python `raise`).
136    Throw,
137
138    // ── Special language patterns ─────────────────────────────────────────
139    /// Go: `if err != nil { return … }` - structurally a Branch but semantically
140    /// an error propagation guard; the engine needs to detect it for auth-guard
141    /// dominator analysis.
142    ErrorGuard,
143    /// Python: `with expr as var:` - a context-manager entry; the callee
144    /// may be a path/file sink.
145    ContextManager,
146    /// Rust: the `?` postfix operator - propagates errors, terminates the
147    /// current scope if the value is `Err`.
148    ErrorPropagation,
149    /// An `await` expression (JS/TS, Python, Rust) or Go's `defer` / `go`
150    /// statement, whose work runs outside the current evaluation.
151    Await,
152
153    // ── Structural ────────────────────────────────────────────────────────
154    /// A block of statements (`{ … }`, an indented block).
155    Block,
156    /// Value-context composite literal: JS/TS `object` used as a value (not a
157    /// statement block), `array`, `object_pattern` in expression position.
158    /// Uniformly-keyed objects and arrays lower to an allocation with one
159    /// store per property/element: field reads resolve through
160    /// field-sensitive heap edges (taint in one property does not pollute
161    /// reads of another), while whole-container consumers see every stored
162    /// value. Objects with unkeyed children (spreads) keep the legacy
163    /// may-analysis union - conservative for taint.
164    Composite,
165    /// An import declaration or call: `import`, `use`, `require`.
166    Import,
167    /// An export declaration (JS/TS only).
168    Export,
169    /// A bare name reference.
170    Identifier,
171    /// A literal value: string, number, boolean or null.
172    Literal,
173
174    // ── Supplementary structural roles ────────────────────────────────────
175    /// A parameter list (`formal_parameters` in JS/TS, `parameters` in
176    /// Python, `parameter_list` in Go/C) or one parameter node a grammar
177    /// spells on its own (`parameter`, `parameter_declaration`,
178    /// `typed_parameter`, `default_parameter`, `positional_separator`, ...).
179    Parameters,
180    /// An argument list: `arguments` (JS/TS/Rust), `argument_list`
181    /// (Python/Go/C), Python's `generator_expression`, or a Rust macro's
182    /// `token_tree`.
183    Arguments,
184    /// A named call argument (`f(secure=False)`): Python `keyword_argument`.
185    /// Distinct from a plain expression so the IR can keep the argument's
186    /// *name* alongside its value - a policy that constrains one flag cannot
187    /// be expressed correctly by position when the caller may omit earlier
188    /// optional arguments.
189    KeywordArgument {
190        /// Field holding the argument name.
191        name_field: Cow<'static, str>,
192        /// Field holding the argument value expression.
193        value_field: Cow<'static, str>,
194    },
195    /// Class / struct / enum / interface / trait definition.
196    ClassDef,
197    /// Binary expression: `a + b`, `x && y`.
198    BinaryOp,
199    /// Unary expression: `!x`, `-x`, `*ptr`.
200    UnaryOp,
201    /// Pattern match: `match` (Rust), `switch` as type/value switch (Go),
202    /// `match` statement (Python), `switch_expression` (JS/TS).
203    Match,
204    /// Rust: `unsafe { … }` block.
205    Unsafe,
206    /// An async block (`async { … }` in Rust, Python `async_block`).
207    ///
208    /// A JS/TS `async function` / Python `async def` is still a
209    /// [`NodeRole::Function`]: only a block *without* a function of its own
210    /// gets this role.
211    AsyncBlock,
212
213    /// Anything the engine does not need to inspect.
214    Other,
215}

313/// Broad origin of tainted data.
314#[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
315pub enum TaintOrigin {
316    /// HTTP request body, query, path or header.
317    UserInput,
318    /// Process environment (`process.env`, `os.environ`, `std::env`).
319    Environment,
320    /// A file read whose path came from the user.
321    FileSystem,
322    /// A query result that may contain injected data.
323    Database,
324    /// An IPC or downstream API response.
325    Network,
326    /// A caller-defined origin, carried verbatim.
327    Custom(String),
328}
329
330impl std::fmt::Display for TaintOrigin {
331    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
332        match self {
333            Self::UserInput => write!(f, "user_input"),
334            Self::Environment => write!(f, "environment"),
335            Self::Database => write!(f, "database"),
336            Self::Network => write!(f, "network"),
337            Self::FileSystem => write!(f, "file_system"),
338            Self::Custom(s) => write!(f, "{}", s),
339        }
340    }
341}
342
343impl From<&str> for TaintOrigin {
344    fn from(s: &str) -> Self {
345        match s {
346            "user_input" | "user" => Self::UserInput,
347            "environment" | "env" => Self::Environment,
348            "database" | "db" => Self::Database,
349            "network" | "net" => Self::Network,
350            "file_system" | "fs" => Self::FileSystem,
351            _ => Self::Custom(s.to_string()),
352        }
353    }
354}

422/// A top-level item declared in a module.
423#[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
424pub struct ModuleItem {
425    /// The declared name, as written in source.
426    pub name: String,
427    /// What kind of item it is.
428    pub kind: ItemKind,
429    /// Its written or inferred visibility.
430    pub visibility: Visibility,
431}
432
433impl ModuleItem {
434    /// Project the `public` items of `items` onto their [`Export`] view.
435    ///
436    /// A language without re-export syntax (C, Go, Python, and Rust's root
437    /// surface) exports exactly its publicly visible declarations, so this
438    /// projection belongs to the shared types rather than to each provider:
439    /// every field of [`Export`] is already on [`ModuleItem`]. Languages that
440    /// *do* have re-export statements (JS/TS `export { x as y } from "…"`)
441    /// still derive exports from the AST, because a `ModuleItem` cannot say
442    /// which module an export came from.
443    ///
444    /// A provider whose language has no re-export syntax returns exactly this
445    /// projection from [`LanguageSpec::extract_exports`].
446    pub(crate) fn public_exports(items: impl IntoIterator<Item = Self>) -> Vec<Export> {
447        items
448            .into_iter()
449            .filter(|item| item.visibility == Visibility::Public)
450            .map(|item| Export {
451                exported_name: item.name.clone(),
452                local_name: item.name,
453                from_module: None,
454                visibility: item.visibility,
455                kind: item.kind,
456            })
457            .collect()
458    }
459}

463/// Everything the frensense engine needs to know about one source language.
464///
465/// Implement this once per language.  The engine never matches raw node-kind
466/// strings anywhere else.
467///
468/// # Object safety
469///
470/// The trait is object-safe: it can be stored as `Arc<dyn LanguageSpec>` in
471/// the [`LanguageRegistry`](crate::registry::LanguageRegistry).
472pub trait LanguageSpec: Send + Sync + 'static {
473    // ── Identity ─────────────────────────────────────────────────────────
474
475    /// Canonical short name used in fingerprints and the FRC bundle.
476    fn name(&self) -> &'static str;
477
478    /// File extensions handled by this spec (lowercase, no dot).
479    fn extensions(&self) -> &'static [&'static str];
480
481    /// The tree-sitter `Language` object for parsing.
482    fn tree_sitter_language(&self) -> tree_sitter::Language;
483
484    /// Rewrites the source into something the grammar can parse, run
485    /// immediately before lowering. The default is the identity: the
486    /// source is parsed as-is.
487    ///
488    /// **Position contract**: the rewrite MUST preserve every byte offset
489    /// — same total length, same newline layout — so tree-sitter nodes and
490    /// every line/column derived from them still point at the original
491    /// file. Single-file-container formats (Vue SFCs, and later Svelte or
492    /// Astro) use this to blank template markup into same-length runs of
493    /// spaces instead of extracting script text to a separate buffer.
494    fn preprocess<'a>(&self, source: &'a str) -> std::borrow::Cow<'a, str> {
495        std::borrow::Cow::Borrowed(source)
496    }
497
498    /// The language key the FactTable and learned bundles key this spec's
499    /// knowledge under. Defaults to [`name`](Self::name); a container
500    /// format delegating to another spec's grammar also delegates its
501    /// vocabulary (a Vue SFC's script is TypeScript, so its facts live in
502    /// the `typescript` section). Registry identity still uses `name()`.
503    fn fact_language(&self) -> &'static str {
504        self.name()
505    }
506
507    // ── Node classification ───────────────────────────────────────────────
508
509    /// Classify a tree-sitter node kind into a [`NodeRole`].
510    ///
511    /// This is the single hot-path entry point for all engine subsystems.
512    /// The returned variant carries the field names needed by the caller so
513    /// no second lookup is required.
514    ///
515    /// ```rust,ignore
516    /// match spec.classify(node.kind()) {
517    ///     NodeRole::Call { callee_field, args_field } => {
518    ///         let callee = node.child_by_field_name(callee_field);
519    ///         // …
520    ///     }
521    ///     NodeRole::Assignment { lhs_field, rhs_field } => { /* … */ }
522    ///     _ => {}
523    /// }
524    /// ```
525    fn classify(&self, kind: &str) -> NodeRole;
526
527    /// Index of the condition among a ternary expression's named children.
528    ///
529    /// JS/TS `a ? b : c` → `[cond, then, else]`, condition at 0.
530    /// Python `then if cond else else_arm` → `[then, cond, else]`,
531    /// condition at 1 (middle). Defaults to 0.
532    fn ternary_cond_index(&self) -> usize {
533        0
534    }
535
536    /// Is this node kind a *property-name* (a named field access) rather
537    /// than a computed index?
538    ///
539    /// Used by the lowering to distinguish `obj.field` (LoadField, the name
540    /// matters for member-path source matching and sink method resolution)
541    /// from `obj[expr]` (LoadElement). The default heuristic recognizes the
542    /// common JS/TS kinds; languages whose attribute/selector property nodes
543    /// have different grammar kinds MUST override this:
544    ///
545    /// - Python `attribute`: the property child is a plain `identifier`
546    /// - Go `selector_expression` / Rust `field_expression`: `field_identifier`
547    ///
548    /// A wrong answer silently turns method calls into CallPointer sites and
549    /// loses sink matching entirely.
550    fn is_property_kind(&self, kind: &str) -> bool {
551        kind.contains("property")
552            || kind == "property_identifier"
553            || kind == "shorthand_property_identifier"
554            || kind == "private_property_identifier"
555    }
556
557    /// Returns `true` if this node is the entry node for a function definition.
558    ///
559    /// Convenience wrapper around [`classify`](Self::classify); the default
560    /// implementation delegates.  Override only when a language has edge-cases
561    /// (e.g. Python `decorated_definition` which wraps the real
562    /// `function_definition`).
563    fn is_function_node(&self, kind: &str) -> bool {
564        matches!(self.classify(kind), NodeRole::Function { .. })
565    }
566
567    /// Returns `true` if this node kind represents a type cast or type assertion (e.g. `x as T`, `(T)x`).
568    fn is_cast(&self, _kind: &str) -> bool {
569        false
570    }
571
572    /// Decorator / attribute call names applied to this function-definition
573    /// node (`@guard(...)`, `#[guard]`, `@app.route(...)`).
574    ///
575    /// Pure grammar mechanism: decorators are placed either as *direct
576    /// children* of the definition (Python `decorated_definition`, JS/TS
577    /// member decorators) or as *preceding sibling attribute nodes*
578    /// (Rust `#[attr]` on `fn`). Their text reliably starts with `@`
579    /// or `#[`, so placement and spelling are recognized structurally
580    /// and no per-language node-kind table is needed. The name is
581    /// stripped to its last `.` segment and cut at the call paren so
582    /// `@group.require_role(user)` yields `require_role`.
583    fn decorator_guard_calls<'a>(
584        &self,
585        node: tree_sitter::Node<'a>,
586        source: &'a str,
587    ) -> Vec<&'a str> {
588        let mut out = Vec::new();
589        let mut visit = |n: tree_sitter::Node<'a>| {
590            if let Some(name) = decorator_call_name(node_text(n, source)) {
591                out.push(name);
592            }
593        };
594        let mut cursor = node.walk();
595        for child in node.children(&mut cursor) {
596            visit(child);
597        }
598        let mut prev = node;
599        while let Some(sib) = prev.prev_sibling() {
600            if is_attribute_text(node_text(sib, source)) {
601                visit(sib);
602            }
603            prev = sib;
604        }
605        out
606    }
607
608    /// Returns `true` if this node kind represents a string template or interpolated string (e.g. `template_string`, `f_string`).
609    fn is_template_string(&self, _kind: &str) -> bool {
610        false
611    }
612
613    /// Returns `true` if this node kind represents an untainted literal fragment of a template string (e.g. `string_fragment`).
614    fn is_template_literal_fragment(&self, _kind: &str) -> bool {
615        false
616    }
617
618    /// Returns `true` if this node kind represents a destructuring pattern (e.g. `object_pattern`, `array_pattern`).
619    fn is_destructuring_pattern(&self, _kind: &str) -> bool {
620        false

978    /// Unwraps declarator wrapper nodes (e.g. `pointer_declarator`, `expression_list`, etc.)
979    /// to locate the leaf name binding node.
980    fn unwrap_declarator_node<'a>(&self, node: Node<'a>) -> Node<'a> {
981        let mut cur = node;
982        if matches!(
983            cur.kind(),
984            "expression_list" | "identifier_list" | "expression_sequence"
985        ) {
986            let mut nc = cur.walk();
987            let named: Vec<_> = cur.named_children(&mut nc).collect();
988            if named.len() == 1 {
989                cur = named[0];
990            }
991        }
992        while let Some(inner) = match cur.kind() {
993            "pointer_declarator"
994            | "array_declarator"
995            | "parenthesized_declarator"
996            | "reference_declarator" => {
997                let mut wc = cur.walk();
998                let mut found = None;
999                for c in cur.children(&mut wc) {
1000                    if c.is_named() && c.kind() != "comment" {
1001                        found = Some(c);
1002                        break;
1003                    }
1004                }
1005                found
1006            }
1007            _ => None,
1008        } {
1009            cur = inner;
1010        }
1011        cur
1012    }

1016    /// The parameter-list node of a function definition, when the grammar
1017    /// exposes one.
1018    ///
1019    /// Follows the [`NodeRole::Function`]'s `params_field`, and keeps walking
1020    /// while the node it lands on is not a [`NodeRole::Parameters`] list. C
1021    /// needs that: its `function_definition` exposes `declarator`, and the
1022    /// list sits one hop further, inside `function_declarator`'s `parameters`
1023    /// field (`pointer_declarator` may wrap it again). Handling the hop here
1024    /// means no consumer carries a C-specific walk.
1025    fn function_params<'t>(&self, fn_node: Node<'t>) -> Option<Node<'t>> {
1026        let NodeRole::Function { params_field, .. } = self.classify(fn_node.kind()) else {
1027            return None;
1028        };
1029        let mut cur = fn_node.child_by_field_name(params_field.as_ref());
1030        while let Some(node) = cur {
1031            if matches!(self.classify(node.kind()), NodeRole::Parameters) {
1032                return Some(node);
1033            }
1034            cur = node
1035                .child_by_field_name("parameters")
1036                .or_else(|| node.child_by_field_name(params_field.as_ref()));
1037        }
1038        None
1039    }

1041    /// The node that names a function definition.
1042    ///
1043    /// Follows the [`NodeRole::Function`]'s `name_field` until it reaches an
1044    /// [`NodeRole::Identifier`]. A grammar that names the function inside a
1045    /// declarator chain (C: `function_definition → declarator →
1046    /// function_declarator → identifier`, with `pointer_declarator` in between
1047    /// for `void *f(void)`) says so by exposing that chain in `name_field`, so
1048    /// no consumer walks it.
1049    ///
1050    /// `None` for a function that binds no name (arrow function, lambda,
1051    /// closure, and a Python `decorated_definition`, whose name belongs to the
1052    /// wrapped definition).
1053    fn function_name_node<'t>(&self, fn_node: Node<'t>) -> Option<Node<'t>> {
1054        let NodeRole::Function { name_field, .. } = self.classify(fn_node.kind()) else {
1055            return None;
1056        };
1057        let field = name_field?;
1058        let mut cur = fn_node.child_by_field_name(field.as_ref());
1059        while let Some(node) = cur {
1060            if matches!(self.classify(node.kind()), NodeRole::Identifier) {
1061                return Some(node);
1062            }
1063            cur = node.child_by_field_name(field.as_ref());
1064        }
1065        None
1066    }

1215/// The binding identifiers of one parameter node, in declaration order.
1216///
1217/// Called through [`LanguageSpec::param_bindings`]; see that method for the
1218/// per-grammar shapes.
1219fn param_bindings<'t>(p: Node<'t>) -> Vec<Node<'t>> {
1220    match p.kind() {
1221        "required_parameter" | "optional_parameter" | "parameter" | "variadic_parameter" => {
1222            param_binding(p).into_iter().collect()
1223        }
1224        "identifier" => vec![p],
1225        "parameter_declaration" | "variadic_parameter_declaration" => {
1226            if let Some(decl) = p.child_by_field_name("declarator") {
1227                return find_param_identifier(decl).into_iter().collect();
1228            }
1229            let mut cursor = p.walk();
1230            p.children_by_field_name("name", &mut cursor).collect()
1231        }
1232        "typed_parameter"
1233        | "default_parameter"
1234        | "typed_default_parameter"
1235        | "list_splat_pattern"
1236        | "dictionary_splat_pattern" => python_param_binding(p).into_iter().collect(),
1237        _ => Vec::new(),
1238    }
1239}
1240
1241/// The binding node of a parameter: a plain identifier, or the INNER
1242/// identifier of a wrapper-shaped pattern (`Query(p)`, `{ body }`, `&mut x`).
1243fn param_binding<'t>(p: Node<'t>) -> Option<Node<'t>> {
1244    let pattern = p.child_by_field_name("pattern").or_else(|| p.child(0))?;
1245    // `...parts` wraps the name: the rest marker is not part of it. Binding
1246    // the parameter under the text `...parts` leaves the body's `parts` an
1247    // undeclared name, so a rest parameter's value never reaches its uses.
1248    if pattern.kind() == "rest_pattern" {
1249        return pattern.named_child(0);
1250    }
1251    Some(extractor_binding(pattern).unwrap_or(pattern))
1252}
1253
1254/// The binding identifier of a Python parameter node.
1255///
1256/// Python wraps annotated and defaulted parameters: `typed_parameter` has no
1257/// `name` field (its first child is the binding), while
1258/// `default_parameter`/`typed_default_parameter` expose `name`. A splat
1259/// parameter (`*args`, `**kwargs`, with or without an annotation) wraps the
1260/// identifier in `list_splat_pattern`/`dictionary_splat_pattern`. Only an
1261/// identifier is a binding: a tuple pattern (`def f((a, b))`) has no single
1262/// name, and naming the parameter after the pattern text would bind a name the
1263/// body never references.
1264fn python_param_binding<'t>(p: Node<'t>) -> Option<Node<'t>> {
1265    let binding = p.child_by_field_name("name").or_else(|| match p.kind() {
1266        "list_splat_pattern" | "dictionary_splat_pattern" => p.named_child(0),
1267        _ => p.child(0),
1268    })?;
1269    match binding.kind() {
1270        "identifier" => Some(binding),
1271        "list_splat_pattern" | "dictionary_splat_pattern" => {
1272            let mut cursor = binding.walk();
1273            let splat = binding
1274                .named_children(&mut cursor)
1275                .find(|c| c.kind() == "identifier");
1276            splat
1277        }
1278        _ => None,
1279    }
1280}
1281
1282/// `Query(p)` (and `State(db)`, `Json(body)`) is a *named* pattern: it has a
1283/// `type` field naming the wrapper plus one binding inside. The binding - not
1284/// the wrapper - is what the body references, so the parameter is named after
1285/// it. Any other pattern (plain identifier, a destructuring pattern with no
1286/// `type` field, `&mut x`) is left alone.
1287fn extractor_binding<'t>(pattern: Node<'t>) -> Option<Node<'t>> {
1288    let wrapper = pattern.child_by_field_name("type")?;
1289    let mut cursor = pattern.walk();
1290    let mut inner = pattern
1291        .named_children(&mut cursor)
1292        .filter(|c| c.id() != wrapper.id());
1293    let first = inner.next()?;
1294    if first.kind() != "identifier" || inner.next().is_some() {
1295        return None;
1296    }
1297    Some(first)
1298}

1300/// Walk a C declarator chain to find the leaf `identifier`.
1301/// Handles `pointer_declarator`, `array_declarator`, etc.
1302fn find_param_identifier<'t>(node: Node<'t>) -> Option<Node<'t>> {
1303    match node.kind() {
1304        "identifier" => Some(node),
1305        _ => node
1306            .child_by_field_name("declarator")
1307            .and_then(find_param_identifier),
1308    }
1309}

1311/// Visit every node of `root`, depth-first and in pre-order (a node before its
1312/// children).
1313///
1314/// Every provider's import/export extractor wants the same thing — "look at
1315/// each node, match on its kind" — and the cursor dance to do it
1316/// (`goto_first_child` / `goto_next_sibling` / `goto_parent` with a labelled
1317/// outer loop) is identical in all of them. Open-coding it once per grammar is
1318/// pure repetition and easy to get subtly wrong (a missing `break 'outer`
1319/// revisits the root forever).
1320///
1321/// It is `pub(crate)`, not `pub`: a provider implementation detail, not part of
1322/// the contract downstream crates consume.
1323pub(crate) fn walk_preorder<'tree>(root: Node<'tree>, mut visit: impl FnMut(Node<'tree>)) {
1324    let mut cursor = root.walk();
1325    'outer: loop {
1326        visit(cursor.node());
1327        if cursor.goto_first_child() {
1328            continue;
1329        }
1330        loop {
1331            if cursor.goto_next_sibling() {
1332                break;
1333            }
1334            if !cursor.goto_parent() {
1335                break 'outer;
1336            }
1337        }
1338    }
1339}

1341/// Does a node's text look like a decorator or attribute definition
1342/// (`@...` or `#[...]`) rather than ordinary statements? Purely textual,
1343/// so placement styles differ per language without a kind table.
1344#[inline]
1345fn is_attribute_text(text: &str) -> bool {
1346    let t = text.trim();
1347    t.starts_with('@') || t.starts_with("#[")
1348}

1350/// Parse the call name out of decorator/attribute text:
1351/// `"@app.route("/x")"` → `route`, `"#[ensure_owner]"` → `ensure_owner`.
1352#[inline]
1353fn decorator_call_name(text: &str) -> Option<&str> {
1354    let t = text.trim();
1355    let inner = match (t.strip_prefix('@'), t.strip_prefix("#[")) {
1356        (Some(s), _) => s,
1357        (_, Some(s)) => s,
1358        (None, None) => return None,
1359    };
1360    let inner = inner.trim();
1361    // Rust `#[...]` closes with `]`; the name is everything before the
1362    // first paren or bracket.
1363    let end = inner.find(['(', '[', '{', ']']).unwrap_or(inner.len());
1364    let name = inner[..end].trim();
1365    if name.is_empty() {
1366        None
1367    } else {
1368        Some(call_last_segment(name))
1369    }
