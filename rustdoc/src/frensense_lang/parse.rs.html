4//!
5//! Source text to syntax tree, with the spec's grammar and preprocessing.
6//!
7//! A [`LanguageSpec`] knows which grammar a file uses and whether the source
8//! has to be rewritten first (a Vue SFC's markup is blanked so the TypeScript
9//! parser sees only its script blocks). Both steps must happen before any
10//! role, import or query answer is meaningful, and the rewrite must preserve
11//! byte offsets. Doing that by hand means every consumer repeats:
12//!
13//! ```text
14//! let rewritten = spec.preprocess(&text);
15//! let mut parser = tree_sitter::Parser::new();
16//! parser.set_language(&spec.tree_sitter_language())?;
17//! let tree = parser.parse(rewritten.as_ref(), None)?;
18//! ```
19//!
20//! [`parse`] is that sequence once. It does not hide tree-sitter: the tree is
21//! a [`Tree`], its nodes are [`Node`](tree_sitter::Node)s, and a caller still
22//! walks them. It removes the setup that has language knowledge in it - which
23//! grammar, and whether to preprocess - so the answer always comes from the
24//! spec.
25//!
26//! ```
27//! use frensense_lang::{parse, spec_for_ext};
28//!
29//! let spec = spec_for_ext("ts").expect("TypeScript is registered");
30//! let parsed = parse(spec, "const id = req.query.id;\n").expect("parses");
31//! assert!(!parsed.has_errors());
32//! assert_eq!(parsed.text(), "const id = req.query.id;\n");
33//! ```
34//!
35//! `parsed.text()` is the text the tree indexes into. Because the position
36//! contract in [`LanguageSpec::preprocess`] preserves offsets, those byte
37//! ranges also address the caller's original string.
38
39use std::borrow::Cow;
40
41use tree_sitter::Tree;
42
43use crate::spec::LanguageSpec;
44
45/// A parsed file: the tree and the exact text its byte ranges index into.
46///
47/// The text is the spec's preprocessed source. For most languages that is the
48/// input unchanged; for a single-file container it is the input with non-code
49/// regions blanked to spaces, at the same offsets.
50#[derive(Debug)]
51pub struct ParsedSource<'a> {
52    tree: Tree,
53    source: Cow<'a, str>,
54}
55
56impl ParsedSource<'_> {
57    /// The parsed text, as the tree sees it.
58    #[must_use]
59    pub fn text(&self) -> &str {
60        self.source.as_ref()
61    }
62
63    /// The tree's root node.
64    #[must_use]
65    pub fn root_node(&self) -> tree_sitter::Node<'_> {
66        self.tree.root_node()
67    }
68
69    /// Did the grammar report an incomplete parse?
70    ///
71    /// tree-sitter does not fail on a syntax error: it returns a tree
72    /// containing `ERROR` or missing nodes. A caller that renders findings
73    /// should say the file was only partially analyzed rather than clean.
74    #[must_use]
75    pub fn has_errors(&self) -> bool {
76        self.tree.root_node().has_error()
77    }
78}
79
80/// Why [`parse`] could not produce a tree.
81#[derive(Debug)]
82#[non_exhaustive]
83pub enum ParseError {
84    /// The grammar rejected the spec's `Language` object.
85    Grammar(tree_sitter::LanguageError),
86    /// tree-sitter returned no tree for the input, which means the parse was
87    /// cancelled.
88    Cancelled,
89}
90
91impl std::fmt::Display for ParseError {
92    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
93        match self {
94            Self::Grammar(err) => write!(f, "cannot use the spec's grammar: {err}"),
95            Self::Cancelled => f.write_str("the parse was cancelled"),
96        }
97    }
98}
99
100impl std::error::Error for ParseError {
101    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
102        match self {
103            Self::Grammar(err) => Some(err),
104            Self::Cancelled => None,
105        }
106    }
107}
108
109impl From<tree_sitter::LanguageError> for ParseError {
110    fn from(err: tree_sitter::LanguageError) -> Self {
111        Self::Grammar(err)
112    }
113}
114
115/// Parse `source` with `spec`'s grammar, after the spec's preprocessing.
116///
117/// # Errors
118///
119/// [`ParseError::Grammar`] when the grammar rejects the spec's `Language`
120/// object, [`ParseError::Cancelled`] when tree-sitter returns no tree.
121pub fn parse<'a>(spec: &dyn LanguageSpec, source: &'a str) -> Result<ParsedSource<'a>, ParseError> {
122    let rewritten = spec.preprocess(source);
123    let mut parser = tree_sitter::Parser::new();
124    parser.set_language(&spec.tree_sitter_language())?;
125    let tree = parser
126        .parse(rewritten.as_ref(), None)
127        .ok_or(ParseError::Cancelled)?;
128    Ok(ParsedSource {
129        tree,
130        source: rewritten,
131    })
132}

134#[cfg(all(test, feature = "typescript"))]
135mod tests {
136    use super::*;
137    use crate::registry::spec_for_ext;
138
139    #[test]
140    fn parses_with_the_specs_grammar() {
141        let spec = spec_for_ext("ts").expect("TypeScript is registered");
142        let parsed = parse(spec, "const id = req.query.id;\n").expect("parses");
143        assert!(!parsed.has_errors());
144        assert_eq!(parsed.text(), "const id = req.query.id;\n");
145        assert!(parsed.root_node().child_count() > 0);
146    }
147
148    #[test]
149    fn a_container_spec_parses_its_blanked_source() {
150        // Vue has no grammar of its own: the spec blanks markup so the
151        // TypeScript parser sees only the script block, at the same offsets.
152        let spec = spec_for_ext("vue").expect("Vue is registered");
153        let source = "<template><p>{{ x }}</p></template>\n<script>\nconst y = 1;\n</script>\n";
154        let parsed = parse(spec, source).expect("parses");
155        assert_eq!(parsed.text().len(), source.len());
156        assert!(!parsed.text().contains("<template>"));
157        assert!(parsed.text().contains("const y = 1;"));
158        assert!(!parsed.has_errors());
159    }
160
161    #[test]
162    fn a_broken_file_reports_errors_instead_of_failing() {
163        let spec = spec_for_ext("ts").expect("TypeScript is registered");
164        let parsed = parse(spec, "function ( {\n").expect("tree-sitter still returns a tree");
165        assert!(parsed.has_errors());
166    }
167}
