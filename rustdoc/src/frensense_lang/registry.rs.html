4//!
5//! A thread-safe registry that maps file extensions and language names to
6//! [`LanguageSpec`] implementations.
7//!
8//! ## Usage
9//!
10//! ```rust
11//! use frensense_lang::registry::LanguageRegistry;
12//!
13//! let reg = LanguageRegistry::global();
14//! let spec = reg.for_extension("ts").expect("TypeScript not registered");
15//! // spec is &dyn LanguageSpec - call any trait method
16//! ```
17//!
18//! ## Adding a new language
19//!
20//! 1. Create a new file in `src/providers/` implementing `LanguageSpec`.
21//! 2. Call `registry.register(Arc::new(YourProvider))` inside
22//!    `LanguageRegistry::build_default()`.
23//! 3. Done.  Every engine subsystem picks it up automatically.
24
25use std::sync::{Arc, OnceLock};
26
27use rustc_hash::FxHashMap;
28
29use crate::spec::LanguageSpec;
30
31// ── Registry ──────────────────────────────────────────────────────────────────
32
33/// Thread-safe registry of all known [`LanguageSpec`] implementations.
34pub struct LanguageRegistry {
35    by_ext: FxHashMap<&'static str, Arc<dyn LanguageSpec>>,
36    by_name: FxHashMap<&'static str, Arc<dyn LanguageSpec>>,
37}
38
39impl std::fmt::Debug for LanguageRegistry {
40    /// Lists the registered language names and extensions, sorted so the
41    /// output is deterministic (the specs themselves are trait objects).
42    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
43        let mut languages: Vec<&str> = self.by_name.keys().copied().collect();
44        languages.sort_unstable();
45        let mut extensions: Vec<&str> = self.by_ext.keys().copied().collect();
46        extensions.sort_unstable();
47        f.debug_struct("LanguageRegistry")
48            .field("languages", &languages)
49            .field("extensions", &extensions)
50            .finish()
51    }
52}
53
54impl LanguageRegistry {
55    // ── Global singleton ──────────────────────────────────────────────────
56
57    /// Returns the process-wide default registry, constructed once on first call.
58    ///
59    /// The default registry includes every language enabled via Cargo features.
60    /// See the private `build_default`.
61    pub fn global() -> &'static Self {
62        static INSTANCE: OnceLock<LanguageRegistry> = OnceLock::new();
63        INSTANCE.get_or_init(Self::build_default)
64    }
65
66    // ── Construction ─────────────────────────────────────────────────────
67
68    /// Build the default registry from all feature-gated providers.
69    fn build_default() -> Self {
70        let mut reg = Self {
71            by_ext: FxHashMap::default(),
72            by_name: FxHashMap::default(),
73        };
74
75        #[cfg(feature = "typescript")]
76        reg.register(Arc::new(crate::providers::javascript::TypeScriptSpec));
77
78        #[cfg(feature = "typescript")]
79        reg.register(Arc::new(crate::providers::javascript::VueSpec));
80
81        #[cfg(feature = "javascript")]
82        reg.register(Arc::new(crate::providers::javascript::JavaScriptSpec));
83
84        #[cfg(feature = "go")]
85        reg.register(Arc::new(crate::providers::go::GoSpec));
86
87        #[cfg(feature = "python")]
88        reg.register(Arc::new(crate::providers::python::PythonSpec));
89
90        #[cfg(feature = "rust-grammar")]
91        reg.register(Arc::new(crate::providers::rust_lang::RustSpec));
92
93        #[cfg(feature = "c-grammar")]
94        reg.register(Arc::new(crate::providers::c_lang::CSpec));
95
96        reg
97    }
98
99    /// Register a [`LanguageSpec`] for all extensions it declares.
100    ///
101    /// Call this before [`global`](Self::global) is first accessed if you need
102    /// to add custom providers (e.g. in tests or downstream crates).
103    pub fn register(&mut self, spec: Arc<dyn LanguageSpec>) {
104        for &ext in spec.extensions() {
105            self.by_ext.insert(ext, Arc::clone(&spec));
106        }
107        self.by_name.insert(spec.name(), spec);
108    }
109
110    // ── Lookups ───────────────────────────────────────────────────────────
111
112    /// Look up the spec for a file extension (lowercase, no leading dot).
113    ///
114    /// Returns `None` when the extension is unknown - the engine should skip
115    /// or fall back gracefully.
116    #[must_use]
117    pub fn for_extension(&self, ext: &str) -> Option<&dyn LanguageSpec> {
118        self.by_ext.get(ext).map(Arc::as_ref)
119    }
120
121    /// Look up the spec by the canonical language name.
122    #[must_use]
123    pub fn for_name(&self, name: &str) -> Option<&dyn LanguageSpec> {
124        self.by_name.get(name).map(Arc::as_ref)
125    }
126
127    /// Look up the spec for a file path, extracting the extension automatically.
128    #[must_use]
129    pub fn for_path(&self, path: &std::path::Path) -> Option<&dyn LanguageSpec> {
130        let ext = path.extension()?.to_str()?.to_lowercase();
131        self.for_extension(ext.as_str())
132    }
133
134    /// All registered language names.
135    pub fn language_names(&self) -> impl Iterator<Item = &'static str> + '_ {
136        self.by_name.keys().copied()
137    }
138
139    /// All registered language specs (deduplicated by language name).
140    pub fn all_specs(&self) -> impl Iterator<Item = &dyn LanguageSpec> + '_ {
141        self.by_name.values().map(|arc| arc.as_ref())
142    }
143}
144
145// ── Convenience free functions ─────────────────────────────────────────────────
146
147/// Convenience: `spec_for_ext("ts")` → `Some(&dyn LanguageSpec)`.
148#[inline]
149pub fn spec_for_ext(ext: &str) -> Option<&'static dyn LanguageSpec> {
150    LanguageRegistry::global().for_extension(ext)
151}
152
153/// Convenience: `spec_for_path(Path::new("src/main.rs"))`.
154#[inline]
155pub fn spec_for_path(path: &std::path::Path) -> Option<&'static dyn LanguageSpec> {
156    LanguageRegistry::global().for_path(path)
157}
158
159/// Convenience: iterate all registered language specs.
160#[inline]
161pub fn all_specs() -> impl Iterator<Item = &'static dyn LanguageSpec> {
162    LanguageRegistry::global().all_specs()
163}
