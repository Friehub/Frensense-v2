5//! The embedded default pack (Phase 5.2 / decision D1, extended in Phase 6).
6//!
7//! `frensense-lang`'s bootstrap policy tables re-shipped as a `.frc`
8//! bundle, embedded in the bundler and merged between the spec seed and
9//! the consumer bundle (seeding order: spec -> default pack -> consumer,
10//! last-wins). Both consumers - the bundler gate's per-family tables and
11//! the CLI scan runner - merge it with [`Provenance::Spec`], so consumer
12//! bundle facts still win and provenance reporting stays truthful.
13//!
14//! Since Phase 6.3a the pack is *generated* by the `frensense-packgen`
15//! crate (which owns the vocabularies); this module only embeds the
16//! committed asset via `include_bytes!` and parses it lazily. The drift
17//! test lives in packgen; `packgen --check` guards it in CI.
18
19use std::sync::OnceLock;
20
21use super::types::{load_bundle, LoadedBundle};
22use frensense_engine::analysis::taint::facts::LearnedFactEntry;
23
24/// Whether `name` is one of the default pack's memory-contract functions:
25/// proposals must not re-propose vocabulary the pack already ships.
26/// Derived from the loaded pack (the shipped truth) rather than the
27/// generator's source tables, so the two can never disagree.
28pub fn is_pack_memory_builtin(name: &str) -> bool {
29    default_pack()
30        .learned_facts
31        .iter()
32        .any(|e| matches!(e, LearnedFactEntry::MemoryContract { name: n, .. } if n == name))
33}
34
35/// Bytes of the committed `assets/frensense-default.frc`.
36pub fn default_bundle_bytes() -> &'static [u8] {
37    include_bytes!("../../assets/frensense-default.frc")
38}
39
40/// BLAKE3 digest of the committed default pack, hex-encoded.
41///
42/// Part of the build identity (`--version --json`): two builds that claim
43/// the same engine version but embed different packs are different
44/// artifacts, and this is how a consumer or the release pipeline tells
45/// them apart.
46#[must_use]
47pub fn default_pack_blake3_hex() -> String {
48    blake3::hash(default_bundle_bytes()).to_hex().to_string()
49}
50
51/// The embedded default pack, parsed once. Panics only if the committed
52/// asset is stale or corrupt - the drift test in this module keeps that
53/// from reaching a release.
54pub fn default_pack() -> &'static LoadedBundle {
55    static PACK: OnceLock<LoadedBundle> = OnceLock::new();
56    PACK.get_or_init(|| {
57        load_bundle(default_bundle_bytes())
58            .unwrap_or_else(|e| panic!("embedded default pack failed to load: {e}"))
59    })
60}

62#[cfg(test)]
63mod tests {
64    use frensense_engine::analysis::taint::config::TaintConfig;
65    use frensense_engine::analysis::taint::facts::{
66        apply_language_entries, FactTable, LearnedFactEntry,
67    };
68
69    /// The wildcard group installs before language groups, and a language
70    /// absent from the scan's language set is filtered out entirely.
71    #[test]
72    fn language_entries_filter_by_language_with_wildcard_first() {
73        let entries = vec![
74            LearnedFactEntry::LanguageSink {
75                language: "*".to_string(),
76                call: "shared.eval".to_string(),
77                role: "execution".to_string(),
78            },
79            LearnedFactEntry::LanguageSink {
80                language: "go".to_string(),
81                call: "os/exec.Command".to_string(),
82                role: "execution".to_string(),
83            },
84        ];
85        let mut config = TaintConfig::default();
86        let mut facts = FactTable::default();
87        apply_language_entries(&mut config, &mut facts, &entries, &["typescript"]);
88        assert!(
89            config.sinks.contains("eval"),
90            "wildcard sink installs for any language"
91        );
92        assert!(
93            !config.sinks.contains("Command"),
94            "go-only sink filtered out of a typescript scan"
95        );
96    }
97}
