5//! # FRC - Frensense Reference Corpus Bundle Format
6//!
7//! A pre-compiled binary format that embeds corpus fingerprints.
8//! This crate handles the `.frc` file envelope (header, versioning, checksum)
9//! while remaining generic over the payload data structure.
10//!
11//! ## Compatibility (Phase 6)
12//!
13//! A bundle outlives the binary that built it, and it stores `LearnedFactEntry`
14//! and `HintKind` values by *variant position*. So a bundle is only decodable
15//! by an engine that speaks the same vocabulary, and the envelope now says so
16//! explicitly instead of leaving the reader to discover it as a serde error:
17//!
18//! * [`BundleHeader::vocabulary`] - the
19//!   [`fact_vocabulary_fingerprint`](frensense_engine::analysis::taint::facts::fact_vocabulary_fingerprint)
20//!   of the engine that wrote it. Checked strictly on load: a mismatch is
21//!   refused with both values named.
22//! * [`BundleHeader::requires_engine`] - an optional coarse range, for
23//!   corpus authors who want a policy bound ("this pack targets 0.9") on top
24//!   of the strict vocabulary check.
25//! * [`BundleHeader::built_by_engine`] - diagnostics only.
26//!
27//! Version 6 bundles carry none of this and still load: the fingerprint field
28//! is absent, so there is nothing to compare, and the reader treats them as
29//! pre-Phase-6 (see [`BundleHeader::is_legacy`]). `docs/FORMATS.md` carries
30//! the table of every artifact and its bump rule.
31
32#![allow(clippy::module_name_repetitions)]

36/// Envelope version. Bumped whenever the header or the payload gains a field
37/// or a variant, because bincode is positional and cannot self-describe.
38///
39/// ## Bump rule
40///
41/// Adding *any* serialized field or enum variant to the header, the payload,
42/// or the fact-entry vocabulary requires a bump here. The engine's
43/// `FACT_VOCABULARY_FINGERPRINT` ratchet catches vocabulary movement; this
44/// number catches layout movement. `docs/FORMATS.md` records both.
45pub const BUNDLE_VERSION: u32 = 8;
46
47pub const BUNDLE_MAGIC: &[u8; 4] = b"FRC1";
48
49/// The oldest bundle version this reader can still decode. Older envelopes
50/// are rejected with an actionable message rather than a parse error.
51pub const MIN_SUPPORTED_VERSION: u32 = 4;

53/// A semver-ish bound on the engine a bundle may be loaded by.
54///
55/// Expressed as `[major, minor, patch]` triples rather than parsed semver
56/// strings, because this type is on the wire: fixed-width arrays bincode
57/// encodes without allocation, and there is no string to parse (or to get
58/// wrong) at load time. Pre-release/build metadata is deliberately ignored -
59/// `0.9.0-beta` and `0.9.0` are the same series for compatibility purposes.
60#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
61pub struct EngineRequirement {
62    /// Inclusive lower bound on the engine version.
63    pub min: [u32; 3],
64    /// Exclusive upper bound. `None` = unbounded above.
65    pub max_exclusive: Option<[u32; 3]>,
66}
67
68impl EngineRequirement {
69    /// The requirement satisfied by every engine in the same minor series as
70    /// `version`: `[major, minor, 0] <= v < [major, minor + 1, 0]`.
71    ///
72    /// A minor bump in 0.x may add fact variants, which invalidates bundles;
73    /// a patch bump may not. So "same series" is the conservative default a
74    /// corpus author reaches for.
75    #[must_use]
76    pub fn same_series(version: [u32; 3]) -> Self {
77        Self {
78            min: [version[0], version[1], 0],
79            max_exclusive: Some([version[0], version[1] + 1, 0]),
80        }
81    }
82
83    /// Whether an engine version satisfies this bound.
84    #[must_use]
85    pub fn accepts(&self, engine: [u32; 3]) -> bool {
86        let above_min = engine >= self.min;
87        let below_max = self.max_exclusive.is_none_or(|max| engine < max);
88        above_min && below_max
89    }
90
91    /// Parse `major.minor.patch`, ignoring any `-pre` / `+build` suffix.
92    /// Returns `None` when the string is not a version, so a caller can fall
93    /// back to the strict vocabulary check rather than guessing.
94    #[must_use]
95    pub fn parse_triple(version: &str) -> Option<[u32; 3]> {
96        let numeric = version.split(['-', '+']).next().unwrap_or(version);
97        let mut parts = numeric.split('.');
98        let next = |parts: &mut core::str::Split<'_, char>| parts.next()?.parse::<u32>().ok();
99        let major = next(&mut parts)?;
100        let minor = next(&mut parts)?;
101        let patch = next(&mut parts).unwrap_or(0);
102        Some([major, minor, patch])
103    }
104}
105
106impl fmt::Display for EngineRequirement {
107    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
108        write!(f, ">={}.{}.0", self.min[0], self.min[1])?;
109        if let Some(max) = self.max_exclusive {
110            write!(f, ",<{}.{}.0", max[0], max[1])?;
111        }
112        Ok(())
113    }
114}

116/// The `.frc` envelope header.
117///
118/// Fields 0..4 are the version-6 layout and are unchanged; the trailing three
119/// are version 7. `read_bundle_parts` branches on `version` before decoding so
120/// that a v6 header is never handed to the v7 shape.
121#[derive(serde::Serialize, serde::Deserialize, Debug, Clone, PartialEq, Eq)]
122pub struct BundleHeader {
123    pub magic: [u8; 4],
124    pub version: u32,
125    pub pattern_count: u32,
126    pub checksum: [u8; 32],
127    /// Fact-vocabulary fingerprint of the engine that wrote this bundle.
128    /// `None` on a v6 (pre-Phase-6) bundle.
129    #[serde(default)]
130    pub vocabulary: Option<u64>,
131    /// Coarse engine range this bundle targets. Optional: the vocabulary
132    /// fingerprint is the strict check; this is policy for corpus authors.
133    #[serde(default)]
134    pub requires_engine: Option<EngineRequirement>,
135    /// Engine version string at write time. Diagnostics only.
136    #[serde(default)]
137    pub built_by_engine: Option<String>,
138    /// Where this bundle came from and what it may ship (v8+).
139    #[serde(default)]
140    pub trust: BundleTrust,
141}
142
143/// Where a bundle came from, and therefore which fact kinds it may ship.
144///
145/// - [`Certified`](Self::Certified): first-party, replay-gated; may ship
146///   sanitizers and Stable policies.
147/// - [`Community`](Self::Community): third-party; may ship sanitizers only
148///   when they are class-scoped and name-scoped, and policies only at the
149///   Hygiene tier.
150/// - [`Custom`](Self::Custom): locally assembled or remotely fetched; may
151///   only add sinks, sources and policies, never sanitizers.
152#[derive(serde::Serialize, serde::Deserialize, Debug, Clone, Copy, PartialEq, Eq, Default)]
153#[serde(rename_all = "snake_case")]
154pub enum BundleTrust {
155    /// First-party, replay-gated corpus bundle.
156    Certified,
157    /// Third-party community bundle.
158    #[default]
159    Community,
160    /// Locally assembled or remotely fetched bundle.
161    Custom,
162}
163
164impl BundleTrust {
165    /// Does this tier permit shipping sanitizer facts at all?
166    #[must_use]
167    pub fn may_ship_sanitizers(self) -> bool {
168        !matches!(self, Self::Custom)
169    }
170
171    /// May a sanitizer of this scope ship under this tier?
172    #[must_use]
173    pub fn may_ship_sanitizer(self, classes: &[String], package: Option<&str>) -> bool {
174        match self {
175            Self::Certified => true,
176            // Community: class-scoped and name-scoped (package/receiver).
177            Self::Community => !classes.is_empty() && package.is_some(),
178            Self::Custom => false,
179        }
180    }
181
182    /// May a policy of this tier ship at `tier`? Only certified bundles may
183    /// publish default-on Stable policies; the others are capped at Hygiene.
184    #[must_use]
185    pub fn may_ship_policy_tier(self, tier: &str) -> bool {
186        match self {
187            Self::Certified => true,
188            Self::Community | Self::Custom => tier.eq_ignore_ascii_case("hygiene"),
189        }
190    }
191}

213/// The v6 header, kept only so old bundles keep loading. Do not use for new
214/// writes.
215///
216/// `Serialize` is derived (not just `Deserialize`) so a test can fabricate a
217/// genuine pre-v7 envelope: hand-writing the four-field bytes is how we
218/// exercise the version branch without depending on a real v6 corpus file.
219#[derive(serde::Serialize, serde::Deserialize)]
220struct BundleHeaderV6 {
221    #[allow(dead_code)]
222    magic: [u8; 4],
223    #[allow(dead_code)]
224    version: u32,
225    #[allow(dead_code)]
226    pattern_count: u32,
227    #[allow(dead_code)]
228    checksum: [u8; 32],
229}

231impl BundleHeader {
232    /// Header for a v8 write, recording the current vocabulary fingerprint,
233    /// the engine that built it, and the certified trust tier.
234    #[must_use]
235    pub fn current(pattern_count: u32, checksum: [u8; 32]) -> Self {
236        Self {
237            magic: *BUNDLE_MAGIC,
238            version: BUNDLE_VERSION,
239            pattern_count,
240            checksum,
241            vocabulary: Some(frensense_engine::analysis::taint::facts::FACT_VOCABULARY_FINGERPRINT),
242            requires_engine: None,
243            built_by_engine: Some(frensense_engine::VERSION.to_string()),
244            trust: BundleTrust::Certified,
245        }
246    }
247
248    /// A v6-shaped header, for tests and for round-tripping an existing
249    /// bundle's metadata through a rewrite.
250    #[must_use]
251    pub fn legacy(pattern_count: u32, checksum: [u8; 32]) -> Self {
252        Self {
253            magic: *BUNDLE_MAGIC,
254            version: 6,
255            pattern_count,
256            checksum,
257            vocabulary: None,
258            requires_engine: None,
259            built_by_engine: None,
260            trust: BundleTrust::Community,
261        }
262    }
263
264    /// Whether this header predates the vocabulary fingerprint. Such a bundle
265    /// loads (nothing to compare) but is not pinned to an engine.
266    #[must_use]
267    pub fn is_legacy(&self) -> bool {
268        self.vocabulary.is_none()
269    }
270}

272/// Serialize a payload into the .frc binary format, recording the current
273/// engine's fact-vocabulary fingerprint.
274///
275/// Use [`write_bundle_pinned`] when the bundle should also carry a coarse
276/// engine range for distribution.
277pub fn write_bundle<T: serde::Serialize>(
278    payload: &T,
279    pattern_count: u32,
280) -> Result<Vec<u8>, String> {
281    write_bundle_with_header(payload, BundleHeader::current(pattern_count, [0; 32]))
282}
283
284/// Like [`write_bundle`], but stamps an engine range so a consumer can refuse
285/// the bundle before decoding it. This is what corpus-repo publishing should
286/// use: the range names the engine the families were replay-gated against.
287pub fn write_bundle_pinned<T: serde::Serialize>(
288    payload: &T,
289    pattern_count: u32,
290    requires_engine: EngineRequirement,
291) -> Result<Vec<u8>, String> {
292    write_bundle_pinned_with_trust(
293        payload,
294        pattern_count,
295        requires_engine,
296        BundleTrust::Certified,
297    )
298}
299
300/// [`write_bundle_pinned`] with an explicit trust tier. A `Community` bundle
301/// is subject to the loader's tier rules (class- and name-scoped sanitizers,
302/// policies capped at Hygiene).
303pub fn write_bundle_pinned_with_trust<T: serde::Serialize>(
304    payload: &T,
305    pattern_count: u32,
306    requires_engine: EngineRequirement,
307    trust: BundleTrust,
308) -> Result<Vec<u8>, String> {
309    let mut header = BundleHeader::current(pattern_count, [0; 32]);
310    header.requires_engine = Some(requires_engine);
311    header.trust = trust;
312    write_bundle_with_header(payload, header)
313}
314
315fn write_bundle_with_header<T: serde::Serialize>(
316    payload: &T,
317    mut header: BundleHeader,
318) -> Result<Vec<u8>, String> {
319    let data = bincode::serialize(payload).map_err(|e| e.to_string())?;
320    let checksum = blake3::hash(&data);
321    header.checksum = *checksum.as_bytes();
322
323    let mut output = Vec::new();
324    let header_bytes = bincode::serialize(&header).map_err(|e| e.to_string())?;
325
326    // Write header length as 4-byte LE
327    output.extend_from_slice(&(header_bytes.len() as u32).to_le_bytes());
328    output.extend_from_slice(&header_bytes);
329    output.extend_from_slice(&data);
330
331    Ok(output)
332}

334/// Hand-write an envelope in the *pre-v7* header shape.
335///
336/// Test-only, for exercising the reader's version branching against a header
337/// that really is 4 fields wide - patching the version inside a v7 header
338/// would leave the extra fields on the wire and prove nothing.
339#[cfg(test)]
340pub(crate) fn write_legacy_envelope(version: u32, payload: &[u8]) -> Vec<u8> {
341    let header = BundleHeaderV6 {
342        magic: *BUNDLE_MAGIC,
343        version,
344        pattern_count: 0,
345        checksum: *blake3::hash(payload).as_bytes(),
346    };
347    let header_bytes = bincode::serialize(&header).expect("legacy header serializes");
348    let mut bytes = (header_bytes.len() as u32).to_le_bytes().to_vec();
349    bytes.extend_from_slice(&header_bytes);
350    bytes.extend_from_slice(payload);
351    bytes
352}

354/// Split the `.frc` envelope: verify magic, gate the version against
355/// [`BUNDLE_VERSION`], verify the checksum, and return the header with the
356/// payload bytes (not yet deserialized).
357///
358/// Compatibility is checked here, before any payload decode, so a bundle that
359/// cannot be understood is refused with a message naming both sides rather
360/// than surfacing as a serde error halfway through a scan.
361///
362/// The version gate rejects *newer* bundles with a clean error; which
363/// payload shape the bytes hold is the caller's decision (see the
364/// version-branched load in `format::load_bundle`).
365pub fn read_bundle_parts(bytes: &[u8]) -> Result<(BundleHeader, &[u8]), String> {
366    if bytes.len() < 4 {
367        return Err("Bundle too small (missing header length)".to_string());
368    }
369
370    let mut len_bytes = [0u8; 4];
371    len_bytes.copy_from_slice(&bytes[0..4]);
372    let header_len = u32::from_le_bytes(len_bytes) as usize;
373
374    if bytes.len() < 4 + header_len {
375        return Err("Bundle truncated (header incomplete)".to_string());
376    }
377
378    let header_data = &bytes[4..4 + header_len];
379
380    // Peek magic + version without committing to a header shape: they are the
381    // first two fields of every version, so the prefix decodes identically.
382    let (magic, version) = peek_header_prefix(header_data)?;
383    if magic != *BUNDLE_MAGIC {
384        return Err("Invalid magic bytes, expected FRC1".to_string());
385    }
386    if version > BUNDLE_VERSION {
387        return Err(format!(
388            "Unsupported bundle version {} (this engine supports up to {})",
389            version, BUNDLE_VERSION
390        ));
391    }
392    if version < MIN_SUPPORTED_VERSION {
393        return Err(format!(
394            "Bundle version {version} is no longer supported (oldest readable is \
395             {MIN_SUPPORTED_VERSION}); rebuild it with a current `frensense bundle`"
396        ));
397    }
398
399    let header: BundleHeader = if version == 7 {
400        // v7 layout: the pre-trust shape, widened with the conservative
401        // community default (no bundle written before v8 was certified).
402        let legacy: BundleHeaderV7 =
403            bincode::deserialize(header_data).map_err(|e| e.to_string())?;
404        BundleHeader {
405            magic: legacy.magic,
406            version: legacy.version,
407            pattern_count: legacy.pattern_count,
408            checksum: legacy.checksum,
409            vocabulary: legacy.vocabulary,
410            requires_engine: legacy.requires_engine,
411            built_by_engine: legacy.built_by_engine,
412            trust: BundleTrust::Community,
413        }
414    } else if version < 7 {
415        // v6-and-earlier layout: decode the frozen shape, then widen it.
416        let legacy: BundleHeaderV6 =
417            bincode::deserialize(header_data).map_err(|e| e.to_string())?;
418        BundleHeader {
419            magic: legacy.magic,
420            version: legacy.version,
421            pattern_count: legacy.pattern_count,
422            checksum: legacy.checksum,
423            vocabulary: None,
424            requires_engine: None,
425            built_by_engine: None,
426            trust: BundleTrust::Community,
427        }
428    } else {
429        bincode::deserialize(header_data).map_err(|e| e.to_string())?
430    };
431
432    let payload_data = &bytes[4 + header_len..];
433    let actual_checksum = blake3::hash(payload_data);
434    if actual_checksum.as_bytes() != &header.checksum {
435        return Err("Bundle checksum mismatch (corrupted data)".to_string());
436    }
437
438    check_engine_compatibility(&header)?;
439
440    Ok((header, payload_data))
441}

443/// Enforce the header's engine compatibility claims against this binary.
444///
445/// The vocabulary fingerprint is the strict check: it is derived from the
446/// serialized fact vocabulary itself, so it moves exactly when a bundle's
447/// bytes would stop decoding. The engine range is a coarser, author-declared
448/// policy bound on top of it.
449///
450/// # Errors
451/// Returns a message naming both sides whenever the bundle cannot be loaded
452/// by this engine.
453pub fn check_engine_compatibility(header: &BundleHeader) -> Result<(), String> {
454    let engine_version = frensense_engine::VERSION;
455    let engine_triple = EngineRequirement::parse_triple(engine_version);
456
457    if let Some(required) = header.requires_engine {
458        match engine_triple {
459            Some(triple) if required.accepts(triple) => {}
460            Some(triple) => {
461                return Err(format!(
462                    "Bundle requires frensense {required} but this engine is \
463                     {engine_version} ({}.{}.{}). Rebuild the bundle with a \
464                     matching engine, or upgrade.",
465                    triple[0], triple[1], triple[2]
466                ));
467            }
468            None => {
469                return Err(format!(
470                    "Bundle declares an engine requirement ({required}) but this \
471                     engine's version {engine_version} is not a parseable version; \
472                     refusing rather than guessing"
473                ));
474            }
475        }
476    }
477
478    // `built_by_engine` is diagnostics only: a bundle written by an older
479    // patch of the same series carries the same vocabulary fingerprint and is
480    // perfectly loadable, so nothing is checked against it here.
481    if let Some(recorded) = header.vocabulary {
482        let current = frensense_engine::analysis::taint::facts::FACT_VOCABULARY_FINGERPRINT;
483        if recorded != current {
484            return Err(format!(
485                "Bundle was built for fact vocabulary {recorded:#018x} but this \
486                 engine speaks {current:#018x}. The engine's fact vocabulary has \
487                 changed since it was written; rebuild the bundle with this engine \
488                 (`frensense bundle <corpus> <out>.frc`)."
489            ));
490        }
491    }
492
493    Ok(())
494}

496/// Decode magic and version from a header without knowing its full shape.
497///
498/// Both are the first fields of every version of [`BundleHeader`], and bincode
499/// 1.x encodes them fixed-width little-endian, so the prefix is stable.
500fn peek_header_prefix(header_data: &[u8]) -> Result<([u8; 4], u32), String> {
501    if header_data.len() < 8 {
502        return Err(format!(
503            "Bundle header truncated ({} bytes, need at least 8)",
504            header_data.len()
505        ));
506    }
507    let mut magic = [0u8; 4];
508    magic.copy_from_slice(&header_data[0..4]);
509    let mut version_bytes = [0u8; 4];
510    version_bytes.copy_from_slice(&header_data[4..8]);
511    Ok((magic, u32::from_le_bytes(version_bytes)))
512}
513
514/// Deserialize a payload from the .frc binary format.
515pub fn read_bundle<T: serde::de::DeserializeOwned>(
516    bytes: &[u8],
517) -> Result<(BundleHeader, T), String> {
518    let (header, payload_data) = read_bundle_parts(bytes)?;
519    let payload: T = bincode::deserialize(payload_data).map_err(|e| e.to_string())?;
520    Ok((header, payload))
521}
