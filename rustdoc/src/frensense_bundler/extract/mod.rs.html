5//! Fact extraction from positive/negative corpus pairs (§9).
6//!
7//! Replaces the old shape/fingerprint pipeline. For every corpus family:
8//!
9//! 1. Scan positives and negatives through the **shared engine harness**
10//!    (`engine::data_flow::scan`) under the language specs of the family's
11//!    own extensions - exactly the tables the consumer CLI scans with.
12//! 2. Compute the delta: sink/sanitizer candidates from how taint moves (or
13//!    fails to move) through each variant.
14//! 3. **Replay gate**: apply candidate facts, re-scan the family; publish a
15//!    fact only if the family separates (positives alert, negatives don't)
16//!    and no other family's negatives newly alert.
17//! 4. Publish with `support` = number of variants that voted. support >= 2 is
18//!    `confirmed`, otherwise `provisional`.

47/// Full extraction: propose per family -> replay-gate -> merged learned table.
48///
49/// Production path: every family is proposed, baselined and gate-trialed
50/// under exactly the language specs of its own files - the same tables the
51/// consumer CLI builds for those files ([`family_tables`]). Harness/CLI
52/// parity matters in both directions: a foreign-language rule must not
53/// silence a positive during learning, and a foreign-language sanitizer
54/// must not quiet a negative the CLI would flag.
55pub fn extract_facts(families: &[Family]) -> (FactTable, Vec<LearnedFact>) {
56    extract_facts_with(families, &family_tables)
57}

59/// Extraction under caller-supplied tables, identical for every family.
60/// Unit tests use this to exercise the gate with synthetic vocabularies no
61/// language spec declares; production goes through [`extract_facts`].
62pub fn extract_facts_with_tables(
63    families: &[Family],
64    config: &TaintConfig,
65    builtin: &FactTable,
66) -> (FactTable, Vec<LearnedFact>) {
67    extract_facts_with(families, &|_| (config.clone(), builtin.clone()))
68}

70/// The tables the consumer CLI will scan a family's files under: the
71/// language specs of the family's own extensions, plus the embedded
72/// default pack (Spec provenance - the CLI runner merges the same pack, so
73/// gate and scan replay under identical baselines).
74pub fn family_tables(f: &Family) -> (TaintConfig, FactTable) {
75    let exts: Vec<&str> = f
76        .positives
77        .iter()
78        .chain(f.negatives.iter())
79        .map(|(_, _, ext)| ext.as_str())
80        .collect();
81    let (mut config, mut facts) =
82        frensense_engine::analysis::taint::facts::tables_from_exts(exts.iter().copied());
83    let pack_entries = crate::format::default_pack().learned_facts.as_slice();
84    facts.merge(
85        &frensense_engine::analysis::taint::facts::fact_table_from_entries_with(
86            pack_entries,
87            frensense_engine::analysis::taint::facts::Provenance::Spec,
88        ),
89    );
90    // Phase 6.2: pack language-keyed sections, same layer and order as the
91    // CLI runner's pack merge. Since Phase 6.2d they are the sole source of
92    // the provider knowledge (the spec seed carries none), so gate and scan
93    // stay in lockstep.
94    frensense_engine::analysis::taint::facts::apply_language_entries(
95        &mut config,
96        &mut facts,
97        pack_entries,
98        &frensense_engine::analysis::taint::facts::languages_for_exts(exts),
99    );
100    (config, facts)
101}

103fn extract_facts_with(
104    families: &[Family],
105    tables_for: &dyn Fn(&Family) -> (TaintConfig, FactTable),
106) -> (FactTable, Vec<LearnedFact>) {
107    // Each family's tables resolve once; the proposal pass, the baselines
108    // and every gate trial all run under them.
109    let fam_tables: FxHashMap<String, (TaintConfig, FactTable)> = families
110        .iter()
111        .map(|f| (f.id.clone(), tables_for(f)))
112        .collect();
113
114    // votes: candidate-key -> (support, families)
115    let mut votes: FxHashMap<String, (Candidate, u32, Vec<String>)> = FxHashMap::default();
116    // Guard alternatives per `check:{rule}:{call}` key: families sharing a
117    // trigger verb propose the check with the helper THEIR negatives use;
118    // a union lets one fact accept every voting family's guard shape.
119    let mut check_guards: FxHashMap<String, std::collections::BTreeSet<String>> =
120        FxHashMap::default();
121
122    // Pass 1: proposals per family (accumulate votes). The trace notes are
123    // kept alongside so a family that teaches nothing can explain itself in
124    // the learn report instead of the pipeline staying silent.
125    let mut family_notes: FxHashMap<String, Vec<String>> = FxHashMap::default();
126    for f in families {
127        let (f_config, f_builtin) = &fam_tables[&f.id];
128        let (cands, notes) = propose_with_trace(f, f_config, f_builtin);
129        family_notes.insert(f.id.clone(), notes);
130        for c in cands {
131            let key = match &c {
132                Candidate::Sink { call, package, .. } => match package {
133                    Some(p) => format!("sink:{p}:{call}"),
134                    None => format!("sink:{call}"),
135                },
136                Candidate::Sanitizer {
137                    call,
138                    package,
139                    guard,
140                    ..
141                } => match package {
142                    Some(p) => format!("san:{p}:{call}:{guard}"),
143                    None => format!("san:{call}:{guard}"),
144                },
145                Candidate::Check { rule, call, .. } => format!("check:{rule}:{call}"),
146                Candidate::Policy { rule, call, .. } => format!("policy:{rule}:{call}"),
147                Candidate::MemoryContract { name, .. } => format!("mem:{name}"),
148                Candidate::WeakCrypto { fact } => format!(
149                    "weak_crypto:{}:{}:{:?}",
150                    fact.rule_id, fact.call, fact.selector_slot
151                ),
152                Candidate::GuardBypass { fact } => format!(
153                    "gb:{:?}:{:?}:{:?}",
154                    fact.containment_callees, fact.credential_sinks, fact.credential_params
155                ),
156                Candidate::SchemaPolicy { fact } => format!(
157                    "sp:{:?}:{:?}:{:?}",
158                    fact.builders, fact.enforcers, fact.bound_keywords
159                ),
160                Candidate::IntegerOverflowRule { rule, .. } => format!("io_rule:{rule}"),
161            };
162
163            // Collect the enforcement helpers for a possible guard union.
164            if let Candidate::Check {
165                unless_guard: Some(guard),
166                ..
167            } = &c
168            {
169                check_guards
170                    .entry(key.clone())
171                    .or_default()
172                    .insert(guard.clone());
173            }
174            let entry = votes.entry(key).or_insert_with(|| (c, 0, Vec::new()));
175            entry.1 += 1;
176            if !entry.2.contains(&f.id) {
177                entry.2.push(f.id.clone());
178            }
179        }
180    }
181
182    // Guard-union consolidation: several families can propose the same
183    // check fact with DIFFERENT enforcement helpers (each learned from its
184    // own negatives). The legacy Check fact carries one helper, so the
185    // first proposal won and the engine then alerted on twins whose guard
186    // used one of the discarded names (registry/network guard variants).
187    // When more than one guard name exists for a key, publish a
188    // generalized Policy whose RequireCall accepts ANY of them executing
189    // before the trigger.
190    for (key, guards) in &check_guards {
191        if guards.len() < 2 {
192            continue;
193        }
194        let Some(entry) = votes.get_mut(key) else {
195            continue;
196        };
197        let Some((rule, call, message, package)) = (match &entry.0 {
198            // A range-check modality cannot merge with the guard
199            // disjunction (require lists are conjunctive): keep the
200            // legacy fact intact.
201            Candidate::Check {
202                rule,
203                call,
204                message,
205                package,
206                unless_range_check: None,
207                ..
208            } => Some((rule.clone(), call.clone(), message.clone(), package.clone())),
209            _ => None,
210        }) else {
211            continue;
212        };
213        entry.0 = Candidate::Policy {
214            rule,
215            call,
216            message,
217            require: vec![PolicyRequirement::RequireCall {
218                any_of: guards.iter().cloned().collect(),
219            }],
220            scope: PolicyScope::Function,
221            package,
222        };
223    }
224
225    // Pass 2: replay gate. Every family is lowered once up front, under
226    // that family's own seeded tables (spec + default pack) so gate
227    // lowering and CLI lowering see the same pack vocabulary; each
228    // candidate trial then re-runs only the taint engine + checker per
229    // family (no re-lowering). Baseline separation is measured under each
230    // family's own language tables; a candidate fact must be validated by
231    // at least one voting family that separates under it, and must NOT
232    // break any family that separated at baseline nor newly-noise any
233    // family whose baseline negative was clean. Only families whose
234    // baseline negative ALREADY alerts are exempt from the regression
235    // check: their noise predates the fact and gating on it would reject
236    // every fact.
237    let prepared: Vec<PreparedFamily> = families
238        .iter()
239        .filter_map(|f| {
240            let Some((_, facts)) = fam_tables.get(&f.id) else {
241                eprintln!("[facts] skip family {}: no seeded tables", f.id);
242                return None;
243            };
244            match PreparedFamily::new(f, facts) {
245                Ok(p) => Some(p),
246                Err(e) => {
247                    eprintln!("[facts] skip family {}: {e}", f.id);
248                    None
249                }
250            }
251        })
252        .collect();
253
254    // Baseline per family under its own language tables: both alert flags,
255    // not just the boolean verdict, so every report line can say whether
256    // the positive is silent or the negative is noisy.
257    let mut baseline: FxHashMap<String, Baseline> = FxHashMap::default();
258    for p in &prepared {
259        let (f_config, f_builtin) = &fam_tables[&p.id];
260        let (pos_alerts, neg_alerts) = p.alert_flags(f_config, f_builtin);
261        baseline.insert(
262            p.id.clone(),
263            Baseline {
264                positive_alerts: pos_alerts,
265                negative_alerts: neg_alerts,
266                separates: pos_alerts && !neg_alerts,
267            },
268        );
269    }
270    if std::env::var("FXDBG").is_ok() {
271        for p in &prepared {
272            if let Some(b) = baseline.get(&p.id) {
273                eprintln!(
274                    "[facts] family {}: baseline positive_alerts={} negative_alerts={} separates={}",
275                    p.id, b.positive_alerts, b.negative_alerts, b.separates
276                );
277            }
278        }
279    }
280
281    let mut learned = FactTable::default();
282    let mut published: Vec<LearnedFact> = Vec::new();
283
284    // Deterministic gate order: `votes` is an `FxHashMap`, and its
285    // iteration order decides equal-key tie-breaks downstream. Sort by
286    // gate tier, then key, so bundle bytes never depend on hasher layout
287    // or insertion history. The tier puts pest controllers first: a
288    // sanitizer that silences findings must be learned before any
289    // check/policy is trialed, or the regression gate measures noise
290    // against a context that the final bundle will never scan.
291    let mut ordered_votes: Vec<(String, Candidate, u32, Vec<String>)> = votes
292        .into_iter()
293        .map(|(key, (cand, support, fams))| (key, cand, support, fams))
294        .collect();
295    ordered_votes.sort_by(|a, b| {
296        let tier = |key: &str| {
297            if key.starts_with("san:") {
298                0
299            } else if key.starts_with("sink:") {
300                1
301            } else {
302                2
303            }
304        };
305        tier(&a.0).cmp(&tier(&b.0)).then_with(|| a.0.cmp(&b.0))
306    });
307    // Sink candidates that separate their voting family, per family. Two
308    // different shapes produce several tainted calls in one positive:
309    //
310    // * the defect followed by an echo of its result (Django `Trunc` then
311    //   `JsonResponse`): both separate, and the later one is not a sink;
312    // * a wrapper followed by the defect (unzipper `Open.buffer` then
313    //   `extract`): only the later one separates, so it is not an echo.
314    //
315    // Separation is what distinguishes them, so the echo test may only look
316    // at siblings that prove it themselves - a candidate that fails
317    // separation cannot explain another's taint.
318    let mut separating_sinks_by_family: std::collections::BTreeMap<
319        String,
320        std::collections::BTreeSet<String>,
321    > = std::collections::BTreeMap::new();
322    for (_, cand, _, fams) in &ordered_votes {
323        let Candidate::Sink { call, .. } = cand else {
324            continue;
325        };
326        for fam in fams {
327            let Some(p) = prepared.iter().find(|p| &p.id == fam) else {
328                continue;
329            };
330            let (f_config, f_builtin) = &fam_tables[&p.id];
331            let mut trial = f_builtin.clone();
332            apply_candidate(&mut trial, cand);
333            if p.separates(f_config, &trial) {
334                separating_sinks_by_family
335                    .entry(fam.clone())
336                    .or_default()
337                    .insert(call.clone());
338            }
339        }
340    }
341    // Candidates already published in this pass (gate context for
342    // regression baselines; see the non-voter regression check below).
343    let mut published_cands: Vec<Candidate> = Vec::new();
344
345    let mut candidate_reports: Vec<CandidateReport> = Vec::new();
346
347    for (key, cand, support, fams) in ordered_votes {
348        // The fact must be validated by at least one voting family that
349        // separates under it, and must not break any family that separated
350        // at baseline. Trials run under each family's own tables.
351        if std::env::var("FXDBG").is_ok() {
352            eprintln!("[gate] candidate key={} support={}", key, support);
353        }
354        // Candidate call names (trigger + guard), a family whose variants
355        // never call any of them cannot change separation under this fact,
356        // so keep its baseline verdict without re-scanning.
357        let cand_calls: Vec<&str> = match &cand {
358            Candidate::Sink { call, .. } => vec![call.as_str()],
359            Candidate::Sanitizer { call, .. } => vec![call.as_str()],
360            Candidate::Check {
361                call, unless_guard, ..
362            } => match unless_guard {
363                Some(g) => vec![call.as_str(), g.as_str()],
364                None => vec![call.as_str()],
365            },
366            // Trigger plus every requirement's call names: a family whose
367            // variants never touch any of them cannot change separation.
368            Candidate::Policy { call, require, .. } => {
369                let mut calls = vec![call.as_str()];
370                for req in require {
371                    for name in req.call_names() {
372                        if !calls.contains(&name) {
373                            calls.push(name);
374                        }
375                    }
376                }
377                calls
378            }
379            Candidate::MemoryContract { name, .. } => vec![name.as_str()],
380            Candidate::WeakCrypto { fact } => vec![fact.call.as_str()],
381            Candidate::GuardBypass { fact } => {
382                let mut calls = Vec::new();
383                for c in &fact.containment_callees {
384                    calls.push(c.as_str());
385                }
386                for s in &fact.credential_sinks {
387                    calls.push(s.as_str());
388                }
389                calls
390            }
391            Candidate::SchemaPolicy { fact } => {
392                let mut calls = Vec::new();
393                for b in &fact.builders {
394                    calls.push(b.as_str());
395                }
396                for e in &fact.enforcers {
397                    calls.push(e.as_str());
398                }
399                for k in &fact.bound_keywords {
400                    calls.push(k.as_str());
401                }
402                calls
403            }
404            // The prover rule fires on allocation shapes, not a named call:
405            // empty = always relevant, every family re-checked for
406            // regression under this fact.
407            Candidate::IntegerOverflowRule { .. } => Vec::new(),
408        };
409
410        let mut reject: Option<String> = None;
411        // Existential vote: at least one voting family must separate under
412        // the fact. The old rule required EVERY voting family to separate
413        // (`voted && !sep` -> reject), which let a voter whose own baseline
414        // never separates (blind positive, noisy negative) permanently veto
415        // the fact its sibling family needs - it can never confirm anything,
416        // so it must not be able to block anything either. No voter
417        // separating still rejects: an unvalidated fact must not publish.
418        let mut any_voter_separates = false;
419        let mut first_failing_voter: Option<(String, bool, bool)> = None;
420        for p in &prepared {
421            let (f_config, f_builtin) = &fam_tables[&p.id];
422            let voted = fams.contains(&p.id);
423            let relevant =
424                voted || cand_calls.is_empty() || cand_calls.iter().any(|c| p.calls.contains(*c));
425            let sep = if relevant {
426                let mut trial = f_builtin.clone();
427                apply_candidate(&mut trial, &cand);
428                p.separates(f_config, &trial)
429            } else {
430                baseline.get(&p.id).map(|b| b.separates).unwrap_or(false)
431            };
432            if voted {
433                if sep {
434                    // A downstream echo separates exactly like the defect whose
435                    // result it consumes. Publishing it teaches a sink that
436                    // fires on any tainted value of that shape, so the family's
437                    // knowledge must ride on the upstream call only.
438                    let echoed = match (&cand, separating_sinks_by_family.get(&p.id)) {
439                        (Candidate::Sink { call, .. }, Some(siblings)) => {
440                            let mut echo_trial = f_builtin.clone();
441                            apply_candidate(&mut echo_trial, &cand);
442                            p.downstream_echo(f_config, &echo_trial, call, siblings)
443                                .map(|upstream| (call.clone(), upstream))
444                        }
445                        _ => None,
446                    };
447                    if let Some((call, upstream)) = echoed {
448                        reject = Some(format!(
449                            "downstream echo: `{call}` only receives a value `{upstream}` already \
450                             carries, so it re-names the same defect with a wider blast radius"
451                        ));
452                        break;
453                    }
454                    any_voter_separates = true;
455                } else if first_failing_voter.is_none() {
456                    // Remember which side still fails: "does not separate"
457                    // alone cannot be acted on in the learn report.
458                    let mut trial = f_builtin.clone();
459                    apply_candidate(&mut trial, &cand);
460                    let (pos_alerts, neg_alerts) = p.alert_flags(f_config, &trial);
461                    first_failing_voter = Some((p.id.clone(), pos_alerts, neg_alerts));
462                }
463            }
464            if !relevant {
465                // The fact cannot change this family at all: its baseline
466                // verdict holds by construction.
467                continue;
468            }
469            // Effective baseline: the family's tables plus every fact
470            // already published in this pass. The final bundle scans that
471            // accumulated context (facts interact: a sibling family's
472            // sanitizer silences a noisy negative before a co-occurring
473            // policy is even considered), and the gate tier order puts
474            // sanitizers first so the context is already learned.
475            let mut base_trial = f_builtin.clone();
476            for pc in &published_cands {
477                apply_candidate(&mut base_trial, pc);
478            }
479            let (base_pos, base_neg) = p.alert_flags(f_config, &base_trial);
480            let mut trial = base_trial.clone();
481            apply_candidate(&mut trial, &cand);
482            let (pos_alerts, neg_alerts) = p.alert_flags(f_config, &trial);
483            // Negative-noise regression: applies to every relevant family,
484            // voters included — no family may confirm a fact whose cost is
485            // its own sibling twin turning noisy. Failed separation stays a
486            // non-veto (the existential vote above).
487            if !base_neg && neg_alerts {
488                reject = Some(format!(
489                    "cross-family regression: '{}' negative was clean at baseline but alerts \
490                     under this fact (positive_alerts={pos_alerts}, negative_alerts={neg_alerts})",
491                    p.id
492                ));
493                break;
494            }
495            if !voted && base_pos && !(pos_alerts && !neg_alerts) {
496                // Cross-family regression: the fact broke a previously
497                // separating family. Reject the fact (shape-thinking creep).
498                reject = Some(format!(
499                    "cross-family regression: '{}' separated at baseline but not under this \
500                     fact (positive_alerts={pos_alerts}, negative_alerts={neg_alerts})",
501                    p.id
502                ));
503                break;
504            }
505        }
506        if reject.is_none() && !any_voter_separates {
507            reject = Some(match first_failing_voter {
508                Some((id, pos_alerts, neg_alerts)) => format!(
509                    "no voting family separates under this fact ('{id}' \
510                     positive_alerts={pos_alerts}, negative_alerts={neg_alerts})"
511                ),
512                None => "no voting family could validate this fact".to_string(),
513            });
514        }
515        // Blast-radius quarantine: a candidate carrying a package may only be
516        // published if the voting family actually imports or declares that package.
517        if reject.is_none() {
518            let cand_pkg = match &cand {
519                Candidate::Sink { package, .. } => package.as_deref(),
520                Candidate::Sanitizer { package, .. } => package.as_deref(),
521                Candidate::Policy { package, .. } => package.as_deref(),
522                // Check candidates carrying a family-declared package are
523                // published as scoped policies: the blast-radius quarantine
524                // must apply to them too.
525                Candidate::Check { package, .. } => package.as_deref(),
526                _ => None,
527            };
528            if let Some(pkg) = cand_pkg {
529                let pkg_supported = fams.iter().any(|fam_id| {
530                    families
531                        .iter()
532                        .find(|f| &f.id == fam_id)
533                        .map_or(false, |f| {
534                            f.metadata.package.as_deref() == Some(pkg) || {
535                                let is_san = matches!(&cand, Candidate::Sanitizer { .. });
536                                let files = if is_san { &f.negatives } else { &f.positives };
537                                files.iter().any(|(_, src, ext)| {
538                                    if let Some(spec) = frensense_lang::spec_for_ext(ext) {
539                                        let mut parser = tree_sitter::Parser::new();
540                                        if parser.set_language(&spec.tree_sitter_language()).is_ok()
541                                        {
542                                            if let Some(tree) = parser.parse(src, None) {
543                                                // Project-scope: candidate packages are
544                                                // already normalized to the engine's
545                                                // token model (`node:crypto` ->
546                                                // `crypto`), so compare the import's
547                                                // normalized token, not the raw one.
548                                                return spec
549                                                    .extract_imports(tree.root_node(), src)
550                                                    .iter()
551                                                    .any(|i| {
552                                                        crate::extract::propose::normalize_pkg(
553                                                            &i.package,
554                                                        ) == pkg
555                                                    });
556                                            }
557                                        }
558                                    }
559                                    false
560                                })
561                            }
562                        })
563                });
564                if !pkg_supported {
565                    reject = Some(format!(
566                        "blast-radius quarantine: candidate package '{pkg}' not imported by voting family"
567                    ));
568                }
569            }
570        }
571        // Sanitizer scope is mandatory: a fact with no class scope cannot be
572        // safely applied (the engine would fall back to the legacy
573        // cross-class cut), and a bare ubiquitous name is rejected unless the
574        // family supplied a package/receiver scope.
575        if reject.is_none() {
576            if let Candidate::Sanitizer {
577                call,
578                package,
579                classes,
580                ..
581            } = &cand
582            {
583                if classes.is_empty() {
584                    reject = Some(
585                        "sanitizer candidate has no class scope: the family must declare `class:`"
586                            .to_string(),
587                    );
588                } else if package.is_none()
589                    && crate::lint::UBIQUITOUS_SANITIZERS
590                        .contains(&call.to_ascii_lowercase().as_str())
591                {
592                    reject = Some(format!(
593                        "sanitizer `{call}` is a ubiquitous name with no package/receiver scope"
594                    ));
595                }
596            }
597        }
598        if let Some(reason) = reject {
599            if std::env::var("FXDBG").is_ok() {
600                eprintln!("[gate] REJECT key={key} support={support} reason={reason}");
601            }
602            candidate_reports.push(CandidateReport {
603                key: key.clone(),
604                support,
605                voted_by: fams.clone(),
606                verdict: "rejected",
607                reason: Some(reason),
608            });
609            continue;
610        }
611        if std::env::var("FXDBG").is_ok() {
612            eprintln!("[gate] PUBLISH key={key} support={support}");
613        }
614        candidate_reports.push(CandidateReport {
615            key: key.clone(),
616            support,
617            voted_by: fams.clone(),
618            verdict: "published",
619            reason: None,
620        });
621
622        apply_candidate(&mut learned, &cand);
623        published_cands.push(cand.clone());
624        let lang = fams.first().and_then(|id| {
625            families
626                .iter()
627                .find(|f| &f.id == id)
628                .and_then(|f| f.language())
629        });
630        // A learned sanitizer is scoped to the classes its voting families
631        // declare. Union (deduplicated, sorted) so a fact proven by two
632        // families covers both; the family on the wire is the first voter,
633        // for `--show-suppressed` provenance.
634        let mut classes: Vec<String> = fams
635            .iter()
636            .filter_map(|id| {
637                families
638                    .iter()
639                    .find(|f| &f.id == id)
640                    .and_then(|f| f.metadata.class.clone())
641            })
642            .collect();
643        classes.sort();
644        classes.dedup();
645        let family = fams.first().cloned();
646        let entry = cand.to_learned_entry_scoped(lang);
647
648        published.push(LearnedFact {
649            entry,
650            classes,
651            family,
652            support,
653            status: if support >= 2 {
654                "confirmed"
655            } else {
656                "provisional"
657            }
658            .into(),
659            families: fams,
660        });
661    }
662
663    emit_learn_report(
664        families,
665        &prepared,
666        &baseline,
667        &family_notes,
668        &candidate_reports,
669        &published,
670    );
671
672    published.sort_by(|a, b| {
673        let ka = fact_key(&a.entry);
674        let kb = fact_key(&b.entry);
675        ka.cmp(&kb)
676    });
677    (learned, published)
678}

680/// Baseline separation evidence for one family (built-in tables only).
681#[derive(Debug, Clone, Copy, Serialize)]
682struct Baseline {
683    positive_alerts: bool,
684    negative_alerts: bool,
685    separates: bool,
686}
687
688/// One candidate's journey through the replay gate. A rejected candidate
689/// used to vanish with no output at all; `reason` is why it vanished.
690#[derive(Debug, Clone, Serialize)]
691struct CandidateReport {
692    key: String,
693    support: u32,
694    voted_by: Vec<String>,
695    verdict: &'static str,
696    reason: Option<String>,
697}
698
699/// Everything the bundler decided about one family.
700#[derive(Debug, Clone, Serialize)]
701struct FamilyReport {
702    id: String,
703    baseline: Baseline,
704    notes: Vec<String>,
705    candidates: Vec<CandidateReport>,
706    learned: Vec<String>,
707}
708
709#[derive(Debug, Serialize)]
710struct LearnSummary {
711    families: usize,
712    candidates: usize,
713    published: usize,
714    rejected: usize,
715    families_not_separating_at_baseline: Vec<String>,
716    families_that_taught_nothing: Vec<String>,
717}
718
719#[derive(Debug, Serialize)]
720struct LearnReport {
721    schema: &'static str,
722    summary: LearnSummary,
723    families: Vec<FamilyReport>,
724}

726/// Per-family outcome log: printed under `FXDBG`, written as JSON when
727/// `FXREPORT=<path>` is set. This is the answer to "why didn't the bundle
728/// learn X": baseline flags say what the engine saw, `notes` say what the
729/// proposal stage did with it, and each candidate's `reason` says why the
730/// replay gate refused it.
731fn emit_learn_report(
732    families: &[Family],
733    prepared: &[PreparedFamily],
734    baseline: &FxHashMap<String, Baseline>,
735    family_notes: &FxHashMap<String, Vec<String>>,
736    candidate_reports: &[CandidateReport],
737    published: &[LearnedFact],
738) {
739    let dbg = std::env::var("FXDBG").is_ok();
740    let report_path = std::env::var("FXREPORT").ok().filter(|p| !p.is_empty());
741    if !dbg && report_path.is_none() {
742        return;
743    }
744
745    let mut ids: Vec<String> = families.iter().map(|f| f.id.clone()).collect();
746    ids.sort();
747    ids.dedup();
748    let prepared_ids: Vec<&str> = prepared.iter().map(|p| p.id.as_str()).collect();
749
750    let mut family_reports = Vec::with_capacity(ids.len());
751    for id in ids {
752        let mut notes = family_notes.get(&id).cloned().unwrap_or_default();
753        if !prepared_ids.contains(&id.as_str()) {
754            notes.push(
755                "family skipped: it could not be lowered, so the replay gate never saw it"
756                    .to_string(),
757            );
758        }
759        let baseline = baseline.get(&id).copied().unwrap_or(Baseline {
760            positive_alerts: false,
761            negative_alerts: false,
762            separates: false,
763        });
764        let fam_candidates: Vec<CandidateReport> = candidate_reports
765            .iter()
766            .filter(|c| c.voted_by.contains(&id))
767            .cloned()
768            .collect();
769        let learned: Vec<String> = published
770            .iter()
771            .filter(|f| f.families.contains(&id))
772            .map(|f| {
773                let (kind, name) = fact_key(&f.entry);
774                format!("{kind}:{name}")
775            })
776            .collect();
777
778        if dbg {
779            let cand_summary = if fam_candidates.is_empty() {
780                "(none proposed)".to_string()
781            } else {
782                fam_candidates
783                    .iter()
784                    .map(|c| {
785                        if c.verdict == "published" {
786                            format!("{}=PUBLISHED", c.key)
787                        } else {
788                            format!("{}=rejected", c.key)
789                        }
790                    })
791                    .collect::<Vec<_>>()
792                    .join(", ")
793            };
794            eprintln!(
795                "[facts] family {id}: baseline positive_alerts={} negative_alerts={} separates={} | candidates: {cand_summary} | learned {} fact(s){}",
796                baseline.positive_alerts,
797                baseline.negative_alerts,
798                baseline.separates,
799                learned.len(),
800                if learned.is_empty() { " <- TAUGHT NOTHING" } else { "" }
801            );
802            if learned.is_empty() {
803                for c in &fam_candidates {
804                    if let Some(r) = &c.reason {
805                        eprintln!("[facts]   rejected {c}: {r}", c = c.key);
806                    }
807                }
808            }
809            for n in &notes {
810                eprintln!("[facts]   note: {n}");
811            }
812        }
813
814        family_reports.push(FamilyReport {
815            id,
816            baseline,
817            notes,
818            candidates: fam_candidates,
819            learned,
820        });
821    }
822
823    let report = LearnReport {
824        schema: "frensense-learn-report/1",
825        summary: LearnSummary {
826            families: family_reports.len(),
827            candidates: candidate_reports.len(),
828            published: candidate_reports
829                .iter()
830                .filter(|c| c.verdict == "published")
831                .count(),
832            rejected: candidate_reports
833                .iter()
834                .filter(|c| c.verdict == "rejected")
835                .count(),
836            families_not_separating_at_baseline: family_reports
837                .iter()
838                .filter(|f| !f.baseline.separates)
839                .map(|f| f.id.clone())
840                .collect(),
841            families_that_taught_nothing: family_reports
842                .iter()
843                .filter(|f| f.learned.is_empty())
844                .map(|f| f.id.clone())
845                .collect(),
846        },
847        families: family_reports,
848    };
849
850    if let Some(path) = report_path {
851        match serde_json::to_string_pretty(&report) {
852            Ok(json) => match std::fs::write(&path, json) {
853                Ok(()) => eprintln!("[facts] learn report written to {path}"),
854                Err(e) => eprintln!("[facts] learn report write failed for {path}: {e}"),
855            },
856            Err(e) => eprintln!("[facts] learn report serialization failed: {e}"),
857        }
858    }
859}
