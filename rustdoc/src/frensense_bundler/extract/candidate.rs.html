14/// A candidate fact proposed by delta analysis.
15///
16/// `#[non_exhaustive]`: an evolving internal DTO. Variant fields (e.g.
17/// `Check.package` in 0.8.0) are added as minor releases; it never escapes the
18/// bundler's functional API, so consumers match with a wildcard arm.
19#[derive(Debug, Clone)]
20#[non_exhaustive]
21pub enum Candidate {
22    /// Call is a dangerous sink at these arg slots.
23    Sink {
24        call: String,
25        package: Option<String>,
26        dangerous: BTreeSet<usize>,
27    },
28    /// Call sanitizes (guard-style or transform).
29    Sanitizer {
30        call: String,
31        package: Option<String>,
32        guard: bool,
33        /// Declared corpus class scope (`bug_classes.json` ids). The bundler
34        /// fills this from the voting family's `class:` automatically.
35        classes: Vec<String>,
36        /// First family that taught the fact, for suppression provenance.
37        family: Option<String>,
38    },
39    /// Call is a policy violation per a corpus-verified non-dataflow rule
40    /// (positive contains the call and alerts-or-should; negative lacks it,
41    /// or enforces it with a guard the positive lacks).
42    Check {
43        rule: String,
44        call: String,
45        message: String,
46        /// Fire only when this guard is absent (shape-b families).
47        unless_guard: Option<String>,
48        /// Fire only when the trigger arg has NO literal range comparison
49        /// (inline enforcement modality).
50        unless_range_check: Option<Vec<String>>,
51        /// Optional package qualification from the family's `// package:`
52        /// metadata. When set, the check is published as a package-scoped
53        /// policy (`PolicyScoped`) so it cannot fire on unrelated files
54        /// that merely share the trigger verb. `None` = unscoped `Check`.
55        package: Option<String>,
56    },
57    /// Generalized co-occurrence policy (the `LearnedFactEntry::Policy`
58    /// shape): trigger + a list of requirements the trigger's scope must
59    /// satisfy.
60    Policy {
61        rule: String,
62        call: String,
63        message: String,
64        require: Vec<PolicyRequirement>,
65        scope: PolicyScope,
66        /// Package qualification: when `Some(pkg)`, the policy only fires
67        /// in files that import `pkg` (mirrors sink/source scoping).
68        package: Option<String>,
69    },
70    /// A custom allocation/deallocation wrapper contract learned from corpus examples.
71    MemoryContract {
72        name: String,
73        returns_fresh: bool,
74        return_capacity: CapacitySpec,
75        consumes_params: Vec<usize>,
76    },
77    /// Weak cryptographic primitive or algorithm selector learned from corpus examples.
78    WeakCrypto { fact: WeakCryptoFact },
79    /// Guard bypass pattern (containment callee, credential sink or credential param).
80    GuardBypass { fact: GuardBypassFact },
81    /// Schema policy rule (number builder, enforcer method, or bound keyword).
82    SchemaPolicy { fact: SchemaPolicyFact },
83    /// Allocation-size integer-overflow rule declared by the family's
84    /// `[frensense] check-rule:` block (corpus extends the prover's rule
85    /// table; the engine's prover stays stable).
86    IntegerOverflowRule {
87        rule: String,
88        wrap_threshold: u128,
89        severity: String,
90        message: String,
91        unbounded_product: bool,
92    },
93}
94
95impl Candidate {
96    pub fn to_learned_entry(&self) -> LearnedFactEntry {
97        self.to_learned_entry_scoped(None)
98    }
99
100    /// Lower a candidate into its wire entry.
101    ///
102    /// Class scope and teaching family are not part of the fact: they are
103    /// published in the payload's `sanitizer_scopes` table (see
104    /// [`crate::fact_extract::LearnedFact`]), so the engine's
105    /// `SanitizerFact` keeps its published shape.
106    pub fn to_learned_entry_scoped(&self, language: Option<&str>) -> LearnedFactEntry {
107        if let Some(lang) = language {
108            match self {
109                Candidate::Sink {
110                    call,
111                    package,
112                    dangerous,
113                } => LearnedFactEntry::LanguageSinkSlots {
114                    language: lang.to_string(),
115                    call: call.clone(),
116                    package: package.clone(),
117                    dangerous_args: dangerous.clone(),
118                    binding_args_safe: false,
119                },
120                Candidate::Sanitizer {
121                    call,
122                    package,
123                    guard,
124                    ..
125                } => LearnedFactEntry::LanguageSanitizer {
126                    language: lang.to_string(),
127                    call: call.clone(),
128                    package: package.clone(),
129                    kind: if *guard {
130                        "allowlist".into()
131                    } else {
132                        "encode".into()
133                    },
134                    guard_style: *guard,
135                },
136                _ => self.to_learned_entry_unscoped(),
137            }
138        } else {
139            self.to_learned_entry_unscoped()
140        }
141    }
142
143    fn to_learned_entry_unscoped(&self) -> LearnedFactEntry {
144        match self {
145            Candidate::Sink {
146                call,
147                package,
148                dangerous,
149            } => LearnedFactEntry::Sink {
150                call: call.clone(),
151                package: package.clone(),
152                dangerous_args: dangerous.clone(),
153                binding_args_safe: false,
154            },
155            Candidate::Sanitizer {
156                call,
157                package,
158                guard,
159                ..
160            } => LearnedFactEntry::Sanitizer {
161                call: call.clone(),
162                package: package.clone(),
163                kind: if *guard {
164                    "allowlist".into()
165                } else {
166                    "encode".into()
167                },
168                guard_style: *guard,
169            },
170            Candidate::Check {
171                rule,
172                call,
173                message,
174                unless_guard,
175                unless_range_check,
176                package,
177            } => {
178                // A family-declared package turns the check into a
179                // package-scoped policy (same engine mechanism as
180                // `Candidate::Policy` + package): the generic presence
181                // trigger can then never collide with unrelated bundle
182                // families that share the verb but not the package import.
183                if let Some(pkg) = package {
184                    let mut require = Vec::new();
185                    if let Some(g) = unless_guard {
186                        require.push(PolicyRequirement::GuardCall { call: g.clone() });
187                    }
188                    if let Some(ops) = unless_range_check {
189                        require.push(PolicyRequirement::RangeCheck { ops: ops.clone() });
190                    }
191                    LearnedFactEntry::PolicyScoped {
192                        rule: rule.clone(),
193                        when_call: call.clone(),
194                        require,
195                        scope: PolicyScope::Function,
196                        message: message.clone(),
197                        severity: "warning".into(),
198                        package: pkg.clone(),
199                    }
200                } else {
201                    LearnedFactEntry::Check {
202                        rule: rule.clone(),
203                        call: call.clone(),
204                        message: message.clone(),
205                        severity: "warning".into(),
206                        unless_guard: unless_guard.clone(),
207                        unless_range_check: unless_range_check.clone(),
208                    }
209                }
210            }
211            Candidate::Policy {
212                rule,
213                call,
214                message,
215                require,
216                scope,
217                package,
218            } => {
219                if let Some(pkg) = package {
220                    LearnedFactEntry::PolicyScoped {
221                        rule: rule.clone(),
222                        when_call: call.clone(),
223                        require: require.clone(),
224                        scope: *scope,
225                        message: message.clone(),
226                        severity: "warning".into(),
227                        package: pkg.clone(),
228                    }
229                } else {
230                    LearnedFactEntry::Policy {
231                        rule: rule.clone(),
232                        when_call: call.clone(),
233                        require: require.clone(),
234                        scope: *scope,
235                        message: message.clone(),
236                        severity: "warning".into(),
237                    }
238                }
239            }
240            Candidate::MemoryContract {
241                name,
242                returns_fresh,
243                return_capacity,
244                consumes_params,
245            } => LearnedFactEntry::MemoryContract {
246                name: name.clone(),
247                returns_fresh: *returns_fresh,
248                return_capacity: return_capacity.clone(),
249                consumes_params: consumes_params.clone(),
250            },
251            Candidate::WeakCrypto { fact } => LearnedFactEntry::WeakCrypto(fact.clone()),
252            Candidate::GuardBypass { fact } => LearnedFactEntry::GuardBypass(fact.clone()),
253            Candidate::SchemaPolicy { fact } => LearnedFactEntry::SchemaPolicy(fact.clone()),
254            Candidate::IntegerOverflowRule {
255                rule,
256                wrap_threshold,
257                severity,
258                message,
259                unbounded_product,
260            } => {
261                // The wire schema is positional: the base `IntegerOverflowRule`
262                // variant keeps its byte layout, and the modality travels as
263                // the trailing `IntegerOverflowUnboundedRule` variant instead
264                // of a new field.
265                let payload: LearnedFactEntry = if *unbounded_product {
266                    LearnedFactEntry::IntegerOverflowUnboundedRule {
267                        rule: rule.clone(),
268                        wrap_threshold: *wrap_threshold,
269                        severity: severity.clone(),
270                        message: message.clone(),
271                    }
272                } else {
273                    LearnedFactEntry::IntegerOverflowRule {
274                        rule: rule.clone(),
275                        wrap_threshold: *wrap_threshold,
276                        severity: severity.clone(),
277                        message: message.clone(),
278                    }
279                };
280                payload
281            }
282        }
283    }
284}

286/// Apply a candidate to a fact table (for replay).
287pub fn apply_candidate(table: &mut FactTable, c: &Candidate) {
288    match c {
289        Candidate::Sink {
290            call,
291            package,
292            dangerous,
293        } => {
294            let mut sig = if dangerous.is_empty() {
295                SinkSignature::all_args(call)
296            } else {
297                SinkSignature::with_args(call, &dangerous.iter().copied().collect::<Vec<_>>())
298            };
299            sig.package = package.clone();
300            match package {
301                Some(pkg) => {
302                    // Package-scoped sink: install ONLY under (package, call),
303                    // exactly as `LearnedFactEntry::apply_with` publishes it.
304                    // A universal alias here would replay the fact with
305                    // semantics no consumer loads: the gate would approve a
306                    // sink that arms on any receiver, while the shipped
307                    // bundle arms only under a resolved origin. Same rule the
308                    // sanitizer arm below spells out.
309                    table
310                        .scoped_sink_signatures
311                        .insert((pkg.clone(), call.clone()), sig);
312                }
313                None => {
314                    table.sink_signatures.insert(call.clone(), sig);
315                }
316            }
317        }
318        Candidate::Sanitizer {
319            call,
320            package,
321            guard,
322            classes,
323            family: _,
324        } => {
325            let fact = SanitizerFact {
326                call: call.clone(),
327                package: package.clone(),
328                kind: if *guard {
329                    "allowlist".into()
330                } else {
331                    "encode".into()
332                },
333                sanitizes_args: Default::default(),
334                guard_style: *guard,
335            };
336            // The trial table replays exactly the shape a consumer installs:
337            // the fact in its package slot, then re-scoped to the classes the
338            // families proved (`FactTable::scope_sanitizer`). Installing it
339            // unscoped here would let the gate approve a fact that cuts
340            // another class once loaded.
341            match package {
342                Some(pkg) => {
343                    table
344                        .scoped_sanitizer_facts
345                        .insert((pkg.clone(), call.clone()), fact);
346                }
347                None => {
348                    table.sanitizer_facts.insert(call.clone(), fact);
349                }
350            }
351            table.scope_sanitizer(call, package.as_deref(), classes);
352        }
353        Candidate::Check {
354            rule,
355            call,
356            message,
357            unless_guard,
358            unless_range_check,
359            package,
360        } => {
361            // Scoped checks are replayed as package-scoped policies so the
362            // trial scan exercises the exact publish shape.
363            if let Some(pkg) = package {
364                let mut require = Vec::new();
365                if let Some(g) = unless_guard {
366                    require.push(PolicyRequirement::GuardCall { call: g.clone() });
367                }
368                if let Some(ops) = unless_range_check {
369                    require.push(PolicyRequirement::RangeCheck { ops: ops.clone() });
370                }
371                table.policy_facts.push((
372                    PolicyFact {
373                        rule: rule.clone(),
374                        when_call: call.clone(),
375                        require,
376                        scope: PolicyScope::Function,
377                        message: message.clone(),
378                        severity: "warning".into(),
379                        package: Some(pkg.clone()),
380                    },
381                    Provenance::Learned,
382                ));
383            } else {
384                table.learned_checks.push((
385                    LearnedCheckFact {
386                        rule: rule.clone(),
387                        call: call.clone(),
388                        message: message.clone(),
389                        severity: "warning".into(),
390                        unless_guard: unless_guard.clone(),
391                        unless_range_check: unless_range_check.clone(),
392                    },
393                    Provenance::Learned,
394                ));
395            }
396        }
397        Candidate::Policy {
398            rule,
399            call,
400            message,
401            require,
402            scope,
403            package,
404        } => {
405            table.policy_facts.push((
406                PolicyFact {
407                    rule: rule.clone(),
408                    when_call: call.clone(),
409                    require: require.clone(),
410                    scope: *scope,
411                    message: message.clone(),
412                    severity: "warning".into(),
413                    package: package.clone(),
414                },
415                Provenance::Learned,
416            ));
417        }
418        Candidate::MemoryContract {
419            name,
420            returns_fresh,
421            return_capacity,
422            consumes_params,
423        } => {
424            let fact = MemoryContractFact {
425                name: name.clone(),
426                returns_fresh: *returns_fresh,
427                return_capacity: return_capacity.clone(),
428                consumes_params: consumes_params.clone(),
429            };
430            if let Some(existing) = table
431                .memory_contracts
432                .iter_mut()
433                .find(|c| c.name == fact.name)
434            {
435                *existing = fact;
436            } else {
437                table.memory_contracts.push(fact);
438            }
439        }
440        Candidate::WeakCrypto { fact } => {
441            if let Some(existing) = table
442                .weak_crypto_rules
443                .iter_mut()
444                .find(|c| c.rule_id == fact.rule_id && c.call == fact.call)
445            {
446                *existing = fact.clone();
447            } else {
448                table.weak_crypto_rules.push(fact.clone());
449            }
450        }
451        Candidate::GuardBypass { fact } => {
452            for c in &fact.containment_callees {
453                table
454                    .containment_callees
455                    .insert(c.clone(), Provenance::Learned);
456            }
457            for s in &fact.credential_sinks {
458                table
459                    .credential_sinks
460                    .insert(s.clone(), Provenance::Learned);
461            }
462            for p in &fact.credential_params {
463                table
464                    .credential_params
465                    .insert(p.clone(), Provenance::Learned);
466            }
467        }
468        Candidate::SchemaPolicy { fact } => {
469            for b in &fact.builders {
470                table.schema_builders.insert(b.clone(), Provenance::Learned);
471            }
472            for e in &fact.enforcers {
473                table.schema_enforcers.insert(e.clone());
474            }
475            for k in &fact.bound_keywords {
476                table.schema_keywords.insert(k.clone(), Provenance::Learned);
477            }
478        }
479        Candidate::IntegerOverflowRule {
480            rule,
481            wrap_threshold,
482            severity,
483            message,
484            unbounded_product,
485        } => {
486            let fact = frensense_engine::analysis::taint::facts::IntegerOverflowRule {
487                rule_id: rule.clone(),
488                wrap_threshold: *wrap_threshold,
489                severity: severity.clone(),
490                message: message.clone(),
491                unbounded_product: *unbounded_product,
492            };
493            if let Some(existing) = table
494                .integer_overflow_rules
495                .iter_mut()
496                .find(|(r, _)| r.rule_id == fact.rule_id && r.wrap_threshold == fact.wrap_threshold)
497            {
498                *existing = (fact, Provenance::Learned);
499            } else {
500                table
501                    .integer_overflow_rules
502                    .push((fact, Provenance::Learned));
503            }
504        }
505    }
506}

508pub fn fact_key(e: &LearnedFactEntry) -> (String, String) {
509    match e {
510        LearnedFactEntry::Source { pattern, .. } => ("source".into(), pattern.clone()),
511        LearnedFactEntry::Sink { call, .. } => ("sink".into(), call.clone()),
512        LearnedFactEntry::Sanitizer { call, .. } => ("san".into(), call.clone()),
513        LearnedFactEntry::Check { rule, call, .. } => ("check".into(), format!("{rule}:{call}")),
514        LearnedFactEntry::Policy {
515            rule, when_call, ..
516        } => ("policy".into(), format!("{rule}:{when_call}")),
517        LearnedFactEntry::PolicyScoped {
518            rule, when_call, ..
519        } => ("policy".into(), format!("{rule}:{when_call}")),
520        LearnedFactEntry::MemoryContract { name, .. } => ("mem".into(), name.clone()),
521        LearnedFactEntry::WeakCrypto(f) => {
522            ("weak_crypto".into(), format!("{}:{}", f.rule_id, f.call))
523        }
524        LearnedFactEntry::GuardBypass(f) => (
525            "guard_bypass".into(),
526            format!(
527                "{:?}:{:?}:{:?}",
528                f.containment_callees, f.credential_sinks, f.credential_params
529            ),
530        ),
531        LearnedFactEntry::SchemaPolicy(f) => (
532            "schema_policy".into(),
533            format!("{:?}:{:?}:{:?}", f.builders, f.enforcers, f.bound_keywords),
534        ),
535        LearnedFactEntry::GrammarRole {
536            language,
537            node_kind,
538            ..
539        } => ("grammar_role".into(), format!("{language}:{node_kind}")),
540        LearnedFactEntry::GrammarFeature {
541            language,
542            node_kind,
543            feature,
544        } => (
545            "grammar_feature".into(),
546            format!("{language}:{node_kind}:{feature:?}"),
547        ),
548        LearnedFactEntry::Allocator { name } => ("allocator".into(), name.clone()),
549        LearnedFactEntry::Deallocator { name } => ("deallocator".into(), name.clone()),
550        LearnedFactEntry::IdorFinderSink { call, .. } => ("idor_sink".into(), call.clone()),
551        LearnedFactEntry::IdorKey { key } => ("idor_key".into(), key.clone()),
552        LearnedFactEntry::Propagator { call, .. } => ("propagator".into(), call.clone()),
553        LearnedFactEntry::GuardDenylistPattern { pattern } => {
554            ("guard_denylist".into(), pattern.clone())
555        }
556        LearnedFactEntry::IntegerOverflowRule { rule, .. }
557        | LearnedFactEntry::IntegerOverflowUnboundedRule { rule, .. } => {
558            ("io_rule".into(), rule.clone())
559        }
560        LearnedFactEntry::Hints { kind, .. } => ("hints".into(), format!("{kind:?}")),
561        LearnedFactEntry::WeakPrimitiveRule { rule_id, .. } => {
562            ("weak_primitive".into(), rule_id.clone())
563        }
564        LearnedFactEntry::InsecureConfigRule { rule_id, .. } => {
565            ("insecure_config".into(), rule_id.clone())
566        }
567        LearnedFactEntry::KeySizeRule { rule_id, call, .. } => {
568            ("key_size".into(), format!("{rule_id}:{call}"))
569        }
570        LearnedFactEntry::SuspiciousHashWrappers { calls } => {
571            ("hash_wrappers".into(), calls.join(","))
572        }
573        LearnedFactEntry::BufferBuiltin { name, .. } => ("buffer_builtin".into(), name.clone()),
574        LearnedFactEntry::LanguageSink { language, call, .. } => {
575            ("language_sink".into(), format!("{language}:{call}"))
576        }
577        LearnedFactEntry::LanguageSinkGlobalOnly { language, call } => (
578            "language_sink_global_only".into(),
579            format!("{language}:{call}"),
580        ),
581        LearnedFactEntry::LanguageSinkSlots { language, call, .. } => {
582            ("language_sink_slots".into(), format!("{language}:{call}"))
583        }
584        LearnedFactEntry::LanguageIdorSink { language, call, .. } => {
585            ("language_idor_sink".into(), format!("{language}:{call}"))
586        }
587        LearnedFactEntry::LanguageSource {
588            language, pattern, ..
589        } => ("language_source".into(), format!("{language}:{pattern}")),
590        LearnedFactEntry::LanguageSanitizer { language, call, .. } => {
591            ("language_sanitizer".into(), format!("{language}:{call}"))
592        }
593        LearnedFactEntry::LanguagePropagator { language, call, .. } => {
594            ("language_propagator".into(), format!("{language}:{call}"))
595        }
596        LearnedFactEntry::LanguageSessionRoot { language, root } => {
597            ("language_session_root".into(), format!("{language}:{root}"))
598        }
599        LearnedFactEntry::LanguageRoutePattern { language, pattern } => (
600            "language_route_pattern".into(),
601            format!("{language}:{pattern}"),
602        ),
603        // Phase 6.5 structural vocabularies: pack-only whole-table
604        // entries, keyed by their contents like the other table-shaped
605        // facts.
606        LearnedFactEntry::StackAllocators { values } => {
607            ("stack_allocators".into(), values.join(","))
608        }
609        LearnedFactEntry::CollectionConstructors { values } => {
610            ("collection_constructors".into(), values.join(","))
611        }
612        LearnedFactEntry::SchemaDescribeMethods { values } => {
613            ("schema_describe_methods".into(), values.join(","))
614        }
615        LearnedFactEntry::NullTokens { values } => ("null_tokens".into(), values.join(",")),
616        LearnedFactEntry::SessionAccessors { values } => {
617            ("session_accessors".into(), values.join(","))
618        }
619        LearnedFactEntry::ReceiverParams { values } => ("receiver_params".into(), values.join(",")),
620        LearnedFactEntry::LanguageAmbiguousVerbs { language, values } => (
621            "language_ambiguous_verbs".into(),
622            format!("{language}:{}", values.join(",")),
623        ),
624    }
