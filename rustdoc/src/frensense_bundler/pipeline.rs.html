11fn format_entry_tag(entry: &LearnedFactEntry) -> String {
12    match entry {
13        LearnedFactEntry::Source { pattern, .. }
14        | LearnedFactEntry::LanguageSource { pattern, .. } => format!("source:{pattern}"),
15        LearnedFactEntry::Sink { call, .. }
16        | LearnedFactEntry::LanguageSink { call, .. }
17        | LearnedFactEntry::LanguageSinkSlots { call, .. } => format!("sink:{call}"),
18        LearnedFactEntry::Sanitizer { call, kind, .. }
19        | LearnedFactEntry::LanguageSanitizer { call, kind, .. } => {
20            format!("sanitizer:{call}({kind})")
21        }
22        LearnedFactEntry::Check {
23            rule,
24            call,
25            unless_guard,
26            ..
27        } => {
28            format!("check:{rule}({call}, guard={unless_guard:?})")
29        }
30        LearnedFactEntry::Policy {
31            rule, when_call, ..
32        } => format!("policy:{rule}({when_call})"),
33        LearnedFactEntry::PolicyScoped {
34            rule,
35            when_call,
36            package,
37            ..
38        } => format!("policy:{rule}({when_call}, pkg={package})"),
39        LearnedFactEntry::MemoryContract {
40            name,
41            returns_fresh,
42            consumes_params,
43            ..
44        } => {
45            format!("mem:{name}(fresh={returns_fresh},consumes={consumes_params:?})")
46        }
47        LearnedFactEntry::WeakCrypto(fact) => {
48            format!(
49                "weak_crypto:{}(call={},slot={:?})",
50                fact.rule_id, fact.call, fact.selector_slot
51            )
52        }
53        LearnedFactEntry::IntegerOverflowRule {
54            rule,
55            wrap_threshold,
56            ..
57        } => format!("io_rule:{rule}(max={wrap_threshold})"),
58        _ => format_structural_entry(entry),
59    }
60}

62fn format_structural_entry(entry: &LearnedFactEntry) -> String {
63    match entry {
64        LearnedFactEntry::GuardBypass(f) => format!("guard_bypass(sinks={:?})", f.credential_sinks),
65        LearnedFactEntry::SchemaPolicy(f) => format!("schema_policy(builders={:?})", f.builders),
66        LearnedFactEntry::GrammarRole {
67            language,
68            node_kind,
69            role,
70        } => {
71            format!("grammar_role:{language}:{node_kind}({role:?})")
72        }
73        LearnedFactEntry::GrammarFeature {
74            language,
75            node_kind,
76            feature,
77        } => {
78            format!("grammar_feature:{language}:{node_kind}({feature:?})")
79        }
80        LearnedFactEntry::Allocator { name } => format!("allocator:{name}"),
81        LearnedFactEntry::Deallocator { name } => format!("deallocator:{name}"),
82        LearnedFactEntry::IdorFinderSink { call, keys } => {
83            format!("idor_sink:{call}(keys={keys:?})")
84        }
85        LearnedFactEntry::IdorKey { key } => format!("idor_key:{key}"),
86        LearnedFactEntry::Propagator {
87            call,
88            input_args,
89            preserves_taint,
90        } => {
91            format!("propagator:{call}(args={input_args:?},taints={preserves_taint})")
92        }
93        LearnedFactEntry::GuardDenylistPattern { pattern } => format!("guard_denylist:{pattern}"),
94        _ => "unknown_fact".to_string(),
95    }
96}

98fn print_published_facts(facts: &[LearnedFact]) {
99    for f in facts {
100        let call = format_entry_tag(&f.entry);
101        eprintln!("  [{}] {} <- {}", f.status, call, f.families.join(", "));
102    }
103}

105/// Compile a corpus directory into an .frc facts bundle.
106pub fn run_facts_pipeline(corpus_dir: &Path, output_path: &Path) -> Result<(), String> {
107    run_facts_pipeline_with_trust(corpus_dir, output_path, BundleTrust::Certified)
108}
109
110/// [`run_facts_pipeline`] with an explicit trust tier recorded in the header.
111pub fn run_facts_pipeline_with_trust(
112    corpus_dir: &Path,
113    output_path: &Path,
114    trust: BundleTrust,
115) -> Result<(), String> {
116    let (bytes, facts) = build_facts_bundle_with_trust(corpus_dir, trust)?;
117
118    let loaded = load_bundle(&bytes)?;
119    eprintln!(
120        "[facts] round-trip OK: {} learned facts in bundle ({} effective at this trust tier, {} sanitizer scopes)",
121        loaded.learned_facts.len(),
122        loaded.effective_learned_facts().len(),
123        loaded.effective_sanitizer_scopes().len()
124    );
125
126    print_published_facts(&facts);
127
128    std::fs::write(output_path, &bytes)
129        .map_err(|e| format!("write {}: {e}", output_path.display()))?;
130    eprintln!(
131        "Facts bundle written to {} ({} bytes)",
132        output_path.display(),
133        bytes.len()
134    );
135    Ok(())
136}
