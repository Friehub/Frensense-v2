17/// The per-fact cross-class canary gate.
18///
19/// A learned sanitizer is proven for the classes it names. For every other
20/// class in which the same call is applied by a corpus family, the fact must
21/// leave that family's vulnerable negative alerting: if it silences it, the
22/// fact is cross-class and the build fails. This is the probe that would have
23/// caught both sanitizer-collapse incidents automatically.
24fn verify_sanitizer_class_scope(
25    families: &[crate::fact_extract::Family],
26    published: &[crate::fact_extract::LearnedFact],
27) -> Result<(), String> {
28    for fact in published {
29        let (entry, call, package) = match &fact.entry {
30            LearnedFactEntry::Sanitizer { call, package, .. }
31            | LearnedFactEntry::LanguageSanitizer { call, package, .. } => {
32                (&fact.entry, call, package)
33            }
34            _ => continue,
35        };
36        let classes = &fact.classes;
37        if classes.is_empty() {
38            return Err(format!(
39                "sanitizer `{call}` has no class scope; refusing to publish a cross-class fact"
40            ));
41        }
42        for family in families {
43            let Some(class) = family.metadata.class.as_deref() else {
44                continue;
45            };
46            if classes.iter().any(|c| c == class || c == "*") {
47                continue;
48            }
49            // Only families in which the sanitizer is actually applied can
50            // be a counter-probe, and only when the vulnerable twin would
51            // alert without the fact.
52            let applies = family
53                .negatives
54                .iter()
55                .any(|(_, src, _)| src.contains(call.as_str()));
56            if !applies {
57                continue;
58            }
59            let exts: Vec<&str> = family
60                .negatives
61                .iter()
62                .map(|(_, _, ext)| ext.as_str())
63                .chain(family.positives.iter().map(|(_, _, ext)| ext.as_str()))
64                .collect();
65            let langs = languages_for_exts(exts);
66            let mut config = TaintConfig::default();
67            let mut facts = FactTable::default();
68            apply_language_entries(
69                &mut config,
70                &mut facts,
71                &crate::format::default_pack().learned_facts,
72                &langs,
73            );
74            let baseline = scan::scan(&family.negatives, &config, &facts);
75            if !crate::extract::gate::alerts(&baseline) {
76                continue;
77            }
78            apply_language_entries(&mut config, &mut facts, &[entry.clone()], &langs);
79            // The trial table must apply the same class scope a consumer
80            // installs, or the probe would approve a fact that cuts another
81            // class once loaded.
82            facts.scope_sanitizer(call, package.as_deref(), classes);
83            let probed = scan::scan(&family.negatives, &config, &facts);
84            if !crate::extract::gate::alerts(&probed) {
85                return Err(format!(
86                    "cross-class sanitizer `{call}` (class(es): {}) silenced family `{}` \
87                     (class {class}); a class-scoped fact must leave every other class alerting",
88                    classes.join(", "),
89                    family.id
90                ));
91            }
92        }
93    }
94    Ok(())
95}

97/// §9 pipeline: corpus pairs → replay-verified learned facts → bundle bytes.
98///
99/// Returns the bundle plus a summary of the published facts for reporting.
100pub fn build_facts_bundle(
101    corpus_dir: &std::path::Path,
102) -> Result<(Vec<u8>, Vec<crate::fact_extract::LearnedFact>), String> {
103    build_facts_bundle_with_trust(corpus_dir, BundleTrust::Certified)
104}
105
106/// [`build_facts_bundle`] with an explicit trust tier. `Community` bundles
107/// are filtered by [`crate::format::LoadedBundle::effective_learned_facts`]
108/// on load exactly like any third-party bundle.
109pub fn build_facts_bundle_with_trust(
110    corpus_dir: &std::path::Path,
111    trust: BundleTrust,
112) -> Result<(Vec<u8>, Vec<crate::fact_extract::LearnedFact>), String> {
113    let families = group_families(corpus_dir)?;
114    eprintln!(
115        "[facts] {} families grouped from {}",
116        families.len(),
117        corpus_dir.display()
118    );
119
120    let (learned_table, published) = extract_facts(&families);
121    verify_sanitizer_class_scope(&families, &published)?;
122
123    // Payload: one metadata pattern per family + the published facts.
124    let mut entries: Vec<frensense_engine::analysis::taint::facts::LearnedFactEntry> =
125        Vec::with_capacity(published.len());
126    let mut policy_meta: Vec<crate::format::PolicyMeta> = Vec::new();
127    let mut sanitizer_scopes: Vec<crate::format::SanitizerScope> = Vec::new();
128    for f in &published {
129        entries.push(f.entry.clone());
130        // A policy's tier/lifecycle are authored family metadata. They ride in
131        // the payload's `policy_meta` table rather than in a wire variant, so
132        // the engine's `PolicyFact` keeps its published shape.
133        if let LearnedFactEntry::Policy { rule, .. } | LearnedFactEntry::PolicyScoped { rule, .. } =
134            &f.entry
135        {
136            let meta = f
137                .families
138                .iter()
139                .filter_map(|id| families.iter().find(|fam| &fam.id == id))
140                .map(|fam| &fam.metadata)
141                .find(|m| m.tier.is_some() || m.status.is_some());
142            if let Some(meta) = meta {
143                policy_meta.push(crate::format::PolicyMeta {
144                    rule: rule.clone(),
145                    tier: meta.tier.clone().unwrap_or_default(),
146                    status: meta.status.clone().unwrap_or_default(),
147                });
148            }
149        }
150        // A sanitizer's class scope and teaching family ride in the payload's
151        // `sanitizer_scopes` table: the consumer applies the scope with
152        // `FactTable::scope_sanitizer`, so the fact itself keeps the engine's
153        // published shape.
154        if let LearnedFactEntry::Sanitizer { call, package, .. }
155        | LearnedFactEntry::LanguageSanitizer { call, package, .. } = &f.entry
156        {
157            if !f.classes.is_empty() || f.family.is_some() {
158                sanitizer_scopes.push(crate::format::SanitizerScope {
159                    call: call.clone(),
160                    package: package.clone(),
161                    classes: f.classes.clone(),
162                    family: f.family.clone(),
163                });
164            }
165        }
166    }
167
168    // Tier transparency: a non-certified bundle is filtered on load, so warn
169    // at build time when facts it just verified will not ship.
170    if !matches!(trust, BundleTrust::Certified) {
171        let mut sanitizers = 0usize;
172        let mut shippable = 0usize;
173        for f in &published {
174            let package = match &f.entry {
175                LearnedFactEntry::Sanitizer { package, .. }
176                | LearnedFactEntry::LanguageSanitizer { package, .. } => package,
177                _ => continue,
178            };
179            sanitizers += 1;
180            if trust.may_ship_sanitizer(&f.classes, package.as_deref()) {
181                shippable += 1;
182            }
183        }
184        if shippable < sanitizers {
185            eprintln!(
186                "[facts] trust {trust:?}: {shippable}/{sanitizers} sanitizer facts ship; the rest lack a \
187                 package/receiver scope the tier requires",
188            );
189        }
190    }
191
192    // One advisory pattern per family, sorted by id for a deterministic
193    // payload. Metadata comes from the family's `[frensense]` comment block
194    // (parsed during grouping); families without a block ship all-None.
195    // `rules` joins the pattern to the findings the family's published
196    // facts cause (checker rule ids + learned sink names), so the consumer
197    // can attach this metadata when one of those findings fires.
198    let mut family_ids: Vec<String> = families.iter().map(|f| f.id.clone()).collect();
199    family_ids.sort();
200    let mut patterns: Vec<BundlePattern> = Vec::with_capacity(family_ids.len());
201    let empty = crate::fact_extract::FamilyMetadata::default();
202    for id in &family_ids {
203        let family = families.iter().find(|f| f.id == *id);
204        let meta = family.map(|f| &f.metadata).unwrap_or(&empty);
205        let mut rules: Vec<String> = published
206            .iter()
207            .filter(|f| f.families.iter().any(|fam| fam == id))
208            .flat_map(|f| f.entry.finding_identities())
209            .collect();
210        rules.sort();
211        rules.dedup();
212        patterns.push(BundlePattern {
213            id: id.clone(),
214            observation: meta.observation.clone(),
215            impact: meta.impact.clone(),
216            improvement: meta.improvement.clone(),
217            cwe: meta.cwe.clone(),
218            cvss: meta.cvss,
219            owasp: meta.owasp.clone(),
220            severity: meta.severity.clone(),
221            rules,
222        });
223    }
224
225    let count = entries.len().max(patterns.len()) as u32;
226    // v6 payload; package-qualified provenance facts for LTS 0.7.0.
227    let payload = BundlePayloadV8 {
228        patterns,
229        learned_facts: entries,
230        policy_pack: Vec::new(),
231        sanitizer_scopes,
232        policy_meta,
233    };
234    let _ = learned_table;
235
236    // A corpus bundle is meant to be handed to other people's scanners, so it
237    // declares the engine series its families were replay-gated against.
238    // Bundles without the range still load (the strict vocabulary fingerprint
239    // is always checked), but a distribution that can say "this targets 0.9"
240    // should.
241    let engine = EngineRequirement::parse_triple(frensense_engine::VERSION).unwrap_or([0, 0, 0]);
242    let bytes = write_bundle_pinned_with_trust(
243        &payload,
244        count,
245        EngineRequirement::same_series(engine),
246        trust,
247    )?;
248    Ok((bytes, published))
249}
